```python
def select_engine(self, context: dict) -> tuple:
    """
    选择引擎，确保所有路径都有返回值和reason。
    
    Returns:
        tuple: (engine_name, reason) engine_name为None表示不工作
    """
    # === 初始化所有可能被引用的变量 ===
    mem_reason = "normal"
    selected_engine = None
    
    try:
        # === 0-10点休息策略（显式标注）===
        current_hour = datetime.now().hour
        if 0 <= current_hour < 10:
            mem_reason = "rest_period_0_to_10"
            logger.info(
                f"select_engine: 当前时间{current_hour}点，"
                f"处于0-10点休息时段，引擎休眠"
            )
            return None, mem_reason
        
        # === 内存检查 ===
        if self.memory_check_enabled:
            memory_usage = context.get("memory_usage", 0)
            if memory_usage > self.memory_threshold:
                mem_reason = "memory_exceeded"
                logger.warning(
                    f"select_engine: 内存{memory_usage}%超过阈值"
                    f"{self.memory_threshold}%，暂停引擎"
                )
                return None, mem_reason
            else:
                mem_reason = "memory_ok"
        
        # === 正常引擎选择逻辑 ===
        selected_engine = self._do_select_engine(context)
        if selected_engine is None:
            mem_reason = "no_available_engine"
            logger.warning("select_engine: 无可用引擎")
        else:
            mem_reason = "engine_selected"
        
        logger.info(f"select_engine result: engine={selected_engine}, reason={mem_reason}")
        return selected_engine, mem_reason
        
    except Exception as e:
        mem_reason = f"select_error: {type(e).__name__}"
        logger.error(f"select_engine异常: {e}", exc_info=True)
        return None, mem_reason
```

### 修改说明
1. **变量预初始化**：在函数开头将`mem_reason`初始化为`"normal"`，确保任何路径都有值
2. **所有分支都return**：每个条件分支都有明确的return语句，不依赖变量穿透
3. **异常兜底**：try-except包裹整个逻辑，异常时也返回reason

```python
# === 在类级别定义休息时段配置 ===
class EngineSelector:
    # 休息时段配置（可配置化，不硬编码）
    REST_PERIOD_START = 0   # 休息开始时间（24小时制）
    REST_PERIOD_END = 10    # 休息结束时间
    REST_PERIOD_ENABLED = True  # 是否启用休息时段
    
    def select_engine(self, context: dict) -> tuple:
        """
        选择引擎。
        
        策略说明：
        - 0-10点为休息时段，引擎休眠，返回(None, "rest_period")
        - 10-24点为工作时段，正常选择引擎
        - 休息时段可通过REST_PERIOD_ENABLED配置关闭
        
        Returns:
            tuple: (engine_name, reason)
                - engine_name: 引擎名称，None表示不工作
                - reason: 不工作原因或工作状态
        """
        current_hour = datetime.now().hour
        current_time = datetime.now().strftime("%H:%M:%S")
        
        # === 休息时段检查（显式策略）===
        if self.REST_PERIOD_ENABLED:
            if self.REST_PERIOD_START <= current_hour < self.REST_PERIOD_END:
                reason = (
                    f"rest_period: 当前{current_time}，"
                    f"处于休息时段({self.REST_PERIOD_START}-{self.REST_PERIOD_END}点)"
                )
                logger.info(f"select_engine: {reason}")
                return None, reason
        
        # === 工作时段正常逻辑 ===
        try:
            engine = self._select_working_engine(context)
            if engine:
                reason = f"working_period: 选中引擎{engine}, 当前{current_time}"
                logger.info(f"select_engine: {reason}")
                return engine, reason
            else:
                reason = f"no_engine_available: 当前{current_time}，无可用引擎"
                logger.warning(f"select_engine: {reason}")
                return None, reason
        except Exception as e:
            reason = f"select_error: {type(e).__name__}: {e}"
            logger.error(f"select_engine异常: {reason}", exc_info=True)
            return None, reason
```

### 修改说明
1. **休息策略显式化**：用类常量`REST_PERIOD_START/END/ENABLED`定义休息时段，不硬编码在逻辑中
2. **返回元组**：始终返回`(engine, reason)`，调用方可以区分"休息"和"出错"
3. **日志明确**：休息时记录具体时段和当前时间，便于排查
4. **可配置**：通过`REST_PERIOD_ENABLED`可以关闭休息时段

## P0-3: `max_ratio` 忽略B模型

### 修复前代码
```python
def check_budget(self):
    # 只检查A模型预算，忽略B模型
    a_ratio = self.a_used / self.a_total if self.a_total > 0 else 1.0
    max_ratio = a_ratio  # 只看A
    
    if max_ratio >= 1.0:
        return False  # A用完就停，但B可能还有额度
    
    return True
```

### 修复后代码
```python
def check_budget(self) -> tuple:
    """
    检查预算是否可用。
    
    策略：任一模型有额度就继续工作。
    只有所有模型都耗尽才停止。
    
    Returns:
        tuple: (can_continue, status_dict)
            - can_continue: True表示至少有一个模型有额度
            - status_dict: 各模型的详细状态
    """
    status = {}
    any_available = False
    
    # === 检查A模型 ===
    if self.a_total > 0:
        a_ratio = self.a_used / self.a_total
        a_available = a_ratio < 1.0
        status['A'] = {
            'used': self.a_used,
            'total': self.a_total,
            'ratio': round(a_ratio, 4),
            'available': a_available,
            'remaining': max(0, self.a_total - self.a_used)
        }
        if a_available:
            any_available = True
    else:
        status['A'] = {'available': False, 'reason': 'total_is_zero'}
    
    # === 检查B模型 ===
    if self.b_total > 0:
        b_ratio = self.b_used / self.b_total
        b_available = b_ratio < 1.0
        status['B'] = {
            'used': self.b_used,
            'total': self.b_total,
            'ratio': round(b_ratio, 4),
            'available': b_available,
            'remaining': max(0, self.b_total - self.b_used)
        }
        if b_available:
            any_available = True
    else:
        status['B'] = {'available': False, 'reason': 'total_is_zero'}
    
    # === 检查C模型（如果有）===
    if hasattr(self, 'c_total') and self.c_total > 0:
        c_ratio = self.c_used / self.c_total
        c_available = c_ratio < 1.0
        status['C'] = {
            'used': self.c_used,
            'total': self.c_total,
            'ratio': round(c_ratio, 4),
            'available': c_available,
            'remaining': max(0, self.c_total - self.c_used)
        }
        if c_available:
            any_available = True
    elif hasattr(self, 'c_total'):
        status['C'] = {'available': False, 'reason': 'total_is_zero'}
    
    # === 计算最大使用率（用于日志/监控）===
    ratios = [s.get('ratio', 1.0) for s in status.values() if 'ratio' in s]
    max_ratio = max(ratios) if ratios else 1.0
    
    logger.info(
        f"check_budget: can_continue={any_available}, "
        f"max_ratio={max_ratio:.4f}, "
        f"A={status.get('A', {}).get('ratio', 'N/A')}, "
        f"B={status.get('B', {}).get('ratio', 'N/A')}"
    )
    
    if not any_available:
        logger.warning(
            f"check_budget: 所有模型预算已耗尽! "
            f"status={status}"
        )
    
    return any_available, status


def get_available_engine(self, status_dict: dict = None) -> str:
    """
    根据预算状态选择可用引擎。
    优先选择剩余额度比例最高的引擎。
    
    Args:
        status_dict: check_budget返回的状态字典，为None时自动调用
    
    Returns:
        str: 引擎名称，None表示无可用引擎
    """
    if status_dict is None:
        _, status_dict = self.check_budget()
    
    best_engine = None
    best_remaining_ratio = -1
    
    for engine_name, engine_status in status_dict.items():
        if not engine_status.get('available', False):
            continue
        
        remaining = engine_status.get('remaining', 0)
        total = engine_status.get('total', 1)
        remaining_ratio = remaining / total if total > 0 else 0
        
        if remaining_ratio > best_remaining_ratio:
            best_remaining_ratio = remaining_ratio
            best_engine = engine_name
    
    if best_engine:
        logger.info(
            f"get_available_engine: 选中{best_engine}, "
            f"剩余比例={best_remaining_ratio:.4f}"
        )
    else:
        logger.warning("get_available_engine: 无可用引擎")
    
    return best_engine
```

### 修改说明
1. **任一可用即继续**：`any_available`只要有一个模型有额度就为True
2. **详细状态返回**：返回`status_dict`包含每个模型的used/total/ratio/available/remaining
3. **智能选择引擎**：`get_available_engine`优先选剩余额度比例最高的模型
4. **防御性编程**：处理`total=0`的情况，避免除零错误
5. **可扩展**：C模型用`hasattr`检查，不存在时跳过

## P0-5: `in_flight_tokens` 只增不减

### 修复前代码
```python
async def call_api(self, request):
    # 只增不减
    self.in_flight_tokens += 1  # 增加但从不减少
    
    try:
        result = await self._do_request(request)
        return result
    except Exception:
        raise
    # 无论成功还是失败，in_flight_tokens都没有减少
```

### 修复后代码
```python
import asyncio
from contextlib import asynccontextmanager
from typing import Optional

class EngineManager:
    def __init__(self):
        self._lock = asyncio.Lock()
        self.in_flight_tokens = 0
        self.max_in_flight = 10  # 最大并发数
        
    @asynccontextmanager
    async def _track_in_flight(self, timeout: float = 30.0):
        """
        上下文管理器：跟踪in-flight请求。
        确保无论成功、失败、超时都会释放token。
        
        Args:
            timeout: 请求超时时间（秒）
        """
        acquired = False
        try:
            # === 获取token（带并发保护）===
            async with self._lock:
                if self.in_flight_tokens >= self.max_in_flight:
                    raise RuntimeError(
                        f"in_flight_tokens达到上限: "
                        f"{self.in_flight_tokens}/{self.max_in_flight}"
                    )
                self.in_flight_tokens += 1
                acquired = True
                current = self.in_flight_tokens
                logger.debug(f"in_flight_tokens +1 → {current}")
            
            # === 执行请求（在锁外，不阻塞其他请求）===
            yield
            
        except asyncio.TimeoutError:
            logger.warning(
                f"请求超时，释放in_flight_token. "
                f"当前: {self.in_flight_tokens}"
            )
            raise
        except Exception as e:
            logger.error(
                f"请求异常: {type(e).__name__}, 释放in_flight_token. "
                f"当前: {self.in_flight_tokens}"
            )
            raise
        finally:
            # === 无论成功/失败/超时，都释放token ===
            if acquired:
                async with self._lock:
                    self.in_flight_tokens = max(0, self.in_flight_tokens - 1)
                    current = self.in_flight_tokens
                logger.debug(f"in_flight_tokens -1 → {current}")
    
    async def call_api(self, request: dict, timeout: float = 30.0) -> dict:
        """
        调用API，自动管理in_flight_tokens。
        
        Args:
            request: 请求参数
            timeout: 超时时间
            
        Returns:
            dict: API响应
        """
        async with self._track_in_flight(timeout=timeout):
            try:
                # 带超时的请求
                result = await asyncio.wait_for(
                    self._do_request(request),
                    timeout=timeout
                )
                return result
            except asyncio.TimeoutError:
                logger.warning(f"API请求超时({timeout}s): {request.get('id', 'N/A')}")
                raise
            except Exception as e:
                logger.error(f"API请求失败: {e}", exc_info=True)
                raise
    
    async def _do_request(self, request: dict) -> dict:
        """实际执行API请求"""
        # 模拟API调用
        await asyncio.sleep(0.1)
        return {"status": "ok", "data": request}
    
    def get_in_flight_count(self) -> int:
        """获取当前in-flight数量（线程安全读取）"""
        return self.in_flight_tokens
    
    async def reset_in_flight(self):
        """紧急重置in_flight_tokens（用于死锁恢复）"""
        async with self._lock:
            old = self.in_flight_tokens
            self.in_flight_tokens = 0
            logger.warning(f"in_flight_tokens紧急重置: {old} → 0")
```

### 修改说明
1. **上下文管理器**：`_track_in_flight`用`@asynccontextmanager`装饰，确保`finally`块一定执行
2. **三路释放**：成功（yield正常返回）、失败（异常被raise）、超时（`asyncio.TimeoutError`）都在finally中释放
3. **并发安全**：增减操作都在`async with self._lock`内
4. **max(0, ...)**：防止异常情况下减到负数
5. **acquired标志**：防止获取token失败时在finally中错误释放
6. **紧急重置**：提供`reset_in_flight`用于死锁恢复

## P0-6: `state` 修改在锁外

### 修复前代码
```python
class StateManager:
    async def update_state(self, key, value):
        # 读在锁内
        async with self._lock:
            current = self.state.get(key, {})
        
        # 修改在锁外！其他协程可能同时修改
        current['updated_at'] = datetime.now()
        current['value'] = value
        
        # 写回在锁内，但current可能已经过期
        async with self._lock:
            self.state[key] = current  # 覆盖了其他协程的修改
```

### 修复后代码
```python
import asyncio
from datetime import datetime
from typing import Any, Dict, Optional
import copy

class StateManager:
    def __init__(self):
        self._lock = asyncio.Lock()
        self.state: Dict[str, Any] = {}
    
    async def update_state(self, key: str, value: Any) -> None:
        """
        原子性更新状态。
        读-改-写全部在锁内完成。
        
        Args:
            key: 状态键
            value: 新值
        """
        async with self._lock:
            # === 读-改-写全部在锁内 ===
            current = self.state.get(key, {})
            
            # 深拷贝避免外部引用修改
            current = copy.deepcopy(current)
            current['updated_at'] = datetime.now().isoformat()
            current['value'] = value
            
            self.state[key] = current
            logger.debug(f"state更新: {key}={value}")
    
    async def update_state_partial(self, key: str, updates: dict) -> None:
        """
        原子性部分更新状态（合并更新）。
        
        Args:
            key: 状态键
            updates: 要更新的字段字典
        """
        async with self._lock:
            # === 读-改-写全部在锁内 ===
            current = self.state.get(key, {})
            current = copy.deepcopy(current)
            
            # 合并更新
            current.update(updates)
            current['updated_at'] = datetime.now().isoformat()
            
            self.state[key] = current
            logger.debug(f"state部分更新: {key}, updates={updates}")
    
    async def get_state(self, key: str, default: Any = None) -> Any:
        """
        原子性读取状态。
        
        Args:
            key: 状态键
            default: 默认值
            
        Returns:
            状态值的深拷贝（避免外部修改影响内部状态）
        """
        async with self._lock:
            value = self.state.get(key, default)
            # 返回深拷贝，避免外部修改影响内部状态
            return copy.deepcopy(value)
    
    async def atomic_update(self, key: str, update_fn) -> Any:
        """
        原子性更新（支持自定义更新函数）。
        
        Args:
            key: 状态键
            update_fn: 接收当前值，返回新值的函数
            
        Returns:
            更新后的值
        """
        async with self._lock:
            current = self.state.get(key)
            current = copy.deepcopy(current)
            
            new_value = update_fn(current)
            
            self.state[key] = new_value
            logger.debug(f"state原子更新: {key}")
            return copy.deepcopy(new_value)
    
    async def batch_update(self, updates: Dict[str, Any]) -> None:
        """
        原子性批量更新多个key。
        所有更新在同一个锁内完成，保证一致性。
        
        Args:
            updates: {key: value} 字典
        """
        async with self._lock:
            now = datetime.now().isoformat()
            for key, value in updates.items():
                current = self.state.get(key, {})
                current = copy.deepcopy(current)
                current['value'] = value
                current['updated_at'] = now
                self.state[key] = current
            logger.debug(f"state批量更新: {list(updates.keys())}")
    
    async def get_and_set(self, key: str, new_value: Any) -> Any:
        """
        原子性获取旧值并设置新值。
        
        Args:
            key: 状态键
            new_value: 新值
            
        Returns:
            之前的旧值
        """
        async with self._lock:
            old_value = self.state.get(key)
            old_value_copy = copy.deepcopy(old_value)
            
            self.state[key] = {
                'value': new_value,
                'updated_at': datetime.now().isoformat()
            }
            
            return old_value_copy
    
    def get_state_sync(self, key: str, default: Any = None) -> Any:
        """
        同步读取状态（用于非async上下文）。
        注意：不保证与async更新的原子性。
        
        Args:
            key: 状态键
            default: 默认值
        """
        value = self.state.get(key, default)
        return copy.deepcopy(value)
```

### 修改说明
1. **读-改-写全在锁内**：`update_state`中获取current、修改、写回全部在`async with self._lock`内
2. **深拷贝隔离**：`copy.deepcopy`防止外部引用修改内部状态
3. **多种更新模式**：
   - `update_state`：整体替换
   - `update_state_partial`：合并更新
   - `atomic_update`：自定义函数更新
   - `batch_update`：批量原子更新
   - `get_and_set`：原子性get-set
4. **读取也加锁**：`get_state`在锁内读取并返回深拷贝
5. **同步读取**：`get_state_sync`用于非async上下文（有注释说明限制）

```python
import os
import json
from typing import Optional, Dict, Any

def load_state(self, path: str) -> Dict[str, Any]:
    """
    加载状态文件。
    确保所有路径都有返回值。
    
    Args:
        path: 状态文件路径
        
    Returns:
        dict: 状态字典。加载失败时返回空字典。
    """
    # === 默认返回值 ===
    default_state: Dict[str, Any] = {}
    
    # === 文件不存在 ===
    if not path:
        logger.warning("load_state: path为空，返回默认状态")
        return default_state
    
    if not os.path.exists(path):
        logger.info(f"load_state: 状态文件不存在: {path}，返回默认状态")
        return default_state
    
    # === 文件存在但为空 ===
    if os.path.getsize(path) == 0:
        logger.warning(f"load_state: 状态文件为空: {path}，返回默认状态")
        return default_state
    
    # === 读取文件 ===
    try:
        with open(path, 'r', encoding='utf-8') as f:
            content = f.read()
    except IOError as e:
        logger.error(f"load_state: 读取文件失败: {path}, error={e}")
        return default_state
    except Exception as e:
        logger.error(f"load_state: 读取文件未知异常: {path}, error={e}")
        return default_state
    
    # === 解析JSON ===
    try:
        state = json.loads(content)
    except json.JSONDecodeError as e:
        logger.error(
            f"load_state: JSON解析失败: {path}, error={e}. "
            f"尝试备份并返回默认状态"
        )
        # 备份损坏的文件
        self._backup_corrupt_state(path, content)
        return default_state
    except Exception as e:
        logger.error(f"load_state: JSON解析未知异常: {path}, error={e}")
        return default_state
    
    # === 验证返回类型 ===
    if not isinstance(state, dict):
        logger.warning(
            f"load_state: 状态不是dict类型: {type(state).__name__}, "
            f"返回默认状态"
        )
        return default_state
    
    # === 验证关键字段 ===
    validated_state = self._validate_state(state)
    
    logger.info(
        f"load_state: 成功加载状态 from {path}, "
        f"keys={list(validated_state.keys())}"
    )
    return validated_state


def _backup_corrupt_state(self, path: str, content: str) -> None:
    """备份损坏的状态文件"""
    try:
        backup_path = f"{path}.corrupt.{int(time.time())}"
        with open(backup_path, 'w', encoding='utf-8') as f:
            f.write(content)
        logger.info(f"损坏的状态文件已备份到: {backup_path}")
    except Exception as e:
        logger.error(f"备份损坏文件失败: {e}")


def _validate_state(self, state: dict) -> dict:
    """
    验证状态字典的关键字段。
    
    Args:
        state: 原始状态字典
        
    Returns:
        dict: 验证后的状态字典
    """
    validated = {}
    
    # 必需字段检查
    required_fields = ['version', 'engines']
    for field in required_fields:
        if field not in state:
            logger.warning(f"load_state: 缺少必需字段: {field}")
            # 不return，继续处理其他字段
    
    # 复制有效字段
    for key, value in state.items():
        try:
            # 简单验证：确保值是可序列化的
            json.dumps(value)
            validated[key] = value
        except (TypeError, ValueError) as e:
            logger.warning(
                f"load_state: 字段'{key}'值不可序列化, "
                f"跳过. error={e}"
            )
    
    return validated


def save_state(self, path: str, state: Dict[str, Any]) -> bool:
    """
    保存状态到文件。
    
    Args:
        path: 状态文件路径
        state: 状态字典
        
    Returns:
        bool: True表示保存成功
    """
    try:
        # 确保目录存在
        dir_path = os.path.dirname(path)
        if dir_path and not os.path.exists(dir_path):
            os.makedirs(dir_path, exist_ok=True)
            logger.info(f"save_state: 创建目录: {dir_path}")
        
        # 原子写入：先写临时文件，再重命名
        tmp_path = f"{path}.tmp"
        with open(tmp_path, 'w', encoding='utf-8') as f:
            json.dump(state, f, ensure_ascii=False, indent=2)
        
        # 原子性重命名
        os.replace(tmp_path, path)
        
        logger.info(f"save_state: 状态已保存到: {path}")
        return True
        
    except Exception as e:
        logger.error(f"save_state: 保存失败: {path}, error={e}", exc_info=True)
        # 清理临时文件
        tmp_path = f"{path}.tmp"
        if os.path.exists(tmp_path):
            try:
                os.remove(tmp_path)
            except Exception:
                pass
        return False
```
