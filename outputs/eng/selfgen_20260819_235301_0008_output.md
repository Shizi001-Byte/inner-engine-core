# 引擎崩溃自动恢复机制 — 完整代码架构方案

## 一、约束映射与设计决策

| 约束 | 工程映射 | 设计决策 |
|------|---------|---------|
| 无记忆修改（不依赖本地持久态） | 状态不落本地磁盘，全部通过可插拔存储后端持久化 | `StorageBackend` 抽象层，支持 Redis / 远程KV / 节点内存 |
| 无全量扫描（拒绝O(N)路由） | 恢复时不能遍历所有节点状态 | 内容寻址（Content-Addressed）+ `scan_prefix` 有限查询 |
| 异构环境自适应 | 存储后端、超时、质量规则按环境/模型动态切换 | `ConfigRegistry` 单例 + 模型类型分治 |
| 不依赖外部守护进程 | 心跳、崩溃检测、恢复全部在引擎进程内 | 引擎内部 asyncio 协程实现心跳 |

### 3.1 配置管理层（避坑：配置修改后必须同步读取点）

```python
import asyncio
import hashlib
import json
import time
import uuid
from dataclasses import dataclass, field
from enum import Enum, auto
from typing import Any, Optional, Callable

# ==================== 配置注册中心 ====================

class ModelType(Enum):
    A_MODEL = "a_model"
    B_TURBO = "b_turbo"


class ConfigRegistry:
    """
    全局配置注册中心（单例）
    避坑1：配置修改后自动失效缓存 + 通知所有读取点，确保立即生效
    避坑2：配置修改后立即验证，不通过则回滚
    """
    _instance = None
    _config: dict = {}
    _version: int = 0
    _listeners: list = []

    @classmethod
    def instance(cls):
        if cls._instance is None:
            cls._instance = cls()
        return cls._instance

    def get(self, key: str, default=None) -> Any:
        # 直接从最新字典读取，无缓存层，避免陈旧
        return self._config.get(key, default)

    def set(self, key: str, value: Any, verify: bool = True) -> bool:
        old_value = self._config.get(key)
        self._config[key] = value
        self._version += 1
        # 避坑：修改后立即验证
        if verify:
            actual = self.get(key)
            if actual != value:
                # 回滚
                self._config[key] = old_value
                self._version -= 1
                return False
        # 立即通知所有读取点
        for listener in list(self._listeners):
            try:
                listener(key, value, self._version)
            except Exception:
                pass
        return True

    def register_listener(self, callback: Callable):
        self._listeners.append(callback)

    @property
    def version(self) -> int:
        return self._version


def init_default_config():
    """初始化预设配置 — 遵循所有避坑经验"""
    reg = ConfigRegistry.instance()

    # 【避坑2】API超时预设：B模型turbo用300秒，A模型用600秒
    reg.set("model_timeout", {
        ModelType.A_MODEL: 600,
        ModelType.B_TURBO: 300,
    })

    # 【坑库预警】B模型废稿误杀：不同模型独立质量规则
    reg.set("quality_check_rules", {
        ModelType.A_MODEL: {
            "check_reasoning_ratio": True,
            "min_reasoning_ratio": 0.15,
            "max_reasoning_ratio": 0.60,
        },
        ModelType.B_TURBO: {
            # B模型(turbo)跳过reasoning占比检测 — 70-82%是正常特性
            "check_reasoning_ratio": False,
            "min_reasoning_ratio": None,
            "max_reasoning_ratio": None,
        },
    })

    # 心跳与恢复配置
    reg.set("heartbeat_interval_sec", 5)
    reg.set("heartbeat_ttl_sec", 30)
    reg.set("crash_grace_period_sec", 15)
    reg.set("max_recovery_attempts", 3)
    reg.set("wal_max_entries", 1000)
    reg.set("checkpoint_interval_nodes", 1)  # 每完成1个节点存一次检查点
```

### 3.2 核心数据结构

```python
class EngineState(Enum):
    BOOTING = auto()
    RUNNING = auto()
    CHECKPOINTING = auto()
    RECOVERING = auto()
    CRASHED = auto()


@dataclass
class Epoch:
    """
    逻辑时钟 — 区分引擎实例世代
    每次 boot 时 generation 递增，崩溃恢复时通过 epoch 判断遗留状态归属
    """
    engine_id: str
    generation: int
    boot_timestamp: float

    def __gt__(self, other: 'Epoch') -> bool:
        if self.engine_id != other.engine_id:
            return self.boot_timestamp > other.boot_timestamp
        return self.generation > other.generation

    def to_dict(self) -> dict:
        return {
            "engine_id": self.engine_id,
            "generation": self.generation,
            "boot_timestamp": self.boot_timestamp,
        }

    @classmethod
    def from_dict(cls, d: dict) -> 'Epoch':
        return cls(**d)


@dataclass
class Checkpoint:
    """
    执行检查点 — 断点续跑的基础单元
    使用内容寻址（state_hash）而非固定路径，支持异构环境
    """
    checkpoint_id: str
    epoch: Epoch
    task_node_id: str
    state_hash: str
    wal_sequence: int
    timestamp: float
    payload: dict

    def compute_hash(self) -> str:
        content = json.dumps(self.payload, sort_keys=True)
        return hashlib.sha256(content.encode()).hexdigest()


@dataclass
class WALEntry:
    """
    预写日志条目
    两阶段：append(prepare) → commit
    崩溃时未 commit 的条目会被丢弃，保证原子性
    """
    sequence: int
    epoch: Epoch
    operation: str  # task_start / task_complete / state_update / checkpoint
    data: dict
    checksum: str
    committed: bool = False

    def compute_checksum(self) -> str:
        content = f"{self.sequence}:{json.dumps(self.data, sort_keys=True)}"
        return hashlib.sha256(content.encode()).hexdigest()[:16]
```

### 3.3 存储后端抽象（异构环境自适应）

```python
class StorageBackend:
    """
    存储后端抽象 — 无本地持久态
    支持多种后端：Redis / 远程KV / 节点内存
    所有方法返回 coroutine，适配异步环境
    """

    async def read(self, key: str) -> Optional[bytes]:
        raise NotImplementedError

    async def write(self, key: str, value: bytes,
                    ttl: Optional[int] = None) -> bool:
        raise NotImplementedError

    async def delete(self, key: str) -> bool:
        raise NotImplementedError

    async def scan_prefix(self, prefix: str, limit: int = 100) -> list:
        """有限前缀查询 — 拒绝全量扫描"""
        raise NotImplementedError


class InMemoryStorageBackend(StorageBackend):
    """内存存储 — 用于测试和同节点恢复"""

    def __init__(self):
        self._store: dict = {}
        self._ttls: dict = {}

    async def read(self, key: str) -> Optional[bytes]:
        entry = self._store.get(key)
        if entry is None:
            return None
        expiry = self._ttls.get(key)
        if expiry and time.time() > expiry:
            del self._store[key]
            del self._ttls[key]
            return None
        return entry

    async def write(self, key: str, value: bytes,
                    ttl: Optional[int] = None) -> bool:
        self._store[key] = value
        if ttl:
            self._ttls[key] = time.time() + ttl
        return True

    async def delete(self, key: str) -> bool:
        self._store.pop(key, None)
        self._ttls.pop(key, None)
        return True

    async def scan_prefix(self, prefix: str, limit: int = 100) -> list:
        results = []
        for key in list(self._store.keys()):
            if key.startswith(prefix):
                results.append((key, self._store[key]))
                if len(results) >= limit:
                    break
        return results


class RedisStorageBackend(StorageBackend):
    """Redis存储 — 用于跨节点恢复"""

    def __init__(self, redis_client):
        self._redis = redis_client

    async def read(self, key: str) -> Optional[bytes]:
        return await self._redis.get(key)

    async def write(self, key: str, value: bytes,
                    ttl: Optional[int] = None) -> bool:
        if ttl:
            return await self._redis.setex(key, ttl, value)
        return await self._redis.set(key, value)

    async def delete(self, key: str) -> bool:
        return await self._redis.delete(key) > 0

    async def scan_prefix(self, prefix: str, limit: int = 100) -> list:
        # 使用 SCAN 而非 KEYS，避免全量扫描
        results = []
        async for key in self._redis.scan_iter(
            match=f"{prefix}*", count=limit
        ):
            val = await self._redis.get(key)
            results.append((key, val))
            if len(results) >= limit:
                break
        return results
```

### 3.4 WAL管理器（状态原子持久化核心）

```python
class WALManager:
    """
    预写日志管理器
    原则：所有状态变更先写WAL(prepare)，确认持久化后commit，再更新内存
    保证：崩溃时可通过回放 committed 条目恢复到最后一致状态
    """

    def __init__(self, storage: StorageBackend, epoch: Epoch):
        self._storage = storage
        self._epoch = epoch
        self._sequence = 0
        self._wal_key = f"wal:{epoch.engine_id}:gen{epoch.generation}"
        self._pending: list = []
        self._lock = asyncio.Lock()

    async def append(self, operation: str, data: dict) -> WALEntry:
        """Prepare阶段：追加WAL条目到存储"""
        async with self._lock:
            entry = WALEntry(
                sequence=self._sequence,
                epoch=self._epoch,
                operation=operation,
                data=data,
                checksum="",
            )
            entry.checksum = entry.compute_checksum()
            self._sequence += 1

            # 写入存储
            key = f"{self._wal_key}:{entry.sequence}"
            await self._storage.write(key, self._serialize(entry))
            self._pending.append(entry)
            return entry

    async def commit(self, entry: WALEntry) -> bool:
        """Commit阶段：标记条目为已提交"""
        entry.committed = True
        key = f"{self._wal_key}:{entry.sequence}"
        await self._storage.write(key, self._serialize(entry))

        # 更新commit point指针
        await self._storage.write(
            f"{self._wal_key}:commit_point",
            str(entry.sequence).encode()
        )
        if entry in self._pending:
            self._pending.remove(entry)
        return True

    async def rollback(self, entry: WALEntry):
        """回滚未提交的条目"""
        await self._storage.delete(f"{self._wal_key}:{entry.sequence}")
        if entry in self._pending:
            self._pending.remove(entry)

    async def replay_from(self, after_sequence: int) -> list:
        """
        从指定位置回放WAL（崩溃恢复用）
        只返回 committed 且 sequence > after_sequence 的条目
        """
        entries = []
        results = await self._storage.scan_prefix(
            f"{self._wal_key}:", limit=1000
        )
        for key, value in results:
            if b"commit_point" in key.encode():
                continue
            entry = self._deserialize(value)
            if entry.sequence > after_sequence and entry.committed:
                # 校验完整性
                expected = entry.compute_checksum()
                if entry.checksum != expected:
                    continue  # 跳过损坏条目
                entries.append(entry)

        entries.sort(key=lambda e: e.sequence)
        return entries

    async def get_last_commit_point(self) -> int:
        """获取最后提交点序号"""
        data = await self._storage.read(f"{self._wal_key}:commit_point")
        if data is None:
            return -1
        return int(data.decode())

    async def compact(self, keep_after: int):
        """压缩WAL — 删除旧条目"""
        results = await self._storage.scan_prefix(
            f"{self._wal_key}:", limit=1000
        )
        for key, _ in results:
            if "commit_point" in key:
                continue
            parts = key.split(":")
            if len(parts) >= 4:
                try:
                    seq = int(parts[-1])
                    if seq < keep_after:
                        await self._storage.delete(key)
                except ValueError:
                    pass

    def _serialize(self, entry: WALEntry) -> bytes:
        return json.dumps({
            "sequence": entry.sequence,
            "operation": entry.operation,
            "data": entry.data,
            "checksum": entry.checksum,
            "committed": entry.committed,
            "epoch": entry.epoch.to_dict(),
        }).encode()

    def _deserialize(self, data: bytes) -> WALEntry:
        d = json.loads(data.decode())
        return WALEntry(
            sequence=d["sequence"],
            epoch=Epoch.from_dict(d["epoch"]),
            operation=d["operation"],
            data=d["data"],
            checksum=d["checksum"],
            committed=d["committed"],
        )
```

### 3.5 心跳检测器（引擎内部实现，无外部守护进程）

```python
class HeartbeatGenerator:
    """
    内部心跳生成器
    - 引擎内部 asyncio 协程，不依赖外部守护进程
    - 心跳写入存储后端，带TTL自动过期
    - 心跳内容：epoch + 当前执行节点 + 资源水位
    """

    def __init__(self, storage: StorageBackend, epoch: Epoch):
        self._storage = storage
        self._epoch = epoch
        self._heartbeat_key = f"heartbeat:{epoch.engine_id}"
        self._running = False
        self._task: Optional[asyncio.Task] = None
        self._current_node: Optional[str] = None
        self._resource_snapshot: dict = {}
        self._config = ConfigRegistry.instance()

    def set_current_node(self, node_id: Optional[str]):
        self._current_node = node_id

    def update_resource_snapshot(self, snapshot: dict):
        self._resource_snapshot.update(snapshot)

    async def start(self):
        if self._running:
            return
        self._running = True
        interval = self._config.get("heartbeat_interval_sec", 5)
        ttl = self._config.get("heartbeat_ttl_sec", 30)

        async def _beat_loop():
            while self._running:
                try:
                    heartbeat = {
                        "epoch": self._epoch.to_dict(),
                        "current_node": self._current_node,
                        "timestamp": time.time(),
                        "resource": self._resource_snapshot,
                    }
                    await self._storage.write(
                        self._heartbeat_key,
                        json.dumps(heartbeat).encode(),
                        ttl=ttl,
                    )
                except Exception as e:
                    # 心跳写入失败不致命，下次重试
                    print(f"[Heartbeat] write failed: {e}")
                await asyncio.sleep(interval)

        self._task = asyncio.create_task(_beat_loop())

    async def stop(self):
        self._running = False
        if self._task:
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
        # 正常停止时清除心跳
        await self._storage.delete(self._heartbeat_key)


class CrashDetector:
    """
    崩溃检测器
    - 检测心跳是否过期（TTL + 宽限期）
    - 引擎重启时自检：是否有上一代的遗留状态
    """

    def __init__(self, storage: StorageBackend):
        self._storage = storage
        self._config = ConfigRegistry.instance()

    async def detect_previous_crash(self, engine_id: str) -> Optional[dict]:
        """
        检测是否有崩溃遗留的心跳
        返回遗留心跳数据（含epoch信息），无则返回None
        """
        heartbeat_key = f"heartbeat:{engine_id}"
        data = await self._storage.read(heartbeat_key)

        if data is None:
            return None  # 无遗留状态

        heartbeat = json.loads(data.decode())
        grace = self._config.get("crash_grace_period_sec", 15)
        hb_time = heartbeat.get("timestamp", 0)

        if time.time() - hb_time > grace:
            return heartbeat  # 心跳过期 → 判定崩溃
        return None  # 心跳仍在有效期内 → 可能有其他实例在运行

    async def find_previous_generation(self, engine_id: str) -> Optional[Epoch]:
        """
        查找上一代引擎的epoch（用于恢复时递增generation）
        有限查询心跳记录，不全量扫描
        """
        heartbeat = await self.detect_previous_crash(engine_id)
        if heartbeat is None:
            return None
        return Epoch.from_dict(heartbeat["epoch"])
```

### 3.6 检查点管理器

```python
class CheckpointManager:
    """
    检查点管理器
    - 内容寻址存储（state_hash 作为 key），不依赖固定路径
    - 支持 latest 指针快速定位
    - 检查点频率可配置
    """

    def __init__(self, storage: StorageBackend, epoch: Epoch):
        self._storage = storage
        self._epoch = epoch
        self._config = ConfigRegistry.instance()
        self._node_count = 0

    async def save_checkpoint(self, task_node_id: str,
                              state: dict, wal_seq: int) -> Checkpoint:
        checkpoint = Checkpoint(
            checkpoint_id=str(uuid.uuid4()),
            epoch=self._epoch,
            task_node_id=task_node_id,
            state_hash="",
            wal_sequence=wal_seq,
            timestamp=time.time(),
            payload=state,
        )
        checkpoint.state_hash = checkpoint.compute_hash()

        # 内容寻址：用 hash 作为存储 key
        ckpt_key = f"ckpt:{self._epoch.engine_id}:{checkpoint.state_hash}"
        data = json.dumps({
            "checkpoint_id": checkpoint.checkpoint_id,
            "epoch": checkpoint.epoch.to_dict(),
            "task_node_id": checkpoint.task_node_id,
            "state_hash": checkpoint.state_hash,
            "wal_sequence": checkpoint.wal_sequence,
            "timestamp": checkpoint.timestamp,
            "payload": checkpoint.payload,
        }).encode()
        await self._storage.write(ckpt_key, data)

        # 更新 latest 指针（O(1) 定位，无需扫描）
        latest_key = f"ckpt:{self._epoch.engine_id}:latest"
        await self._storage.write(
            latest_key, checkpoint.state_hash.encode()
        )

        return checkpoint

    async def load_latest_checkpoint(self, engine_id: str) -> Optional[Checkpoint]:
        """从 latest 指针加载最新检查点 — O(1) 复杂度"""
        latest_key = f"ckpt:{engine_id}:latest"
        state_hash = await self._storage.read(latest_key)
        if state_hash is None:
            return None

        ckpt_key = f"ckpt:{engine_id}:{state_hash.decode()}"
        data = await self._storage.read(ckpt_key)
        if data is None:
            return None

        d = json.loads(data.decode())
        return Checkpoint(
            checkpoint_id=d["checkpoint_id"],
            epoch=Epoch.from_dict(d["epoch"]),
            task_node_id=d["task_node_id"],
            state_hash=d["state_hash"],
            wal_sequence=d["wal_sequence"],
            timestamp=d["timestamp"],
            payload=d["payload"],
        )

    async def load_checkpoint_by_epoch(self, engine_id: str,
                                       epoch: Epoch) -> Optional[Checkpoint]:
        """加载指定epoch的检查点（跨世代恢复）"""
        # 有限前缀查询
        results = await self._storage.scan_prefix(
            f"ckpt:{engine_id}:", limit=100
        )
        for key, data in results:
            if key.endswith(":latest"):
                continue
            d = json.loads(data.decode())
            ckpt_epoch = Epoch.from_dict(d["epoch"])
            if ckpt_epoch.generation == epoch.generation:
                return Checkpoint(
                    checkpoint_id=d["checkpoint_id"],
                    epoch=ckpt_epoch,
                    task_node_id=d["task_node_id"],
                    state_hash=d["state_hash"],
                    wal_sequence=d["wal_sequence"],
                    timestamp=d["timestamp"],
                    payload=d["payload"],
                )
        return None
```

### 3.7 质量检测器（坑库预警：B模型废稿误杀）

```python
class QualityChecker:
    """
    质量检测器
    【坑库预警】B模型reasoning占比70-82%被误判为废稿
    避坑：不同模型必须有独立的质量检测规则，B模型(turbo)跳过reasoning占比检测
    """

    def __init__(self):
        self._config = ConfigRegistry.instance()

    def check_output(self, output: dict,
                     model_type: ModelType) -> tuple:
        """
        检查模型输出质量
        返回: (is_valid, reason)
        """
        rules = self._config.get("quality_check_rules", {}).get(
            model_type, {}
        )

        # reasoning占比检测 — 仅当该模型需要检测时
        if rules.get("check_reasoning_ratio", False):
            reasoning_text = output.get("reasoning", "")
            full_text = output.get("content", "")
            if full_text:
                ratio = len(reasoning_text) / len(full_text)
                min_r = rules.get("min_reasoning_ratio", 0)
                max_r = rules.get("max_reasoning_ratio", 1)
                if ratio < min_r:
                    return False, (
                        f"reasoning_ratio={ratio:.2f} "
                        f"below min {min_r}"
                    )
                if ratio > max_r:
                    return False, (
                        f"reasoning_ratio={ratio:.2f} "
                        f"above max {max_r}"
                    )
        # B模型turbo: check_reasoning_ratio=False → 跳过检测，直接通过
        # 不再误杀 reasoning 占比 70-82% 的正常B模型输出

        return True, "pass"
```

```python
class APICaller:
    """
    API调用管理器
    【避坑2】根据任务类型预设合理timeout
    B模型turbo用300秒，A模型用600秒
    """

    def __init__(self):
        self._config = ConfigRegistry.instance()

    async def call_model(self, prompt: str,
                         model_type: ModelType) -> dict:
        timeouts = self._config.get("model_timeout", {})
        timeout = timeouts.get(model_type, 300)

        try:
            result = await asyncio.wait_for(
                self._do_api_call(prompt, model_type),
                timeout=timeout,
            )
            return result
        except asyncio.TimeoutError:
            raise APITimeoutError(
                f"API timeout after {timeout}s for {model_type.value}"
            )

    async def _do_api_call(self, prompt: str,
                           model_type: ModelType) -> dict:
        """
        实际API调用实现
        此处为模拟，生产环境替换为真实HTTP调用
        """
        await asyncio.sleep(0.1)  # 模拟网络延迟
        # B模型turbo正常会返回高reasoning占比(70-82%)
        if model_type == ModelType.B_TURBO:
            return {
                "content": "Final answer content here.",
                "reasoning": "R" * 150 + " Detailed reasoning process " + "R" * 100,
            }
        return {
            "content": "Analysis and final answer content here.",
            "reasoning": "Brief reasoning.",
        }
```

### 3.9 断点续跑协调器

```python
class ResumptionCoordinator:
    """
    断点续跑协调器
    1. 检测崩溃遗留 → 2. 加载检查点 → 3. 回放WAL → 4. 返回恢复上下文
    """

    def __init__(self, storage: StorageBackend):
        self._storage = storage
        self._config = ConfigRegistry.instance()

    async def attempt_recovery(self, engine_id: str) -> dict:
        """尝试从崩溃中恢复，返回恢复上下文"""
        max_attempts = self._config.get("max_recovery_attempts", 3)

        # 1. 检测崩溃遗留
        crash_detector = CrashDetector(self._storage)
        crashed_heartbeat = await crash_detector.detect_previous_crash(
            engine_id
        )

        if crashed_heartbeat is None:
            return {"recovered": False, "reason": "no_crash_detected"}

        prev_epoch = Epoch.from_dict(crashed_heartbeat["epoch"])

        # 2. 加载最新检查点
        ckpt_mgr = CheckpointManager(self._storage, prev_epoch)
        checkpoint = await ckpt_mgr.load_latest_checkpoint(engine_id)

        if checkpoint is None:
            # 无检查点 → 可能需要从头开始
            return {
                "recovered": True,
                "from_scratch": True,
                "previous_epoch": prev_epoch,
                "checkpoint": None,
            }

        # 3. 回放WAL（检查点之后的已提交操作）
        wal = WALManager(self._storage, prev_epoch)
        last_commit = await wal.get_last_commit_point()
        replayed_count = 0

        if last_commit > checkpoint.wal_sequence:
            entries = await wal.replay_from(checkpoint.wal_sequence)
            for entry in entries:
                checkpoint.payload = self._apply_wal_entry(
                    checkpoint.payload, entry
                )
                replayed_count += 1

        return {
            "recovered": True,
            "from_scratch": False,
            "previous_epoch": prev_epoch,
            "checkpoint": checkpoint,
            "wal_entries_replayed": replayed_count,
        }

    def _apply_wal_entry(self, state: dict,
                         entry: WALEntry) -> dict:
        """将WAL条目应用到状态"""
        if entry.operation == "state_update":
            state.update(entry.data)
        elif entry.operation == "task_complete":
            node_id = entry.data.get("node_id")
            if node_id:
                state[f"node_{node_id}_result"] = entry.data.get(
                    "result", {}
                )
        return state
```

### 3.10 引擎主控（整合所有组件）

```python
class DigitalLifeEngine:
    """
    数字生命引擎主控
    整合：心跳检测 + WAL原子持久化 + 检查点 + 崩溃恢复
    不依赖外部守护进程，全部内部实现
    """

    def __init__(self, engine_id: str, storage: StorageBackend,
                 task_graph: Optional[dict] = None):
        self._engine_id = engine_id
        self._storage = storage
        self._state = EngineState.BOOTING
        self._task_graph = task_graph or {}
        self._config = ConfigRegistry.instance()
        self._quality_checker = QualityChecker()
        self._api_caller = APICaller()

        self._epoch: Optional[Epoch] = None
        self._heartbeat: Optional[HeartbeatGenerator] = None
        self._wal: Optional[WALManager] = None
        self._checkpoint_mgr: Optional[CheckpointManager] = None
        self._recovery = ResumptionCoordinator(storage)

        self._current_node: Optional[str] = None
        self._engine_state_data: dict = {}

    async def boot(self):
        """引擎启动 — 先尝试崩溃恢复"""
        self._state = EngineState.BOOTING

        # 1. 尝试崩溃恢复
        recovery = await self._recovery.attempt_recovery(self._engine_id)

        if recovery["recovered"]:
            prev_epoch = recovery["previous_epoch"]
            self._epoch = Epoch(
                engine_id=self._engine_id,
                generation=prev_epoch.generation + 1,
                boot_timestamp=time.time(),
            )
            if recovery.get("checkpoint"):
                ckpt = recovery["checkpoint"]
                self._engine_state_data = ckpt.payload
                self._current_node = self._get_next_node(ckpt.task_node_id)
                print(
                    f"[Engine] Recovered: gen={self._epoch.generation}, "
                    f"replayed={recovery.get('wal_entries_replayed', 0)} "
                    f"WAL entries, resuming from after {ckpt.task_node_id}"
                )
            else:
                self._current_node = self._find_entry_node()
                print(
                    f"[Engine] Recovered (no checkpoint), "
                    f"gen={self._epoch.generation}, starting fresh"
                )
        else:
            self._epoch = Epoch(
                engine_id=self._engine_id,
                generation=0,
                boot_timestamp=time.time(),
            )
            self._current_node = self._find_entry_node()
            print(f"[Engine] Fresh boot, gen=0")

        # 2. 初始化组件（使用新epoch）
        self._wal = WALManager(self._storage, self._epoch)
        self._heartbeat = HeartbeatGenerator(self._storage, self._epoch)
        self._heartbeat.set_current_node(self._current_node or "init")
        self._checkpoint_mgr = CheckpointManager(
            self._storage, self._epoch
        )

        # 3. 启动心跳
        await self._heartbeat.start()

        self._state = EngineState.RUNNING
        print(f"[Engine] Running, epoch={self._epoch.to_dict()}")

    async def run(self):
        """主运行循环"""
        await self.boot()

        try:
            while self._state == EngineState.RUNNING:
                if self._current_node is None:
                    print("[Engine] All tasks completed, entering idle")
                    break

                node = self._task_graph.get(self._current_node)
                if node is None:
                    print(
                        f"[Engine] Node {self._current_node} "
                        f"not in graph, stopping"
                    )
                    break

                await self._execute_node(node)
                self._current_node = node.get("next")
                self._heartbeat.set_current_node(
                    self._current_node or "done"
                )

        except Exception as e:
            print(f"[Engine] Crash: {e}")
            self._state = EngineState.CRASHED
            # 心跳协程随进程退出自然停止
            # TTL到期后其他实例可检测到崩溃并恢复
            raise
        finally:
            if self._heartbeat:
                await self._heartbeat.stop()

    async def _execute_node(self, node: dict):
        """执行单个任务节点 — 含WAL + 检查点 + 质量检测"""
        node_id = node["id"]
        model_type = node.get("model_type", ModelType.B_TURBO)

        # WAL: 记录任务开始
        start_entry = await self._wal.append("task_start", {
            "node_id": node_id,
            "model_type": model_type.value,
        })

        try:
            # 调用API（超时已按模型类型预设）
            result = await self._api_caller.call_model(
                node["prompt"], model_type
            )

            # 质量检测（B模型跳过reasoning占比检测）
            is_valid, reason = self._quality_checker.check_output(
                result, model_type
            )
            if not is_valid:
                print(
                    f"[Engine] Quality check FAILED for "
                    f"{node_id}: {reason}"
                )
                await self._wal.append("quality_check_failed", {
                    "node_id": node_id, "reason": reason
                })
                # 不误杀：记录后继续（非致命）
            else:
                print(f"[Engine] Quality check PASSED for {node_id}")

            # WAL: 提交任务开始 + 记录任务完成
            await self._wal.commit(start_entry)

            complete_entry = await self._wal.append(
                "task_complete", {
                    "node_id": node_id,
                    "result": result,
                }
            )
            await self._wal.commit(complete_entry)

            # 更新引擎状态
            self._engine_state_data[f"node_{node_id}_result"] = result

            # 保存检查点
            await self._checkpoint_mgr.save_checkpoint(
                task_node_id=node_id,
                state=self._engine_state_data,
                wal_seq=complete_entry.sequence,
            )

            # 更新心跳资源快照
            self._heartbeat.update_resource_snapshot({
                "last_completed_node": node_id,
                "wal_sequence": complete_entry.sequence,
                "nodes_completed": len([
                    k for k in self._engine_state_data
                    if k.endswith("_result")
                ]),
            })

            print(f"[Engine] Node {node_id} completed ✓")

        except APITimeoutError as e:
            print(f"[Engine] API timeout for {node_id}: {e}")
            await self._wal.rollback(start_entry)
            raise
        except Exception as e:
            print(f"[Engine] Node {node_id} failed: {e}")
            await self._wal.rollback(start_entry)
            raise

    def _find_entry_node(self) -> Optional[str]:
        for nid, node in self._task_graph.items():
            if node.get("entry", False):
                return nid
        return next(iter(self._task_graph), None)

    def _get_next_node(self, completed_node_id: str) -> Optional[str]:
        """获取已完成节点的下一个节点"""
        node = self._task_graph.get(completed_node_id)
        if node:
            return node.get("next")
        return None

    @property
    def state(self) -> EngineState:
        return self._state

    @property
    def epoch(self) -> Optional[Epoch]:
        return self._epoch
```

## 四、崩溃恢复流程时序

```
正常运行期:
  Engine.boot() → Heartbeat.start() → 每5s写心跳(TTL=30s)
  _execute_node():
    WAL.append("task_start")     ← prepare
    APICaller.call_model()      ← timeout=B:300s/A:600s
    QualityChecker.check()      ← B模型跳过reasoning检测
    WAL.commit("task_start")    ← commit
    WAL.append("task_complete") ← prepare
    WAL.commit("task_complete") ← commit
    CheckpointManager.save()    ← 内容寻址存储
    Heartbeat.update_resource()

崩溃发生 (OOM/断电/网络重置):
  → 进程退出，心跳TTL到期(30s后消失)
  → WAL中最后未commit的条目保留但标记committed=False
  → 最新Checkpoint保留在存储中

恢复启动:
  Engine.boot()
    → ResumptionCoordinator.attempt_recovery()
      → CrashDetector.detect_previous_crash()  ← 心跳过期?
      → CheckpointManager.load_latest()        ← O(1)指针定位
      → WALManager.replay_from(ckpt.wal_seq)   ← 回放committed条目
      → 返回恢复上下文
    → Epoch.generation += 1
    → 从 checkpoint.task_node_id 的下一节点继续
    → Heartbeat.start() (新epoch)
```

## 五、关键设计决策说明

| 决策 | 理由 | 对应约束 |
|------|------|---------|
| WAL 两阶段提交 | append→commit 之间崩溃，未commit条目被丢弃，保证原子性 | 状态原子持久化 |
| 内容寻址存储 | 用 state_hash 做 key，不依赖固定路径，任意节点可恢复 | 异构环境自适应 |
| 心跳内部协程 | `asyncio.create_task` 在引擎进程内运行，不依赖外部进程 | 无外部守护进程 |
| TTL 自动过期 | 崩溃后心跳30s内自动消失，无需手动清理 | 无记忆修改 |
| `scan_prefix(limit)` | 只做有限前缀查询，拒绝 O(N) 全量扫描 | 无全量扫描 |
| 模型分治质量规则 | B模型turbo跳过reasoning占比检测，避免误杀70-82%正常输出 | 坑库预警 |
| 超时按模型预设 | B_turbo=300s, A=600s，写入即验证生效 | 避坑经验 |
| ConfigRegistry单例 | 配置修改后通知所有监听器 + 立即验证回读 | 避坑经验 |
