## 一、系统总览：灵物归藏体系

> **核心隐喻**：每一个被引擎产出的任务，都是一株"灵物"。归档不是"存储"，而是"炼化归藏"——灵物经鉴定灵根、铭刻符文、归位洞天后，其信息能量反哺引擎算力，形成代谢闭环。

```
任务产出 → 灵气探查(特征提取) → 灵根鉴定(分类) → 符文铭刻(打标签) → 洞天归位(存目录) → 修为反哺(模型迭代)
                                                                                              ↓
                                                                                    系统境界提升
                                                                                              ↓
                                                                              虚又返实：分类能力增强
                                                                                              ↓
                                                                                    下一批任务归档
```

### 2.2 鉴定流水线

```python
class 灵根鉴定器:
    """三层鉴定：关键词共鸣 → 语义Embedding → 上下文确证"""
    
    def 鉴定(self, 任务: Task) -> 鉴定结果:
        # 第一层：灵气探查 —— 关键词快速共鸣
        关键词命中 = self.关键词扫描(任务)
        候选灵根 = self.筛选候选(关键词命中)
        
        # 第二层：灵根共鸣 —— Embedding语义匹配
        if 候选灵根.置信度 < 0.85:
            语义向量 = self.embedding模型.编码(任务.内容)
            候选灵根 = self.余弦匹配(语义向量, 候选灵根)
        
        # 第三层：属性判定 —— 上下文工程指标确证
        上下文特征 = self.提取工程指标(任务)
        最终灵根 = self.加权决策(候选灵根, 上下文特征)
        
        # 判定结果分级
        if 最终灵根.置信度 >= 0.85:
            return 鉴定结果(灵根=最终灵根, 状态="灵根纯净", 可归档=True)
        elif 最终灵根.置信度 >= 0.60:
            return 鉴定结果(灵根=最终灵根, 状态="灵根不稳", 需二次确认=True)
        else:
            return 鉴定结果(灵根=None, 状态="未鉴定灵物", 入暂存区=True)
```

### 2.3 多灵根判定（复合任务）

当一个任务同时具备多个灵根特征时：

| 灵根组合 | 含义 | 归档策略 |
|---------|------|---------|
| 金+木 | 架构驱动的功能开发 | 归入金（主灵根），木为副灵根标记 |
| 火+水 | 高吞吐数据流优化 | 归入火（主灵根），水为副灵根标记 |
| 雷+土 | 紧急基础设施修复 | 归入雷（优先级最高） |
| 风+任意 | 文档相关任务 | 始终归入风，记录关联灵根 |

**主灵根判定规则**：`max(置信度 × 紧急度权重 × 复杂度权重)`

### 3.2 铭刻规则引擎

```python
class 符文炉:
    RULES = [
        # 天阶符文规则
        Rule(
            condition=lambda t: t.urgency == "critical" and t.impact_radius == "system-wide",
            action=lambda t: t.add_rune("天劫", "CRITICAL"),
            desc="系统级紧急任务铭刻天劫符"
        ),
        Rule(
            condition=lambda t: t.deprecated == True,
            action=lambda t: t.add_rune("轮回", "DEPRECATED"),
            desc="废弃任务铭刻轮回符"
        ),
        Rule(
            condition=lambda t: t.status == "experimental",
            action=lambda t: t.add_rune("试炼", "EXPERIMENTAL"),
            desc="实验性任务铭刻试炼符"
        ),
        
        # 地阶符文规则（继承灵根鉴定结果）
        Rule(
            condition=lambda t: t.灵根 is not None,
            action=lambda t: t.add_rune(t.灵根.符文名, t.灵根.代号),
            desc="按灵根属性铭刻对应符文"
        ),
        
        # 人阶符文规则
        Rule(
            condition=lambda t: t.timestamp is not None,
            action=lambda t: t.add_rune("甲子", t.timestamp.isoformat()),
            desc="时间戳铭刻为甲子符"
        ),
        Rule(
            condition=lambda t: len(t.dependencies) > 0,
            action=lambda t: t.add_rune("因果", [d.id for d in t.dependencies]),
            desc="依赖关系铭刻为因果符"
        ),
        Rule(
            condition=lambda t: t.git_branch is not None,
            action=lambda t: t.add_rune("灵脉", t.git_branch),
            desc="Git分支铭刻为灵脉符"
        ),
        
        # 复合规则：特定组合触发特殊符文
        Rule(
            condition=lambda t: t.has_rune("火") and t.has_rune("雷"),
            action=lambda t: t.add_rune("渡劫", "UPGRADE"),
            desc="火+雷组合 → 触发渡劫符（性能紧急优化）"
        ),
    ]
    
    def 铭刻(self, 灵物) -> List[符文]:
        for rule in self.RULES:
            if rule.condition(灵物):
                rule.action(灵物)
        
        # 计算符文完整度
        灵物.符文完整度 = len(灵物.已铭刻符文) / self.最大符文数
        return 灵物.已铭刻符文
```

### 4.2 归位算法

```python
class 洞天图:
    """洞天福地映射与归位引擎"""
    
    灵根_洞天映射 = {
        "金": "须弥山",
        "木": "万木林",
        "水": "碧水潭",
        "火": "炼丹炉",
        "土": "厚土界",
        "雷": "天劫海",
        "冰": "玄冰宫",
        "风": "听风阁",
    }
    
    def 归位(self, 灵物) -> str:
        if 灵物.状态 == "未鉴定":
            return self.暂存(灵物, "未鉴定/")
        
        主洞天 = self.灵根_洞天映射[灵物.主灵根]
        子目录 = self.细分归位(灵物)
        灵名 = self.生成灵名(灵物)
        
        完整路径 = f"/vault/{主洞天}/{子目录}/{灵名}"
        
        # 写入文件
        self.文件系统.写入(完整路径, 灵物.序列化())
        
        # 写入元数据旁车文件
        self.写入元数据(完整路径 + ".meta.json", {
            "task_id": 灵物.task_id,
            "灵根": 灵物.主灵根,
            "副灵根": 灵物.副灵根,
            "符文": 灵物.已铭刻符文,
            "灵根纯度": 灵物.置信度,
            "符文完整度": 灵物.符文完整度,
            "归档时间": 灵物.归档时间,
            "修为贡献": 灵物.修为贡献,
            "状态": 灵物.状态,
            "来源": 灵物.来源,
            "关联灵物": 灵物.关联ID列表,
        })
        
        return 完整路径
    
    def 生成灵名(self, 灵物) -> str:
        """生成归档文件名：{task_id}_{简称}_{灵根代号}.md"""
        简称 = self.提取简称(灵物.标题, max_length=30)
        代号 = 灵物.主灵根[:1].upper()
        return f"{灵物.task_id}_{简称}_{代号}.md"
    
    def 细分归位(self, 灵物) -> str:
        """根据符文和复杂度细分子目录"""
        if 灵物.has_rune("轮回"):
            return "枯荣园" if 灵物.主灵根 == "木" else "余烬"
        if 灵物.has_rune("试炼"):
            return "青苗圃" if 灵物.主灵根 == "木" else "天机阁"
        if 灵物.复杂度 > 0.8:
            return self.高复杂度子目录(灵物.主灵根)
        return self.默认子目录(灵物.主灵根)
```

### 5.3 渡劫验证机制

```python
def 渡劫(self, 目标境界) -> 渡劫结果:
    """境界突破时的验证考试"""
    
    # 1. 雷劫降临 —— 随机抽取已归档任务作为考题
    考题 = random.sample(self.历史归档, k=100)
    
    # 2. 天劫考验 —— 重新鉴定，与原标注对比
    正确数 = 0
    for 灵物 in 考题:
        新鉴定 = self.鉴定器.鉴定(灵物.原始内容)
        if 新鉴定.灵根 == 灵物.原始灵根:
            正确数 += 1
    
    实际准确率 = 正确数 / 100
    通过线 = 境界阈值表[目标境界]["准确率"]
    
    # 3. 判定
    if 实际准确率 >= 通过线:
        self.境界 = 目标境界
        self.解锁能力(目标境界)
        return 渡劫结果(成功=True, 新境界=目标境界, 准确率=实际准确率)
    else:
        self.修为值 *= 0.9  # 修为衰减10%
        return 渡劫结果(成功=False, 准确率=实际准确率, 差距=通过线-实际准确率)
```

### 6.3 反哺算力计算

```python
def 计算修为贡献(self, 灵物) -> float:
    """单个归档任务对系统的修为贡献"""
    
    基础分 = 10  # 每次归档基础贡献
    
    # 灵根纯度加成（分类置信度越高，贡献越大）
    纯度加成 = 基础分 * 灵物.置信度 * 2
    
    # 符文完整度加成
    符文加成 = 基础分 * 灵物.符文完整度 * 1.5
    
    # 洞天契合度加成（是否归入正确目录）
    契合加成 = 基础分 * 灵物.洞天契合度 * 1.5
    
    # 复杂度加成（复杂任务贡献更大）
    复杂度加成 = 基础分 * 灵物.复杂度 * 3
    
    # 人工校正惩罚（需要人工干预的任务贡献减半）
    惩罚 = 0.5 if 灵物.需人工校正 else 1.0
    
    总贡献 = (基础分 + 纯度加成 + 符文加成 + 契合加成 + 复杂度加成) * 惩罚
    
    return 总贡献

def 反哺系统(self, 灵物):
    """归档完成后反哺系统"""
    修为 = self.计算修为贡献(灵物)
    self.总修为值 += 修为
    self.修为账本.记录(灵物.task_id, 修为, datetime.now())
    
    # 检查是否达到模型微调阈值
    if self.自上次微调以来样本数() >= 100:
        self.触发模型微调()
    
    # 检查境界突破条件
    self.检查突破()
```

## 七、完整执行流程

```python
class 归藏引擎:
    """自动归档系统主入口"""
    
    def __init__(self):
        self.鉴定器 = 灵根鉴定器()
        self.符文炉 = 符文炉()
        self.洞天图 = 洞天图()
        self.修为系统 = 修为系统()
        self.境界 = 境界.炼气
        self.总修为值 = 0.0
    
    def 归档(self, 任务: Task) -> 归档结果:
        """
        完整归档流程：鉴定 → 铭刻 → 归位 → 反哺
        """
        # Step 1: 灵根鉴定（分类）
        鉴定结果 = self.鉴定器.鉴定(任务)
        灵物 = 灵物(任务, 鉴定结果)
        
        # Step 2: 符文铭刻（打标签）
        self.符文炉.铭刻(灵物)
        
        # Step 3: 洞天归位（存目录）
        归档路径 = self.洞天图.归位(灵物)
        
        # Step 4: 修为反哺（模型迭代）
        self.修为系统.反哺系统(灵物)
        
        # Step 5: 记录归档日志
        self.记录归档日志(灵物, 归档路径)
        
        return 归档结果(
            task_id=灵物.task_id,
            路径=归档路径,
            灵根=灵物.主灵根,
            符文=灵物.已铭刻符文,
            修为贡献=灵物.修为贡献,
            系统境界=self.境界,
            系统修为=self.总修为值
        )
    
    def 批量归档(self, 任务列表: List[Task]) -> List[归档结果]:
        """批量归档，自动去重和依赖排序"""
        
        # 按因果符（依赖关系）拓扑排序
        排序后 = self.拓扑排序(任务列表)
        
        结果列表 = []
        for 任务 in 排序后:
            结果 = self.归档(任务)
            结果列表.append(结果)
        
        # 批量归档后触发统计
        self.更新仪表盘()
        
        return 结果列表
```
