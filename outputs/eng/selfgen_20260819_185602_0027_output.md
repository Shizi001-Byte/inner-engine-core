# 任务类型: eng

## 一、设计目标

本方案面向「生态型 / 轻边际成本供给主体」的 AI 调度/运营引擎，核心目标是：

- **不依赖外部守护进程**：Guardian 是引擎自带的子进程，不是 systemd/supervisor。
- **崩溃后可自愈**：通过内置心跳检测、原子持久化、租约机制，实现断点续跑。
- **防超时**：按模型设置 API 超时，A 模型 600 秒，B 模型 300 秒。
- **防空转**：崩溃后不从头重跑，从 checkpoint 继续。
- **避免误杀 B 模型**：B 模型 reasoning 占比 70%-82% 是正常特性，质检跳过。

## 二、总体架构

```text
┌───────────────────────────────────────────────────────┐
│ Engine 进程组（无外部守护进程）                          │
│                                                       │
│  ┌──────────────┐      心跳/拉起      ┌──────────────┐ │
│  │   Worker     │ ◄────────────────► │   Guardian   │ │
│  │ 主引擎/执行器 │                    │ 内部监督进程   │ │
│  └──────┬───────┘                    └──────┬───────┘ │
│         │  WAL / Snapshot / Lease            │         │
│         ▼                                    │         │
│  ┌──────────────────────────┐                │         │
│  │ StateStore               │                │         │
│  │  wal.log                 │                │         │
│  │  snapshot-latest.json    │                │         │
│  │  heartbeat/*.json        │                │         │
│  └──────────────────────────┘                │         │
└──────────────────────────────────────────────┴─────────┘
```

- **Worker**：负责任务调度、API 调用、状态推进。
- **Guardian**：负责监控 Worker 心跳，发现 Worker 崩溃后拉起新 Worker。
- **Worker 也监控 Guardian**：Guardian 异常退出时，Worker 自己重新拉起 Guardian。
- **StateStore**：所有状态变更先写 WAL，再更新内存；定期做原子快照。

## 三、状态原子持久化

### 1. WAL 设计

所有状态变更都追加到 WAL，使用「长度 + JSON + CRC」防止半包写坏。

```python
# core/wal.py
import os, json, struct, zlib, time

class WAL:
    def __init__(self, path: str):
        self.path = path
        self.fd = open(path, "a+b")
        self.seq = self._repair_tail()

    def _repair_tail(self) -> int:
        """扫描到第一个损坏/半截记录处，截断尾部。"""
        seq = 0
        offset = 0
        self.fd.seek(0)
        while True:
            header = self.fd.read(4)
            if len(header) < 4:
                break
            (length,) = struct.unpack(">I", header)
            body = self.fd.read(length)
            if len(body) < length:
                break
            crc = self.fd.read(4)
            if len(crc) < 4:
                break
            if zlib.crc32(body) != struct.unpack(">I", crc)[0]:
                break
            rec = json.loads(body)
            seq = rec["seq"]
            offset = self.fd.tell()
        self.fd.truncate(offset)
        self.fd.flush()
        os.fsync(self.fd.fileno())
        return seq

    def append(self, op: str, payload: dict, epoch: int) -> dict:
        self.seq += 1
        rec = {
            "seq": self.seq,
            "epoch": epoch,
            "ts": time.time(),
            "op": op,
            "payload": payload,
        }
        body = json.dumps(rec, ensure_ascii=False).encode()
        self.fd.write(struct.pack(">I", len(body)))
        self.fd.write(body)
        self.fd.write(struct.pack(">I", zlib.crc32(body)))
        self.fd.flush()
        os.fsync(self.fd.fileno())
        return rec

    def replay(self, after_seq: int = 0):
        self.fd.seek(0)
        while True:
            header = self.fd.read(4)
            if len(header) < 4:
                break
            (length,) = struct.unpack(">I", header)
            body = self.fd.read(length)
            if len(body) < length:
                break
            crc = self.fd.read(4)
            if len(crc) < 4:
                break
            if zlib.crc32(body) != struct.unpack(">I", crc)[0]:
                break
            rec = json.loads(body)
            if rec["seq"] > after_seq:
                yield rec
```

### 2. 原子快照

快照使用「临时文件 + fsync + rename」，保证快照文件要么不存在，要么完整。

```python
# core/store.py
import os, json, time

def atomic_write(path, data: bytes):
    tmp = path + f".tmp.{os.getpid()}"
    with open(tmp, "wb") as f:
        f.write(data)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, path)
    dir_fd = os.open(os.path.dirname(path), os.O_DIRECTORY)
    os.fsync(dir_fd)
    os.close(dir_fd)


class StateStore:
    def __init__(self, data_dir: str, epoch: int):
        self.data_dir = data_dir
        self.epoch = epoch
        self.wal = WAL(os.path.join(data_dir, "wal.log"))
        self.tasks = {}
        self._load_snapshot()

    def _load_snapshot(self):
        snap_path = os.path.join(self.data_dir, "snapshot-latest.json")
        after_seq = 0
        if os.path.exists(snap_path):
            with open(snap_path) as f:
                snap = json.load(f)
            self.tasks = {t["task_id"]: t for t in snap["tasks"]}
            after_seq = snap["seq"]
        for rec in self.wal.replay(after_seq=after_seq):
            self.apply(rec)

    def append(self, op: str, payload: dict) -> dict:
        return self.wal.append(op, payload, self.epoch)

    def apply(self, rec: dict):
        op = rec["op"]
        payload = rec["payload"]
        if op == "task_update":
            task_id = payload["task_id"]
            task = self.tasks.setdefault(task_id, {"task_id": task_id})
            task.update(payload["changes"])
            task["updated_at"] = time.time()
        elif op == "lease_acquire":
            task = self.tasks[payload["task_id"]]
            task["lease"] = payload["lease"]
            task["state"] = "RUNNING"
        elif op == "lease_renew":
            task = self.tasks[payload["task_id"]]
            task["lease"] = payload["lease"]
        elif op == "lease_release":
            task = self.tasks[payload["task_id"]]
            task["lease"] = None
            task["state"] = "READY"

    def snapshot(self):
        data = {
            "seq": self.wal.seq,
            "ts": time.time(),
            "tasks": list(self.tasks.values()),
        }
        atomic_write(
            os.path.join(self.data_dir, "snapshot-latest.json"),
            json.dumps(data, ensure_ascii=False).encode(),
        )
```

## 四、心跳检测与内部互监

### 1. 心跳文件

Worker 和 Guardian 各自维护一个心跳文件，格式统一：

```json
{
  "role": "worker",
  "pid": 12345,
  "epoch": 3,
  "ts": 1710000000.0
}
```

```python
# core/heartbeat.py
import os, json, time

def read_heartbeat(path):
    try:
        with open(path) as f:
            return json.load(f)
    except Exception:
        return None

def write_heartbeat(path, role, pid, epoch):
    data = {
        "role": role,
        "pid": pid,
        "epoch": epoch,
        "ts": time.time(),
    }
    tmp = path + ".tmp"
    with open(tmp, "w") as f:
        json.dump(data, f)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, path)

def pid_alive(pid):
    if pid <= 0:
        return False
    try:
        os.kill(pid, 0)
    except ProcessLookupError:
        return False
    except PermissionError:
        return True
    return True
```

### 2. Guardian 监督 Worker

Guardian 是 Worker 自己拉起的子进程，不是外部守护进程。

```python
# core/guardian.py
import os, sys, signal, subprocess, time
from .heartbeat import read_heartbeat, write_heartbeat, pid_alive

class Guardian:
    def __init__(self, worker_pid, data_dir, epoch):
        self.worker_pid = worker_pid
        self.data_dir = data_dir
        self.epoch = epoch
        self.worker_hb = os.path.join(data_dir, "heartbeat", "worker.json")
        self.guardian_hb = os.path.join(data_dir, "heartbeat", "guardian.json")
        self.heartbeat_interval = 5
        self.deadline = 15

    def loop(self):
        while True:
            write_heartbeat(self.guardian_hb, "guardian", os.getpid(), self.epoch)
            hb = read_heartbeat(self.worker_hb)
            if self._is_stale(hb):
                self._restart_worker()
            time.sleep(self.heartbeat_interval)

    def _is_stale(self, hb):
        if not hb:
            return True
        if hb.get("epoch") != self.epoch:
            return True
        return time.time() - hb.get("ts", 0) > self.deadline

    def _restart_worker(self):
        if self.worker_pid and pid_alive(self.worker_pid):
            os.kill(self.worker_pid, signal.SIGTERM)
            time.sleep(3)
            if pid_alive(self.worker_pid):
                os.kill(self.worker_pid, signal.SIGKILL)
                time.sleep(1)

        self.epoch += 1
        epoch_path = os.path.join(self.data_dir, "epoch")
        with open(epoch_path, "w") as f:
            f.write(str(self.epoch))
            f.flush()
            os.fsync(f.fileno())

        proc = subprocess.Popen(
            [
                sys.executable, "-m", "engine.main",
                "--worker",
                "--epoch", str(self.epoch),
                "--recover",
                "--data-dir", self.data_dir,
            ],
            cwd=os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
        )
        self.worker_pid = proc.pid
```

### 3. Worker 监督 Guardian

Worker 自己启动一个后台线程，持续检查 Guardian 心跳；如果 Guardian 失联，则重新拉起 Guardian。

```python
# core/worker.py
import os, sys, subprocess, threading, time
from .heartbeat import read_heartbeat, write_heartbeat, pid_alive
from .store import StateStore
from .task_manager import TaskManager
from .recovery import Recovery

class Worker:
    def __init__(self, epoch, recover, data_dir):
        self.epoch = epoch
        self.data_dir = data_dir
        self.store = StateStore(data_dir, epoch)
        self.tm = TaskManager(self.store)
        if recover:
            Recovery(self.store, self.tm).run()

        self.worker_hb = os.path.join(data_dir, "heartbeat", "worker.json")
        self.guardian_hb = os.path.join(data_dir, "heartbeat", "guardian.json")
        self.heartbeat_interval = 5
        self.deadline = 15

    def start_heartbeat_thread(self):
        def _loop():
            while True:
                write_heartbeat(self.worker_hb, "worker", os.getpid(), self.epoch)
                time.sleep(self.heartbeat_interval)
        threading.Thread(target=_loop, daemon=True).start()

    def start_guardian_watchdog(self):
        def _loop():
            while True:
                self._ensure_guardian()
                time.sleep(self.heartbeat_interval)
        threading.Thread(target=_loop, daemon=True).start()

    def _ensure_guardian(self):
        hb = read_heartbeat(self.guardian_hb)
        if hb and time.time() - hb.get("ts", 0) <= self.deadline:
            return

        if hb and hb.get("pid") and pid_alive(hb["pid"]):
            os.kill(hb["pid"], 9)

        subprocess.Popen(
            [
                sys.executable, "-m", "engine.main",
                "--guardian",
                "--worker-pid", str(os.getpid()),
                "--epoch", str(self.epoch),
                "--data-dir", self.data_dir,
            ],
            cwd=os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
        )

    def run(self):
        self.start_heartbeat_thread()
        self.start_guardian_watchdog()

        while True:
            task = self.tm.next_ready_task()
            if task:
                owner = f"node-{os.uname().nodename}:{os.getpid()}:{self.epoch}"
                if self.tm.acquire(task["task_id"], owner, lease_duration=30, epoch=self.epoch):
                    Executor(self.tm).run(task)
            else:
                time.sleep(0.2)
```

## 五、崩溃后断点续跑

### 1. 任务状态机

```text
PENDING → READY → RUNNING → CALLING_API → RUNNING → SUCCEEDED
                      │                         │
                      └──────── FAILED ←─────────┘
```

崩溃恢复时：

- `RUNNING` / `CALLING_API` 且持有者已死/租约过期 → 回退到 `READY`
- 任务 `attempts + 1`
- 如果超过 `max_attempts` → `FAILED`

### 2. 恢复扫描

```python
# core/recovery.py
import time
from .heartbeat import pid_alive

class Recovery:
    def __init__(self, store, task_manager):
        self.store = store
        self.tm = task_manager

    def run(self):
        for task in self.store.tasks.values():
            if task["state"] in ("RUNNING", "CALLING_API"):
                lease = task.get("lease")
                if lease and self._lease_alive(lease):
                    continue

                task["attempts"] += 1
                if task["attempts"] > task.get("max_attempts", 3):
                    self.tm.update(
                        task["task_id"],
                        state="FAILED",
                        lease=None,
                        error="crash_recovery_max_attempts",
                    )
                else:
                    self.tm.update(
                        task["task_id"],
                        state="READY",
                        lease=None,
                    )
            elif task["state"] == "PENDING":
                self.tm.update(task["task_id"], state="READY")

    def _lease_alive(self, lease):
        if lease.get("epoch", -1) < self.store.epoch:
            return False
        pid = lease.get("pid")
        if pid and not pid_alive(pid):
            return False
        return lease.get("expires_at", 0) > time.time()
```

### 3. 任务级 Checkpoint

Executor 通过 `task["progress"]["step"]` 判断上次执行到哪一步。

```python
# core/executor.py
import time

class Executor:
    def __init__(self, tm, settings, quality_guard):
        self.tm = tm
        self.settings = settings
        self.quality_guard = quality_guard

    def run(self, task):
        step = task["progress"].get("step", "preprocess")

        if step == "preprocess":
            # 模拟预处理
            self.tm.update(task["task_id"], progress={
                "step": "calling_api",
                "preprocessed": True,
            })
            step = "calling_api"

        if step == "calling_api":
            self.tm.update(task["task_id"], state="CALLING_API")

            timeout = self.settings.timeout_by_model.get(
                task["model"],
                self.settings.default_timeout,
            )

            request_id = f"{task['task_id']}-{task['attempts']}-{int(time.time())}"

            self.tm.update(task["task_id"], progress={
                "step": "calling_api",
                "request_id": request_id,
            })

            try:
                response = call_ai_api(
                    task,
                    request_id=request_id,
                    timeout=timeout,
                )
            except TimeoutError:
                self.tm.update(task["task_id"], state="FAILED", error="api_timeout")
                return

            if not self.quality_guard.check(task, response):
                self.tm.update(task["task_id"], state="FAILED", error="quality_check_failed")
                return

            self.tm.update(task["task_id"], state="RUNNING", progress={
                "step": "postprocess",
                "response_ref": save_response(task["task_id"], response),
            })
            step = "postprocess"

        if step == "postprocess":
            response = load_response(task["progress"]["response_ref"])
            result = postprocess(response)
            self.tm.update(task["task_id"], state="SUCCEEDED", result=result, lease=None)
```

## 七、避坑清单

### 1. 配置必须集中读取，改后立即验证

```python
# core/config.py
from dataclasses import dataclass, field

@dataclass
class Settings:
    heartbeat_interval: int = 5
    heartbeat_deadline: int = 15
    lease_duration: int = 30
    snapshot_interval: int = 60
    max_attempts: int = 3
    default_timeout: int = 300

    timeout_by_model: dict = field(default_factory=lambda: {
        "A": 600,
        "B": 300,
    })

    skip_reasoning_ratio_models: list = field(default_factory=lambda: ["B"])
    max_reasoning_ratio: float = 0.9

    def validate(self):
        assert self.timeout_by_model["A"] >= 600
        assert self.timeout_by_model["B"] >= 300
```

所有模块只从 `Settings` 读取，不散落硬编码常量。

### 2. B 模型质检跳过 reasoning 占比检测

```python
# core/quality.py
class QualityGuard:
    def __init__(self, settings):
        self.settings = settings

    def check(self, task, response):
        if task["model"] in self.settings.skip_reasoning_ratio_models:
            return True

        reasoning_ratio = response.get("reasoning_ratio", 0)
        return reasoning_ratio <= self.settings.max_reasoning_ratio
```

### 3. API 超时按模型区分

- B 模型：300 秒
- A 模型：600 秒

在 `Executor` 中强制使用：

```python
timeout = settings.timeout_by_model[task["model"]]
response = call_ai_api(task, timeout=timeout)
```

## 八、总结

这套方案通过 **Worker + Guardian 互监** 实现不依赖外部守护进程的崩溃拉起；通过 **WAL + Snapshot** 保证状态原子持久化；通过 **任务级 checkpoint + 租约恢复** 实现断点续跑；通过 **配置中心 + 模型级超时 + B 模型质检豁免** 规避历史踩坑。

对于整机宕机场景，可在此基础上扩展为多节点对等恢复：节点间通过租约感知彼此心跳，任务从故障节点迁移到存活节点，仍然不需要外部守护进程。
