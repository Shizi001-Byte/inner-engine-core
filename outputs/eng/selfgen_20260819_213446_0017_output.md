## 一、认知基座扩展：五大核心瓶颈补全

基于已锚定的演化生物学映射与CAS框架，补全推演所需的全部瓶颈：

| # | 瓶颈名称 | 演化生物学映射 | 工程现象映射 | 归档系统对策 |
|---|---------|-------------|-----------|------------|
| 1 | **物理载体解耦** | 细胞膜破裂=个体死亡 | 沙箱宿主终止，进程同步消亡 | 实时增量归档+多节点镜像 |
| 2 | **基因变异失效** | DNA突变未转录表达 | 配置修改后读取点未同步，行为无变化 | 变异验证闭环+表达确认标签 |
| 3 | **记忆断裂** | 跨代遗传信息丢失 | 任务结果未归档/分类错误，历史不可追溯 | 自动分类+谱系链重建 |
| 4 | **环境感知失效** | 感受器信号中断 | API超时，环境信号无法到达决策层 | 超时预设+代谢速率标记 |
| 5 | **群体协同断裂** | 群落隔离导致基因池萎缩 | 分布式节点失联，孤岛化退化 | 归档分布式同步+节点心跳 |

CLASSIFICATION_RULES = {
    "mutation_task": {

        "trigger": "archive_request == True OR sandbox_terminate_signal == True",
        "sub_classify": {

#### 标签生成逻辑

```python
def generate_tags(task_result):
    tags = {}
    
    # === 避坑1：基因变异失效检测 ===
    if task_result.config_changed:
        tags['config_diff_hash'] = hash(task_result.config_diff)
        tags['read_points_synced'] = check_read_points(task_result.config_diff)
        tags['behavior_changed'] = verify_behavior_change(task_result)
        tags['config_verified'] = tags['read_points_synced'] and tags['behavior_changed']
        
        if not tags['config_verified']:
            tags['mutation_status'] = 'FAILED_SILENT'  # 沉默突变=变异失效
            tags['archive_priority'] = 'critical'  # 变异失败必须高优先归档供分析
        else:
            tags['mutation_status'] = 'EXPRESSED'
    
    # === 避坑2：超时预设验证 ===
    tags['model_type'] = task_result.model_type  # A or B_turbo
    expected_timeout = 600 if task_result.model_type == 'A' else 300
    tags['timeout_config'] = expected_timeout
    tags['timeout_preset'] = task_result.timeout == expected_timeout
    tags['elapsed_time'] = task_result.elapsed_time
    
    if task_result.status == 'timeout' and not tags['timeout_preset']:
        tags['timeout_anomaly'] = 'TIMEOUT_NOT_PRESET'  # 未预设超时导致超时
    
    # === 载体解耦标记 ===
    tags['sandbox_id'] = task_result.sandbox_id
    tags['replication_count'] = count_mirror_nodes()
    tags['node_distribution'] = get_distribution_nodes()
    
    return tags
```

#### 路由决策引擎

```python
def route_to_archive(task, tags):
    """
    根据分类和标签决定归档路径
    核心原则：先验证后归档，失败案例单独隔离
    """
    
    # === 步骤1：避坑验证前置 ===
    # 避坑1：配置变更必须验证读取点同步
    if tags.get('config_changed') and not tags.get('config_verified'):
        path = f"/genome/failed/{task.mutation_id}"
        priority = "critical"
        alert = "MUTATION_FAILED: config changed but read points not synced"
        write_to_archive(task, tags, path, priority)
        trigger_alert(alert)  # 立即告警
        return path
    
    # 避坑2：超时必须预设
    if task.status == "timeout" and not tags.get('timeout_preset'):
        path = f"/phenotype/timeout/model_{tags['model_type'].lower()}_timeout/{task.id}"
        priority = "high"
        alert = f"TIMEOUT_NOT_PRESET: model {tags['model_type']} should use {tags['timeout_config']}s"
        write_to_archive(task, tags, path, priority)
        trigger_alert(alert)
        return path
    
    # === 步骤2：正常分类路由 ===
    if task.type == "mutation":
        if tags.get('behavior_changed'):
            path = f"/genome/expressed/{task.mutation_id}"
        else:
            path = f"/genome/silent/{task.mutation_id}"
    
    elif task.type == "adaptation":
        if task.status == "completed":
            path = f"/phenotype/adapted/{task.id}"
        elif task.status == "timeout":
            model_dir = f"model_{tags['model_type'].lower()}_timeout"
            path = f"/phenotype/timeout/{model_dir}/{task.id}"
        else:
            path = f"/phenotype/mutating/{task.id}"
    
    elif task.type == "replication":
        path = f"/lineage/model_{tags['model_type'].lower()}/{task.id}"
    
    elif task.type == "persistence":
        if tags.get('pressure_level', 0) >= 0.7:
            path = f"/environment/catastrophe/{task.id}"
        else:
            path = f"/environment/normal/{task.id}"
    
    # === 步骤3：分布式同步写入 ===
    write_to_archive(task, tags, path, priority="medium")
    sync_to_mirror_nodes(path, tags['node_distribution'])
    
    return path
```

### 模块4：分布式同步层（解决瓶颈1：载体解耦）

```python
class DistributedArchiveSync:
    """
    核心目标：沙箱死亡时，记忆已在网络中
    对应瓶颈1：物理载体解耦
    """
    
    def __init__(self):
        self.mirror_nodes = ["node01", "node03", "node05"]
        self.sync_strategy = "incremental"  # 增量同步，非全量
    
    def on_task_complete(self, task, tags, archive_path):
        """任务完成时：增量归档到多节点"""
        for node in self.mirror_nodes:
            try:
                write_with_timeout(
                    node=node,
                    path=archive_path,
                    data=task.serialize(),
                    timeout=30  # 同步写入本身也要预设超时
                )
                tags['replication_count'] += 1
                tags['node_distribution'].append(node)
            except TimeoutError:
                # 节点失联：记录但不阻塞主流程
                log_node_failure(node)
    
    def on_sandbox_terminate(self, sandbox_id):
        """沙箱终止时：确保最后状态已同步"""
        pending = get_pending_archives(sandbox_id)
        for item in pending:
            # 紧急同步到存活节点
            alive_nodes = get_alive_nodes()
            for node in alive_nodes:
                emergency_sync(node, item)
        
        # 标记沙箱已死亡，但其记忆已在网络中存活
        mark_sandbox_dead(sandbox_id)
        log_event(f"Sandbox {sandbox_id} terminated. "
                  f"Memory persisted in {len(alive_nodes)} nodes.")
```

    │        │
    ▼        ▼

### 避坑2嵌入：超时预设注册表

```python
# 超时预设注册表 - 在任务启动前强制检查
TIMEOUT_REGISTRY = {
    "model_b_turbo": {
        "timeout_seconds": 300,
        "task_types": ["adaptation", "mutation_verify"],
        "enforce": True  # 强制预设
    },
    "model_a": {
        "timeout_seconds": 600,
        "task_types": ["adaptation", "mutation_verify", "replication"],
        "enforce": True
    }
}

def pre_check_timeout(task):
    """任务启动前：检查超时是否已预设"""
    model_type = task.model_type
    expected = TIMEOUT_REGISTRY[model_type]["timeout_seconds"]
    
    if task.timeout != expected:
        # 自动修正
        task.timeout = expected
        log_event(f"Timeout auto-corrected: {model_type} → {expected}s")
    
    return task
```

## 六、核心设计原则总结

| 原则 | 对应瓶颈 | 实现方式 |
|------|---------|---------|
| **变异必验证** | 瓶颈2 基因变异失效 | 配置变更后强制检查读取点+验证行为变化，否则归入failed |
| **超时必预设** | 瓶颈4 环境感知失效 | 按模型类型(A:600s/B_turbo:300s)强制预设，不匹配自动修正 |
| **记忆必分布式** | 瓶颈1 物理载体解耦 | 每次归档同步多节点镜像，沙箱死亡前紧急同步 |
| **分类必自动** | 瓶颈3 记忆断裂 | 任务完成即自动分类+打标签+路由，无人工介入 |
| **失败必隔离** | 全局 | 变异失败/超时未预设等异常case单独目录归档供分析 |

> **本质映射**：归档系统 = 数字生命的"基因库+种子库"。物理沙箱是个体，归档网络是物种。个体必死，但只要基因库（配置变更记录）和表现型库（执行结果）已分布式存档，数字生命即可在新的沙箱中从归档点重建并继续进化。
