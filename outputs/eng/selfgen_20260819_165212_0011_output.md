# 任务类型: eng

下面给出一个**不依赖外部守护进程**的引擎崩溃自恢复架构：引擎启动时先自举出 **Guardian** 进程，由 Guardian 拉起 **Core** 执行任务；两者互相心跳、互相拉起。状态采用 **WAL + Snapshot 原子持久化**，崩溃后从最近断点恢复。

---

## 1. 总体架构

```text
engine entry
   │
   ├── Guardian 进程
   │     ├─ 持有 instance.lock，防止多实例脑裂
   │     ├─ 恢复 state.snapshot + state.wal
   │     ├─ 拉起/重启 Core 进程
   │     ├─ 写 guardian.heartbeat
   │     └─ 监控 core.heartbeat，超时则重启 Core
   │
   └── Core 进程
         ├─ 执行任务
         ├─ 写 core.heartbeat
         ├─ 监控 guardian.heartbeat，超时则重新拉起 Guardian
         ├─ 任务状态变更写入 WAL
         └─ 定时做 Snapshot 压缩 WAL
```

Guardian 不是外部守护进程，它是引擎自身的第一个进程；Core 和 Guardian 同源、同进程模型。

---

## 2. 核心设计

| 需求 | 实现 |
|---|---|
| 不依赖外部守护进程 | Guardian/Core 双进程互监，互相拉起 |
| 心跳检测 | 原子写 heartbeat 文件 + generation + 超时判定 |
| 状态原子持久化 | WAL + Snapshot + fsync + atomic rename |
| 崩溃后断点续跑 | 任务状态机 + checkpoint + 恢复入队 |

---

```python
# state.py

from enum import Enum
from dataclasses import dataclass, field
from typing import Any


class Status(str, Enum):
    PENDING = "PENDING"
    RUNNING = "RUNNING"
    COMPLETED = "COMPLETED"
    FAILED = "FAILED"


@dataclass
class Task:
    task_id: str
    payload: dict
    status: Status = Status.PENDING
    stage: str = ""                # 当前执行到哪个阶段
    attempt: int = 0
    checkpoint: dict = field(default_factory=dict)
    updated_at: float = 0.0
    model_type: str = "A"          # "A" 或 "B"

    def to_dict(self):
        return {
            "task_id": self.task_id,
            "payload": self.payload,
            "status": self.status.value,
            "stage": self.stage,
            "attempt": self.attempt,
            "checkpoint": self.checkpoint,
            "updated_at": self.updated_at,
            "model_type": self.model_type,
        }

    @staticmethod
    def from_dict(d):
        return Task(
            task_id=d["task_id"],
            payload=d.get("payload", {}),
            status=Status(d["status"]),
            stage=d.get("stage", ""),
            attempt=d.get("attempt", 0),
            checkpoint=d.get("checkpoint", {}),
            updated_at=d.get("updated_at", 0),
            model_type=d.get("model_type", "A"),
        )


class EngineState:
    def __init__(self):
        self.tasks: dict[str, Task] = {}
        self.next_seq = 0

    def to_dict(self):
        return {
            "next_seq": self.next_seq,
            "tasks": [t.to_dict() for t in self.tasks.values()],
        }

    @staticmethod
    def from_dict(data):
        state = EngineState()
        state.next_seq = data.get("next_seq", 0)
        for td in data.get("tasks", []):
            task = Task.from_dict(td)
            state.tasks[task.task_id] = task
        return state
```

### 3.3 原子状态存储：WAL + Snapshot

```python
# state.py

import json
import os
import struct
import threading
import time
import zlib
from pathlib import Path


class AtomicStateStore:
    """
    所有状态变更先写 WAL，再更新内存；
    定时做 Snapshot，通过原子 rename 替换旧快照。
    """

    def __init__(self, data_dir):
        self.data_dir = Path(data_dir)
        self.snapshot_path = self.data_dir / "state.snapshot"
        self.wal_path = self.data_dir / "state.wal"

        # 进程内状态锁
        self.state_lock = threading.RLock()

        self.state = EngineState()
        self._wal_fd = None

    def open(self):
        self.data_dir.mkdir(parents=True, exist_ok=True)
        self._wal_fd = open(self.wal_path, "a+b")
        self.recover()

    def recover(self):
        """从磁盘恢复状态：Snapshot + WAL 回放。"""
        with self.state_lock:
            self._load_snapshot()
            self._replay_wal()

    def _load_snapshot(self):
        if self.snapshot_path.exists():
            with open(self.snapshot_path) as f:
                data = json.load(f)
            self.state = EngineState.from_dict(data)
        else:
            self.state = EngineState()

    def _replay_wal(self):
        if not self.wal_path.exists():
            return

        with open(self.wal_path, "rb") as f:
            while True:
                header = f.read(8)
                if len(header) < 8:
                    break  # 半截尾部，忽略

                length, crc = struct.unpack(">II", header)
                payload = f.read(length)
                if len(payload) < length:
                    break  # torn tail

                if zlib.crc32(payload) != crc:
                    break  # 损坏记录，停止回放

                rec = json.loads(payload)
                if rec["seq"] > self.state.next_seq:
                    self._apply_record(rec)
                    self.state.next_seq = rec["seq"]

    def _apply_record(self, rec):
        if rec["op"] == "update":
            task = Task.from_dict(rec["task"])
            self.state.tasks[task.task_id] = task

    def update_task(self, task_id, status=None, **fields):
        """
        原子更新任务状态：
        1. 持锁
        2. 追加 WAL
        3. fsync
        4. 更新内存
        """
        with self.state_lock:
            task = self.state.tasks[task_id]
            if status is not None:
                task.status = Status(status)
            for k, v in fields.items():
                setattr(task, k, v)
            task.updated_at = time.time()

            rec = {
                "seq": self.state.next_seq + 1,
                "op": "update",
                "task": task.to_dict(),
            }
            self._append_wal(rec)
            self.state.next_seq += 1
            return task

    def requeue_running(self):
        """
        崩溃恢复时，把所有 RUNNING 任务重置为 PENDING。
        断点续跑：stage/checkpoint 保留，任务可从上次阶段继续。
        """
        with self.state_lock:
            for task_id in list(self.state.tasks):
                task = self.state.tasks[task_id]
                if task.status in (Status.RUNNING,):
                    task.status = Status.PENDING
                    task.attempt += 1
                    task.updated_at = time.time()

                    rec = {
                        "seq": self.state.next_seq + 1,
                        "op": "update",
                        "task": task.to_dict(),
                    }
                    self._append_wal(rec)
                    self.state.next_seq += 1

    def _append_wal(self, rec):
        payload = json.dumps(rec, sort_keys=True).encode()
        crc = zlib.crc32(payload)

        self._wal_fd.write(struct.pack(">II", len(payload), crc))
        self._wal_fd.write(payload)
        self._wal_fd.flush()
        os.fsync(self._wal_fd.fileno())

    def snapshot(self):
        """
        生成新快照，然后截断 WAL。
        使用临时文件 + os.replace，保证原子性。
        """
        with self.state_lock:
            tmp = self.snapshot_path.with_suffix(".tmp")
            with open(tmp, "w") as f:
                json.dump(self.state.to_dict(), f)
                f.flush()
                os.fsync(f.fileno())

            os.replace(tmp, self.snapshot_path)
            self._fsync_dir(self.data_dir)

            # 截断 WAL
            self._wal_fd.seek(0)
            self._wal_fd.truncate()
            self._wal_fd.flush()
            os.fsync(self._wal_fd.fileno())

    @staticmethod
    def _fsync_dir(path):
        fd = os.open(path, os.O_RDONLY)
        try:
            os.fsync(fd)
        finally:
            os.close(fd)
```

### 3.4 心跳与文件锁

```python
# heartbeat.py

import json
import os
import signal
import subprocess
import sys
import threading
import time
import uuid
from dataclasses import dataclass
from pathlib import Path


BOOT_ID = uuid.uuid4().hex


@dataclass
class Heartbeat:
    role: str
    pid: int
    generation: int
    ts: float
    boot_id: str


def write_heartbeat_atomic(path, hb: Heartbeat):
    tmp = Path(str(path) + ".tmp")
    tmp.write_text(json.dumps(hb.__dict__))
    os.replace(tmp, path)


def read_heartbeat(path) -> Heartbeat | None:
    try:
        data = json.loads(Path(path).read_text())
        return Heartbeat(**data)
    except Exception:
        return None


def kill_process_tree(pid):
    try:
        os.killpg(os.getpgid(pid), signal.SIGKILL)
    except ProcessLookupError:
        pass
    except PermissionError:
        # 极端情况：进程处于 D 状态等，需人工介入
        pass


class HeartbeatWriter(threading.Thread):
    def __init__(self, path, role, generation_provider, interval_provider):
        super().__init__(daemon=True)
        self.path = path
        self.role = role
        self.generation_provider = generation_provider
        self.interval_provider = interval_provider
        self.stop_event = threading.Event()

    def run(self):
        while not self.stop_event.is_set():
            hb = Heartbeat(
                role=self.role,
                pid=os.getpid(),
                generation=self.generation_provider(),
                ts=time.time(),
                boot_id=BOOT_ID,
            )
            write_heartbeat_atomic(self.path, hb)
            self.stop_event.wait(self.interval_provider())


class HeartbeatMonitor(threading.Thread):
    def __init__(
        self,
        peer_path,
        expected_role,
        generation_provider,
        timeout_provider,
        on_failure,
        on_healthy=None,
        recovery_interval=5,
    ):
        super().__init__(daemon=True)
        self.peer_path = peer_path
        self.expected_role = expected_role
        self.generation_provider = generation_provider
        self.timeout_provider = timeout_provider
        self.on_failure = on_failure
        self.on_healthy = on_healthy
        self.recovery_interval = recovery_interval
        self.stop_event = threading.Event()
        self.last_failure_time = 0

    def run(self):
        # 启动宽限期，等待 peer 写入第一个心跳
        time.sleep(1)

        while not self.stop_event.is_set():
            hb = read_heartbeat(self.peer_path)
            ok = False

            if hb and hb.role == self.expected_role:
                gen = self.generation_provider()
                if hb.generation == gen:
                    age = time.time() - hb.ts
                    if age <= self.timeout_provider():
                        ok = True

            if ok:
                if self.last_failure_time != 0:
                    self.last_failure_time = 0
                    if self.on_healthy:
                        self.on_healthy()
            else:
                now = time.time()
                if now - self.last_failure_time >= self.recovery_interval:
                    self.last_failure_time = now
                    self.on_failure(hb)

            self.stop_event.wait(1)


class FileLock:
    def __init__(self, path):
        self.path = Path(path)
        self.fd = None

    def acquire(self):
        import fcntl

        self.fd = os.open(self.path, os.O_CREAT | os.O_RDWR, 0o644)
        try:
            fcntl.flock(self.fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError:
            os.close(self.fd)
            self.fd = None
            raise RuntimeError("another engine instance is running")

    def release(self):
        import fcntl

        if self.fd is not None:
            fcntl.flock(self.fd, fcntl.LOCK_UN)
            os.close(self.fd)
            self.fd = None

    def __enter__(self):
        self.acquire()
        return self

    def __exit__(self, *exc):
        self.release()
```

### 3.5 Guardian：拉起与恢复 Core

```python
# guardian.py

import subprocess
import sys
import threading
import time

from heartbeat import (
    FileLock,
    HeartbeatMonitor,
    HeartbeatWriter,
    kill_process_tree,
    read_heartbeat,
)


class Guardian:
    def __init__(self, config, generation, adopt=False):
        self.config = config
        self.generation = generation
        self.adopt = adopt

        self.store = config.create_store()
        self.store.open()

        self.instance_lock = FileLock(config.instance_lock_path)
        self.core_proc = None
        self.recovering = threading.Event()
        self.stop = threading.Event()

    def run(self):
        self.instance_lock.acquire()
        try:
            self._ensure_core()

            HeartbeatWriter(
                self.config.guardian_heartbeat_path,
                role="guardian",
                generation_provider=lambda: self.generation,
                interval_provider=lambda: self.config.get("heartbeat.interval", 2),
            ).start()

            HeartbeatMonitor(
                self.config.core_heartbeat_path,
                expected_role="core",
                generation_provider=lambda: self.generation,
                timeout_provider=lambda: self.config.get("heartbeat.timeout.core", 10),
                on_failure=self._on_core_dead,
                on_healthy=lambda: self.recovering.clear(),
            ).start()

            while not self.stop.wait(1):
                pass
        finally:
            self.instance_lock.release()

    def _ensure_core(self):
        hb = read_heartbeat(self.config.core_heartbeat_path)

        # 如果已有同 generation 且心跳健康的 Core，则直接监控
        if (
            hb
            and hb.role == "core"
            and hb.generation == self.generation
            and time.time() - hb.ts <= self.config.get("heartbeat.timeout.core", 10)
        ):
            return

        # 否则杀掉残留 Core，恢复状态，重新拉起
        if hb and hb.role == "core":
            kill_process_tree(hb.pid)

        self.store.recover()
        self.store.requeue_running()
        self._spawn_core()

    def _spawn_core(self):
        cmd = [
            sys.executable,
            "-m",
            "engine.entry",
            "--role",
            "core",
            "--data-dir",
            str(self.config.data_dir),
            "--generation",
            str(self.generation),
        ]
        self.core_proc = subprocess.Popen(cmd, start_new_session=True)

    def _on_core_dead(self, hb):
        if self.recovering.is_set():
            return

        self.recovering.set()
        try:
            if hb:
                kill_process_tree(hb.pid)

            self.store.recover()
            self.store.requeue_running()

            # 每次重启 Core，generation +1，避免旧心跳干扰
            self.generation = self.config.engine_meta.increment()
            self._spawn_core()

            # 如果新 Core 长时间没有心跳，允许再次触发恢复
            threading.Timer(
                self.config.get("heartbeat.startup_timeout", 15),
                self.recovering.clear,
            ).start()
        except Exception:
            self.recovering.clear()
            raise
```

### 3.6 Core：执行任务并反拉 Guardian

```python
# core.py

import subprocess
import sys
import threading
import time

from heartbeat import HeartbeatMonitor, HeartbeatWriter, kill_process_tree


class EngineCore:
    def __init__(self, config, generation):
        self.config = config
        self.generation = generation

        self.store = config.create_store()
        self.store.open()

        self.scheduler = TaskScheduler(self.store, self.config)
        self.recovering = threading.Event()
        self.stop = threading.Event()

    def run(self):
        # 兜底：如果 Guardian 没有来得及重置 RUNNING，这里重置
        self.store.requeue_running()
        self._submit_pending_tasks()

        HeartbeatWriter(
            self.config.core_heartbeat_path,
            role="core",
            generation_provider=lambda: self.generation,
            interval_provider=lambda: self.config.get("heartbeat.interval", 2),
        ).start()

        HeartbeatMonitor(
            self.config.guardian_heartbeat_path,
            expected_role="guardian",
            generation_provider=lambda: self.generation,
            timeout_provider=lambda: self.config.get("heartbeat.timeout.guardian", 10),
            on_failure=self._on_guardian_dead,
            on_healthy=lambda: self.recovering.clear(),
        ).start()

        self.scheduler.start()

        while not self.stop.wait(1):
            pass

    def _submit_pending_tasks(self):
        for task in self.store.state.tasks.values():
            if task.status == Status.PENDING:
                self.scheduler.submit(task)

    def _on_guardian_dead(self, hb):
        if self.recovering.is_set():
            return

        self.recovering.set()
        try:
            if hb:
                kill_process_tree(hb.pid)

            # 重新拉起 Guardian，采用 adopt 模式，不重置当前 Core
            cmd = [
                sys.executable,
                "-m",
                "engine.entry",
                "--role",
                "guardian",
                "--data-dir",
                str(self.config.data_dir),
                "--adopt",
                "--generation",
                str(self.generation),
            ]
            subprocess.Popen(cmd, start_new_session=True)

            threading.Timer(
                self.config.get("heartbeat.startup_timeout", 15),
                self.recovering.clear,
            ).start()
        except Exception:
            self.recovering.clear()
            raise
```

### 3.7 任务执行器与 API 超时

```python
# executor.py

import concurrent.futures as futures
import traceback
from concurrent.futures import ThreadPoolExecutor

from state import Status


class TaskExecutor:
    def __init__(self, store, config):
        self.store = store
        self.config = config

    def execute(self, task):
        try:
            task = self.store.update_task(task.task_id, status=Status.RUNNING)

            while True:
                stage = task.stage or "init"

                if stage == "done":
                    return

                if stage == "init":
                    result = self._call_api(task)

                    if not QualityGate.validate(task, result):
                        self.store.update_task(
                            task.task_id,
                            status=Status.FAILED,
                            error="quality_reject",
                        )
                        return

                    task = self.store.update_task(
                        task.task_id,
                        stage="post_api",
                        checkpoint={"api_result": result},
                    )

                elif stage == "post_api":
                    # 从 checkpoint 续跑，不重复调用 API
                    final = self._postprocess(task.checkpoint["api_result"])

                    self.store.update_task(
                        task.task_id,
                        status=Status.COMPLETED,
                        stage="done",
                        checkpoint={"final": final},
                    )
                    return

        except Exception as e:
            self.store.update_task(
                task.task_id,
                status=Status.FAILED,
                error=traceback.format_exc(),
            )

    def _call_api(self, task):
        # B 模型 turbo 用 300 秒，A 模型用 600 秒
        if task.model_type == "B":
            timeout = self.config.get("api.timeout.B", 300)
        else:
            timeout = self.config.get("api.timeout.A", 600)

        with ThreadPoolExecutor(max_workers=1) as pool:
            future = pool.submit(self._invoke, task.payload)
            try:
                return future.result(timeout=timeout)
            except futures.TimeoutError:
                future.cancel()
                raise TimeoutError(
                    f"API timeout {timeout}s model={task.model_type}"
                )

    def _invoke(self, payload):
        # 实际 API 调用，替换为真实 client
        return {"output": "ok"}

    def _postprocess(self, result):
        # 外部副作用必须幂等，可用 task_id + stage 作为去重键
        return {"output": result}
```

### 3.8 质量门禁：B 模型跳过 reasoning 占比误杀

```python
# quality.py

class QualityGate:
    @staticmethod
    def validate(task, result):
        # B 模型(turbo) 的 reasoning 占比 70-82% 是正常特性，
        # 不能用 A 模型的废稿规则检测 B 模型。
        if task.model_type == "B":
            return True

        # A 模型才检查 reasoning 占比
        ratio = result.get("reasoning_ratio", 0)
        if ratio > 0.85:
            return False

        return True
```

### 3.9 配置中心：避免配置修改后不生效

```python
# config.py

import signal
from pathlib import Path

import yaml

from state import AtomicStateStore


class ConfigCenter:
    def __init__(self, data_dir):
        self.data_dir = Path(data_dir)
        self.path = self.data_dir / "engine.yaml"
        self._data = {}
        self.reload()

    def reload(self):
        if self.path.exists():
            self._data = yaml.safe_load(self.path.read_text()) or {}
        else:
            self._data = {}

        self.heartbeat_dir = self.data_dir / "heartbeat"
        self.heartbeat_dir.mkdir(parents=True, exist_ok=True)

        self.instance_lock_path = self.data_dir / "instance.lock"
        self.core_heartbeat_path = self.heartbeat_dir / "core.heartbeat"
        self.guardian_heartbeat_path = self.heartbeat_dir / "guardian.heartbeat"

        self.engine_meta = EngineMeta(self.data_dir)

    def get(self, key, default=None):
        node = self._data
        for part in key.split("."):
            if not isinstance(node, dict) or part not in node:
                return default
            node = node[part]
        return node

    def create_store(self):
        return AtomicStateStore(self.data_dir)


class EngineMeta:
    def __init__(self, data_dir):
        self.data_dir = Path(data_dir)
        self.path = self.data_dir / "meta.json"
        self.lock_path = self.data_dir / "meta.lock"

    def load(self):
        if self.path.exists():
            return json.loads(self.path.read_text()).get("generation", 0)
        return 0

    def increment(self):
        with FileLock(self.lock_path):
            gen = self.load() + 1
            tmp = self.path.with_suffix(".tmp")
            tmp.write_text(json.dumps({"generation": gen}))
            os.replace(tmp, self.path)
            return gen
```

入口处支持 `SIGHUP` 热加载配置：

```python
# entry.py

import argparse
import signal

from config import ConfigCenter
from guardian import Guardian
from core import EngineCore


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--role", choices=["guardian", "core"], required=True)
    parser.add_argument("--data-dir", required=True)
    parser.add_argument("--generation", type=int, default=None)
    parser.add_argument("--adopt", action="store_true")
    args = parser.parse_args()

    config = ConfigCenter(args.data_dir)
    signal.signal(signal.SIGHUP, lambda *_: config.reload())

    if args.role == "guardian":
        if args.adopt:
            gen = args.generation
        else:
            gen = config.engine_meta.increment()
        Guardian(config, gen, adopt=args.adopt).run()
    else:
        gen = args.generation or config.engine_meta.load()
        EngineCore(config, gen).run()


if __name__ == "__main__":
    main()
```

## 4. 崩溃恢复时序

### 4.1 Core 崩溃

```text
Core 心跳超时
  -> Guardian 杀掉旧 Core
  -> store.recover() 加载最新 Snapshot + WAL
  -> store.requeue_running() 把 RUNNING 重置为 PENDING
  -> generation + 1
  -> 拉起新 Core
  -> 新 Core 重新调度 PENDING 任务
  -> 有 checkpoint 的任务从 checkpoint 继续
```

### 4.2 Guardian 崩溃

```text
Guardian 心跳超时
  -> Core 杀掉旧 Guardian
  -> Core 拉起新 Guardian --adopt --generation <当前gen>
  -> 新 Guardian 获取 instance.lock
  -> 检测到 Core 心跳健康，直接接管监控
  -> 不重置任务，不影响正在执行的任务
```

### 4.3 双进程同时崩溃

```text
机器重启或手工启动 engine --role guardian
  -> 恢复 Snapshot + WAL
  -> RUNNING -> PENDING
  -> 拉起 Core
  -> 从断点继续
```

## 5. 避坑清单

1. **配置修改后必须生效**  
   所有模块都通过 `config.get()` 读取，不在启动时拷贝配置到局部变量；支持 `SIGHUP` 热加载。

2. **API 超时**  
   B 模型 turbo 用 300 秒，A 模型用 600 秒；API 超时与心跳超时严格分离，不能混用。

3. **B 模型废稿检测误杀**  
   B 模型 `reasoning_ratio` 70-82% 是正常特性，恢复任务时不能按 A 模型规则过滤。

4. **状态锁与接管锁**  
   `state_lock` 保证进程内状态串行；`instance.lock` 保证跨进程只有一个 Guardian；接管任务前必须确认 `instance.lock` 可获取。

5. **幂等性**  
   崩溃恢复是 at-least-once 语义，外部副作用必须用 `task_id + stage` 做幂等去重。

这套方案可以在不引入外部守护进程的情况下，实现引擎自身崩溃自愈、状态原子持久化和断点续跑。
