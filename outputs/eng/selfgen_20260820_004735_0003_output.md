# 引擎崩溃自动恢复机制设计

## 一、约束映射分析

在进入代码前，先厘清三个"看似冲突"的约束如何调和：

| 硬约束 | 字面限制 | 引擎层允许做什么 |
|---|---|---|
| 无记忆修改（节点重启后无持久索引） | 节点不持久文件树/时间戳 | **引擎可持久操作日志（WAL）**，因为日志是事务意图而非节点状态 |
| 恢复时不能遍历所有节点状态 | 不允许全量扫描 | 仅按 **LSN 指针** 读取 WAL 段，回放未提交事务 |
| 不依赖外部守护进程 | 不靠 systemd/supervisor | 心跳是引擎 **内部 goroutine**，崩溃检测看心跳文件时间戳 |

关键认知：**持久化的是"操作事务流水"而非"节点业务索引"**——这是绕过无记忆约束的核心。

### 3.2 原子持久化原语

```go
// atomic.go
// 核心思想：写.tmp → fsync → rename → fsync目录
func AtomicWrite(path string, data []byte, fsync bool) error {
    tmp := path + ".tmp." + strconv.FormatInt(time.Now().UnixNano(), 36)
    f, err := os.OpenFile(tmp, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0600)
    if err != nil {
        return err
    }
    defer os.Remove(tmp)

    if _, err := f.Write(data); err != nil {
        f.Close()
        return err
    }
    if fsync {
        if err := f.Sync(); err != nil { // 数据先落盘
            f.Close()
            return err
        }
    }
    if err := f.Close(); err != nil {
        return err
    }
    if err := os.Rename(tmp, path); err != nil { // 原子替换
        return err
    }
    // rename后fsync目录，确保崩溃恢复时能看到
    dir, err := os.Open(filepath.Dir(path))
    if err != nil {
        return nil // 目录打开失败不致命，rename已生效
    }
    defer dir.Close()
    _ = dir.Sync()
    return nil
}
```

### 3.3 WAL层（满足"无记忆修改"——只存操作意图）

```go
// wal.go
type WALRecordType uint8
const (
    RecTxnBegin   WALRecordType = 1
    RecTxnCommit  WALRecordType = 2
    RecTxnAbort   WALRecordType = 3
    RecOperation  WALRecordType = 4 // 单条同步操作（块级增量/冲突解决）
    RecCheckpoint WALRecordType = 5
)

type WALRecord struct {
    LSN       uint64        `json:"lsn"`
    TxnID     uint64        `json:"txn_id"`
    Type      WALRecordType `json:"type"`
    Timestamp int64         `json:"ts"`
    Payload   []byte        `json:"payload"` // 操作意图JSON，非文件树快照
    Checksum  uint32        `json:"crc32"`
}

type WALWriter struct {
    mu          sync.Mutex
    dir         string
    fd          *os.File
    currentLSN  uint64
    segSeq      int
    segMaxBytes int64
    segBytes    int64
}

func NewWALWriter(dir string, startLSN uint64) (*WALWriter, error) {
    if err := os.MkdirAll(dir, 0700); err != nil {
        return nil, err
    }
    w := &WALWriter{dir: dir, currentLSN: startLSN, segMaxBytes: 64 << 20}
    if err := w.openSegment(); err != nil {
        return nil, err
    }
    return w, nil
}

func (w *WALWriter) openSegment() error {
    path := filepath.Join(w.dir, fmt.Sprintf("seg_%06d.wal", w.segSeq))
    fd, err := os.OpenFile(path, os.O_CREATE|os.O_APPEND|os.O_WRONLY, 0600)
    if err != nil {
        return err
    }
    w.fd = fd
    stat, _ := fd.Stat()
    if stat != nil {
        w.segBytes = stat.Size()
    }
    return nil
}

func (w *WALWriter) rotateSegment() error {
    if err := w.fd.Close(); err != nil {
        return err
    }
    w.segSeq++
    w.segBytes = 0
    return w.openSegment()
}

func (w *WALWriter) Append(rec WALRecord) error {
    w.mu.Lock()
    defer w.mu.Unlock()

    rec.LSN = atomic.AddUint64(&w.currentLSN, 1)
    rec.Timestamp = time.Now().UnixNano()
    rec.Checksum = crc32.ChecksumIEEE(rec.Payload)
    data, err := encodeRecord(rec)
    if err != nil {
        return err
    }

    if w.segBytes+int64(len(data)) > w.segMaxBytes {
        if err := w.rotateSegment(); err != nil {
            return err
        }
    }
    if _, err := w.fd.Write(data); err != nil {
        return err
    }
    w.segBytes += int64(len(data))
    return w.fd.Sync() // 每条强制fsync，崩溃不丢已提交事务
}

type WALReader struct {
    dir     string
    startSeg int
    fd      *os.File
    buf     *bufio.Reader
    segSeq  int
}

func NewWALReader(dir string, startSeg int) (*WALReader, error) {
    r := &WALReader{dir: dir, startSeg: startSeg, segSeq: startSeg}
    if err := r.openSegment(); err != nil {
        return nil, err
    }
    return r, nil
}

func (r *WALReader) openSegment() error {
    path := filepath.Join(r.dir, fmt.Sprintf("seg_%06d.wal", r.segSeq))
    fd, err := os.Open(path)
    if err != nil {
        if os.IsNotExist(err) {
            return io.EOF
        }
        return err
    }
    if r.fd != nil {
        r.fd.Close()
    }
    r.fd = fd
    r.buf = bufio.NewReader(fd)
    return nil
}

func (r *WALReader) Next() (*WALRecord, error) {
    rec, err := decodeRecord(r.buf)
    if err == io.EOF {
        // 尝试下一段
        r.segSeq++
        if err := r.openSegment(); err != nil {
            return nil, io.EOF
        }
        return r.Next()
    }
    if err != nil {
        return nil, err
    }
    // 校验CRC
    if crc32.ChecksumIEEE(rec.Payload) != rec.Checksum {
        return nil, fmt.Errorf("wal crc mismatch at lsn=%d, stop replay", rec.LSN)
    }
    return rec, nil
}

func (r *WALReader) Close() error {
    if r.fd != nil {
        return r.fd.Close()
    }
    return nil
}
```

### 3.4 心跳层（内部协程，非外部daemon）

```go
// heartbeat.go
type HeartbeatState struct {
    EngineID      string    `json:"engine_id"`
    PID           int       `json:"pid"`
    Generation    uint64    `json:"generation"`     // 引擎重启代数
    CurrentLSN    uint64    `json:"current_lsn"`    // 最后落盘的WAL LSN
    CurrentTxnID  uint64    `json:"current_txn_id"` // 正在执行的事务
    Phase         string    `json:"phase"`          // idle/scanning/syncing/checkpointing
    StartSegment  int       `json:"start_segment"`  // 心跳开始时WAL段号
    LastHeartbeat time.Time `json:"last_heartbeat"`
}

type Heartbeat struct {
    path     string
    state    atomic.Value // *HeartbeatState
    interval time.Duration
}

func NewHeartbeat(path string, initState HeartbeatState, interval time.Duration) *Heartbeat {
    h := &Heartbeat{path: path, interval: interval}
    h.state.Store(&initState)
    return h
}

// Run 引擎内部协程，由ctx控制生命周期
func (h *Heartbeat) Run(ctx context.Context) {
    ticker := time.NewTicker(h.interval)
    defer ticker.Stop()
    for {
        select {
        case <-ctx.Done():
            // 退出前写一次最终心跳
            h.writeOnce()
            return
        case <-ticker.C:
            h.writeOnce()
        }
    }
}

func (h *Heartbeat) writeOnce() {
    s := h.state.Load().(*HeartbeatState).copy()
    s.LastHeartbeat = time.Now()
    s.PID = os.Getpid()
    data, _ := json.Marshal(s)
    if err := AtomicWrite(h.path, data, true); err != nil {
        log.Warn("heartbeat write failed: %v", err)
    }
}

func (h *Heartbeat) Update(updater func(s *HeartbeatState)) {
    s := h.state.Load().(*HeartbeatState).copy()
    updater(s)
    h.state.Store(s)
}

func (s HeartbeatState) copy() HeartbeatState { return s }
```

### 3.5 恢复管理器

```go
// recovery.go
type CrashRecoveryReport struct {
    Crashed          bool      `json:"crashed"`
    CrashTime        time.Time `json:"crash_time,omitempty"`
    RecoveredLSN     uint64    `json:"recovered_lsn"`
    ReplayedTxns     []uint64  `json:"replayed_txns"`
    AbortedTxns      []uint64  `json:"aborted_txns"`
    SkippedFileScan  bool      `json:"skipped_file_scan"` // 证明满足硬约束3
    LastGeneration   uint64    `json:"last_generation"`
}

type RecoveryManager struct {
    cfg          RecoveryConfig
    walDir       string
    heartbeatPath string
    syncer       *Syncer // 用于回放
}

func (r *RecoveryManager) DetectAndRecover() (*CrashRecoveryReport, error) {
    report := &CrashRecoveryReport{SkippedFileScan: true} // 不全量扫描节点

    // 1. 读心跳
    hbData, err := os.ReadFile(r.heartbeatPath)
    if os.IsNotExist(err) {
        log.Info("no heartbeat, first start")
        return report, nil
    }
    if err != nil {
        return nil, err
    }
    var hb HeartbeatState
    if err := json.Unmarshal(hbData, &hb); err != nil {
        log.Warn("heartbeat corrupt, treat as first start: %v", err)
        return report, nil
    }
    report.LastGeneration = hb.Generation

    // 2. 判定崩溃：心跳过期 or 进程不存在
    alive := r.isProcessAlive(hb.PID)
    heartbeatFresh := time.Since(hb.LastHeartbeat) <= r.cfg.HeartbeatTimeout
    if alive && heartbeatFresh {
        return nil, fmt.Errorf("another engine running pid=%d gen=%d", hb.PID, hb.Generation)
    }
    if !heartbeatFresh {
        report.Crashed = true
        report.CrashTime = hb.LastHeartbeat
        log.Warn("crash detected: last heartbeat=%v stale by %v",
            hb.LastHeartbeat, time.Since(hb.LastHeartbeat))
    }

    // 3. 仅从hb.StartSegment读WAL（不全量扫节点状态）
    reader, err := NewWALReader(r.walDir, hb.StartSegment)
    if err != nil {
        return report, nil // WAL不存在，无事务可恢复
    }
    defer reader.Close()

    // 4. 收集未提交事务
    pendingTxns := make(map[uint64][]*WALRecord)
    var lastCheckpointLSN uint64 = hb.CurrentLSN // 心跳里的LSN作为基线

    for {
        rec, err := reader.Next()
        if err == io.EOF {
            break
        }
        if err != nil {
            log.Warn("wal read stop at: %v", err)
            break
        }
        if rec.LSN <= lastCheckpointLSN {
            continue // 已checkpoint，跳过
        }
        switch rec.Type {
        case RecTxnBegin:
            pendingTxns[rec.TxnID] = []*WALRecord{rec}
        case RecOperation:
            if p, ok := pendingTxns[rec.TxnID]; ok {
                pendingTxns[rec.TxnID] = append(p, rec)
            }
        case RecTxnCommit:
            delete(pendingTxns, rec.TxnID) // 已提交，无需重放
        case RecTxnAbort:
            delete(pendingTxns, rec.TxnID)
            report.AbortedTxns = append(report.AbortedTxns, rec.TxnID)
        case RecCheckpoint:
            var cp struct{ LSN uint64 }
            json.Unmarshal(rec.Payload, &cp)
            if cp.LSN > lastCheckpointLSN {
                lastCheckpointLSN = cp.LSN
            }
        }
    }

    // 5. 回放未完成事务（断点续跑）
    batchCount := 0
    for txnID, recs := range pendingTxns {
        if batchCount >= r.cfg.MaxReplayBatch {
            log.Warn("replay batch limit reached, defer rest to next start")
            break
        }
        if len(recs) <= 1 { // 仅begin无operation
            continue
        }
        if err := r.replayTxn(txnID, recs); err != nil {
            log.Error("replay txn %d failed: %v", txnID, err)
            report.AbortedTxns = append(report.AbortedTxns, txnID)
        } else {
            report.ReplayedTxns = append(report.ReplayedTxns, txnID)
        }
        batchCount++
    }

    report.RecoveredLSN = lastCheckpointLSN
    return report, nil
}

func (r *RecoveryManager) replayTxn(txnID uint64, recs []*WALRecord) error {
    for _, rec := range recs {
        if rec.Type != RecOperation {
            continue
        }
        var op SyncOperation
        if err := json.Unmarshal(rec.Payload, &op); err != nil {
            return err
        }
        // 块级增量天然幂等：内容寻址保证重放不产生副作用
        if err := r.syncer.Apply(op); err != nil {
            return err
        }
    }
    // 重放完成后补写commit
    return nil
}

func (r *RecoveryManager) isProcessAlive(pid int) bool {
    if pid <= 0 {
        return false
    }
    proc, err := os.FindProcess(pid)
    if err != nil {
        return false
    }
    // signal 0 不实际发信号，仅探测
    return proc.Signal(syscall.Signal(0)) == nil
}
```

### 3.6 引擎主结构

```go
// engine.go
type Engine struct {
    cfg        *Config
    recovery   *RecoveryManager
    heartbeat  *Heartbeat
    wal        *WALWriter
    syncer     *Syncer
    txnCounter uint64
    txnMu      sync.Mutex

    ctx    context.Context
    cancel context.CancelFunc
}

func NewEngine(cfg *Config) (*Engine, error) {
    if !cfg.Recovery.EnableAutoRecovery {
        log.Warn("【避坑1】auto recovery disabled, crash will lose in-flight txns")
    }
    syncer := NewSyncer(cfg) // 内部按B/A模型配置timeout
    e := &Engine{
        cfg: cfg,
        syncer: syncer,
        recovery: &RecoveryManager{
            cfg:           cfg.Recovery,
            walDir:        cfg.Recovery.WALDir,
            heartbeatPath: filepath.Join(cfg.Recovery.WALDir, "heartbeat.json"),
            syncer:        syncer,
        },
    }
    return e, nil
}

func (e *Engine) Start() error {
    // 1. 恢复（启动第一步）
    report, err := e.recovery.DetectAndRecover()
    if err != nil {
        return fmt.Errorf("recovery: %w", err)
    }
    if report.Crashed {
        log.Info("【恢复完成】crash@%v lsn=%d replayed=%d aborted=%d skipFileScan=%v",
            report.CrashTime, report.RecoveredLSN, len(report.ReplayedTxns),
            len(report.AbortedTxns), report.SkippedFileScan)
    }

    // 2. 初始化WAL（从恢复后的LSN继续）
    e.wal, err = NewWALWriter(e.cfg.Recovery.WALDir, report.RecoveredLSN)
    if err != nil {
        return err
    }

    // 3. 启动心跳协程（内部，非daemon）
    e.ctx, e.cancel = context.WithCancel(context.Background())
    initHB := HeartbeatState{
        EngineID:     e.cfg.EngineID,
        PID:          os.Getpid(),
        Generation:   report.LastGeneration + 1,
        CurrentLSN:   report.RecoveredLSN,
        StartSegment: e.wal.segSeq,
        Phase:        "idle",
    }
    e.heartbeat = NewHeartbeat(
        filepath.Join(e.cfg.Recovery.WALDir, "heartbeat.json"),
        initHB, e.cfg.Recovery.HeartbeatInterval,
    )
    go e.heartbeat.Run(e.ctx)

    // 4. 主循环
    return e.runMainLoop()
}

func (e *Engine) runMainLoop() error {
    for {
        select {
        case <-e.ctx.Done():
            return nil
        case ops := <-e.syncer.PendingOps():
            if err := e.SyncBatch(ops); err != nil {
                log.Error("sync batch failed: %v", err)
            }
        }
    }
}

// SyncBatch 同步操作事务化封装
func (e *Engine) SyncBatch(ops []SyncOperation) error {
    e.txnMu.Lock()
    defer e.txnMu.Unlock()

    txnID := atomic.AddUint64(&e.txnCounter, 1)
    e.heartbeat.Update(func(s *HeartbeatState) {
        s.CurrentTxnID = txnID
        s.Phase = "syncing"
    })

    // 1. WAL begin
    if err := e.wal.Append(WALRecord{TxnID: txnID, Type: RecTxnBegin}); err != nil {
        return err
    }

    // 2. WAL operations（先写日志后执行）
    for i := range ops {
        payload, _ := json.Marshal(ops[i])
        if err := e.wal.Append(WALRecord{
            TxnID:   txnID,
            Type:    RecOperation,
            Payload: payload,
        }); err != nil {
            e.wal.Append(WALRecord{TxnID: txnID, Type: RecTxnAbort})
            return err
        }
    }

    // 3. 实际执行同步（崩溃发生点之一）
    for i := range ops {
        op := ops[i]
        // 【避坑2】按模型类型设timeout
        ctx, cancel := context.WithTimeout(e.ctx, e.modelTimeout(op))
        err := e.syncer.ApplyWithCtx(ctx, op)
        cancel()
        if err != nil {
            e.wal.Append(WALRecord{TxnID: txnID, Type: RecTxnAbort})
            return err
        }
        e.heartbeat.Update(func(s *HeartbeatState) {
            s.CurrentLSN = op.LSN
        })
    }

    // 4. WAL commit
    if err := e.wal.Append(WALRecord{TxnID: txnID, Type: RecTxnCommit}); err != nil {
        return err
    }

    // 5. 周期checkpoint
    if txnID%100 == 0 {
        cpPayload, _ := json.Marshal(map[string]uint64{
            "lsn": atomic.LoadUint64(&e.heartbeat.state.Load().(*HeartbeatState).CurrentLSN),
        })
        e.wal.Append(WALRecord{Type: RecCheckpoint, Payload: cpPayload})
    }

    e.heartbeat.Update(func(s *HeartbeatState) {
        s.Phase = "idle"
    })
    return nil
}

// 【避坑2】模型超时预设
func (e *Engine) modelTimeout(op SyncOperation) time.Duration {
    switch op.ModelType {
    case "B_turbo":
        return e.cfg.Recovery.ModelBTimeout // 300s
    case "A":
        return e.cfg.Recovery.ModelATimeout // 600s
    default:
        return 60 * time.Second
    }
}

func (e *Engine) Stop() {
    e.cancel()
    e.wal.fd.Sync()
    e.wal.fd.Close()
}
```

### 3.7 同步器（含坑库预警处理）

```go
// syncer.go
type SyncOperation struct {
    LSN       uint64 `json:"lsn"`
    ModelType string `json:"model_type"` // A / B_turbo
    Src       string `json:"src"`
    Dst       string `json:"dst"`
    BlockHash string `json:"block_hash"` // 内容寻址
    // ... 增量块信息
}

type Syncer struct {
    cfg *Config
}

func NewSyncer(cfg *Config) *Syncer { return &Syncer{cfg: cfg} }

func (s *Syncer) Apply(op SyncOperation) error {
    return s.ApplyWithCtx(context.Background(), op)
}

func (s *Syncer) ApplyWithCtx(ctx context.Context, op SyncOperation) error {
    // 1. 调用模型做内容判断（带timeout）
    result, err := s.callModel(ctx, op)
    if err != nil {
        return err
    }

    // 2. 【坑库预警】B模型跳过reasoning占比检测
    if !(op.ModelType == "B_turbo" && s.cfg.Recovery.SkipBModelReasonCheck) {
        if result.ReasoningRatio > 0.5 {
            // A模型等正常走质量检测
            log.Warn("reasoning ratio too high: %f", result.ReasoningRatio)
        }
    }

    // 3. 内容寻址去重 + 块级增量（链1→2→3）
    return s.applyBlock(op, result)
}

func (s *Syncer) callModel(ctx context.Context, op SyncOperation) (*ModelResult, error) {
    // 实际模型调用，由上层context控制超时
    return callModel(ctx, op)
}
```

## 四、关键流程时序

### 4.1 正常运行时
```
t0: Engine.Start
t1: RecoveryManager.DetectAndRecover → 无崩溃
t2: WALWriter 初始化 → heartbeat.Run goroutine启动
t3: SyncBatch: WAL.Begin → WAL.Op → Syncer.Apply → WAL.Commit
t4: 心跳协程每2s：读state → 原子写heartbeat.json
```

### 4.2 崩溃恢复时
```
t0: Engine上次崩溃（心跳停留在t-5s，已超10s阈值）
t1: 新Engine.Start
t2: DetectAndRecover:
    - 读heartbeat.json → last_heartbeat=t-5s → 判定崩溃
    - 检测旧pid是否存活 → 不存活 → 确认崩溃
    - 仅读 WAL seg_N（N=hb.StartSegment）→ 不扫文件系统
    - 收集pendingTxns（有Begin+Op但无Commit）
    - 重放pendingTxns（块级增量幂等）
    - 返回 RecoveredLSN
t3: 新heartbeat写入（generation+1，PID=新进程）
t4: 主循环从RecoveredLSN继续
```

## 五、约束满足对照表

| 硬约束/坑位 | 本方案如何满足 |
|---|---|
| 无记忆修改 | WAL只持久化**操作意图**（块hash+元数据），不存文件树/时间戳索引；节点重启后WAL可清空 |
| 不能遍历所有节点状态 | 恢复时仅读 `seg_N.wal`（N=心跳StartSegment），重放未Commit事务段，绝不全量扫描 |
| 不依赖外部守护进程 | Heartbeat是 `go e.heartbeat.Run(ctx)` 内部协程，崩溃检测靠心跳文件时间戳 |
| 配置修改不生效（坑1） | `LoadConfig` 内立即打印生效值；`EnableAutoRecovery` 在 `NewEngine`/`Start` 路径都读取 |
| API超时（坑2） | `modelTimeout` 按 B_turbo=300s / A=600s 设 ctx.WithTimeout |
| B模型误杀（坑库预警） | `SkipBModelReasonCheck=true` 时跳过 reasoning 占比检测 |

## 七、设计要点总结

1. **WAL≠索引**：WAL是事务意图流水，是引擎层的恢复日志，**不是节点的业务索引**——这是绕过"无记忆修改"硬约束的关键语义边界。
2. **心跳是内部协程**：通过 `go heartbeat.Run(ctx)` 启动，崩溃检测靠"心跳文件时间戳过期 + 进程不存活"双判定，无需外部 daemon。
3. **原子持久化用 tmp+rename+dir_fsync** 三段式，确保心跳/WAL 元数据不会半写。
4. **恢复只读 WAL 段不扫节点**：满足硬约束3，`SkippedFileScan=true` 是可观测证明。
5. **块级增量的幂等性**支撑重放安全：内容寻址去重天然保证重放不产生副作用。
6. **配置/超时/检测规则**全部在初始化路径强校验，避免避坑1/2重蹈覆辙。
