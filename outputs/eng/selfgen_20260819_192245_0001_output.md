# 任务池自进化系统：从硬编码到动态生成

## 一、问题诊断

```
现状: SYSTEM_TASKS = [41个硬编码模板]
     pick_system_task → random.choice(SYSTEM_TASKS)
     
1451次执行 → 平均每模板重复35次 → 信息熵趋零 → 局部最优锁死
```

**根因链**：输入端固化 → 产出同质化 → 认知池污染 → 反哺上下文重复 → 进一步固化。CAS系统的「消耗→产出→反哺」循环在输入端就断了。

```python
    desc = gap["description"]

# ============================================================
def _generate_new_templates(count: int = GEN_BATCH_SIZE) -> List[Dict]:
    """
    自动生成新任务模板
    分配比例: 40%认知缺口 + 30%坑库 + 30%弱维度
    不足部分用自由探索任务补齐
    """
    gaps = _collect_cognitive_gaps()
    pitfalls = _collect_pitfall_signals()
    weak_dims = _collect_weak_dimensions()
    
    gap_n = min(len(gaps), max(1, count * 2 // 5))
    pit_n = min(len(pitfalls), max(1, count * 3 // 10))
    dim_n = min(len(weak_dims), count - gap_n - pit_n)
    
    new_templates = []
    
    for gap in gaps[:gap_n]:
        new_templates.append(_gen_from_cognitive_gap(gap))
    
    for pit in pitfalls[:pit_n]:
        new_templates.append(_gen_from_pitfall(pit))
    
    for weak in weak_dims[:dim_n]:
        new_templates.append(_gen_from_weak_dim(weak))
    
    # 补齐: 自由探索任务 (保证多样性)
    while len(new_templates) < count:
        prompt = f"""【自由探索任务】基于当前系统状态，选择一个你认为最需要关注的领域进行深度推演。
任务要求：
1. 识别当前系统最紧迫的问题（按优先级排序，至少3个）
2. 分析根因（至少3层深度）
3. 给出可执行的解决方案
4. 产出结构化认知卡
输出格式：[问题识别] [根因分析] [解决方案] [认知产出]"""
        new_templates.append({
            "id": _gen_template_id(prompt),
            "prompt": prompt,
            "category": "free_exploration",
            "source": "auto_generated",
            "target_domain": "auto_detected",
            "created_at": time.time(),
            "use_count": 0,
            "max_uses": TEMPLATE_MAX_USES
        })
    
    # 统计 (对接坑库预警#2: bonus_count仅系统自触发)
    for t in new_templates:
        _task_pool_stats["auto_gen_count"] += 1
        _task_pool_stats["by_source"][t["source"]] += 1
        _task_pool_stats["by_category"][t["category"]] += 1
    
    print(f"[TaskPool] 自动生成{len(new_templates)}个模板 "
          f"(缺口{gap_n}/坑库{pit_n}/弱维度{dim_n}/自由{len(new_templates)-gap_n-pit_n-dim_n})")
    
    return new_templates[:count]
```

### 3.5 动态淘汰机制

```python
def _prune_exhausted_templates() -> int:
    """
    移除已耗尽的模板 (use_count >= max_uses)
    返回移除数量
    """
    global SYSTEM_TASKS
    
    before = len(SYSTEM_TASKS)
    exhausted = [t for t in SYSTEM_TASKS if t["use_count"] >= t["max_uses"]]
    SYSTEM_TASKS = [t for t in SYSTEM_TASKS if t["use_count"] < t["max_uses"]]
    pruned = before - len(SYSTEM_TASKS)
    
    _task_pool_stats["prune_count"] += pruned
    
    if pruned > 0:
        # 记录淘汰详情
        for t in exhausted:
            print(f"[TaskPool] 模板耗尽淘汰: id={t['id'][:8]} "
                  f"category={t['category']} uses={t['use_count']}/{t['max_uses']}")
        print(f"[TaskPool] 本次淘汰{pruned}个, 剩余{len(SYSTEM_TASKS)}个活跃模板")
    
    return pruned


def _replenish_if_needed():
    """活跃模板不足时自动补充"""
    active_count = len(SYSTEM_TASKS)
    
    if active_count >= MIN_ACTIVE_TEMPLATES:
        return
    
    needed = MIN_ACTIVE_TEMPLATES - active_count
    gen_count = min(needed + GEN_BATCH_SIZE, MAX_POOL_SIZE - len(SYSTEM_TASKS))
    
    if gen_count <= 0:
        print(f"[TaskPool] 任务池已满({len(SYSTEM_TASKS)}/{MAX_POOL_SIZE})，跳过补充")
        return
    
    new_templates = _generate_new_templates(gen_count)
    
    existing_ids = {t["id"] for t in SYSTEM_TASKS}
    added = 0
    for t in new_templates:
        if t["id"] not in existing_ids:
            SYSTEM_TASKS.append(t)
            existing_ids.add(t["id"])
            added += 1
    
    print(f"[TaskPool] 补充{added}个模板, 池大小: {len(SYSTEM_TASKS)}")
```

### 3.6 核心函数：替换 `pick_system_task`

```python
# ============================================================
# 核心函数: pick_system_task (替换原始 random.choice 版本)
# ============================================================

def pick_system_task() -> Optional[Dict[str, Any]]:
    """
    从任务池选取系统任务
    
    流程:
    1. 清理耗尽模板 (use_count >= 5)
    2. 补充活跃模板 (低于15个时自动生成)
    3. 加权随机选择 (偏好使用次数少的模板→提升多样性)
    4. 递增使用计数
    5. 每10次调用刷新认知缓存 (对接坑库预警#1)
    
    返回: Dict {id, prompt, category, source, use_count, max_uses, ...}
          失败返回None
    """
    with _task_pool_lock:
        # Step 1: 淘汰耗尽模板
        _prune_exhausted_templates()
        
        # Step 2: 补充活跃模板
        _replenish_if_needed()
        
        # Step 3: 确保池非空
        if not SYSTEM_TASKS:
            print("[TaskPool] 池空! 紧急生成...")
            emergency = _generate_new_templates(GEN_BATCH_SIZE)
            SYSTEM_TASKS.extend(emergency)
        
        if not SYSTEM_TASKS:
            print("[TaskPool] CRITICAL: 无法生成任何模板!")
            return None
        
        # Step 4: 加权随机选择
        # 权重 = 1/(use_count+1), 使用次数越少权重越高 → 多样性保障
        weights = [1.0 / (t["use_count"] + 1) for t in SYSTEM_TASKS]
        total = sum(weights)
        if total == 0:
            selected = random.choice(SYSTEM_TASKS)
        else:
            weights = [w / total for w in weights]
            r = random.random()
            cumulative = 0.0
            selected = SYSTEM_TASKS[0]
            for i, w in enumerate(weights):
                cumulative += w
                if r <= cumulative:
                    selected = SYSTEM_TASKS[i]
                    break
        
        # Step 5: 递增使用计数
        selected["use_count"] += 1
        _task_pool_stats["pick_count"] += 1  # 对接坑库预警#2: 真实task_count
        
        # Step 6: 认知缓存刷新 (每10个任务, 对接坑库预警#1)
        if _task_pool_stats["pick_count"] % REFRESH_INTERVAL == 0:
            _refresh_cognitive_cache()
        
        # 返回任务 (调用方取 selected["prompt"] 作为实际prompt)
        return selected.copy()  # 返回副本防止外部修改


def _refresh_cognitive_cache():
    """刷新认知缓存, 确保新认知参与注入 (对接坑库预警#1)"""
    try:
        from cognitive_pool import refresh_cache
        refresh_cache()
        print(f"[TaskPool] 认知缓存已刷新 (第{_task_pool_stats['pick_count']}次选取)")
    except (ImportError, AttributeError, Exception) as e:
        print(f"[TaskPool] 认知缓存刷新跳过: {e}")
```

```python
def restore_task_pool_snapshot(snapshot: Dict):

    global SYSTEM_TASKS, _task_pool_stats
    
    with _task_pool_lock:
        tasks = snapshot.get("system_tasks", [])

        _task_pool_stats["auto_gen_count"] = stats.get("auto_gen_count", 0)
        _task_pool_stats["prune_count"] = stats.get("prune_count", 0)
        
        print(f"[TaskPool] 从快照恢复: {len(SYSTEM_TASKS)}个模板, "
              f"已执行{stats.get('pick_count', 0)}次, "
              f"已生成{stats.get('auto_gen_count', 0)}个, "
              f"已淘汰{stats.get('prune_count', 0)}个")


def get_task_pool_status() -> Dict:
    """获取任务池状态 (用于监控面板)"""
    with _task_pool_lock:
        active = [t for t in SYSTEM_TASKS if t["use_count"] < t["max_uses"]]
        exhausted = [t for t in SYSTEM_TASKS if t["use_count"] >= t["max_uses"]]
        
        return {
            "pool_size": len(SYSTEM_TASKS),
            "active_count": len(active),
            "exhausted_count": len(exhausted),
            "seed_remaining": len([t for t in active if t["source"] == "hardcoded"]),
            "auto_gen_active": len([t for t in active if t["source"] == "auto_generated"]),
            "pick_count": _task_pool_stats["pick_count"],
            "auto_gen_count": _task_pool_stats["auto_gen_count"],
            "prune_count": _task_pool_stats["prune_count"],
            "avg_use_count": round(
                sum(t["use_count"] for t in SYSTEM_TASKS) / max(len(SYSTEM_TASKS), 1), 2
            ),
            "by_category": dict(_task_pool_stats["by_category"]),
            "diversity_ratio": round(
                len([t for t in active if t["source"] == "auto_generated"]) / max(len(active), 1), 4
            )
        }
```

## 七、CAS视角下的闭环验证

```
输入端自进化闭环:

  认知缺口 ──→ 生成缺口填补任务 ──→ 引擎执行 ──→ 新认知入池
     ↑                                              │
     │                                              │
     └──── 认知池刷新(每10任务) ←─────── 新认知参与注入
                                                    │
  坑库发现 ──→ 生成修复任务 ──→ 引擎执行 ──→ 坑被修复 ──→ 新坑入库
     ↑                                              │
     └──────────────────────────────────────────────┘
     
  九维弱维度 ──→ 生成补强任务 ──→ 引擎执行 ──→ 维度提升 ──→ 新弱点暴露
     ↑                                              │
     └──────────────────────────────────────────────┘
```

**关键断点修复**：原系统的自进化在输入端就断了（41个固定模板无限重复）。现在通过「信号采集→模板生成→耗尽淘汰→认知刷新」四步闭环，使得每次引擎执行的输入都是基于系统当前状态的动态产物，从根源上打破局部最优。
