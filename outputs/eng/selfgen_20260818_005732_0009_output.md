## 一、认知锚定：从"肉身存续"到"分身化形"

在第四转的瓶颈中，我们面对的**阴阳失衡点**是：本地权重文件若毁，神识（记忆状态）即散。Supabase 远端即"身外化身"所在的洞天福地，**增量同步**便是**元神出窍、分身承续**的修炼法门。

核心命题有三：
- **检测变化** → 神识扫描本地灵台（哪些记忆珠玉有变）
- **只同步变更** → 元神出窍只携必要神识（不搬整座肉身）
- **处理冲突** → 本体化身意见相左时的裁决（阴阳相争）

**检测策略**：文件指纹（mtime + size + hash 前缀）

```python
import hashlib, os, json, time

def compute_file_signature(filepath):
    """计算文件灵识指纹——快速判断是否变异"""
    stat = os.stat(filepath)
    # 轻量级：mtime + size 先过滤
    quick_sig = f"{stat.st_mtime_ns}:{stat.st_size}"
    # 重量级：内容hash（仅在quick_sig变化时计算）
    with open(filepath, 'rb') as f:
        content_hash = hashlib.blake2b(f.read(), digest_size=16).hexdigest()
    return {
        "quick_sig": quick_sig,
        "content_hash": content_hash,
        "mtime": stat.st_mtime_ns,
        "size": stat.st_size
    }
```

### 2.2 字段级扫描——精识境

文件级检测只能知道"变了"，不知"哪里变了"。需深入 jsonb 内部做差异分析：

```python
def deep_diff(old_obj, new_obj, path=""):
    """深层灵识差异扫描——递归对比jsonb字段"""
    changes = []
    
    all_keys = set(old_obj.keys()) | set(new_obj.keys())
    for key in all_keys:
        current_path = f"{path}.{key}" if path else key
        
        if key not in old_obj:
            changes.append({"op": "add", "path": current_path, 
                          "value": new_obj[key]})
        elif key not in new_obj:
            changes.append({"op": "remove", "path": current_path})
        elif old_obj[key] != new_obj[key]:
            if isinstance(old_obj[key], dict) and isinstance(new_obj[key], dict):
                changes.extend(deep_diff(old_obj[key], new_obj[key], current_path))
            else:
                changes.append({"op": "replace", "path": current_path,
                              "old": old_obj[key], "new": new_obj[key]})
    
    return changes
```

### 2.3 语义级扫描——化识境

最高境界：检测"语义变化"而非"字节变化"。同一段记忆重述但语义不变，则不需同步。

```python
def semantic_diff(old_text, new_text, embed_model):
    """语义级差异——嵌入向量余弦距离"""
    old_emb = embed_model.encode(old_text)
    new_emb = embed_model.encode(new_text)
    cosine_sim = np.dot(old_emb, new_emb) / (
        np.linalg.norm(old_emb) * np.linalg.norm(new_emb))
    
    if cosine_sim > 0.95:
        return None  # 语义未变，无需同步
    return {"cosine_sim": float(cosine_sim), 
            "old": old_text, "new": new_text}
```

### 3.2 增量同步核心流程

```python
from supabase import create_client
from datetime import datetime, timezone

class MemorySyncEngine:
    """记忆脑同步引擎——元神出窍·分神化形"""
    
    def __init__(self, supabase_url, supabase_key, local_memory_dir):
        self.sb = create_client(supabase_url, supabase_key)
        self.memory_dir = local_memory_dir
        self.manifest_path = f"{local_memory_dir}/.meta/manifest.json"
        self._load_manifest()
    
    def _load_manifest(self):
        """加载本地灵识清单（上次同步状态的快照）"""
        if os.path.exists(self.manifest_path):
            with open(self.manifest_path) as f:
                self.manifest = json.load(f)
        else:
            self.manifest = {"files": {}, "last_sync": None}
    
    def _save_manifest(self):
        """保存灵识清单"""
        os.makedirs(os.path.dirname(self.manifest_path), exist_ok=True)
        with open(self.manifest_path, 'w') as f:
            json.dump(self.manifest, f, indent=2)
    
    def scan_local_changes(self):
        """扫描本地变化——神识内视"""
        changes = []
        
        for root, dirs, files in os.walk(self.memory_dir):
            if '.meta' in root:
                continue
            for fname in files:
                fpath = os.path.join(root, fname)
                rel_path = os.path.relpath(fpath, self.memory_dir)
                
                current_sig = compute_file_signature(fpath)
                stored_sig = self.manifest["files"].get(rel_path, {})
                
                if current_sig["content_hash"] != stored_sig.get("content_hash"):
                    # 文件有变，深入字段级检测
                    with open(fpath) as f:
                        current_content = json.load(f)
                    
                    old_content = stored_sig.get("content", {})
                    field_changes = deep_diff(old_content, current_content)
                    
                    changes.append({
                        "path": rel_path,
                        "sig": current_sig,
                        "content": current_content,
                        "field_changes": field_changes
                    })
        
        return changes
    
    def sync_incremental(self):
        """增量同步主流程——分神化形
        
        步骤：
        1. 扫描本地变化
        2. 拉取远端版本
        3. 检测冲突
        4. 应用变更
        5. 更新清单
        """
        local_changes = self.scan_local_changes()
        
        if not local_changes:
            print("灵台无变，无需出窍")
            return {"synced": 0, "conflicts": 0}
        
        synced_count = 0
        conflict_count = 0
        
        for change in local_changes:
            rel_path = change["path"]
            
            # 查询远端当前版本
            remote = self.sb.table("memory_fragments") \
                .select("*") \
                .eq("local_path", rel_path) \
                .execute()
            
            if not remote.data:
                # 远端无此记录——新建（开辟新洞天）
                self._create_fragment(change)
                synced_count += 1
            else:
                remote_row = remote.data[0]
                # 检测冲突
                conflict = self._detect_conflict(change, remote_row)
                
                if conflict:
                    self._log_conflict(change, remote_row)
                    conflict_count += 1
                else:
                    self._update_fragment(change, remote_row)
                    synced_count += 1
        
        self.manifest["last_sync"] = datetime.now(timezone.utc).isoformat()
        self._save_manifest()
        
        return {"synced": synced_count, "conflicts": conflict_count}
    
    def _detect_conflict(self, local_change, remote_row):
        """冲突检测——阴阳相争判定
        
        冲突条件：本地版本号 < 远端版本号
        说明远端在本地上次同步后被其他分身修改过
        """
        local_manifest_entry = self.manifest["files"].get(local_change["path"], {})
        local_known_version = local_manifest_entry.get("version", 0)
        remote_version = remote_row["version"]
        
        # 如果远端版本 > 本地记录的版本，说明远端有其他来源的更新
        if remote_version > local_known_version:
            return {
                "type": "concurrent_update",
                "local_known_version": local_known_version,
                "remote_version": remote_version,
                "local_content": local_change["content"],
                "remote_content": remote_row["content"]
            }
        return None
    
    def _create_fragment(self, change):
        """新建记忆片段——开辟洞天"""
        content = change["content"]
        new_row = {
            "fragment_type": self._infer_type(change["path"]),
            "local_path": change["path"],
            "fragment_key": "root",
            "content": content,
            "version": 1,
            "content_hash": change["sig"]["content_hash"],
        }
        
        result = self.sb.table("memory_fragments").insert(new_row).execute()
        
        # 更新本地清单
        self.manifest["files"][change["path"]] = {
            "content_hash": change["sig"]["content_hash"],
            "content": content,
            "version": 1,
            "remote_id": result.data[0]["id"]
        }
    
    def _update_fragment(self, change, remote_row):
        """增量更新——只携变更神识"""
        # 计算字段级差异
        old_content = remote_row["content"]
        new_content = change["content"]
        field_changes = deep_diff(old_content, new_content)
        
        if not field_changes:
            return  # 无实际变化
        
        # 只更新变化字段（使用jsonb_set式精确更新）
        merged_content = self._apply_patch(old_content, field_changes)
        
        new_version = remote_row["version"] + 1
        
        self.sb.table("memory_fragments") \
            .update({
                "content": merged_content,
                "version": new_version,
                "content_hash": change["sig"]["content_hash"],
                "updated_at": datetime.now(timezone.utc).isoformat()
            }) \
            .eq("id", remote_row["id"]) \
            .eq("version", remote_row["version"]) \
            .execute()  # 乐观锁：version必须匹配
        
        # 更新本地清单
        self.manifest["files"][change["path"]] = {
            "content_hash": change["sig"]["content_hash"],
            "content": new_content,
            "version": new_version,
            "remote_id": remote_row["id"]
        }
    
    def _apply_patch(self, target, changes):
        """应用增量补丁——神识融合"""
        result = json.loads(json.dumps(target))  # 深拷贝
        
        for change in changes:
            path_parts = change["path"].split(".")
            obj = result
            
            for part in path_parts[:-1]:
                if part not in obj:
                    obj[part] = {}
                obj = obj[part]
            
            last_key = path_parts[-1]
            
            if change["op"] == "add" or change["op"] == "replace":
                obj[last_key] = change.get("value", change.get("new"))
            elif change["op"] == "remove":
                obj.pop(last_key, None)
        
        return result
    
    def _log_conflict(self, change, remote_row):
        """记录冲突——阴阳相争入册"""
        self.sb.table("sync_conflicts").insert({
            "fragment_id": remote_row["id"],
            "local_version": self.manifest["files"]
                .get(change["path"], {}).get("version", 0),
            "remote_version": remote_row["version"],
            "local_content": change["content"],
            "remote_content": remote_row["content"],
            "conflict_type": "concurrent_update",
            "status": "unresolved"
        }).execute()
    
    def _infer_type(self, path):
        """推断记忆类型"""
        if "episodic" in path: return "episodic"
        if "semantic" in path: return "semantic"
        if "procedural" in path: return "procedural"
        return "unknown"
```

## 四、冲突处理：阴阳裁决三策

### 4.1 三种裁决策略

```python
class ConflictResolver:
    """冲突裁决器——阴阳相争，三策定夺"""
    
    def __init__(self, supabase_client):
        self.sb = supabase_client
    
    def resolve_conflict(self, conflict_id, strategy="merge"):
        """
        三策：
        - last_write_wins: 后写覆盖（弱阳策略）
        - merge: 字段级合并（阴阳调和）
        - manual: 人工裁决（天道仲裁）
        """
        conflict = self.sb.table("sync_conflicts") \
            .select("*, memory_fragments(*)") \
            .eq("id", conflict_id) \
            .single().execute().data
        
        if strategy == "last_write_wins":
            return self._resolve_lww(conflict)
        elif strategy == "merge":
            return self._resolve_merge(conflict)
        elif strategy == "manual":
            return self._resolve_manual(conflict)
    
    def _resolve_lww(self, conflict):
        """后写覆盖——比较时间戳，强者胜"""
        local_time = conflict["local_content"].get("_updated_at", "1970-01-01")
        remote_time = conflict["remote_content"].get("_updated_at", "1970-01-01")
        
        winner = "local" if local_time > remote_time else "remote"
        winning_content = (conflict["local_content"] if winner == "local" 
                          else conflict["remote_content"])
        
        new_version = max(conflict["local_version"], 
                         conflict["remote_version"]) + 1
        
        self.sb.table("memory_fragments") \
            .update({
                "content": winning_content,
                "version": new_version,
                "updated_at": datetime.now(timezone.utc).isoformat()
            }) \
            .eq("id", conflict["fragment_id"]).execute()
        
        self.sb.table("sync_conflicts") \
            .update({
                "status": "resolved",
                "resolution": {"strategy": "lww", "winner": winner}
            }) \
            .eq("id", conflict["id"]).execute()
        
        return {"resolved": True, "strategy": "lww", "winner": winner}
    
    def _resolve_merge(self, conflict):
        """字段级合并——阴阳调和，各取所长
        
        核心逻辑：
        - 双方都改了同一字段 → 取时间戳新者
        - 只一方改了某字段 → 保留该方修改
        - 双方都新增了不同字段 → 都保留
        """
        local = conflict["local_content"]
        remote = conflict["remote_content"]
        
        merged = json.loads(json.dumps(remote))  # 以远端为基础
        
        for key in set(list(local.keys()) + list(remote.keys())):
            if key not in remote:
                # 远端没有，本地新增 → 保留
                merged[key] = local[key]
            elif key not in local:
                # 本地没有，远端新增 → 保留远端
                pass
            elif local[key] != remote[key]:
                # 双方都改了 → 取时间戳新者
                local_t = local.get(f"{key}_updated_at", "1970-01-01")
                remote_t = remote.get(f"{key}_updated_at", "1970-01-01")
                if local_t > remote_t:
                    merged[key] = local[key]
        
        new_version = max(conflict["local_version"],
                         conflict["remote_version"]) + 1
        
        self.sb.table("memory_fragments") \
            .update({
                "content": merged,
                "version": new_version,
                "updated_at": datetime.now(timezone.utc).isoformat()
            }) \
            .eq("id", conflict["fragment_id"]).execute()
        
        self.sb.table("sync_conflicts") \
            .update({
                "status": "resolved",
                "resolution": {"strategy": "merge", "merged_content": merged}
            }) \
            .eq("id", conflict["id"]).execute()
        
        return {"resolved": True, "strategy": "merge", "merged": merged}
```

```python
class RealtimeSync:
    """实时同步——聚散由心，化身变化即时感知"""
    
    def __init__(self, supabase_url, supabase_key, on_change_callback):
        self.sb = create_client(supabase_url, supabase_key)
        self.callback = on_change_callback
    
    def start_listening(self):
        """监听远端变化——化身感知本体动向"""
        def handle_change(payload):
            """
            payload结构:
            {
                "type": "UPDATE",
                "table": "memory_fragments",
                "record": { ... new record ... },
                "old_record": { ... old record ... }
            }
            """
            change_type = payload["type"]
            record = payload["record"]
            
            if change_type in ("INSERT", "UPDATE"):
                self.callback("remote_update", record)
            elif change_type == "DELETE":
                self.callback("remote_delete", payload["old_record"])
        
        # Supabase Realtime订阅
        self.sb.channel("memory_fragments_changes") \
            .on_postgres_changes(
                event="*",
                schema="public",
                table="memory_fragments",
                callback=handle_change
            ) \
            .subscribe()
```

```python
class OfflineBuffer:
    """离线缓冲——断境中暂存神识，待重连后续传"""
    
    def __init__(self, buffer_path=".meta/offline_buffer.jsonl"):
        self.buffer_path = buffer_path
        os.makedirs(os.path.dirname(buffer_path), exist_ok=True)
    
    def enqueue(self, change):
        """入队——暂存未同步的变更"""
        with open(self.buffer_path, 'a') as f:
            f.write(json.dumps({
                **change,
                "queued_at": datetime.now(timezone.utc).isoformat()
            }) + "\n")
    
    def drain(self):
        """出队——重连后批量同步"""
        if not os.path.exists(self.buffer_path):
            return []
        
        changes = []
        with open(self.buffer_path, 'r') as f:
            for line in f:
                changes.append(json.loads(line.strip()))
        
        # 清空缓冲
        os.remove(self.buffer_path)
        return changes
```

## 五、同步主循环：周天运转

```python
class MemoryBrainSync:
    """记忆脑同步主控——周天运转，聚散由心"""
    
    def __init__(self, sb_url, sb_key, memory_dir):
        self.engine = MemorySyncEngine(sb_url, sb_key, memory_dir)
        self.resolver = ConflictResolver(self.engine.sb)
        self.buffer = OfflineBuffer(f"{memory_dir}/.meta/offline_buffer.jsonl")
        self.realtime = RealtimeSync(sb_url, sb_key, self._on_remote_change)
        self.sync_interval = 30  # 秒
    
    def run_forever(self):
        """主循环——周天不息"""
        # 先处理离线缓冲
        buffered = self.buffer.drain()
        for change in buffered:
            self._apply_single_change(change)
        
        # 启动实时监听
        self.realtime.start_listening()
        
        # 定时全量扫描
        while True:
            try:
                result = self.engine.sync_incremental()
                print(f"同步完成: {result['synced']}条更新, "
                      f"{result['conflicts']}条冲突")
                
                # 自动处理冲突（merge策略）
                if result["conflicts"] > 0:
                    self._auto_resolve_conflicts()
                    
            except Exception as e:
                # 网络异常 → 入缓冲
                print(f"同步失败，进入离线缓冲: {e}")
                local_changes = self.engine.scan_local_changes()
                for change in local_changes:
                    self.buffer.enqueue(change)
            
            time.sleep(self.sync_interval)
    
    def _on_remote_change(self, event_type, record):
        """远端变化回调——化身感知，反向同步到本地"""
        local_path = f"{self.engine.memory_dir}/{record['local_path']}"
        
        if event_type == "remote_update":
            # 远端更新 → 写回本地
            os.makedirs(os.path.dirname(local_path), exist_ok=True)
            with open(local_path, 'w') as f:
                json.dump(record["content"], f, indent=2)
            
            # 更新本地清单
            self.engine.manifest["files"][record["local_path"]] = {
                "content_hash": record["content_hash"],
                "content": record["content"],
                "version": record["version"],
                "remote_id": record["id"]
            }
            self.engine._save_manifest()
            
        elif event_type == "remote_delete":
            if os.path.exists(local_path):
                os.remove(local_path)
```

## 六、修证小结

| 修炼阶段 | 技术映射 | 核心机制 |
|---------|---------|---------|
| **神识扫描** | 增量检测 | mtime过滤→hash验证→字段级diff→语义级diff |
| **分神化形** | 只同步变更 | JSON Patch精确更新，乐观锁version控制 |
| **阴阳裁决** | 冲突处理 | LWW/字段级merge/人工裁决三策 |
