### 1.2 印记生成（三阶功法）

**练气期 · 快速指纹**（mtime + size，O(1)）：
```javascript
function quickSignature(stat) {
  return `${stat.mtimeMs}|${stat.size}`;
}
```

**筑基期 · 内容指纹**（SHA-256，准确但O(n)）：
```javascript
async function contentHash(filePath) {
  const hash = crypto.createHash('sha256');
  await pipeline(createReadStream(filePath), hash);
  return `sha256:${hash.digest('hex')}`;
}
```

**金丹期 · 块级指纹**（rolling hash，细粒度增量）：
```javascript
// 将文件切分为固定块，每块独立hash
// 变更只影响邻近块，远处块印记不变
async function blockSignatures(filePath, blockSize = 2048) {
  const sigs = [];
  const fd = await open(filePath, 'r');
  const buf = Buffer.alloc(blockSize);
  let offset = 0;
  while (true) {
    const { bytesRead } = await fd.read(buf, 0, blockSize, offset);
    if (bytesRead === 0) break;
    sigs.push({
      offset,
      weak: adler32(buf.slice(0, bytesRead)),  // rolling hash，快速比对
      strong: crypto.createHash('md5').update(buf.slice(0, bytesRead)).digest('hex')
    });
    offset += bytesRead;
  }
  await fd.close();
  return sigs;
}
```

## 二、增量萃取层：只同步变更部分

### 2.1 差异计算

**决策树**：
```
文件新增  → 直接上传（payload + 元数据）
文件删除 → 软删除标记（tombstone，保留7天后物理清除）
文件修改 ↓
  ├ 快速指纹未变 → 丢弃（mtime变了但size+mtimeMs相同，防误报）
  ├ 内容hash未变 → 丢弃（内容指纹未变，仅touch）
  └ 内容hash变化 ↓
      ├ 小文件（<64KB）→ 直接全量上传，patch开销大于全量
      └ 大文件 → 块级diff，只传变更块
```

### 2.2 变更包封装

```typescript
interface SyncPatch {
  file_id: string;          // 内容hash派生的稳定ID
  file_path: string;
  base_version: number;      // 基于云端当前版本
  base_hash: string;         // 基线内容hash
  new_hash: string;
  ops: Array<
    | { type: 'full_replace'; content: string; reason: 'small_file' | 'binary' }
    | { type: 'block_patch'; offset: number; remove: number; insert: string }
    | { type: 'delete' }
    | { type: 'rename'; new_path: string }
  >;
  client_id: string;
  client_clock: number;     // 客户端逻辑时钟
  timestamp: string;
```

### 3.1 版本控制双轨制

**多端协同 · vector clock**：
```javascript
// 每条记录维护 { client_id: counter } 映射
function incrementClock(clientId) {
  return prev => ({
    ...prev,
    [clientId]: (prev[clientId] || 0) + 1
  });
}

function compareClock(a, b) {
  // 返回 'before' | 'after' | 'concurrent'（并发=需要渡劫）
  let aBefore = false, aAfter = false;
  for (const k of new Set([...Object.keys(a), ...Object.keys(b)])) {
    if ((a[k]||0) < (b[k]||0)) aBefore = true;
    if ((a[k]||0) > (b[k]||0)) aAfter = true;
  }
  if (aBefore && aAfter) return 'concurrent';
  if (aBefore) return 'before';
  if (aAfter) return 'after';
  return 'equal';
}
```

### 3.2 冲突分类与渡劫功法

| 冲突类型 | 天劫等级 | 渡劫策略 | 自动化程度 |
|---------|---------|---------|-----------|
| 修改 vs 修改（同字段） | 小劫 | 三路合并（base + A + B） | 全自动 |
| 修改 vs 修改（异字段） | 微劫 | 字段级合并，无需base | 全自动 |
| 修改 vs 删除 | 中劫 | 修改优先，保留tombstone + 标记 | 半自动 |
| 结构性冲突（schema不兼容） | 大劫 | 分叉保留，标记"分歧识海" | 人工 |
| 语义级冲突（矛盾内容） | 心魔劫 | LLM仲裁 + 人工复核 | 半自动 |

### 3.3 三路合并实现

```javascript
async function threeWayMerge(base, local, remote) {
  // diff3 算法：找出base→local和base→remote的变更
  const localDiff  = diffMatchPatch.diff_main(base, local);
  const remoteDiff = diffMatchPatch.diff_main(base, remote);
  
  // 无重叠区域：双方都接受
  // 有重叠区域：标记为冲突块，需仲裁
  const merged = diffMerge(localDiff, remoteDiff, {
    onConflict: (localPart, remotePart) => {
      // 记录到 conflict_log，保留双方版本
      return {
        conflict: true,
        local: localPart,
        remote: remotePart,
        needs_review: true
      };
    }
  });
  return merged;
}
```

```javascript
function nextRetryDelay(retryCount) {
  const base = 1000;          // 1s
  const max = 60000;          // 60s cap
  const exp = Math.min(base * Math.pow(2, retryCount), max);
  const jitter = exp * (0.8 + Math.random() * 0.4);  // ±20%
  return jitter;
}

// 超过10次 → 渡劫失败，进入"识海封印区"
```

## 六、架构决策摘要

| 决策点 | 选择 | 理由 |
|--------|------|------|
| 监听方式 | chokidar原生事件 | 比polling省CPU，比fs.watch稳定 |
| 印记策略 | 三阶渐进 | 小文件用hash，大文件用块签名，按需升级 |
| 冲突策略 | 混合：LWW + 三路合并 + LLM仲裁 | 简单场景自动，复杂场景半自动，语义场景智能 |
| 存储分离 | 小→Database，大→Storage | 结构化查询与二进制存储各取所长 |
| 离线队列 | 本地SQLite | ACID保证，断电不丢 |
| 实时通道 | postgres_changes | 无需额外消息中间件，Supabase原生 |
| 版本控制 | vector clock | 支持多端并发，比单一version更鲁棒 |
