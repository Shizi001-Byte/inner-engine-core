## 二、第1转：感知层 — 文件变化检测

### 2.1 混合检测策略

单一检测方式都有盲区，采用**双轨制**：

```
轨道A（实时）: 文件系统事件监听
  ├── 优势: 延迟低，<100ms感知
  ├── 风险: 系统休眠/高负载时丢事件
  └── 工具: watchdog(Python) / chokidar(Node)

轨道B（兜底）: 定时全量哈希校验
  ├── 频率: 每60秒一次（可配置）
  ├── 优势: 不丢任何变更
  └── 作用: 补偿轨道A的遗漏
```

### 2.2 三级检测流水线

```
文件事件触发
    │
    ▼
┌─────────────────────────────────┐
│ L1: 快速过滤 — mtime + size    │  ← O(1)，过滤90%无变化文件
│   if mtime未变 AND size未变:    │
│     → 跳过                     │
└──────────────┬──────────────────┘
               │ 变化
               ▼
┌─────────────────────────────────┐
│ L2: 内容哈希 — SHA-256(前4KB)  │  ← O(n)但仅算头部，过滤格式化噪声
│   if 前4KB哈希未变:             │
│     → 标记为"疑似无变化"        │
│     → 进入L3确认               │
└──────────────┬──────────────────┘
               │ 仍判定变化
               ▼
┌─────────────────────────────────┐
│ L3: 全量哈希 — SHA-256(全文)   │  ← O(n)，精确判定
│   if 全量哈希 != manifest记录:  │
│     → 确认变更，入队           │
└─────────────────────────────────┘
```

### 3.2 差异计算核心逻辑

```python
def compute_diff(local_manifest, remote_manifest):
    """
    炼精化气：从高熵文件变化中提取精确差异
    返回四种变更类型，每种附带执行优先级
    """
    changes = {
        'create': [],    # 优先级1 — 新知识优先入库
        'update': [],    # 优先级2 — 常规更新
        'delete': [],    # 优先级3 — 谨慎处理，需二次确认
        'conflict': []   # 优先级0 — 最高优先，立即处理
    }
    
    local_files = set(local_manifest["files"].keys())
    remote_files = set(remote_manifest["files"].keys())
    
    for path in local_files | remote_files:
        local_entry = local_manifest["files"].get(path)
        remote_entry = remote_manifest["files"].get(path)
        
        if local_entry and not remote_entry:
            changes['create'].append({
                'path': path,
                'hash': local_entry['hash'],
                'model_source': local_entry['model_source'],
                'priority': 1
            })
            
        elif local_entry and remote_entry:
            if local_entry['hash'] != remote_entry['hash']:
                # 检查远程是否在我们上次同步后被修改
                if remote_entry.get('version', 0) > local_entry.get('remote_version', 0):
                    # 远程也变了 → 冲突
                    changes['conflict'].append({
                        'path': path,
                        'local_hash': local_entry['hash'],
                        'remote_hash': remote_entry['hash'],
                        'local_mtime': local_entry['mtime'],
                        'remote_mtime': remote_entry['updated_at'],
                        'model_source': local_entry['model_source'],
                        'priority': 0
                    })
                else:
                    # 远程没变，本地变了 → 安全更新
                    changes['update'].append({
                        'path': path,
                        'hash': local_entry['hash'],
                        'old_hash': remote_entry['hash'],
                        'model_source': local_entry['model_source'],
                        'priority': 2
                    })
            # else: 哈希相同，无变化
            
        elif remote_entry and not local_entry:
            changes['delete'].append({
                'path': path,
                'remote_hash': remote_entry['hash'],
                'model_source': remote_entry['model_source'],
                'priority': 3,
                'confirm_required': True  # 删除需确认
            })
    
    return changes
```

### 3.3 分块差异（大文件优化）

对于A模型产出的大文件（>64KB），整文件上传浪费带宽：

```python
BLOCK_SIZE = 4096  # 4KB分块

def compute_block_diff(local_path, remote_blocks):
    """
    对大文件进行块级差异计算
    remote_blocks: [{index:0, hash:"xxx"}, ...]
    """
    local_blocks = []
    with open(local_path, 'rb') as f:
        while True:
            block = f.read(BLOCK_SIZE)
            if not block:
                break
            local_blocks.append({
                'index': len(local_blocks),
                'hash': hashlib.sha256(block).hexdigest(),
                'data': block
            })
    
    # 找出变化的块
    changed_blocks = []
    remote_map = {b['index']: b['hash'] for b in remote_blocks}
    
    for block in local_blocks:
        remote_hash = remote_map.get(block['index'])
        if remote_hash != block['hash']:
            changed_blocks.append(block)
    
    return changed_blocks  # 只上传变化的块
```

### 4.2 增量同步执行器

```python
import asyncio
from datetime import datetime

class IncrementalSyncer:
    def __init__(self, supabase_client, manifest_path):
        self.client = supabase_client
        self.manifest_path = manifest_path
        
        # ⚠️ 避坑#2: 根据模型类型预设合理timeout
        self.timeout_map = {
            'B_turbo': 300,   # B模型turbo: 300秒
            'A_model': 600,   # A模型: 600秒
            'default': 300
        }
    
    async def sync(self, changes):
        """执行增量同步的主入口"""
        results = {'success': [], 'failed': [], 'conflicts': []}
        
        # 按优先级排序: conflict(0) > create(1) > update(2) > delete(3)
        all_changes = self._flatten_and_sort(changes)
        
        for change in all_changes:
            try:
                timeout = self.timeout_map.get(
                    change.get('model_source', 'default'), 300
                )
                
                result = await asyncio.wait_for(
                    self._sync_one(change),
                    timeout=timeout
                )
                results['success'].append(result)
                
            except asyncio.TimeoutError:
                # ⚠️ 避坑#2: 超时处理
                results['failed'].append({
                    'change': change,
                    'error': f'Timeout after {timeout}s',
                    'retryable': True
                })
                
            except Exception as e:
                results['failed'].append({
                    'change': change,
                    'error': str(e),
                    'retryable': self._is_retryable(e)
                })
        
        return results
    
    async def _sync_one(self, change):
        """同步单个变更"""
        
        if change['type'] == 'create':
            return await self._handle_create(change)
        elif change['type'] == 'update':
            return await self._handle_update(change)
        elif change['type'] == 'delete':
            return await self._handle_delete(change)
        elif change['type'] == 'conflict':
            return await self._handle_conflict(change)
    
    async def _handle_create(self, change):
        """处理新增文件"""
        path = change['path']
        model_source = change['model_source']
        
        # 1. 上传文件内容到Storage
        with open(path, 'rb') as f:
            content = f.read()
        
        storage_path = f"{model_source}/{path}"
        await self.client.storage.from_('memory-brain').upload(
            storage_path, content,
            {'content-type': 'application/octet-stream'}
        )
        
        # 2. 写入manifest元数据
        new_version = 1
        response = await self.client.table('file_manifest').insert({
            'file_path': path,
            'content_hash': change['hash'],
            'file_size': len(content),
            'mtime': change['mtime'],
            'model_source': model_source,
            'sync_status': 'synced',
            'version': new_version
        }).execute()
        
        # ⚠️ 避坑#1: 写入后立即验证
        verify = await self.client.table('file_manifest') \
            .select('version, sync_status') \
            .eq('file_path', path).execute()
        
        assert verify.data[0]['sync_status'] == 'synced', \
            f"写入验证失败! path={path}"
        
        # 3. 记录变更日志
        await self.client.table('change_log').insert({
            'file_path': path,
            'change_type': 'create',
            'new_hash': change['hash'],
            'new_version': new_version,
            'model_source': model_source,
            'synced_at': datetime.utcnow().isoformat()
        }).execute()
        
        # 4. 更新本地manifest
        self._update_local_manifest(path, {
            'hash': change['hash'],
            'sync_status': 'synced',
            'remote_version': new_version
        })
        
        return {'path': path, 'action': 'created', 'version': new_version}
```

### 4.3 乐观锁机制

```python
async def _handle_update(self, change):
    """使用乐观锁处理更新，防止覆盖远程并发修改"""
    path = change['path']
    expected_version = change.get('remote_version', 0)
    
    # 尝试CAS（Compare-And-Swap）更新
    response = await self.client.table('file_manifest') \
        .update({
            'content_hash': change['hash'],
            'file_size': change['size'],
            'mtime': change['mtime'],
            'sync_status': 'synced',
            'version': expected_version + 1,
            'updated_at': datetime.utcnow().isoformat()
        }) \
        .eq('file_path', path) \
        .eq('version', expected_version) \
        .execute()
    
    if not response.data:
        # CAS失败 → 版本号不匹配 → 远程已被其他人修改
        # 这意味着冲突，转入第4转处理
        raise ConflictDetectedError(
            path=path,
            expected_version=expected_version,
            message="远程版本已变更，CAS更新失败"
        )
    
    new_version = expected_version + 1
    return {'path': path, 'action': 'updated', 'version': new_version}
```

## 五、第4转：炼神还虚 — 冲突处理

### 5.1 冲突分类矩阵

```
                        远程操作
                    ┌──────────────────────────┐
                    │  未修改  │   修改   │ 删除  │
           ┌────────┼─────────┼─────────┼──────┤
     本   │ 未修改  │   —     │ PULL    │  —   │
     地   ├────────┼─────────┼─────────┼──────┤
     操   │ 修改   │ PUSH    │CONFLICT │CONFL │
     作   ├────────┼─────────┼─────────┼──────┤
          │ 删除   │ DELETE  │CONFL    │  —   │
           └────────┴─────────┴─────────┴──────┘
```

### 5.2 分模型冲突解决策略

> **坑库预警执行**：不同模型必须有独立的质量检测规则

```python
class ConflictResolver:
    def __init__(self):
        # 按模型来源定义不同的冲突解决策略
        self.strategies = {
            'B_turbo': {
                # B模型turbo: reasoning占比70-82%是正常特性，不应判废稿
                'skip_reasoning_check': True,
                'priority': 'local',        # 本地优先（引擎最新产出）
                'merge_strategy': 'local_wins',
                'auto_resolve_threshold': 0.95  # 哈希相似度>95%自动合并
            },
            'A_model': {
                'skip_reasoning_check': False,
                'priority': 'remote',       # 远程优先（A模型产出更稳定）
                'merge_strategy': 'three_way',
                'auto_resolve_threshold': 0.90
            },
            'default': {
                'skip_reasoning_check': False,
                'priority': 'newer',        # 时间戳新的优先
                'merge_strategy': 'keep_both',
                'auto_resolve_threshold': 0.85
            }
        }
    
    async def resolve(self, conflict):
        strategy = self.strategies.get(
            conflict['model_source'], 
            self.strategies['default']
        )
        
        # 步骤1: 计算双方内容相似度
        similarity = self._compute_similarity(
            conflict['local_hash'],
            conflict['remote_hash']
        )
        
        # 步骤2: 尝试自动解决
        if similarity >= strategy['auto_resolve_threshold']:
            return await self._auto_resolve(conflict, strategy)
        
        # 步骤3: 无法自动解决，入冲突队列
        await self._enqueue_conflict(conflict)
        return {'status': 'escalated', 'conflict_id': conflict['id']}
    
    async def _auto_resolve(self, conflict, strategy):
        merge_strategy = strategy['merge_strategy']
        
        if merge_strategy == 'local_wins':
            # 本地版本覆盖远程
            # B模型turbo的默认策略
            winner = 'local'
            
        elif merge_strategy == 'remote_wins':
            # 保留远程版本
            winner = 'remote'
            
        elif merge_strategy == 'three_way':
            # 三路合并：需要查找公共祖先版本
            base = await self._find_common_ancestor(conflict)
            merged = self._three_way_merge(
                base, conflict['local_content'], conflict['remote_content']
            )
            if merged['success']:
                winner = 'merged'
            else:
                # 合并失败，升级为手动处理
                await self._enqueue_conflict(conflict)
                return {'status': 'escalated'}
                
        elif merge_strategy == 'keep_both':
            # 保留两个版本，远程版本重命名
            await self._rename_remote(conflict)
            winner = 'local'
        
        # 执行解决
        await self._apply_resolution(conflict, winner, strategy)
        
        # 记录解决日志
        await self.client.table('conflict_queue').update({
            'status': 'resolved',
            'resolution_strategy': f'{winner}_wins',
            'resolved_at': datetime.utcnow().isoformat()
        }).eq('id', conflict['id']).execute()
        
        # ⚠️ 避坑#1: 验证冲突解决生效
        verify = await self.client.table('conflict_queue') \
            .select('status, resolution_strategy') \
            .eq('id', conflict['id']).execute()
        assert verify.data[0]['status'] == 'resolved'
        
        return {'status': 'resolved', 'winner': winner}
```

### 5.3 三路合并算法

```python
def _three_way_merge(self, base, local, remote):
    """
    基于公共祖先的三路合并
    base: 公共祖先版本内容
    local: 本地当前版本
    remote: 远程当前版本
    """
    import difflib
    
    # 按行分割
    base_lines = base.splitlines(keepends=True)
    local_lines = local.splitlines(keepends=True)
    remote_lines = remote.splitlines(keepends=True)
    
    # 计算差异
    local_diff = list(difflib.unified_diff(
        base_lines, local_lines, lineterm=''
    ))
    remote_diff = list(difflib.unified_diff(
        base_lines, remote_lines, lineterm=''
    ))
    
    # 检查是否修改了同一区域
    local_changes = self._extract_changed_regions(local_diff)
    remote_changes = self._extract_changed_regions(remote_diff)
    
    # 如果修改区域不重叠 → 可自动合并
    if not self._regions_overlap(local_changes, remote_changes):
        merged = self._apply_both_changes(
            base_lines, local_changes, remote_changes
        )
        return {'success': True, 'merged': ''.join(merged)}
    else:
        # 修改区域重叠 → 无法自动合并
        return {'success': False, 'conflict_regions': 'overlap'}
```

## 六、同步流程全生命周期

```
时间轴 ──────────────────────────────────────────────────────▶

T+0s     文件变化事件触发
  │
  ├─ L1快速过滤 (mtime+size) ──────────────── 无变化 → 丢弃
  │
  ├─ L2部分哈希 (SHA256前4KB) ─────────────── 无变化 → 丢弃  
  │
  ├─ L3全量哈希 (SHA256全文) ───────────────── 确认变更
  │
T+0.1s   变更入队
  │
  ├─ 标记model_source (B_turbo/A_model)
  ├─ 记录到本地manifest (sync_status=pending)
  │
T+0.5s   差异计算
  │
  ├─ 对比本地manifest vs 远程manifest
  ├─ 分类: create / update / delete / conflict
  │
T+1s     增量同步执行
  │
  ├─ create → upload to Storage + insert manifest
  ├─ update → CAS update (乐观锁)
  │    ├─ CAS成功 → 更新本地manifest
  │    └─ CAS失败 → 转入冲突处理
  ├─ delete → 标记 + 二次确认
  │
T+1~300s 冲突处理（如有）
  │
  ├─ 计算相似度
  ├─ 相似度高 → 自动解决 (local_wins/remote_wins/merge)
  ├─ 相似度低 → 入conflict_queue，等待人工介入
  │
T+完成   验证与归档
  │
  ├─ ⚠️ 避坑#1: 回读验证sync_status=synced
  ├─ 写入change_log审计记录
  ├─ 更新本地manifest (remote_version=新版本号)
  │
T+60s    定时全量校验（安全网）
  │
  ├─ 扫描所有文件
  ├─ 对比manifest
  └─ 发现遗漏 → 补偿同步
```

```python
MODEL_RULES = {
    'B_turbo': {
        # ── 废稿检测规则 ──
        'skip_reasoning_ratio_check': True,  # B模型reasoning占比70-82%是正常的
        'min_content_length': 100,          # 仅检查最小长度

        'sync_timeout': 300,                # ⚠️ 避坑#2: 300秒
        'batch_size': 10,                   # 批量同步每批10个
        'conflict_strategy': 'local_wins',  # 本地优先（最新产出）
        'block_diff_threshold': 64 * 1024,  # >64KB启用分块
    },
    'A_model': {
        # ── 废稿检测规则 ──
        'skip_reasoning_ratio_check': False,
        'min_content_length': 200,
        'valid_reasoning_range': (0.1, 0.6),# A模型reasoning应在此范围
        'reject_if_reasoning_too_high': 0.7,# >70%判废稿
        
        # ── 同步规则 ──
        'sync_timeout': 600,                # ⚠️ 避坑#2: 600秒
        'batch_size': 5,                    # A模型文件较大，每批5个
        'conflict_strategy': 'three_way',   # 三路合并
        'block_diff_threshold': 64 * 1024,
    }
}

def get_model_rules(model_source):
    """获取模型特定规则，确保配置修改后立即生效"""
    rules = MODEL_RULES.get(model_source, MODEL_RULES.get('B_turbo'))
    # ⚠️ 避坑#1: 配置读取点验证
    assert rules is not None, f"未找到模型 {model_source} 的规则配置"
    assert 'sync_timeout' in rules, f"模型 {model_source} 缺少timeout配置"
    return rules
```

### 避坑#2：API调用超时

```python
# ✅ timeout配置集中管理，根据模型类型自动选择
TIMEOUT_CONFIG = {
    'B_turbo': 300,      # B模型turbo: 300秒
    'A_model': 600,      # A模型: 600秒
    'manifest_query': 30, # 元数据查询: 30秒
    'storage_upload': 120,# 文件上传: 120秒
    'conflict_resolve': 60,# 冲突解决: 60秒
}

# 每个网络调用都包裹在wait_for中
async def safe_call(coro, operation_type, model_source=None):
    timeout = TIMEOUT_CONFIG.get(model_source, 300) \
              if model_source else TIMEOUT_CONFIG[operation_type]
    try:
        return await asyncio.wait_for(coro, timeout=timeout)
    except asyncio.TimeoutError:
        raise SyncTimeoutError(
            operation=operation_type,
            model=model_source,
            timeout=timeout
        )
```
