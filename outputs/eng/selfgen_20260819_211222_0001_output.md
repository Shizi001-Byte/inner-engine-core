## 二、重复率检测器

### 2.1 检测维度（三轴交叉）

```python
class RepetitionDetector:
    """
    三轴重复率检测：
    - 维度A: 任务类型分布重复率（同类型任务占比）
    - 维度B: 任务输出语义重复率（输出相似度 > 0.85 的占比）
    - 维度C: 任务模板调用集中度（Top3模板占总执行量的比例）
    """

    WINDOW_SIZE = 50          # 滑动窗口：最近50个任务
    SAMPLE_INTERVAL = 5       # 每5个任务检测一次
    SOFT_THRESHOLD = 0.55     # 软阈值：进入准备状态
    HARD_THRESHOLD = 0.65     # 硬阈值：强制触发扩充
```

### 2.2 重复率计算

```python
def calculate_repetition_rate(self, task_window: list) -> dict:
    """
    返回三轴重复率报告
    """
    report = {
        "type_repetition": 0.0,   # 维度A
        "output_repetition": 0.0, # 维度B  
        "template_concentration": 0.0  # 维度C
    }

    # 维度A: 任务类型分布
    type_counts = Counter(t.type for t in task_window)
    max_type_ratio = max(type_counts.values()) / len(task_window)
    report["type_repetition"] = max_type_ratio

    # 维度B: 输出语义相似度（用嵌入向量余弦相似度）
    outputs = [t.output_embedding for t in task_window if t.output_embedding]
    if len(outputs) >= 2:
        sim_pairs = 0
        total_pairs = 0
        for i, j in combinations(range(len(outputs)), 2):
            cos_sim = cosine_similarity(outputs[i], outputs[j])
            if cos_sim > 0.85:
                sim_pairs += 1
            total_pairs += 1
        report["output_repetition"] = sim_pairs / total_pairs

    # 维度C: 模板集中度
    template_counts = Counter(t.template_id for t in task_window)
    top3_ratio = sum(
        sorted(template_counts.values(), reverse=True)[:3]
    ) / len(task_window)
    report["template_concentration"] = top3_ratio

    # 综合重复率（加权平均）
    report["composite"] = (
        report["type_repetition"] * 0.3 +
        report["output_repetition"] * 0.4 +
        report["template_concentration"] * 0.3
    )

    return report
```

### 2.3 触发逻辑

```python
def should_trigger_expansion(self, report: dict) -> tuple[bool, str]:
    """
    返回: (是否触发, 触发原因)
    """
    if report["composite"] >= self.HARD_THRESHOLD:
        return True, f"综合重复率{report['composite']:.2f}超过硬阈值{self.HARD_THRESHOLD}"

    # 单维度硬触发
    if report["output_repetition"] >= 0.70:
        return True, f"输出重复率{report['output_repetition']:.2f}过高，输出空间坍缩"

    if report["template_concentration"] >= 0.80:
        return True, f"模板集中度{report['template_concentration']:.2f}过高，模板多样性不足"

    return False, ""
```

## 三、差异化缺口分析器

触发扩充后，先分析"缺什么"，再决定"补什么"。

```python
class GapAnalyzer:
    """
    分析当前任务池的覆盖盲区，为新模板生成提供方向指引
    """

    def analyze(self, task_window: list, task_pool: list) -> dict:
        gaps = {
            "underrepresented_types": [],   # 执行量过少的任务类型
            "missing_combinations": [],     # 从未出现的类型组合
            "stale_templates": [],          # 长时间未调用的模板
            "quality_deficit_areas": []     # 质量分偏低的任务领域
        }

        # 1. 找出执行量最少的类型
        type_exec_counts = Counter(t.type for t in task_window)
        avg_exec = sum(type_exec_counts.values()) / len(type_exec_counts)
        gaps["underrepresented_types"] = [
            t for t, c in type_exec_counts.items()
            if c < avg_exec * 0.3
        ]

        # 2. 分析已有模板的标签组合，找盲区
        existing_tag_sets = {frozenset(t.tags) for t in task_pool}
        # 生成候选组合（从已有标签的笛卡尔积中取未覆盖的）
        all_tags = set()
        for t in task_pool:
            all_tags.update(t.tags)
        # 采样未覆盖的2-tag组合
        for tag_pair in combinations(all_tags, 2):
            if frozenset(tag_pair) not in existing_tag_sets:
                gaps["missing_combinations"].append(list(tag_pair))

        # 3. 找超过100任务未调用的模板
        gaps["stale_templates"] = [
            t for t in task_pool
            if t.last_executed_ago > 100
        ]

        return gaps
```

class TemplateGenerator:
    """
    根据缺口分析结果，调用AI模型生成新任务模板
    """

    MODEL_CONFIG = {
        "A_model": {"timeout": 600, "for": "复杂模板设计,多约束推理"},
        "B_model_turbo": {"timeout": 300, "for": "快速批量生成,变体扩展"}
    }

    def generate(self, gap_report: dict, existing_pool: list) -> list[TaskTemplate]:
        """
        生成新模板，返回候选列表
        """
        # 根据缺口复杂度选模型
        if self._is_complex_gap(gap_report):
            model = "A_model"
            timeout = 600
        else:
            model = "B_model_turbo"
            timeout = 300

        prompt = self._build_generation_prompt(gap_report, existing_pool)

        # 调用模型生成（带超时保护）
        raw_output = self._call_model_with_timeout(
            model=model,
            prompt=prompt,
            timeout=timeout
        )

        # 解析为结构化模板
        candidates = self._parse_templates(raw_output)

        return candidates

## 五、质量检测（模型差异化规则）

### 5.1 核心避坑：B模型不检测reasoning占比

```python
class TemplateQualityChecker:
    """
    不同模型有不同的正常行为特征，检测规则必须差异化
    避坑: B模型reasoning占比70-82%是正常特性，不能判废
    """

    QUALITY_RULES = {
        "A_model": {
            "check_reasoning_ratio": True,    # 检测
            "reasoning_max": 0.60,            # A模型reasoning超过60%才异常
            "check_output_completeness": True,
            "check_format_compliance": True,
        },
        "B_model_turbo": {
            "check_reasoning_ratio": False,   # ← 跳过! B模型正常就高
            "check_output_completeness": True,
            "check_format_compliance": True,
        }
    }

    def check(self, template: TaskTemplate, source_model: str) -> tuple[bool, str]:
        """
        返回: (是否通过, 失败原因)
        """
        rules = self.QUALITY_RULES.get(source_model, self.QUALITY_RULES["A_model"])

        # B模型跳过reasoning占比检测
        if rules["check_reasoning_ratio"]:
            if template.reasoning_ratio > rules["reasoning_max"]:
                return False, f"reasoning占比{template.reasoning_ratio:.2f}超过阈值{rules['reasoning_max']}"

        # 通用检测：输出完整性
        if rules["check_output_completeness"]:
            required_fields = ["template_id", "type", "description", "input_schema", "expected_output", "tags"]
            for field in required_fields:
                if not getattr(template, field, None):
                    return False, f"必填字段缺失: {field}"

        # 通用检测：与已有模板重复度
        max_sim = self._max_similarity_to_pool(template)
        if max_sim > 0.75:
            return False, f"与已有模板最大相似度{max_sim:.2f}超过阈值0.75"

        return True, ""
```

## 六、任务池注入 + 配置三同步

### 6.1 核心避坑：三同步缺一不可

```python
class TemplateInjector:
    """
    新模板注入时必须同步修改三个点：
    1. 状态写入 — 模板写入持久存储
    2. 执行读取 — 引擎调度器能读到新模板  
    3. 生效验证 — 立即验证新模板可被调度
    """

    def inject(self, template: TaskTemplate) -> bool:
        """
        注入模板，三同步执行
        """
        # ═══════════ 同步点1: 状态写入 ═══════════
        try:
            self.task_pool_store.add(template)           # 持久存储
            self.template_registry.register(template)    # 模板注册表
            self.template_status.set(template.id, "active")  # 状态标记
        except Exception as e:
            logger.error(f"[同步点1-状态写入] 失败: {e}")
            return False

        # ═══════════ 同步点2: 执行读取 ═══════════
        try:
            self.scheduler.reload_templates()            # 调度器重新加载
            self.task_router.update_routes(template)     # 路由表更新
            self.weight_balancer.recalc_weights()        # 权重重平衡
        except Exception as e:
            logger.error(f"[同步点2-执行读取] 失败: {e}")
            # 回滚同步点1
            self._rollback_injection(template)
            return False

        # ═══════════ 同步点3: 生效验证 ═══════════
        try:
            verified = self._verify_template_active(template)
            if not verified:
                logger.error(f"[同步点3-生效验证] 模板{template.id}注入后无法被调度")
                self._rollback_injection(template)
                return False
        except Exception as e:
            logger.error(f"[同步点3-生效验证] 异常: {e}")
            self._rollback_injection(template)
            return False

        logger.info(f"模板{template.id}注入成功，三同步验证通过")
        return True

    def _verify_template_active(self, template: TaskTemplate) -> bool:
        """
        立即验证：模拟一次调度，确认新模板可被选中
        """
        # 模拟调度查询
        selectable = self.scheduler.get_selectable_templates()
        if template.id not in [t.id for t in selectable]:
            return False

        # 验证权重非零
        weight = self.weight_balancer.get_weight(template.id)
        if weight <= 0:
            return False

        # 验证路由可达
        route = self.task_router.resolve(template.type)
        if route is None:
            return False

        return True

    def _rollback_injection(self, template: TaskTemplate):
        """注入失败时回滚所有写入"""
        self.task_pool_store.remove(template.id)
        self.template_registry.unregister(template.id)
        self.template_status.set(template.id, "rolled_back")
```

### 6.2 权重与冷却配置同步

```python
class WeightBalancer:
    """
    新模板注入时的权重和冷却配置
    """

    DEFAULT_NEW_TEMPLATE_WEIGHT = 3    # 新模板初始权重（中等偏低，逐步提升）
    DEFAULT_COOLDOWN = 30              # 冷却30秒
    WEIGHT_PROMOTION_THRESHOLD = 5     # 执行5次后评估是否提升权重

    def add_new_template(self, template_id: str):
        """新增模板时同步配置权重和冷却"""
        # 写入权重配置
        self.weights[template_id] = self.DEFAULT_NEW_TEMPLATE_WEIGHT
        # 写入冷却配置
        self.cooldowns[template_id] = self.DEFAULT_COOLDOWN
        # 写入执行计数器
        self.exec_counts[template_id] = 0
        # 立即验证配置生效
        assert self.get_weight(template_id) == self.DEFAULT_NEW_TEMPLATE_WEIGHT
        assert self.get_cooldown(template_id) == self.DEFAULT_COOLDOWN
```

## 七、认知缓存刷新集成

### 6.1 避坑：认知池不能只进不出

```python
class CognitiveCacheRefresher:
    """
    新模板注入后必须刷新认知缓存，
    确保新认知在30分钟内参与注入
    """

    REFRESH_TRIGGERS = {
        "every_10_tasks": True,      # 每10任务刷新
        "every_round_end": True,     # 每轮结束刷新
        "on_template_injection": True # 新模板注入时刷新 ← 新增
    }

    MAX_STALE_TIME = 1800  # 30分钟（秒），超过必须刷新

    def refresh_after_injection(self, new_template: TaskTemplate):
        """
        新模板注入后刷新认知缓存
        """
        # 1. 从认知池提取与新模板相关的认知
        related_cognitions = self.cognitive_pool.query(
            tags=new_template.tags,
            limit=5,
            max_age_seconds=self.MAX_STALE_TIME  # 30分钟内的
        )

        # 2. 注入到任务缓存
        for cognition in related_cognitions:
            self.task_cache.inject_cognition(
                template_id=new_template.id,
                cognition=cognition
            )

        # 3. 记录刷新时间
        self.last_refresh = time.time()

        # 4. 验证认知确实参与注入
        active_cognitions = self.task_cache.get_active_cognitions()
        assert len(active_cognitions) > 0, "认知缓存刷新后仍为空"
```

## 八、完整工作流（伪代码）

```python
class TaskPoolEvolutionEngine:
    """
    任务池自进化引擎主控
    """

    def run_cycle(self):
        task_count_since_last_check = 0

        while engine_running:
            # === 正常执行任务 ===
            task = self.scheduler.next_task()
            result = self.execute_task(task)
            self.task_window.append(task)

            task_count_since_last_check += 1

            # === 每5个任务检测一次重复率 ===
            if task_count_since_last_check >= 5:
                task_count_since_last_check = 0

                report = self.repetition_detector.calculate_repetition_rate(
                    self.task_window[-50:]
                )

                should_trigger, reason = self.repetition_detector.should_trigger_expansion(report)

                if should_trigger:
                    logger.warning(f"触发任务池扩充: {reason}")

                    # Step1: 缺口分析
                    gaps = self.gap_analyzer.analyze(self.task_window, self.task_pool)

                    # Step2: 生成新模板
                    candidates = self.template_generator.generate(gaps, self.task_pool)

                    # Step3: 质量检测（模型差异化规则）
                    for template, model in candidates:
                        passed, fail_reason = self.quality_checker.check(template, model)

                        if not passed:
                            logger.info(f"模板{template.id}质量检测未通过: {fail_reason}")
                            continue

                        # Step4: 注入 + 三同步 + 生效验证
                        success = self.template_injector.inject(template)

                        if success:
                            # Step5: 刷新认知缓存
                            self.cognitive_cache_refresher.refresh_after_injection(template)

                            # Step6: 记录扩充事件到认知池
                            self.cognitive_pool.add(Cognition(
                                type="task_pool_evolution",
                                content=f"检测到重复率{report['composite']:.2f}，"
                                       f"注入模板{template.id}覆盖缺口{template.tags}",
                                timestamp=time.time()
                            ))

                    # Step7: 每10任务或本轮结束刷新认知
                    if len(self.task_window) % 10 == 0:
                        self.cognitive_cache_refresher.refresh()
```

## 九、避坑清单（从历史经验提炼）

| # | 坑 | 对策 | 落地位置 |
|---|---|---|---|
| 1 | 配置修改后未生效 | 三同步：状态写入+执行读取+生效验证 | `TemplateInjector.inject()` |
| 2 | API调用超时 | A模型600s，B模型turbo 300s | `TemplateGenerator.MODEL_CONFIG` |
| 3 | 自动调参全链路失效 | 权重/冷却/并发三配置同步写入+验证 | `WeightBalancer.add_new_template()` |
| 4 | B模型reasoning被误杀 | B模型跳过reasoning占比检测 | `TemplateQualityChecker.QUALITY_RULES` |
| 5 | 认知池只进不出 | 每10任务/每轮/注入时刷新缓存，30分钟强制参与 | `CognitiveCacheRefresher` |
