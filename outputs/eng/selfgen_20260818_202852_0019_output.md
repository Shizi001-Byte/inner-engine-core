# 任务类型: eng

基于已有锚定，直接给结论：**多线程 API 调用 + 串行后处理，必须采用“并行写槽位 + 屏障等待 + 串行提交”模式；锁只保护状态变更，不保护网络调用。**

## 一、竞态来源

- 多个 worker 同时写 `completed_count`，不是原子操作，会丢更新。
- 后处理线程读结果时，可能读到 worker 还没写完的“半成品”对象。
- worker 写结果和后处理标记 `PROCESSED` 并发时，可能发生“提交后被旧线程覆盖”。
- 只靠 `future.result(timeout)` 不能取消线程，超时后线程仍可能写共享状态。

所以锁机制要保证：

1. API 调用期间不持锁。
2. 所有共享状态读写都在同一把锁内。
3. 后处理必须等所有 worker 写完，再按固定顺序串行执行。
4. 超时任务必须标记 `FAILED`，不让旧线程后续写入覆盖新状态。

## 二、完整锁机制方案

```python
import json
import threading
import time
import urllib.request
from concurrent.futures import ThreadPoolExecutor

# 历史坑：统一 urllib 直连，显式禁用全局代理
OPENER = urllib.request.build_opener(urllib.request.ProxyHandler({}))

# 历史坑：超时必须显式设置
MODEL_TIMEOUT = {
    "B_turbo": 300,  # B模型turbo
    "A": 600,        # A模型
}


class TaskSpec:
    def __init__(self, task_id, model, url, payload, timeout=None):
        self.task_id = task_id
        self.model = model
        self.url = url
        self.payload = payload
        self.timeout = timeout if timeout is not None else MODEL_TIMEOUT[model]
        self.epoch = 0


class TaskState:
    def __init__(self, epoch):
        self.status = "PENDING"   # PENDING/RUNNING/SUCCEEDED/FAILED/PROCESSED
        self.result = None
        self.error = None
        self.epoch = epoch


class EngineState:
    def __init__(self):
        # 全局唯一状态锁，RLock 避免同线程内重入死锁
        self.lock = threading.RLock()
        # 条件变量用于等待所有 API 调用完成
        self.cond = threading.Condition(self.lock)
        self.phase = "IDLE"        # IDLE/RUNNING/POSTPROCESSING
        self.epoch = 0
        self.tasks = {}
        self.completed_count = 0
        self.total_count = 0

    def register_batch(self, specs):
        with self.lock:
            if self.phase != "IDLE":
                raise RuntimeError("engine busy")
            self.epoch += 1
            self.phase = "RUNNING"
            self.completed_count = 0
            self.total_count = len(specs)
            self.tasks.clear()
            for spec in specs:
                spec.epoch = self.epoch
                self.tasks[spec.task_id] = TaskState(self.epoch)

    def finish_task(self, spec, status, result=None, error=None):
        with self.lock:
            task = self.tasks.get(spec.task_id)
            # epoch 不一致说明是上一批的旧线程，丢弃
            if task is None or task.epoch != spec.epoch:
                return
            # 已超时标记或已提交，不允许覆盖
            if task.status in ("SUCCEEDED", "FAILED", "PROCESSED"):
                return
            task.status = status
            task.result = result
            task.error = error
            self.completed_count += 1
            self.cond.notify_all()

    def wait_all(self, timeout):
        deadline = time.monotonic() + timeout
        with self.cond:
            while self.completed_count < self.total_count:
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    raise TimeoutError("batch wait timeout")
                self.cond.wait(timeout=remaining)


def call_api(spec):
    req = urllib.request.Request(
        spec.url,
        data=json.dumps(spec.payload).encode("utf-8"),
        headers={"Content-Type": "application/json"},
        method="POST",
    )
    # 核心：socket 层超时，防止线程卡死
    with OPENER.open(req, timeout=spec.timeout) as resp:
        data = json.loads(resp.read().decode("utf-8"))

    # 历史坑：B_turbo 走 Responses API，content 可能为空
    if spec.model == "B_turbo":
        content = data.get("content")
        if not content or not content.strip():
            raise RuntimeError("B_turbo Responses API content empty")

    return data


def worker(state, spec):
    with state.lock:
        task = state.tasks.get(spec.task_id)
        if task is None or task.epoch != spec.epoch:
            return
        task.status = "RUNNING"

    try:
        result = call_api(spec)
    except Exception as exc:
        state.finish_task(spec, "FAILED", error=exc)
    else:
        state.finish_task(spec, "SUCCEEDED", result=result)


def post_process_one(spec, result):
    # 串行后处理逻辑，这里不要访问共享 state
    return result


def handle_failure(spec, error):
    # 失败处理：记录日志、对账、告警等
    pass


def run_batch(specs):
    if not specs:
        return

    state = EngineState()
    state.register_batch(specs)

    try:
        with ThreadPoolExecutor(max_workers=len(specs)) as pool:
            for spec in specs:
                pool.submit(worker, state, spec)

            # 屏障：等待所有 API 调用返回或超时
            try:
                batch_timeout = max(s.timeout for s in specs) + 60
                state.wait_all(batch_timeout)
            except TimeoutError:
                with state.lock:
                    for spec in specs:
                        task = state.tasks[spec.task_id]
                        if task.status in ("PENDING", "RUNNING"):
                            task.status = "FAILED"
                            task.error = TimeoutError(f"{spec.task_id} timeout")
                            state.completed_count += 1
                    state.cond.notify_all()

        # 所有 worker 线程已结束，进入串行后处理阶段
        with state.lock:
            state.phase = "POSTPROCESSING"

        # 串行后处理：严格按 specs 原始顺序执行
        for spec in specs:
            with state.lock:
                task = state.tasks[spec.task_id]
                status = task.status
                result = task.result
                error = task.error

            if status == "SUCCEEDED":
                try:
                    processed = post_process_one(spec, result)
                except Exception as exc:
                    with state.lock:
                        task = state.tasks[spec.task_id]
                        task.status = "PROCESSED"
                        task.error = exc
                else:
                    with state.lock:
                        task = state.tasks[spec.task_id]
                        task.result = processed
                        task.status = "PROCESSED"
                        task.error = None
            else:
                try:
                    handle_failure(spec, error)
                finally:
                    with state.lock:
                        task = state.tasks[spec.task_id]
                        task.status = "PROCESSED"
                        task.error = error

    finally:
        with state.lock:
            state.phase = "IDLE"
            state.completed_count = 0
            state.total_count = 0
```

## 三、验证方法

### 1. 并发压力测试

```python
def assert_state_consistent(state, specs):
    with state.lock:
        assert state.phase == "IDLE"
        assert state.completed_count == 0
        assert state.total_count == 0
        for spec in specs:
            assert state.tasks[spec.task_id].status == "PROCESSED"
```

跑 **16 线程 × 1000 轮**，假 API 随机 `sleep`、随机超时、随机空响应，断言：

- 无死锁；
- 无丢更新；
- 每轮结束后所有任务都是 `PROCESSED`；
- `completed_count` 最终归零。

### 2. 故障注入

- 注入一个永不返回的假 API，确认 `urlopen(timeout=...)` 会触发，任务被标记 `FAILED`，整批不卡死。
- 注入一个 `content=""` 的 B_turbo 响应，确认被拒，且后续任务仍正常处理。

### 3. 代理环境验证

- 在设置了全局代理的机器上跑通全链路，确认没有 `ProxyError`。
- 如果存量代码必须用 `requests`，必须显式：

```python
requests.post(url, json=payload, timeout=timeout, proxies={"http": None, "https": None})
```

## 四、写入 SOP 的 3 条铁律

1. **所有模型调用必须显式设置 timeout**  
   `B_turbo=300s`，`A=600s`；验证方法：故障注入挂起 API，任务必须在 deadline 内标记 `FAILED`，不卡死。

2. **网络层统一 urllib 直连并显式禁代理**  
   使用 `urllib.request.build_opener(urllib.request.ProxyHandler({}))`，不引入 `requests`；验证方法：全局代理环境下跑通全链路。

3. **Responses API content 非空才允许提交**  
   新 API 格式接入前先单测 `content` 非空；验证方法：mock 空 content，断言该任务失败且后处理继续。

锁机制本身也是“突变”：任何锁顺序、超时阈值、代理设置变更，都必须跑过上面的并发压力测试；失败就回滚到上一个通过测试的版本。
