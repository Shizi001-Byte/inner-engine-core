```python
# 防坑1核心：配置管理器
class ConfigManager:
    """单一配置源 + 修改即验证 + 全局通知"""
    
    _readers = {}  # 配置读取点注册表
    
    def register_reader(self, module, config_key, callback):
        """每个模块必须注册自己的配置读取点"""
        self._readers.setdefault(config_key, []).append({
            'module': module,
            'callback': callback,
            'confirmed': False
        })
    
    def update_config(self, config_key, changes):
        """修改配置后立即验证生效"""
        # 1. 写入新配置
        self._write(config_key, changes)
        # 2. 重新加载
        self._reload(config_key)
        # 3. 通知所有注册的读取点
        for reader in self._readers.get(config_key, []):
            reader['callback'](self._get(config_key))
        # 4. 立即验证：对比实际值与期望值
        for key, expected in changes.items():
            actual = self._get_nested(config_key, key)
            assert actual == expected, \
                f"配置未生效！{key}: 期望{expected}, 实际{actual}"
        # 5. 验证所有读取点已确认
        for reader in self._readers.get(config_key, []):
            assert reader.get('confirmed'), \
                f"模块{reader['module']}未确认配置更新"
```

#### 归档完整性校验

```python
class ArchiveVerifier:
    def verify_archive(self, task_id, expected_path):
        """归档后立即验证"""
        checks = {
            'file_exists': os.path.exists(expected_path),
            'file_not_empty': os.path.getsize(expected_path) > 0,
            'checksum_matches': self._verify_checksum(expected_path),
            'frontmatter_valid': self._validate_frontmatter(expected_path),
            'index_updated': self._check_index(task_id),
            'tags_written': self._check_tags(task_id),
            'config_version_recorded': self._check_config_version(task_id)
        }
        all_passed = all(checks.values())
        if not all_passed:
            failed = [k for k, v in checks.items() if not v]
            self._alert(f"归档校验失败: {task_id}, 失败项: {failed}")
        return all_passed, checks
```

## 五、防坑措施总结

### 防坑1：配置修改后未生效

| 措施 | 实现方式 |
|------|----------|
| **单一配置源** | 所有配置统一存放在 `_config/` 目录 |
| **读取点注册** | 每个模块必须向ConfigManager注册自己的配置读取点 |
| **修改即通知** | 配置变更后自动通知所有注册的读取点 |
| **修改即验证** | 配置更新后立即注入测试数据验证行为变化 |
| **配置版本化** | 每个归档文件记录归档时的config_version |
| **失败即回滚** | 验证不通过自动回滚到上一版本配置 |

### 防坑2：API调用超时

| 任务类型 | 超时设定 | 重试次数 | 降级策略 |
