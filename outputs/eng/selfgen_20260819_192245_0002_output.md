```python
def test_restore_from_snapshot():
    # 1. 先创建快照
    original_state = {"tasks": [1,2,3], "counter": 42, "wal_seq": 100}
    save_test_state(original_state)
    snapshot_path = create_snapshot()
    
    # 2. 篡改当前状态（模拟状态丢失/损坏）

    save_test_state(corrupted_state)
    
    # 3. 执行恢复
    restore_from_snapshot(snapshot_path)
    
    # 4. 验证恢复结果

        print(f"✅ 损坏快照正确抛异常: {type(e).__name__}")
```

```python
def test_state_loss_recovery():
    """完整的状态丢失→恢复流程验证"""
    
    # === Phase 1: 正常运行，积累状态 ===
    engine_state = {
        "task_queue": [{"id": 1, "status": "done"}, {"id": 2, "status": "running"}],
        "metrics": {"total_generated": 150, "success_rate": 0.87},
        "archived_items": [...],  # 归档系统状态
        "wal_sequence": 230,
        "snapshot_version": 5
    }
    save_state(engine_state)
    
    # 创建快照
    snapshot = create_snapshot()
    print(f"快照已创建: {snapshot}")
    
    # === Phase 2: 模拟崩溃，状态丢失 ===
    # 删除或清空状态文件
    clear_state_dir()
    
    # 验证状态确实丢失
    assert load_state() is None or load_state() == {}, "状态应该已清空"
    print("状态已清空，模拟崩溃场景")
    
    # === Phase 3: 执行恢复 ===
    # 恢复应该自动找到最新快照
    restore_result = restore_from_snapshot(latest=True)  # 或指定路径
    
    # === Phase 4: 验证恢复结果 ===
    recovered_state = load_state()
    
    # 逐字段验证
    assert recovered_state["task_queue"] == engine_state["task_queue"], "任务队列恢复失败"
    assert recovered_state["metrics"] == engine_state["metrics"], "指标恢复失败"
    assert recovered_state["wal_sequence"] == engine_state["wal_sequence"], "WAL序列号恢复失败"
    
    # 关键：恢复后系统能继续正常工作
    # 模拟恢复后的第一次操作
    new_task = {"id": 3, "status": "pending"}
    recovered_state["task_queue"].append(new_task)
    recovered_state["wal_sequence"] += 1
    save_state(recovered_state)
    
    # 再创建快照验证恢复后系统正常
    new_snapshot = create_snapshot()
    assert os.path.exists(new_snapshot), "恢复后创建快照失败"
    
    print("✅ 状态丢失→恢复流程验证通过")
```

### 此场景的关键风险

```
风险：WAL和Snapshot之间的间隙
现象：快照保存了WAL seq=100，之后又有seq=101~105的WAL日志
      如果restore只恢复snapshot不回放WAL，则丢失101~105的操作
检查：restore后是否需要回放快照之后的WAL日志

风险：恢复时机
现象：Guardian进程检测到Core崩溃，自动调用restore
      如果Core还在写数据时被restore，会导致状态混乱
检查：restore前是否有锁或进程状态检查
```

## 验证点4：快照只保留最近5个

### 检查清单

```python
def test_snapshot_retention():
    """验证快照轮转策略"""
    snapshot_dir = get_snapshot_dir()
    
    # 清理后创建7个快照
    clear_dir(snapshot_dir)
    
    snapshots_created = []
    for i in range(7):
        update_test_state({"iteration": i})
        path = create_snapshot()
        snapshots_created.append(path)
        # 确保时间戳不同（文件名不冲突）
        time.sleep(1.1)
    
    # 检查磁盘上实际保留的快照数量
    remaining = list_snapshots(snapshot_dir)
    
    print(f"创建快照数: {len(snapshots_created)}")
    print(f"磁盘保留数: {len(remaining)}")
    print(f"保留的快照: {remaining}")
    
    # 核心断言
    assert len(remaining) == 5, f"应保留5个，实际保留{len(remaining)}个"
    
    # 验证保留的是最新的5个（不是最旧的）
    # 第1、2个快照应该已被删除
    for old_snapshot in snapshots_created[:2]:
        assert not os.path.exists(old_snapshot), f"旧快照未被清理: {old_snapshot}"
    
    # 第3~7个快照应该保留
    for recent_snapshot in snapshots_created[2:]:
        assert os.path.exists(recent_snapshot), f"新快照被误删: {recent_snapshot}"
    
    print("✅ 快照轮转策略验证通过：保留最近5个，旧的已清理")
```

### 关键风险

```
风险：时间戳精度不足导致文件名冲突
现象：同一秒内创建两个快照，第二个覆盖第一个
预防：文件名包含毫秒或UUID

风险：清理逻辑在创建前还是创建后
现象：如果"先清理再创建"，最多只有4个快照
      如果"先创建再清理"，可能有6个临时存在
检查：确认是哪种策略，对Guardian的判断有无影响

风险：快照大小不一，5个可能占用过多磁盘
预防：增加总大小上限（如500MB），与数量限制取严格
```
