## 一、竞态条件根因分析

### 1.1 问题场景还原

```
Thread-1 ──► API Call ──┬──► PostProcess(读state, 改state, 写state) ──► ...
Thread-2 ──► API Call ──┤
Thread-3 ──► API Call ──┘
                         ▲
                    串行后处理区
                    竞态发生点
```

**核心矛盾**：API调用是IO密集型可并发，但后处理涉及共享状态的读-改-写（RMW），必须串行。

### 1.2 竞态条件矩阵

| # | 竞态类型 | 触发条件 | 后果 |
|---|---------|---------|------|
| R1 | **RMW竞态** | Thread-A读state时Thread-B正在写 | 基于过期state做后处理，状态丢失 |
| R2 | **结果顺序错乱** | 先完成的API后处理被后完成的插入 | 后处理依赖调用顺序时逻辑断裂 |
| R3 | **计数器丢失** | `_task_counter++` 非原子 | task_id重复，结果覆盖 |
| R4 | **后处理交错** | 多线程同时进入后处理临界区 | 全局状态被并发修改，数据撕裂 |
| R5 | **PID文件竞态映射** | 检查"是否已处理"与"标记已处理"非原子 | 同一结果被处理两次 |

## 二、三层锁架构设计

```
┌─────────────────────────────────────────────────────────┐
│                    V7.5 方舟效能引擎                      │
├─────────────┬───────────────┬───────────────────────────┤
│  L1 细粒度层  │  L2 串行化层    │  L3 状态一致性层           │
│  (计数器/队列)  │  (后处理锁)     │  (全局状态读写锁)          │
│  Lock         │  RLock        │  ReadWriteLock           │
│  原子操作       │  条件变量       │  乐观版本号               │
├─────────────┼───────────────┼───────────────────────────┤
│  无死锁风险     │  可重入        │  读多写少优化               │
│  持有时间极短   │  持有=后处理时长  │  CAS式提交                │
└─────────────┴───────────────┴───────────────────────────┘
```

**锁排序规则**（死锁预防的基石）：

```
获取顺序: L1 → L2 → L3
释放顺序: L3 → L2 → L1
永不反向获取，永不跨层持有
```

### 3.1 核心引擎类

```python
"""
V7.5方舟效能引擎 - 状态一致性锁机制
多线程API调用 + 串行后处理 + 状态一致性保证
"""

import threading
import queue
import time
import json
import urllib.request
import urllib.error
from contextlib import contextmanager
from typing import Any, Dict, List, Optional, Tuple, Callable
from dataclasses import dataclass, field
from enum import Enum
import traceback


class TaskStatus(Enum):
    PENDING = "pending"
    API_CALLING = "api_calling"
    QUEUED = "queued"
    POST_PROCESSING = "post_processing"
    DONE = "done"
    ERROR = "error"


@dataclass(order=True)
class TaskResult:
    """优先级队列元素，实现FIFO+优先级"""
    priority: int          # 优先级（错误=99，正常=0）
    sequence: int          # 分配序号（FIFO保证）
    task_id: int = field(compare=False)
    status: TaskStatus = field(compare=False)
    content: Any = field(compare=False, default=None)
    raw_response: Any = field(compare=False, default=None)
    error: Optional[str] = field(compare=False, default=None)
    timestamp: float = field(compare=False, default_factory=time.time)


class ReadWriteLock:
    """
    读写锁：多读单写
    - 读操作可并发（API结果读取）
    - 写操作独占（全局状态更新）
    """
    def __init__(self):
        self._cond = threading.Condition(threading.Lock())
        self._readers = 0
        self._writer = False
        self._waiting_writers = 0

    @contextmanager
    def read_lock(self):
        self._acquire_read()
        try:
            yield
        finally:
            self._release_read()

    @contextmanager
    def write_lock(self):
        self._acquire_write()
        try:
            yield
        finally:
            self._release_write()

    def _acquire_read(self):
        with self._cond:
            while self._writer or self._waiting_writers > 0:
                self._cond.wait()
            self._readers += 1

    def _release_read(self):
        with self._cond:
            self._readers -= 1
            if self._readers == 0:
                self._cond.notify_all()

    def _acquire_write(self):
        with self._cond:
            self._waiting_writers += 1
            while self._readers > 0 or self._writer:
                self._cond.wait()
            self._waiting_writers -= 1
            self._writer = True

    def _release_write(self):
        with self._cond:
            self._writer = False
            self._cond.notify_all()


class ArkEngineV75:
    """
    V7.5方舟效能引擎
    
    架构: 生产者(多线程API调用) → 优先级队列 → 消费者(单线程串行后处理)
    
    线程模型:
        - N个Worker线程: 并发API调用，结果入队
        - 1个PostProcessor线程: 串行后处理，保证状态一致性
        - 主线程: 调度与等待
    
    锁层次:
        L1: _counter_lock (任务ID分配，持有<1μs)
        L2: _post_process_lock (后处理串行化，持有=后处理时长)
        L3: _rw_lock (全局状态读写，读多写少优化)
    
    CAS映射:
        - 乐观版本号 = 基因代际标记
        - 状态回滚 = 基因断裂修复
        - 原子提交 = 个体存活确认后写入PID文件
    """

    # 避坑1: timeout必须预设且验证
    TIMEOUT_MAP = {
        'turbo': 300,   # B模型turbo: 300秒
        'A': 600,       # A模型: 600秒
    }

    def __init__(self, config: Dict[str, Any]):
        # ── 配置 ──
        self._config = config
        # 避坑: 配置修改后立即验证生效
        self._validate_config()

        # ── L1: 细粒度锁 ──
        self._counter_lock = threading.Lock()
        self._task_counter = 0

        # ── L2: 串行化锁 ──
        self._post_process_lock = threading.RLock()  # 可重入，支持嵌套调用

        # ── L3: 状态一致性锁 ──
        self._rw_lock = ReadWriteLock()
        self._global_state: Dict[str, Any] = {}
        self._state_version: int = 0  # 乐观锁版本号

        # ── 结果队列（线程安全，PriorityQueue内部有锁）──
        self._result_queue: queue.PriorityQueue = queue.PriorityQueue()

        # ── 后处理消费者 ──
        self._shutdown_event = threading.Event()
        self._post_processor_thread: Optional[threading.Thread] = None
        self._post_process_callback: Optional[Callable] = None

        # ── 统计（原子操作保护）──
        self._stats_lock = threading.Lock()
        self._stats = {
            'total_calls': 0,
            'success_calls': 0,
            'error_calls': 0,
            'post_processed': 0,
            'post_process_errors': 0,
        }

    # ═══════════════════════════════════════════════════════
    #  公开接口
    # ═══════════════════════════════════════════════════════

    def start(self, post_process_callback: Optional[Callable] = None):
        """启动后处理消费者线程"""
        self._post_process_callback = post_process_callback
        self._shutdown_event.clear()
        self._post_processor_thread = threading.Thread(
            target=self._post_process_loop,
            daemon=True,
            name="ArkPostProcessor"
        )
        self._post_processor_thread.start()

    def shutdown(self, timeout: float = 10.0):
        """优雅关闭：等待队列消费完毕"""
        # 等待队列清空
        self._result_queue.join()
        self._shutdown_event.set()
        if self._post_processor_thread and self._post_processor_thread.is_alive():
            self._post_processor_thread.join(timeout=timeout)

    def execute_batch(
        self,
        tasks: List[Dict[str, Any]]
    ) -> Dict[str, Any]:
        """
        批量执行：多线程API调用 + 自动串行后处理
        
        Args:
            tasks: 任务列表，每个task包含:
                - url: API地址
                - payload: 请求体
                - model_type: 'turbo' 或 'A'
                - api_key: 认证密钥
                - priority: 优先级(可选，默认0)
        
        Returns:
            执行结果摘要
        """
        if not self._post_processor_thread or not self._post_processor_thread.is_alive():
            self.start()

        threads: List[threading.Thread] = []
        expected_count = len(tasks)

        # ── Phase 1: 并发API调用（生产者）──
        for task in tasks:
            t = threading.Thread(
                target=self._api_call_worker,
                args=(task,),
                daemon=True,
                name=f"ArkWorker-{task.get('id', 'unknown')}"
            )
            threads.append(t)
            t.start()

        # ── 等待所有API调用完成 ──
        max_timeout = max(
            self._get_timeout(t.get('model_type', 'turbo'))
            for t in tasks
        ) + 30  # 30秒缓冲

        for t in threads:
            t.join(timeout=max_timeout)
            if t.is_alive():
                # 避坑2: API超时必须处理
                self._log(f"线程超时未结束: {t.name}, timeout={max_timeout}s")

        # ── 等待后处理消费完毕 ──
        self._result_queue.join()

        # ── 返回摘要 ──
        with self._stats_lock:
            stats = dict(self._stats)

        with self._rw_lock.read_lock():
            state_snapshot = dict(self._global_state)
            state_version = self._state_version

        return {
            'stats': stats,
            'state': state_snapshot,
            'state_version': state_version,
            'expected': expected_count,
        }

    # ═══════════════════════════════════════════════════════
    #  Phase 1: 并发API调用（生产者）
    # ═══════════════════════════════════════════════════════

    def _api_call_worker(self, task: Dict[str, Any]):
        """工作线程：调用API，结果入队"""
        # L1: 分配任务ID（原子操作，持有<1μs）
        task_id = self._allocate_task_id()

        with self._stats_lock:
            self._stats['total_calls'] += 1

        model_type = task.get('model_type', 'turbo')
        # 避坑1: 配置修改后立即验证
        timeout = self._get_timeout(model_type)
        priority = task.get('priority', 0)

        try:
            # ── 无锁区：IO密集的API调用 ──
            raw_response = self._invoke_api(task, timeout)

            # 避坑3: Responses API content为空检测
            content = self._extract_content(raw_response)
            if not content:
                raise ValueError(
                    f"content为空，reasoning可能耗尽output_tokens"
                )

            result = TaskResult(
                priority=priority,
                sequence=task_id,
                task_id=task_id,
                status=TaskStatus.QUEUED,
                content=content,
                raw_response=raw_response,
            )

            with self._stats_lock:
                self._stats['success_calls'] += 1

        except Exception as e:
            result = TaskResult(
                priority=99,  # 错误优先处理
                sequence=task_id,
                task_id=task_id,
                status=TaskStatus.ERROR,
                error=f"{type(e).__name__}: {str(e)}",
            )

            with self._stats_lock:
                self._stats['error_calls'] += 1

        # ── 入队（PriorityQueue内部线程安全）──
        self._result_queue.put(result)

    def _invoke_api(
        self,
        task: Dict[str, Any],
        timeout: int
    ) -> Dict[str, Any]:
        """
        API调用
        
        避坑3: 统一用urllib直连，不引入requests
        避坑4: 如必须用requests，显式proxies=None
        """
        url = task['url']
        payload = task['payload']
        api_key = task.get('api_key', '')

        data = json.dumps(payload).encode('utf-8')
        req = urllib.request.Request(
            url,
            data=data,
            headers={
                'Content-Type': 'application/json',
                'Authorization': f'Bearer {api_key}',
            }
        )

        # 避坑3: 显式不使用代理，直连
        proxy_handler = urllib.request.ProxyHandler({})
        opener = urllib.request.build_opener(proxy_handler)

        try:
            with opener.open(req, timeout=timeout) as resp:
                body = resp.read().decode('utf-8')
                return json.loads(body)
        except urllib.error.HTTPError as e:
            raise Exception(f"HTTP {e.code}: {e.reason}")
        except urllib.error.URLError as e:
            raise Exception(f"URL Error: {e.reason}")
        except Exception as e:
            raise Exception(f"API调用异常: {e}")

    def _extract_content(self, response: Dict[str, Any]) -> str:
        """
        提取content
        避坑4: 新API格式必须先验证content非空
        """
        # 兼容多种API格式
        content = (
            response.get('content')
            or response.get('output', {}).get('content', '')
            if isinstance(response.get('output'), dict)
            else response.get('content', '')
        )

        if not content and isinstance(response, dict):
            # 尝试OpenAI兼容格式
            choices = response.get('choices', [])
            if choices and isinstance(choices, list):
                content = (
                    choices[0]
                    .get('message', {})
                    .get('content', '')
                )

        # 避坑4: content为空时主动报错，不静默放过
        if not content:
            # 检查是否reasoning耗尽了tokens
            reasoning = response.get('reasoning', '')
            output_tokens = response.get('output_tokens', 0)
            if reasoning and output_tokens > 0:
                raise ValueError(
                    f"reasoning耗尽output_tokens: "
                    f"reasoning_len={len(str(reasoning))}, "
                    f"output_tokens={output_tokens}"
                )
            raise ValueError("content为空且无明显原因")

        return content

    # ═══════════════════════════════════════════════════════
    #  Phase 2: 串行后处理（消费者）
    # ═══════════════════════════════════════════════════════

    def _post_process_loop(self):
        """
        后处理消费者：单线程串行处理
        天然保证后处理串行，无需额外锁
        """
        while not self._shutdown_event.is_set():
            try:
                # 阻塞获取，1秒超时以便检查shutdown
                result: TaskResult = self._result_queue.get(timeout=1.0)
            except queue.Empty:
                continue

            try:
                self._do_post_process(result)
            except Exception as e:
                self._log(f"后处理异常: {e}\n{traceback.format_exc()}")
                with self._stats_lock:
                    self._stats['post_process_errors'] += 1
            finally:
                # 标记任务完成（让join()能正确返回）
                self._result_queue.task_done()

                with self._stats_lock:
                    self._stats['post_processed'] += 1

    def _do_post_process(self, result: TaskResult):
        """
        串行后处理：在消费者单线程中执行
        
        L2: _post_process_lock 保证可重入（回调中可能再次调用）
        L3: _rw_lock.write_lock 保证状态原子更新
        """
        # L2: 串行化锁（可重入）
        with self._post_process_lock:
            # ── 状态事务（乐观锁模式）──
            # 避坑5: PID文件竞态映射——先备份再修改，失败回滚
            with self._rw_lock.write_lock():
                # 读取当前版本
                expected_version = self._state_version
                backup_state = dict(self._global_state)

                try:
                    # 应用变更
                    if result.status == TaskStatus.ERROR:
                        self._global_state.setdefault('errors', []).append({
                            'task_id': result.task_id,
                            'error': result.error,
                            'timestamp': result.timestamp,
                        })
                        self._global_state['last_error_time'] = result.timestamp
                    else:
                        self._global_state[result.task_id] = {
                            'content': result.content,
                            'timestamp': result.timestamp,
                        }
                        self._global_state['last_success_time'] = result.timestamp

                    # 原子提交：版本号递增
                    self._state_version += 1

                except Exception:
                    # 回滚到备份状态
                    self._global_state = backup_state
                    self._state_version = expected_version
                    raise

            # ── 用户自定义回调（锁外执行，避免长时间持锁）──
            if self._post_process_callback:
                # 回调中如需访问状态，使用读锁
                with self._rw_lock.read_lock():
                    state_snapshot = dict(self._global_state)
                    state_version = self._state_version

                self._post_process_callback(
                    result.task_id,
                    result.status.value,
                    result.content if result.status != TaskStatus.ERROR else result.error,
                    state_snapshot,
                    state_version,
                )

    # ═══════════════════════════════════════════════════════
    #  辅助方法
    # ═══════════════════════════════════════════════════════

    def _allocate_task_id(self) -> int:
        """L1: 原子分配任务ID"""
        with self._counter_lock:
            self._task_counter += 1
            return self._task_counter

    def _get_timeout(self, model_type: str) -> int:
        """
        获取超时时间
        避坑1: 配置修改后立即验证
        避坑2: 根据模型类型预设合理timeout
        """
        timeout = self.TIMEOUT_MAP.get(model_type, 300)
        # 避坑1: 配置修改后必须验证生效
        if not isinstance(timeout, int) or timeout <= 0:
            raise ValueError(f"timeout配置异常: model_type={model_type}, timeout={timeout}")
        return timeout

    def _validate_config(self):
        """配置验证：修改后立即验证生效"""
        required_keys = ['engine_name']
        for key in required_keys:
            if key not in self._config:
                raise ValueError(f"缺少必需配置项: {key}")

        # 验证timeout_map
        for model_type, timeout in self.TIMEOUT_MAP.items():
            if not isinstance(timeout, (int, float)) or timeout <= 0:
                raise ValueError(
                    f"timeout配置异常: {model_type}={timeout}"
                )

    def _log(self, msg: str):
        """日志输出"""
        print(f"[ArkEngine] {time.strftime('%H:%M:%S')} {msg}")

    # ═══════════════════════════════════════════════════════
    #  状态查询接口（读锁保护）
    # ═══════════════════════════════════════════════════════

    def get_state(self) -> Tuple[Dict[str, Any], int]:
        """获取全局状态快照（读锁保护）"""
        with self._rw_lock.read_lock():
            return dict(self._global_state), self._state_version

    def get_stats(self) -> Dict[str, int]:
        """获取统计信息"""
        with self._stats_lock:
            return dict(self._stats)
```

### 3.2 死锁预防矩阵

```
┌──────────────────────────────────────────────────────────────┐
│                    锁获取顺序矩阵                              │
├──────────┬──────────┬──────────┬──────────┬──────────────────┤
│          │  L1计数器 │ L2后处理锁 │ L3读写锁  │  约束            │
├──────────┼──────────┼──────────┼──────────┼──────────────────┤
│ Worker   │    ✓     │    ✗     │    ✗     │  仅L1，API无锁    │
│ PostProc │    ✗     │    ✓     │    ✓     │  L2→L3，顺序获取  │
│ 回调     │    ✗     │    ✓(重入) │    ✓(读)  │  L2已持有，L3读锁 │
│ 查询接口  │    ✗     │    ✗     │    ✓(读)  │  仅L3读锁         │
├──────────┴──────────┴──────────┴──────────┴──────────────────┤
│  规则:                                                        │
│  1. 永不反向获取: L3→L2→L1 禁止                                │
│  2. 持有L2时可重入(RLock)，可获取L3读锁                        │
│  3. Worker线程只碰L1，PostProcessor只碰L2+L3                   │
│  4. 锁持有时间: L1<1μs, L2=后处理时长, L3写=状态更新            │
└──────────────────────────────────────────────────────────────┘
```

### 3.3 状态一致性保证机制

```python
"""
乐观锁 + 状态事务 = CAS式原子提交

映射: PID文件竞态避坑
  - 旧方案: 检查PID文件 → 发现已存在 → 退出（竞态！）
  - 新方案: 备份状态 → 修改 → 版本号校验 → 原子提交
  
状态提交流程:
  1. 获取写锁（独占）
  2. 备份当前状态 + 版本号
  3. 应用变更
  4. 版本号 +1（原子提交）
  5. 释放写锁
  
  异常路径: 回滚到备份状态 + 备份版本号
  
CAS映射:
  expected_version = 当前版本号（读取时）
  new_version = expected_version + 1
  提交条件: 在写锁保护下，版本号未被其他线程修改
"""

# 状态事务的另一种实现（如果后处理逻辑更复杂）
@contextmanager
def state_transaction(self):
    """
    显式状态事务
    用法:
        with engine.state_transaction() as state:
            state['key'] = value  # 直接修改
            # 退出with块时自动提交或回滚
    """
    with self._rw_lock.write_lock():
        backup = dict(self._global_state)
        backup_version = self._state_version
        try:
            yield self._global_state
            # 成功: 版本号递增（原子提交）
            self._state_version += 1
        except Exception:
            # 失败: 回滚
            self._global_state = backup
            self._state_version = backup_version
            raise
```

| 竞态 | 消除机制 | 验证方法 |
|------|---------|---------|

### 4.2 压力测试方案

```python
def stress_test():
    """竞态条件压力测试"""
    engine = ArkEngineV75({'engine_name': 'ArkV75-Stress'})
    engine.start()

    # 10线程 × 100轮 = 1000次API调用
    tasks = []
    for round_i in range(100):
        for thread_i in range(10):
            tasks.append({
                'id': f'r{round_i}t{thread_i}',
                'url': 'https://api.example.com/v1/completions',
                'payload': {'model': 'turbo', 'messages': []},
                'model_type': 'turbo',
                'api_key': 'sk-test',
            })

    result = engine.execute_batch(tasks)

    # ── 验证点1: task_id连续无重复 ──
    state = result['state']
    task_ids = [k for k in state.keys() if isinstance(k, int)]
    assert len(task_ids) == len(set(task_ids)), "task_id有重复!"
    assert max(task_ids) == len(task_ids), "task_id有跳号!"

    # ── 验证点2: 状态版本号 == 成功处理数 ──
    assert result['state_version'] == result['stats']['post_processed'], \
        f"版本号不匹配: {result['state_version']} vs {result['stats']['post_processed']}"

    # ── 验证点3: 无数据丢失 ──
    assert result['stats']['total_calls'] == len(tasks), "调用次数不匹配"

    engine.shutdown()
    print("✅ 压力测试通过")
```
