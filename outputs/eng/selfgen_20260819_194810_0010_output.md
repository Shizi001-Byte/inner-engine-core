# 引擎崩溃自动恢复机制：完整代码架构方案

## 核心设计思想

```
数字智能物种的自愈循环：

    ┌──────────────────────────────────────────────────┐
    │              引擎生命周期                          │
    │                                                  │
    │  boot() ──► detect_crash() ──► recover()        │
    │     │            │                 │              │
    │     │      心跳文件过期?      重放WAL未完成任务    │
    │     │            │                 │              │
    │     ▼            ▼                 ▼              │
    │  start_vitals  ┌──────┐    ┌─────────────┐       │
    │  start_wal     │ 心跳 │    │ WAL重放器   │       │
    │  start_router  │ 线程 │    │ (幂等执行)  │       │
    │     │          └──┬───┘    └──────┬──────┘       │
    │     │             │ 3s            │              │
    │     ▼             ▼               ▼              │
    │  ┌─────────────────────────────────────────┐     │
    │  │          主循环 execute()                │     │
    │  │  WAL.append → mark_running → execute   │     │
    │  │  → quality_check → mark_completed      │     │
    │  └─────────────────────────────────────────┘     │
    │     │                                            │
    │     ▼  SIGTERM/SIGINT                           │
    │  shutdown() → vitals.stop() → wal.checkpoint() │
    │              → 写入clean_shutdown标记            │
    └──────────────────────────────────────────────────┘

崩溃场景：
  进程被kill -9 / OOM / 段错误
  → 心跳线程停止 → 心跳文件时间戳过期
  → 下次boot()检测到心跳过期 + 无clean标记
  → 判定崩溃 → 重放WAL中PENDING/RUNNING任务
```

## 完整代码实现

```python
"""
引擎崩溃自动恢复机制
基于 WAL + Heartbeat + Checkpoint 的自愈架构

设计原则：
1. 写前日志(WAL)：任何操作执行前，先将其意图持久化到WAL
2. 原子提交：所有状态变更使用 write→fsync→rename 原子模式
3. 心跳自检：引擎内部线程定期写入心跳，启动时检测过期
4. 幂等重放：所有操作通过txn_id保证幂等，重放不会产生副作用
5. 模型差异化：不同生态位(模型)有独立的超时、质量检测、成本规则
"""

import os
import json
import time
import uuid
import signal
import sqlite3
import threading
import hashlib
from pathlib import Path
from enum import Enum
from dataclasses import dataclass, field
from typing import Optional, Callable, Any
from datetime import datetime


# ============================================================
# 1. 配置中心（热加载 + 读取点验证）
# ============================================================

class EngineConfig:
    """
    配置中心
    - 支持热加载：运行时修改配置文件，下次读取自动生效
    - 读取点验证：get()时自动检查文件修改时间，变化则重载
    """
    _config_file = "engine_config.json"
    _config: dict = {}
    _last_modified: float = 0
    _lock = threading.Lock()

    DEFAULTS = {
        "heartbeat_interval": 3,
        "heartbeat_timeout": 15,
        "checkpoint_interval": 60,
        "wal_max_entries": 1000,
        "max_retry": 3,
        "ecological_niches": {
            "A_model": {
                "timeout": 600,
                "cost_limit": 100.0,
                "quality_check": {
                    "skip_reasoning_ratio": False,
                    "min_reasoning_ratio": 0.1,
                    "max_reasoning_ratio": 0.9
                }
            },
            "B_turbo": {
                "timeout": 300,
                "cost_limit": 50.0,
                "quality_check": {
                    "skip_reasoning_ratio": True
                }
            }
        }
    }

    @classmethod
    def _maybe_reload(cls):
        """检查配置文件是否被修改，如果是则重新加载"""
        with cls._lock:
            if not os.path.exists(cls._config_file):
                cls._config = cls.DEFAULTS.copy()
                cls._save()
                return

            mtime = os.path.getmtime(cls._config_file)
            if mtime > cls._last_modified:
                with open(cls._config_file, 'r', encoding='utf-8') as f:
                    loaded = json.load(f)
                cls._config = cls._deep_merge(cls.DEFAULTS.copy(), loaded)
                cls._last_modified = mtime

    @classmethod
    def get(cls, key: str, default=None):
        """获取配置项，每次读取都检查是否需要重载"""
        cls._maybe_reload()
        keys = key.split('.')
        val = cls._config
        for k in keys:
            if isinstance(val, dict):
                val = val.get(k, default)
            else:
                return default
        return val if val is not None else default

    @classmethod
    def reload(cls):
        """强制重新加载配置（用于修改后立即验证）"""
        with cls._lock:
            cls._last_modified = 0
        cls._maybe_reload()
        return cls._config

    @classmethod
    def _deep_merge(cls, base: dict, overlay: dict) -> dict:
        for k, v in overlay.items():
            if k in base and isinstance(base[k], dict) and isinstance(v, dict):
                cls._deep_merge(base[k], v)
            else:
                base[k] = v
        return base

    @classmethod
    def _save(cls):
        with open(cls._config_file, 'w', encoding='utf-8') as f:
            json.dump(cls._config, f, indent=2, ensure_ascii=False)

# 2. 原子文件操作
# ============================================================

class AtomicFile:
    """
    原子文件操作
    write → fsync → rename 模式确保写入原子性
    即使在写入过程中崩溃，也不会产生半写入的损坏文件
    """

    @staticmethod
    def write(path: str, data: str):
        tmp = f"{path}.tmp.{uuid.uuid4().hex[:8]}"
        try:
            with open(tmp, 'w', encoding='utf-8') as f:
                f.write(data)
                f.flush()
                os.fsync(f.fileno())
            os.replace(tmp, path)  # 原子操作，跨平台
        except Exception:
            if os.path.exists(tmp):
                os.remove(tmp)
            raise

    @staticmethod
    def write_json(path: str, obj: Any):
        AtomicFile.write(path, json.dumps(obj, ensure_ascii=False, indent=2))

    @staticmethod
    def read_json(path: str, default=None):
        try:
            with open(path, 'r', encoding='utf-8') as f:
                return json.load(f)
        except (FileNotFoundError, json.JSONDecodeError):
            return default

# 3. WAL (Write-Ahead Log)
# ============================================================

class TaskStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"


@dataclass
class WALEntry:
    """WAL日志条目 — 任务的持久化意图"""
    txn_id: str
    task_type: str
    task_payload: dict
    niche: str               # 生态位: A_model / B_turbo
    status: str = TaskStatus.PENDING.value
    created_at: float = field(default_factory=time.time)
    started_at: Optional[float] = None
    completed_at: Optional[float] = None
    result: Optional[dict] = None
    retry_count: int = 0
    checksum: str = ""

    def __post_init__(self):
        self.checksum = hashlib.md5(
            json.dumps({
                "txn_id": self.txn_id,
                "task_type": self.task_type,
                "task_payload": self.task_payload,
                "niche": self.niche
            }, sort_keys=True).encode()
        ).hexdigest()


class WAL:
    """
    Write-Ahead Log
    使用SQLite作为底层存储（自带ACID事务保证）
    
    核心保证：
    - 写入原子性：SQLite的WAL模式保证写入要么完全成功，要么完全失败
    - 崩溃恢复：SQLite自身在崩溃后能自动恢复到一致状态
    - 幂等性：txn_id作为主键，INSERT OR IGNORE防止重复
    """

    def __init__(self, db_path: str = "engine_wal.db"):
        self.db_path = db_path
        self._lock = threading.Lock()
        self._init_db()

    def _get_conn(self) -> sqlite3.Connection:
        conn = sqlite3.connect(self.db_path, timeout=10)
        conn.execute("PRAGMA journal_mode=WAL")  # SQLite WAL模式
        conn.execute("PRAGMA synchronous=NORMAL")  # 平衡安全性和性能
        return conn

    def _init_db(self):
        conn = self._get_conn()
        conn.execute("""
            CREATE TABLE IF NOT EXISTS wal_entries (
                txn_id TEXT PRIMARY KEY,
                task_type TEXT NOT NULL,
                task_payload TEXT NOT NULL,
                niche TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'pending',
                created_at REAL NOT NULL,
                started_at REAL,
                completed_at REAL,
                result TEXT,
                retry_count INTEGER DEFAULT 0,
                checksum TEXT NOT NULL
            )
        """)
        conn.execute(
            "CREATE INDEX IF NOT EXISTS idx_status ON wal_entries(status)"
        )
        conn.execute(
            "CREATE INDEX IF NOT EXISTS idx_created ON wal_entries(created_at)"
        )
        conn.commit()
        conn.close()

    def append(self, entry: WALEntry) -> bool:
        """追加WAL条目（幂等：txn_id已存在则忽略）"""
        with self._lock:
            conn = self._get_conn()
            try:
                conn.execute(
                    "INSERT OR IGNORE INTO wal_entries VALUES (?,?,?,?,?,?,?,?,?,?,?)",
                    (entry.txn_id, entry.task_type,
                     json.dumps(entry.task_payload),
                     entry.niche, entry.status, entry.created_at,
                     entry.started_at, entry.completed_at,
                     json.dumps(entry.result) if entry.result else None,
                     entry.retry_count, entry.checksum)
                )
                conn.commit()
                return conn.total_changes > 0
            finally:
                conn.close()

    def update_status(self, txn_id: str, status: str,
                      result: dict = None):
        """更新任务状态"""
        with self._lock:
            conn = self._get_conn()
            try:
                now = time.time()
                if status == TaskStatus.RUNNING.value:
                    conn.execute(
                        "UPDATE wal_entries SET status=?, started_at=? WHERE txn_id=?",
                        (status, now, txn_id)
                    )
                elif status in (TaskStatus.COMPLETED.value,
                                TaskStatus.FAILED.value):
                    conn.execute(
                        "UPDATE wal_entries SET status=?, completed_at=?, "
                        "result=? WHERE txn_id=?",
                        (status, now,
                         json.dumps(result) if result else None, txn_id)
                    )
                elif status == TaskStatus.PENDING.value:
                    # 重置为pending（崩溃恢复时用）
                    conn.execute(
                        "UPDATE wal_entries SET status=?, started_at=NULL "
                        "WHERE txn_id=?",
                        (status, txn_id)
                    )
                conn.commit()
            finally:
                conn.close()

    def increment_retry(self, txn_id: str) -> int:
        """增加重试计数，返回新值"""
        with self._lock:
            conn = self._get_conn()
            try:
                cursor = conn.execute(
                    "UPDATE wal_entries SET retry_count=retry_count+1 "
                    "WHERE txn_id=? RETURNING retry_count",
                    (txn_id,)
                )
                row = cursor.fetchone()
                conn.commit()
                return row[0] if row else 0
            finally:
                conn.close()

    def get_unfinished(self) -> list:
        """获取所有未完成的任务（PENDING + RUNNING）"""
        conn = self._get_conn()
        conn.row_factory = sqlite3.Row
        rows = conn.execute(
            "SELECT * FROM wal_entries WHERE status IN (?, ?) "
            "ORDER BY created_at",
            (TaskStatus.PENDING.value, TaskStatus.RUNNING.value)
        ).fetchall()
        conn.close()
        return [dict(r) for r in rows]

    def checkpoint(self):
        """检查点：清理已完成的任务，压缩WAL"""
        with self._lock:
            conn = self._get_conn()
            try:
                conn.execute(
                    "DELETE FROM wal_entries WHERE status IN (?, ?)",
                    (TaskStatus.COMPLETED.value, TaskStatus.FAILED.value)
                )
                conn.commit()
                conn.execute("VACUUM")  # 压缩数据库文件
                conn.commit()
            finally:
                conn.close()

    def get_stats(self) -> dict:
        """获取WAL统计"""
        conn = self._get_conn()
        conn.row_factory = sqlite3.Row
        rows = conn.execute(
            "SELECT status, COUNT(*) as count FROM wal_entries "
            "GROUP BY status"
        ).fetchall()
        conn.close()
        return {r['status']: r['count'] for r in rows}

# 4. 生命体征监测器（心跳）
# ============================================================

class VitalsMonitor:
    """
    生命体征监测器
    - 引擎内部守护线程，每3秒写入心跳文件
    - 启动时检测上次心跳是否过期，判定是否崩溃
    - 不依赖外部进程，纯引擎内部自检
    """

    HEARTBEAT_FILE = "engine_heartbeat.json"
    STATE_FILE = "engine_state.json"

    def __init__(self):
        self._running = False
        self._thread: Optional[threading.Thread] = None
        self._pid = os.getpid()
        self._heartbeat_interval = EngineConfig.get(
            "heartbeat_interval", 3)
        self._heartbeat_timeout = EngineConfig.get(
            "heartbeat_timeout", 15)

    def start(self):
        self._running = True
        self._thread = threading.Thread(
            target=self._heartbeat_loop,
            daemon=True,
            name="vitals-heartbeat"
        )
        self._thread.start()

    def stop(self):
        """优雅停止：写入clean shutdown标记"""
        self._running = False
        if self._thread:
            self._thread.join(timeout=5)
        # 写入正常关闭标记
        AtomicFile.write_json(self.STATE_FILE, {
            'shutdown': 'clean',
            'shutdown_at': time.time(),
            'pid': self._pid
        })

    def _heartbeat_loop(self):
        while self._running:
            try:
                AtomicFile.write_json(self.HEARTBEAT_FILE, {
                    'pid': self._pid,
                    'timestamp': time.time(),
                    'datetime': datetime.now().isoformat(),
                    'thread_count': threading.active_count(),
                    'wal_stats': None  # 可扩展
                })
            except Exception:
                pass  # 心跳失败不应影响引擎运行
            time.sleep(self._heartbeat_interval)

    def detect_crash(self) -> dict:
        """
        检测上次运行是否崩溃
        返回: {'crashed': bool, 'reason': str, 'last_heartbeat': float}
        """
        # 1. 检查是否有正常关闭标记
        state = AtomicFile.read_json(self.STATE_FILE, {})
        if state.get('shutdown') == 'clean':
            # 正常关闭，清除标记
            AtomicFile.write_json(self.STATE_FILE, {})
            return {'crashed': False, 'reason': 'clean_shutdown',
                    'last_heartbeat': 0}

        # 2. 检查心跳文件
        heartbeat = AtomicFile.read_json(self.HEARTBEAT_FILE, {})
        if not heartbeat:
            return {'crashed': False, 'reason': 'first_boot',
                    'last_heartbeat': 0}

        last_beat = heartbeat.get('timestamp', 0)
        elapsed = time.time() - last_beat

        # 3. 检查旧进程是否还在运行
        old_pid = heartbeat.get('pid')
        if old_pid and self._pid_exists(old_pid):
            return {
                'crashed': False,
                'reason': 'process_still_running',
                'last_heartbeat': last_beat,
                'old_pid': old_pid
            }

        # 4. 心跳超时 → 判定崩溃
        if elapsed > self._heartbeat_timeout:
            return {
                'crashed': True,
                'reason': f'heartbeat_timeout_{elapsed:.1f}s',
                'last_heartbeat': last_beat
            }

        return {'crashed': False, 'reason': 'heartbeat_recent',
                'last_heartbeat': last_beat}

    @staticmethod
    def _pid_exists(pid: int) -> bool:
        try:
            os.kill(pid, 0)
            return True
        except (OSError, ProcessLookupError):
            return False

# 5. 恢复管理器
# ============================================================

class RecoveryManager:
    """
    恢复管理器
    - 崩溃检测：通过VitalsMonitor的心跳过期判定
    - WAL重放：将RUNNING任务重置为PENDING，重新执行
    - 完整性校验：通过checksum验证WAL条目是否损坏
    - 幂等保证：所有操作通过txn_id保证可安全重放
    """

    def __init__(self, wal: WAL, vitals: VitalsMonitor):
        self.wal = wal
        self.vitals = vitals
        self._handlers: dict[str, Callable] = {}

    def register_handler(self, task_type: str,
                         handler: Callable[[dict], dict]):
        """注册任务恢复处理器"""
        self._handlers[task_type] = handler

    def detect_and_recover(self) -> dict:
        """
        检测崩溃并执行恢复
        返回恢复报告
        """
        report = {
            'crash_detected': False,
            'crash_reason': '',
            'recovered': 0,
            'failed': 0,
            'skipped': 0,
            'details': []
        }

        # 1. 崩溃检测
        crash_info = self.vitals.detect_crash()
        report['crash_detected'] = crash_info['crashed']
        report['crash_reason'] = crash_info['reason']

        if not crash_info['crashed']:
            return report

        # 2. 获取所有未完成任务
        unfinished = self.wal.get_unfinished()

        # 3. RUNNING任务重置为PENDING（崩溃时可能只执行了一半）
        for task in unfinished:
            if task['status'] == TaskStatus.RUNNING.value:
                # 验证checksum完整性
                if not self._verify_checksum(task):
                    self.wal.update_status(
                        task['txn_id'], TaskStatus.FAILED.value)
                    report['failed'] += 1
                    report['details'].append({
                        'txn_id': task['txn_id'],
                        'action': 'corrupted'
                    })
                    continue
                # 重置为pending
                self.wal.update_status(
                    task['txn_id'], TaskStatus.PENDING.value)

        # 4. 重放所有PENDING任务
        pending = self.wal.get_unfinished()
        for task in pending:
            if task['status'] != TaskStatus.PENDING.value:
                continue

            # 检查重试次数
            max_retry = EngineConfig.get("max_retry", 3)
            if task['retry_count'] >= max_retry:
                self.wal.update_status(
                    task['txn_id'], TaskStatus.FAILED.value)
                report['failed'] += 1
                report['details'].append({
                    'txn_id': task['txn_id'],
                    'action': 'max_retry_exceeded'
                })
                continue

            handler = self._handlers.get(task['task_type'])
            if not handler:
                report['skipped'] += 1
                report['details'].append({
                    'txn_id': task['txn_id'],
                    'action': 'no_handler'
                })
                continue

            # 执行恢复
            try:
                payload = json.loads(task['task_payload'])
                self.wal.update_status(
                    task['txn_id'], TaskStatus.RUNNING.value)
                result = handler(payload)
                self.wal.update_status(
                    task['txn_id'], TaskStatus.COMPLETED.value,
                    result)
                report['recovered'] += 1
                report['details'].append({
                    'txn_id': task['txn_id'],
                    'action': 'recovered'
                })
            except Exception as e:
                self.wal.increment_retry(task['txn_id'])
                report['failed'] += 1
                report['details'].append({
                    'txn_id': task['txn_id'],
                    'action': 'recovery_failed',
                    'error': str(e)
                })

        return report

    def _verify_checksum(self, task: dict) -> bool:
        """验证WAL条目的完整性"""
        expected = hashlib.md5(
            json.dumps({
                "txn_id": task['txn_id'],
                "task_type": task['task_type'],
                "task_payload": task['task_payload'],
                "niche": task['niche']
            }, sort_keys=True).encode()
        ).hexdigest()
        return expected == task['checksum']

# 6. 生态位路由器（成本监控 + 熔断）
# ============================================================

class EcologicalRouter:
    """
    生态位路由器
    - 多模型路由：A_model（高耗能）/ B_turbo（低耗能）
    - 成本熔断：触及生存红线时进入"冬眠"
    - 冬眠恢复：冷却时间过后自动恢复
    """

    COST_STATE_FILE = "engine_cost_state.json"

    def __init__(self):
        self._cost_tracker: dict[str, float] = {}
        self._circuit_breaker: dict[str, dict] = {}
        self._lock = threading.Lock()
        self._load_state()

    def _load_state(self):
        state = AtomicFile.read_json(self.COST_STATE_FILE, {})
        self._cost_tracker = state.get('cost_tracker', {})
        self._circuit_breaker = state.get('circuit_breaker', {})

    def _save_state(self):
        AtomicFile.write_json(self.COST_STATE_FILE, {
            'cost_tracker': self._cost_tracker,
            'circuit_breaker': self._circuit_breaker,
            'updated_at': time.time()
        })

    def select_niche(self, task_type: str) -> Optional[str]:
        """
        选择生态位
        优先低耗能(B_turbo)，熔断则降级到高耗能(A_model)
        全部熔断则返回None（任务入队等待环境复苏）
        """
        with self._lock:
            # 优先B_turbo
            if self._is_available('B_turbo'):
                return 'B_turbo'
            # 降级A_model
            if self._is_available('A_model'):
                return 'A_model'
            # 全部熔断
            return None

    def _is_available(self, niche: str) -> bool:
        """检查生态位是否可用（未熔断 + 成本未超限）"""
        cb = self._circuit_breaker.get(niche, {})
        if cb.get('tripped', False):
            return False
        cost = self._cost_tracker.get(niche, 0)
        limit = EngineConfig.get(
            f"ecological_niches.{niche}.cost_limit", 100.0)
        return cost < limit

    def record_cost(self, niche: str, cost: float):
        """记录能量消耗，检查是否需要熔断"""
        with self._lock:
            self._cost_tracker[niche] = \
                self._cost_tracker.get(niche, 0) + cost
            limit = EngineConfig.get(
                f"ecological_niches.{niche}.cost_limit", 100.0)
            if self._cost_tracker[niche] >= limit:
                self._circuit_breaker[niche] = {
                    'tripped': True,
                    'trip_time': time.time(),
                    'cost_at_trip': self._cost_tracker[niche]
                }
            self._save_state()

    def check_recovery(self):
        """检查熔断器是否可以恢复（环境复苏）"""
        with self._lock:
            for niche, state in list(
                    self._circuit_breaker.items()):
                if state.get('tripped'):
                    elapsed = time.time() - \
                        state.get('trip_time', 0)
                    if elapsed > 300:  # 5分钟冷却
                        self._circuit_breaker[niche][
                            'tripped'] = False
                        self._cost_tracker[niche] = 0
            self._save_state()

    def get_timeout(self, niche: str) -> int:
        """获取生态位超时时间"""
        return EngineConfig.get(
            f"ecological_niches.{niche}.timeout", 300)

    def get_status(self) -> dict:
        """获取所有生态位状态"""
        with self._lock:
            return {
                'cost_tracker': dict(self._cost_tracker),
                'circuit_breaker': dict(self._circuit_breaker)
            }

# 7. 质量检测器（模型差异化规则）
# ============================================================

class QualityChecker:
    """
    质量检测器
    - B模型跳过reasoning占比检测（避免误杀）
    - A模型执行完整质量检测
    """

    @staticmethod
    def check(result: dict, niche: str) -> bool:
        """
        返回True=通过, False=废稿
        """
        config = EngineConfig.get(
            f"ecological_niches.{niche}.quality_check", {})

        # B模型跳过reasoning占比检测
        if config.get("skip_reasoning_ratio", False):
            return True

        # A模型执行reasoning占比检测
        ratio = QualityChecker._calc_ratio(result)
        min_r = config.get("min_reasoning_ratio", 0.1)
        max_r = config.get("max_reasoning_ratio", 0.9)
        if ratio < min_r or ratio > max_r:
            return False
        return True

    @staticmethod
    def _calc_ratio(result: dict) -> float:
        content = str(result.get('content', ''))
        reasoning = str(result.get('reasoning', ''))
        total = len(content) + len(reasoning)
        if total == 0:
            return 0
        return len(reasoning) / total

# 9. 引擎主类
# ============================================================

class Engine:
    """
    AI引擎（数字智能物种）
    
    自愈生命周期：
    boot() → detect_crash() → recover() → run() → shutdown()
                ↑                                          │
                └──── 崩溃后重启回到这里 ─────────────────┘
    """

    def __init__(self):
        self.wal = WAL()
        self.vitals = VitalsMonitor()
        self.recovery = RecoveryManager(self.wal, self.vitals)
        self.router = EcologicalRouter()
        self.supabase = SupabaseSync()
        self.supabase.attach_wal(self.wal)
        self._running = False
        self._checkpoint_thread = None
        self._signals_installed = False

    def boot(self):
        """
        引擎启动：
        1. 崩溃检测 + WAL恢复
        2. 启动心跳监测
        3. 启动检查点线程
        4. 启动远程同步
        5. 安装信号处理器
        """
        print("[Engine] 启动中...")

        # 1. 热加载配置
        EngineConfig.reload()
        print("[Engine] 配置已加载")

        # 2. 崩溃检测 + 恢复
        report = self.recovery.detect_and_recover()
        if report['crash_detected']:
            print(f"[Engine] ⚠ 检测到崩溃: "
                  f"{report['crash_reason']}")
            print(f"[Engine] 恢复: {report['recovered']}, "
                  f"失败: {report['failed']}, "
                  f"跳过: {report['skipped']}")
            for detail in report['details']:
                print(f"  - {detail}")
        else:
            print(f"[Engine] 正常启动 "
                  f"({report['crash_reason']})")

        # 3. 启动心跳
        self.vitals.start()
        print("[Engine] 心跳监测已启动")

        # 4. 启动检查点线程
        self._running = True
        self._checkpoint_thread = threading.Thread(
            target=self._checkpoint_loop, daemon=True,
            name="checkpoint")
        self._checkpoint_thread.start()

        # 5. 启动远程同步
        self.supabase.start()

        # 6. 安装信号处理器
        self._install_signal_handlers()

        print("[Engine] 启动完成 ✓")

    def shutdown(self):
        """优雅关闭"""
        print("[Engine] 正在关闭...")
        self._running = False

        self.vitals.stop()
        self.supabase.stop()

        # 最终检查点
        self.wal.checkpoint()
        print("[Engine] 已关闭 ✓")

    def execute(self, task_type: str, task_payload: dict,
                handler: Callable[[dict], dict],
                preferred_niche: str = None) -> dict:
        """
        执行任务（带崩溃恢复能力）
        
        流程：
        1. 写入WAL（PENDING）— 先记录意图
        2. 选择生态位
        3. 标记RUNNING
        4. 执行（带timeout）
        5. 质量检测
        6. 标记COMPLETED
        """
        # 1. 选择生态位
        niche = preferred_niche or \
            self.router.select_niche(task_type)
        if niche is None:
            # 全部熔断，任务入队等待
            txn_id = f"{task_type}_{uuid.uuid4().hex[:12]}"
            entry = WALEntry(
                txn_id=txn_id,
                task_type=task_type,
                task_payload=task_payload,
                niche='B_turbo',
                status=TaskStatus.PENDING.value
            )
            self.wal.append(entry)
            raise RuntimeError(
                "所有生态位已熔断，任务已入队等待复苏")

        # 2. 写入WAL
        txn_id = f"{task_type}_{uuid.uuid4().hex[:12]}"
        entry = WALEntry(
            txn_id=txn_id,
            task_type=task_type,
            task_payload=task_payload,
            niche=niche
        )
        self.wal.append(entry)

        # 3. 标记运行中
        self.wal.update_status(txn_id, TaskStatus.RUNNING.value)

        # 4. 执行
        try:
            timeout = self.router.get_timeout(niche)
            result = self._execute_with_timeout(
                handler, task_payload, timeout)

            # 5. 质量检测
            if not QualityChecker.check(result, niche):
                retry = self.wal.increment_retry(txn_id)
                max_retry = EngineConfig.get("max_retry", 3)
                if retry < max_retry:
                    # 切换生态位重试
                    new_niche = 'A_model' \
                        if niche == 'B_turbo' else 'B_turbo'
                    self.wal.update_status(
                        txn_id, TaskStatus.FAILED.value)
                    return self.execute(
                        task_type, task_payload,
                        handler, new_niche)
                else:
                    self.wal.update_status(
                        txn_id, TaskStatus.FAILED.value)
                    raise RuntimeError(
                        f"质量检测失败，已达最大重试次数")

            # 6. 标记完成
            self.wal.update_status(
                txn_id, TaskStatus.COMPLETED.value, result)

            # 记录成本
            cost = result.get('cost', 0.01)
            self.router.record_cost(niche, cost)

            return result

        except Exception as e:
            retry = self.wal.increment_retry(txn_id)
            max_retry = EngineConfig.get("max_retry", 3)
            if retry < max_retry:
                self.wal.update_status(
                    txn_id, TaskStatus.PENDING.value)
                raise RuntimeError(
                    f"任务失败(重试{retry}/{max_retry}): {e}")
            else:
                self.wal.update_status(
                    txn_id, TaskStatus.FAILED.value)
                raise

    def _execute_with_timeout(self, handler: Callable,
                                payload: dict,
                                timeout: int) -> dict:
        """带超时的任务执行"""
        result_box = {}
        error_box = {}

        def worker():
            try:
                result_box['result'] = handler(payload)
            except Exception as e:
                error_box['error'] = e

        thread = threading.Thread(target=worker)
        thread.start()
        thread.join(timeout=timeout)

        if thread.is_alive():
            raise TimeoutError(
                f"任务执行超时（{timeout}秒）")

        if 'error' in error_box:
            raise error_box['error']

        return result_box.get('result', {})

    def _checkpoint_loop(self):
        """定期检查点：清理WAL + 检查熔断恢复"""
        interval = EngineConfig.get(
            "checkpoint_interval", 60)
        while self._running:
            time.sleep(interval)
            try:
                self.wal.checkpoint()
                self.router.check_recovery()
            except Exception:
                pass

    def _install_signal_handlers(self):
        """安装信号处理器，实现优雅关闭"""
        if self._signals_installed:
            return

        def handler(signum, frame):
            print(f"\n[Engine] 收到信号 {signum}")
            self.shutdown()
            os._exit(0)

        signal.signal(signal.SIGINT, handler)
        signal.signal(signal.SIGTERM, handler)
        self._signals_installed = True
```

## 架构总结

```
┌─────────────────────────────────────────────────────────────────┐
│                        Engine（引擎主类）                        │
│                                                                 │
│  boot()                                                         │
│    ├─ EngineConfig.reload()        配置热加载                   │
│    ├─ RecoveryManager.detect_and_recover()                      │
│    │    ├─ VitalsMonitor.detect_crash()  心跳过期检测            │
│    │    ├─ WAL.get_unfinished()          获取未完成任务          │
│    │    ├─ RUNNING → PENDING             重置中断任务            │
│    │    └─ 重放PENDING任务               幂等执行               │
│    ├─ VitalsMonitor.start()         心跳线程启动               │
│    ├─ checkpoint线程启动            定期清理WAL                │
│    ├─ SupabaseSync.start()          远程状态同步               │
│    └─ signal handlers              SIGINT/SIGTERM优雅关闭     │
│                                                                 │
│  execute(task)                                                  │
│    ├─ EcologicalRouter.select_niche()  选择生态位               │
│    │    └─ B_turbo优先 → A_model降级 → None熔断                │
│    ├─ WAL.append(entry)               写前日志(PENDING)        │
│    ├─ WAL.update_status(RUNNING)       标记运行中              │
│    ├─ _execute_with_timeout()          执行(300s/600s)         │
│    ├─ QualityChecker.check()          质量检测                 │
│    │    └─ B模型跳过reasoning占比检测                          │
│    └─ WAL.update_status(COMPLETED)    标记完成                │
│                                                                 │
│  shutdown()                                                     │
│    ├─ VitalsMonitor.stop()           写入clean标记             │
│    ├─ SupabaseSync.stop()            停止远程同步              │
│    └─ WAL.checkpoint()               最终清理                  │
└─────────────────────────────────────────────────────────────────┘

崩溃恢复时序：

  t0: 引擎正常运行，心跳每3s写入
  t1: 进程被kill -9，心跳停止
  t2: 重启 → boot() → detect_crash()
      ├─ STATE_FILE无clean标记
      ├─ HEARTBEAT_FILE时间戳过期(>15s)
      ├─ 旧PID不存在
      └─ 判定: CRASHED
  t3: WAL.get_unfinished() → 5个PENDING + 3个RUNNING
  t4: 3个RUNNING → 验证checksum → 重置为PENDING
  t5: 重放8个PENDING任务（幂等执行）
  t6: 恢复完成，引擎正常运行
```

**关键设计决策**：

| 决策 | 选择 | 原因 |
|------|------|------|
| WAL底层存储 | SQLite (WAL模式) | 自带ACID，崩溃后自动恢复到一致状态 |
| 原子文件写入 | write→fsync→rename | 跨平台原子操作，防止半写入 |
| 心跳检测 | 内部守护线程 | 不依赖外部进程，纯引擎自检 |
| 任务幂等 | txn_id主键 + INSERT OR IGNORE | 安全重放，不会产生副作用 |
| 超时控制 | B_turbo=300s, A_model=600s | 避坑历史经验 |
| 质量检测 | 按生态位差异化配置 | B模型跳过reasoning占比检测 |
| 熔断恢复 | 5分钟冷却后自动恢复 | 环境复苏机制 |
