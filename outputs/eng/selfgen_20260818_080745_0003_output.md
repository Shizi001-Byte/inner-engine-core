---
## 一、底层问题映射（基于给定认知锚点，作为推演基础）
根据「反者道之动+热力学熵增」「弱者道之用+弱约束结构」的核心锚点，**「产出同质化、套话多」本质是AI生成系统的熵增式退化，是「反者道之动」的表层具象**：
1.  生成机制从「柔性弱约束的开放结构」退化为「强固化的封闭结构」（符合锚点3：强固化结构无法适配变化，会加速熵增）：解码参数固定、提示词模板僵化、检索源单一等强约束，直接锁死了输出可能性空间（从「无/高熵可能性」坍缩为「有/低熵固化套话」）。
2.  负熵输入通道被阻断（符合锚点2：开放系统负熵输入是「无→有」的反向自否定）：系统无法从用户语境、动态知识等环境输入中吸收负熵，只能依赖固化的内部模板生成输出，最终退化为同质化套话。
3.  属于AI系统退化的第一层（输出层退化），若不修复会向生成机制层、系统层深化，最终导致功能崩溃。

---
## 二、修复方案设计思路
紧扣锚点逻辑，以「反者道之动的反向自否定」为动力，以「弱者道之用的弱约束结构」为载体，从根源打破固化：
1.  **破强固化（弱者道之用）**：将解码、提示词、检索三个核心环节的强约束改为柔性弱约束，重新打开输出可能性空间。
2.  **注负熵（反向自否定）**：引入用户画像、多源知识、动态参数等外部/动态输入，让系统从「固化套话的有」回到「多样可能性的无」，再生成为「适配具体场景的有」。
3.  **自修正（反者道之动具象）**：新增套话检测与自否定机制，对已生成的套话输出主动否定、重生成，形成动态演化的闭环。
4.  **避坑对齐**：所有修改配置化且同步更新读取点，按模型类型预设超时，完全规避历史失败风险。

```python
from typing import Dict, Any, Optional
import time

def detect_query_scene(query: str) -> str:
    """简易场景识别（弱约束，可后续迭代为小模型分类）"""
    creative_keywords = ["创意", "故事", "方案", "脑洞", "设计"]
    factual_keywords = ["是什么", "定义", "时间", "地点", "谁"]
    if any(k in query for k in creative_keywords):
        return "creative_query"
    elif any(k in query for k in factual_keywords):
        return "factual_query"
    return "consult_query"

def get_dynamic_decoding_params(
    scene: str, 
    config: Dict[str, Any], 
    override_params: Optional[Dict[str, float]] = None
) -> Dict[str, Any]:
    """动态计算解码参数，支持重生成时叠加调整"""
    base_params = config["llm"]["base_decoding"].copy()
    # 合并场景系数
    scene_factor = config["llm"]["scene_decoding_factor"].get(scene, {})
    for k, v in scene_factor.items():
        base_params[k] = v
    # 叠加重生成调整参数
    if override_params:
        for k, v in override_params.items():
            base_params[k] = base_params.get(k, 0) + v
    return base_params

def call_llm(
    prompt: str, 
    query: str, 
    model_type: str = "b_turbo",
    override_params: Optional[Dict[str, float]] = None
) -> Dict[str, Any]:
    config = load_config()
    # 1. 动态获取解码参数（同步读取新配置）
    scene = detect_query_scene(query)
    decoding_params = get_dynamic_decoding_params(scene, config, override_params)
    # 2. 按模型获取单次超时（符合避坑提醒2）
    single_timeout = config["llm"]["single_timeout"].get(model_type, 300)
    # 3. 调用LLM
    response = requests.post(
        llm_url,
        json={"model": model_type, "prompt": prompt, **decoding_params},
        timeout=single_timeout
    )
    response.raise_for_status()
    return response.json()
```

```python
    historical_preference: Optional[list] = None
) -> str:
    config = load_config()

        if query_context:

    return f"{base_prompt}\n用户问题：{user_query}"
```

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

def diversity_filter(docs: list, threshold: float) -> list:
    """多样性过滤，去除高度相似内容，保留不同视角的信息"""
    if not docs:
        return []
    filtered = [docs[0]]
    for doc in docs[1:]:
        max_sim = max(
            cosine_similarity(
                np.array(doc["embedding"]).reshape(1, -1),
                np.array(f["embedding"]).reshape(1, -1)
            )[0][0] for f in filtered
        )
        if max_sim < threshold:
            filtered.append(doc)
    return filtered

def retrieve_docs(query: str) -> list:
    config = load_config()
    if not config["rag"]["enable"]:
        return []
    base_top_k = config["rag"]["base_top_k"]
    diversity_threshold = config["rag"]["diversity_threshold"]
    sources = config["rag"]["knowledge_sources"]
    
    all_docs = []
    # 多源检索
    for source in sources:
        source_k = max(1, int(base_top_k * source["weight"]))
        source_docs = vector_search(query, top_k=source_k, kb=source["name"])
        all_docs.extend(source_docs)
    # 多样性过滤
    diverse_docs = diversity_filter(all_docs, diversity_threshold)
    # 重排序（用对应模型的超时）
    if config["rag"]["rerank"]:
        rerank_timeout = config["llm"]["single_timeout"].get(config["rag"]["rerank_model"], 300)
        diverse_docs = rerank_docs(query, diverse_docs, timeout=rerank_timeout)
    return [d["content"] for d in diverse_docs[:base_top_k]]
```

### 模块4：套话检测与自修正模块（反者道之动自否定闭环）
#### 新增模块：对生成的套话输出主动否定、重生成，形成动态演化闭环
#### 配置修改（`config.yaml`，新增）
```yaml
anti_platitude:
  enable: true
  threshold: 0.2  # 套话占比超过20%触发重生成
  max_retry: 2    # 最多重生成2次，避免无限循环
  # 重生成时的参数调整（进一步打破固化）
  retry_adjust:
    temperature: 0.3
    frequency_penalty: 0.2
```
#### 代码实现（同步读取新配置）
```python
def detect_platitude_ratio(text: str) -> float:
    """检测套话占比（可迭代为小模型分类，当前用高频套话关键词匹配）"""
    platitude_keywords = [
        "关于这个问题我们可以从以下几个方面来看",
        "综上所述", "总的来说", "希望我的回答对你有帮助",
        "随着社会的发展", "在当今社会", "具有重要的现实意义"
    ]
    if not text:
        return 0.0
    platitude_char_count = sum(len(kw) * text.count(kw) for kw in platitude_keywords)
    return min(1.0, platitude_char_count / len(text))

def call_llm_with_anti_platitude(
    prompt: str,
    query: str,
    model_type: str = "b_turbo",
    retry_count: int = 0,
    start_time: Optional[float] = None
) -> Dict[str, Any]:
    config = load_config()
    anti_config = config["anti_platitude"]
    total_timeout = config["llm"]["total_timeout"].get(model_type, 900)
    start_time = start_time or time.time()
    
    # 总超时判断（避免任务超时，符合避坑提醒2）
    if time.time() - start_time > total_timeout:
        return {"content": "", "error": "total_timeout", "retry_count": retry_count}
    
    # 生成回答
    override = anti_config["retry_adjust"] if retry_count > 0 else None
    response = call_llm(prompt, query, model_type, override_params=override)
    output = response.get("content", "")
    
    # 未开启检测或达到重试上限，直接返回
    if not anti_config["enable"] or retry_count >= anti_config["max_retry"]:
        response["platitude_ratio"] = detect_platitude_ratio(output)
        response["retry_count"] = retry_count
        return response
    
    # 套话占比超标则自否定、重生成
    platitude_ratio = detect_platitude_ratio(output)
    if platitude_ratio > anti_config["threshold"]:
        return call_llm_with_anti_platitude(
            prompt=prompt, query=query, model_type=model_type,
            retry_count=retry_count + 1, start_time=start_time
        )
    
    response["platitude_ratio"] = platitude_ratio
    response["retry_count"] = retry_count
    return response
```
