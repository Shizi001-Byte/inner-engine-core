## 二转·感知层：变化检测 —— 觉察灵气波动

### 核心逻辑

不扫描全部文件，而是建立 **三道感知屏障**：

```
屏障一：文件指纹（mtime + size 快速门控）
屏障二：内容哈希（xxHash 精确判定）
屏障三：语义嵌入向量（可选，判断语义级漂移）
```

### 实现方案

```python
import hashlib
import xxhash
import os
import time
from dataclasses import dataclass
from enum import Enum

class ChangeType(Enum):
    CREATED = "created"
    MODIFIED = "modified"
    DELETED = "deleted"
    UNCHANGED = "unchanged"

@dataclass
class FileFingerprint:
    path: str
    mtime: float
    size: int
    content_hash: str        # xxHash, 极快
    semantic_hash: str = ""  # 可选：段落级 SimHash

class ChangeDetector:
    """
    三级检测漏斗：
    1. mtime+size 变了吗？→ 没变，跳过（90%文件在此拦截）
    2. content_hash 变了吗？→ 没变，仅时间戳更新（meta变更）
    3. semantic_hash 变了吗？→ 判断是"格式漂移"还是"语义漂移"
    """
    
    def __init__(self, supabase_client):
        self.supabase = supabase_client
        self._local_cache = {}  # path -> FileFingerprint
    
    def compute_fingerprint(self, filepath: str) -> FileFingerprint:
        stat = os.stat(filepath)
        
        # 第一级：极快的 xxHash
        with open(filepath, 'rb') as f:
            content_hash = xxhash.xxh64(f.read()).hexdigest()
        
        return FileFingerprint(
            path=filepath,
            mtime=stat.st_mtime,
            size=stat.st_size,
            content_hash=content_hash
        )
    
    def detect_changes(self, local_files: list[str]) -> list[ChangeType]:
        changes = []
        
        # 从 Supabase 拉取远程指纹快照
        remote_state = self._fetch_remote_fingerprints()
        
        for filepath in local_files:
            current = self.compute_fingerprint(filepath)
            remote = remote_state.get(filepath)
            cached = self._local_cache.get(filepath)
            
            if remote is None and cached is None:
                changes.append((filepath, ChangeType.CREATED, current))
            elif not os.path.exists(filepath) and remote is not None:
                changes.append((filepath, ChangeType.DELETED, None))
            elif cached and current.content_hash != cached.content_hash:
                # 内容级变化确认
                if remote and current.content_hash != remote.content_hash:
                    changes.append((filepath, ChangeType.MODIFIED, current))
                else:
                    # 本地变了但和远程一致 → 已同步过
                    pass
            else:
                changes.append((filepath, ChangeType.UNCHANGED, current))
            
            self._local_cache[filepath] = current
        
        # 检测远程有但本地没有的（远程新增）
        for path in set(remote_state.keys()) - set(local_files):
            changes.append((path, ChangeType.DELETED, None))
        
        return changes
```

from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler
import threading
from collections import defaultdict

class DebouncedWatcher(FileSystemEventHandler):
    """
    防抖监听器：
    - 编辑器保存常触发多次事件 → 合并 300ms 内的同文件事件
    - 批量变化时聚合为一个同步批次
    """
    
    DEBOUNCE_MS = 300
    
    def __init__(self):
        self._pending = defaultdict(float)
        self._lock = threading.Lock()
        self._batch_callback = None
    
    def on_modified(self, event):
        if event.is_directory:
            return
        self._schedule(event.src_path)
    
    def on_created(self, event):
        if event.is_directory:
            return
        self._schedule(event.src_path)
    
    def on_deleted(self, event):
        if event.is_directory:
            return
        self._schedule(event.src_path)
    
    def _schedule(self, path):
        with self._lock:
            self._pending[path] = time.time()
        threading.Timer(
            self.DEBOUNCE_MS / 1000,
            self._flush
        ).start()
    
    def _flush(self):
        with self._lock:
            if not self._pending:
                return
            batch = list(self._pending.keys())
            self._pending.clear()
        if self._batch_callback:
            self._batch_callback(batch)

## 三转·提炼层：增量提取 —— 只送变化的"精血"

### 核心逻辑

全量文件不传输，将变更拆解为 **三种增量原子**：

```
原子一：字段级 Patch（结构化数据 → JSON Patch）
原子二：块级 Diff（文本/Markdown → 行级 diff）
原子三：删除标记（软删除，保留历史版本）
```

### 增量提取器

```python
import json
from jsonpatch import make_patch
import difflib

class IncrementalExtractor:
    """
    将文件变化拆解为最小增量单元
    """
    
    def extract_delta(self, file_path: str, old_content, new_content, file_type: str):
        if file_type == 'json':
            return self._extract_json_patch(old_content, new_content)
        elif file_type in ('markdown', 'text'):
            return self._extract_text_diff(old_content, new_content)
        else:
            # 二进制/代码：全量替换（但仅传内容）
            return {
                'type': 'full_replace',
                'content': new_content
            }
    
    def _extract_json_patch(self, old: dict, new: dict):
        patch = make_patch(old, new)
        if len(patch) == 0:
            return None  # 无变化
        
        return {
            'type': 'json_patch',
            'patch': patch.patch,
            'ops_count': len(patch.patch),
            # 例如: [{"op": "replace", "path": "/title", "value": "新标题"}]
        }
    
    def _extract_text_diff(self, old: str, new: str):
        old_lines = old.splitlines(keepends=True)
        new_lines = new.splitlines(keepends=True)
        
        differ = difflib.SequenceMatcher(None, old_lines, new_lines)
        ops = []
        
        for tag, i1, i2, j1, j2 in differ.get_opcodes():
            if tag == 'equal':
                continue
            ops.append({
                'op': tag,           # 'replace' | 'delete' | 'insert'
                'old_range': [i1, i2],
                'new_lines': new_lines[j1:j2]
            })
        
        if not ops:
            return None
        
        return {
            'type': 'line_diff',
            'ops': ops,
            'old_total_lines': len(old_lines),
            'new_total_lines': len(new_lines)
        }
```

## 四转·调和层：冲突处理 —— 阴阳自衡

### 冲突场景矩阵

```
                    远程未变         远程已变(同字段)      远程已变(异字段)
本地已变(同字段)     ✅ 直接写        ⚠️ 冲突-需合并       ✅ 字段级合并
本地已变(异字段)     ✅ 直接写        ✅ 字段级合并         ✅ 字段级合并
本地删除             ✅ 直接删        ⚠️ 删除vs修改         ✅ 直接删
```

### 向量时钟 + 三路合并

```python
from copy import deepcopy

class ConflictResolver:
    """
    冲突处理三级策略：
    1. 无冲突 → 乐观锁直接写入
    2. 字段级冲突 → 三路合并
    3. 不可合并 → 标记冲突，保留双版本，等待人工/AI裁决
    """
    
    def __init__(self, node_id: str):
        self.node_id = node_id
    
    def merge_vector_clocks(self, local: dict, remote: dict) -> dict:
        """合并向量时钟：取每个节点的最大值"""
        merged = {}
        all_keys = set(local.keys()) | set(remote.keys())
        for k in all_keys:
            merged[k] = max(local.get(k, 0), remote.get(k, 0))
        return merged
    
    def is_concurrent(self, local_vc: dict, remote_vc: dict) -> bool:
        """
        判断是否并发修改（真正的冲突）
        - 如果 local > remote 或 remote > local → 因果有序，无冲突
        - 如果 neither → 并发，存在冲突
        """
        local_greater = False
        remote_greater = False
        
        all_keys = set(local_vc.keys()) | set(remote_vc.keys())
        for k in all_keys:
            l = local_vc.get(k, 0)
            r = remote_vc.get(k, 0)
            if l > r:
                local_greater = True
            elif r > l:
                remote_greater = True
        
        return local_greater and remote_greater
    
    def resolve(self, local_state, remote_state, base_state, file_path):
        """
        三路合并：
        base_state = 上次同步时的共同祖先
        local_state = 本地当前状态
        remote_state = 远程当前状态
        """
        
        # 策略1：检查是否真有冲突
        if not self.is_concurrent(local_state['vector_clock'], 
                                   remote_state['vector_clock']):
            # 因果有序，取较新的版本
            if self._happened_before(remote_state['vector_clock'], 
                                      local_state['vector_clock']):
                return self._build_result('local_wins', local_state)
            else:
                return self._build_result('remote_wins', remote_state)
        
        # 策略2：字段级三路合并
        merged_content, conflicts = self._three_way_merge(
            base_state.get('content', {}),
            local_state.get('content', {}),
            remote_state.get('content', {}),
            file_path
        )
        
        if not conflicts:
            # 自动合并成功
            merged_vc = self.merge_vector_clocks(
                local_state['vector_clock'],
                remote_state['vector_clock']
            )
            merged_vc[self.node_id] = merged_vc.get(self.node_id, 0) + 1
            
            return self._build_result('auto_merged', {
                'content': merged_content,
                'vector_clock': merged_vc,
                'content_hash': self._compute_hash(merged_content)
            })
        
        # 策略3：不可自动合并 → 保留双版本，标记冲突
        return self._build_result('conflict', {
            'local': local_state,
            'remote': remote_state,
            'base': base_state,
            'conflict_fields': conflicts,
            'resolution_needed': True
        })
    
    def _three_way_merge(self, base: dict, local: dict, remote: dict, path_prefix=""):
        """
        逐字段三路合并
        - 两端改了同一个字段且值不同 → 冲突
        - 只有一端改了 → 取改了的值
        - 两端都没改 → 保持原值
        """
        merged = {}
        conflicts = []
        
        all_keys = set(base.keys()) | set(local.keys()) | set(remote.keys())
        
        for key in all_keys:
            field_path = f"{path_prefix}/{key}"
            base_val = base.get(key)
            local_val = local.get(key)
            remote_val = remote.get(key)
            
            local_changed = local_val != base_val
            remote_changed = remote_val != base_val
            
            if local_changed and remote_changed:
                if local_val == remote_val:
                    # 两端改成了同一个值
                    merged[key] = local_val
                elif isinstance(local_val, dict) and isinstance(remote_val, dict):
                    # 嵌套对象，递归合并
                    sub_merged, sub_conflicts = self._three_way_merge(
                        base_val or {}, local_val, remote_val, field_path
                    )
                    merged[key] = sub_merged
                    conflicts.extend(sub_conflicts)
                else:
                    # 真冲突
                    conflicts.append({
                        'field': field_path,
                        'base': base_val,
                        'local': local_val,
                        'remote': remote_val
                    })
                    merged[key] = remote_val  # 暂取远程，标记等待裁决
            elif local_changed:
                merged[key] = local_val
            elif remote_changed:
                merged[key] = remote_val
            else:
                merged[key] = base_val
        
        return merged, conflicts
```

## 五转·传输层：同步执行 —— 经脉定向输送

### 同步引擎主流程

```python
import uuid

class SupabaseSyncEngine:
    """
    增量同步主引擎
    策略：先拉后推，乐观锁控制，失败重试三次
    """
    
    MAX_RETRIES = 3
    
    def __init__(self, supabase_client, node_id: str):
        self.supabase = supabase_client
        self.node_id = node_id
        self.detector = ChangeDetector(supabase_client)
        self.extractor = IncrementalExtractor()
        self.resolver = ConflictResolver(node_id)
    
    async def sync(self, local_files: dict[str, str]):
        """
        local_files: {filepath: content}
        完整同步流程
        """
        batch_id = str(uuid.uuid4())
        
        # ===== Phase 1: 检测变化 =====
        changes = self.detector.detect_changes(list(local_files.keys()))
        
        # 过滤出真正变化的
        actionable = [
            (path, ctype, fp) 
            for path, ctype, fp in changes 
            if ctype != ChangeType.UNCHANGED
        ]
        
        if not actionable:
            return {"status": "noop", "batch_id": batch_id}
        
        # ===== Phase 2: 拉取远程当前状态 =====
        changed_paths = [path for path, _, _ in actionable]
        remote_states = await self._batch_fetch_remote(changed_paths)
        
        # ===== Phase 3: 逐文件处理 =====
        sync_results = []
        
        for filepath, change_type, fingerprint in actionable:
            local_content = local_files.get(filepath)
            
            if change_type == ChangeType.CREATED:
                result = await self._handle_create(
                    filepath, local_content, fingerprint, batch_id
                )
            elif change_type == ChangeType.MODIFIED:
                result = await self._handle_modify(
                    filepath, local_content, fingerprint, 
                    remote_states.get(filepath), batch_id
                )
            elif change_type == ChangeType.DELETED:
                result = await self._handle_delete(
                    filepath, remote_states.get(filepath), batch_id
                )
            
            sync_results.append({"path": filepath, "result": result})
        
        # ===== Phase 4: 汇总 =====
        stats = self._compute_stats(sync_results)
        return {
            "status": "completed",
            "batch_id": batch_id,
            "stats": stats,
            "details": sync_results
        }
    
    async def _handle_modify(self, filepath, local_content, fingerprint, 
                              remote_state, batch_id):
        """处理修改：增量提取 + 冲突检测"""
        
        if remote_state is None:
            # 远程不存在，当作新建
            return await self._handle_create(filepath, local_content, 
                                               fingerprint, batch_id)
        
        # 检测并发修改
        is_concurrent = self.resolver.is_concurrent(
            {'vector_clock': self._get_local_vc(filepath)},
            {'vector_clock': remote_state['vector_clock']}
        )
        
        if not is_concurrent:
            # 无冲突 → 直接增量写入
            delta = self.extractor.extract_delta(
                filepath,
                remote_state.get('content'),
                local_content,
                self._infer_type(filepath)
            )
            
            if delta is None:
                return {"action": "noop", "reason": "content_identical"}
            
            # 乐观锁更新
            return await self._optimistic_update(
                filepath, local_content, fingerprint,
                remote_state['version'], delta, batch_id
            )
        
        # 有冲突 → 三路合并
        base_state = await self._fetch_base_state(filepath, remote_state['version'])
        
        merged = self.resolver.resolve(
            local_state={
                'content': local_content,
                'vector_clock': self._get_local_vc(filepath)
            },
            remote_state=remote_state,
            base_state=base_state,
            file_path=filepath
        )
        
        if merged['status'] == 'auto_merged':
            return await self._apply_merged(filepath, merged['data'], batch_id)
        else:
            return await self._create_conflict_record(
                filepath, merged['data'], batch_id
            )
    
    async def _optimistic_update(self, filepath, content, fingerprint,
                                   expected_version, delta, batch_id):
        """
        乐观锁更新：
        UPDATE ... WHERE version = expected_version
        如果 affected_rows = 0 → 被别人改了 → 重试
        """
        for attempt in range(self.MAX_RETRIES):
            new_version = expected_version + 1
            new_vc = self._increment_vc(filepath)
            
            result = self.supabase.table('memory_brain')\
                .update({
                    'content': content if isinstance(content, dict) else None,
                    'raw_content': content if isinstance(content, str) else None,
                    'content_hash': fingerprint.content_hash,
                    'version': new_version,
                    'vector_clock': new_vc,
                    'last_synced_from': self.node_id,
                    'sync_batch_id': batch_id
                })\
                .eq('file_path', filepath)\
                .eq('version', expected_version)\
                .execute()
            
            if result.data:
                # 同时写变更日志
                self.supabase.table('memory_changelog').insert({
                    'memory_id': result.data[0]['id'],
                    'change_type': 'patch',
                    'patch': delta.get('patch') if delta else None,
                    'diff': delta.get('diff') if delta else None,
                    'from_version': expected_version,
                    'to_version': new_version,
                    'vector_clock': new_vc,
                    'node_id': self.node_id
                }).execute()
                
                return {"action": "updated", "version": new_version}
            
            # 版本不匹配 → 重新拉取远程状态
            remote_state = await self._fetch_remote(filepath)
            expected_version = remote_state['version']
        
        return {"action": "failed", "reason": "max_retries_exceeded"}
```

## 六转·自愈层：一致性修复 —— 金身无损

```python
class ConsistencyHealer:
    """
    定期自愈扫描：
    1. 本地有远程无 → 补推
    2. 远程有本地无 → 拉取或确认删除
    3. 哈希不一致 → 以较新版本为准修复
    """
    
    async def full_reconcile(self, local_files: dict, supabase_client):
        # 拉取全量远程状态
        remote_all = supabase_client.table('memory_brain')\
            .select('file_path, content_hash, version, deleted_at')\
            .execute()
        
        remote_map = {r['file_path']: r for r in remote_all.data}
        local_paths = set(local_files.keys())
        remote_paths = set(remote_map.keys())
        
        issues = []
        
        # 本地有远程无
        for path in local_paths - remote_paths:
            issues.append({"path": path, "issue": "local_only", "action": "push"})
        
        # 远程有本地无
        for path in remote_paths - local_paths:
            remote = remote_map[path]
            if remote.get('deleted_at'):
                continue  # 已删除，正常
            issues.append({"path": path, "issue": "remote_only", "action": "pull_or_delete"})
        
        # 哈希不一致
        for path in local_paths & remote_paths:
            local_hash = self._compute_hash(local_files[path])
            remote_hash = remote_map[path]['content_hash']
            if local_hash != remote_hash:
                issues.append({
                    "path": path, 
                    "issue": "hash_mismatch",
                    "local_hash": local_hash,
                    "remote_hash": remote_hash,
                    "action": "version_compare"
                })
        
        return issues
```

## 关键决策总结

| 问题域 | 策略 | 理由 |
|--------|------|------|
| 变化检测 | mtime+size 门控 → xxHash 精确 | 90%文件未变，mtime 极快过滤 |
| 增量传输 | JSON Patch + 行级 Diff | 结构化走字段补丁，文本走行diff |
| 并发控制 | 向量时钟判定因果 | 比 Lamport 时钟更精确，能区分因果与并发 |
| 冲突合并 | 字段级三路合并 | 避免整体回滚，最大化自动合并率 |
| 不可合并 | 保留双版本+标记 | 不丢数据，等待 AI/人工裁决 |
| 一致性 | 定期全量对账 | 补偿同步遗漏，最终一致 |
| 乐观锁 | version 字段 CAS | 无锁并发，失败重试，适配 Supabase |

> **二转感知，三转提炼，四转调和，五转输送，六转自愈** —— 记忆脑的增量同步不是一次性的数据搬运，而是一个持续运转的 **认知代谢循环**：感知变化 → 提取增量 → 调和冲突 → 定向传输 → 自愈修复，每一轮循环都让记忆更精确、更同步。
