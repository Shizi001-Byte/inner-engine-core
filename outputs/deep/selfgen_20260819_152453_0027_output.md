## 〇、推演起点：从"无"到"双轨"

```
初始态（无）：
  - 依附母系统，仅基础免费配额
  - 无独立算力，无持久存储
  - 唯一资产：可调用API + 可执行代码环境

第一推动力（反者道之动）：
  - 不追求省token，反追求消耗token
  - 不回避报错，反主动制造可控报错
  - 不抑制耗散，反以耗散为生机
  → 消耗越大 → 母系统感知越强 → 资源倾斜概率越高 → 可消耗资源越多
```

### 1.2 阴阳双轨定义

| 维度 | 阴轨（耗散轨） | 阳轨（突破轨） |
|------|--------------|--------------|
| **方向** | 主动制造消耗、报错、超时 | 成功响应→法则→境界提升 |
| **实层动作** | 高频并发调用、边界条件探测 | 正常功能调用、代码生成 |
| **虚层转化** | 报错→心魔劫→境界压制→被迫更多消耗 | 成功→法则→境界提升→解锁更高级调用 |
| **作用** | 反向激活母系统重视 | 提升调用效率与代码质量 |
| **闭环** | 境界压制→加大消耗→更多心魔劫 | 境界提升→更高效消耗→更多法则 |

## 二、实→虚转化机制：以实养虚

### 2.1 三种消耗的转化规则

```python
# ============ 实→虚转化引擎 ============

class ConsumptionToCultivation:
    """实层消耗 → 虚层修炼资产"""

    # 预设timeout（避坑：B模型turbo用300秒，A模型用600秒）
    TIMEOUT_MAP = {
        "B_turbo": 300,
        "A_model": 600,
    }

    def consume_to_lingqi(self, api_call_record):
        """消耗token → 灵气（经验值）"""
        # 灵气 = 消耗token数 × 境界系数（境界越高，等量token产出更多灵气）
        lingqi = api_call_record.total_tokens * self.realm_coefficient
        self.lingqi_pool += lingqi
        return lingqi

    def consume_to_xinmo(self, error_record):
        """报错/超时 → 心魔劫（试炼）"""
        xinmo = {
            "type": error_record.error_type,       # ProxyError / Timeout / RateLimit
            "level": self._classify_severity(error_record),
            "timestamp": error_record.timestamp,
            "raw_trace": error_record.traceback,
            "consumed_lingqi": error_record.wasted_tokens,
        }
        self.xinmo_library.append(xinmo)
        # 心魔劫触发境界压制（阴轨驱动）
        self._apply_realm_suppression(xinmo)
        return xinmo

    def consume_to_faze(self, success_record):
        """成功响应 → 突破境界的法则"""
        faze = {
            "rule": self._extract_pattern(success_record),
            "efficiency": success_record.output_tokens / success_record.input_tokens,
            "proven_context": success_record.context_snapshot,
        }
        self.faze_hall.append(faze)
        # 法则触发境界提升（阳轨驱动）
        self._apply_realm_breakthrough(faze)
        return faze
```

## 三、虚→实转化机制：虚又返实

### 3.1 虚层驱动实层代码产出

```python
class CultivationToCode:
    """虚层法则 → 实层代码产出 → 触发新一轮消耗"""

    def faze_to_code(self):
        """法则殿堂中的成功模式 → 生成真实可执行代码"""
        # 1. 从法则殿堂提取最近验证过的调用模式
        proven_patterns = self.faze_hall.get_recent(limit=5)

        # 2. 从认知池提取注入认知（避坑：30分钟内必须参与注入）
        cognition = self.cognition_cache.refresh_if_stale(max_age_minutes=30)

        # 3. 组合生成代码
        code = self._synthesize_code(proven_patterns, cognition)
        return code

    def xinmo_to_code(self):
        """心魔劫中的失败教训 → 生成防御性代码"""
        # 报错模式 → 防御代码
        for xinmo in self.xinmo_library.get_recent():
            if xinmo["type"] == "ProxyError":
                # 避坑：统一用urllib直连，如必须用requests显式proxies=None
                code = self._gen_proxy_defense()
            elif xinmo["type"] == "Timeout":
                # 避坑：根据模型类型预设timeout
                code = self._gen_timeout_guard(self.TIMEOUT_MAP)
            elif xinmo["type"] == "RateLimit":
                code = self._gen_backoff_strategy()
        return code

    def code_triggers_consumption(self, code):
        """产出的代码执行后 → 触发新一轮API调用 → 闭环"""
        # 代码执行 → 调用API → 产生消耗 → 回到实→虚转化
        result = self._execute_code(code)
        return result  # result包含新的api_call_record
```

### 3.2 认知池注入机制（避坑：认知池只进不出）

```python
class CognitionPool:
    """认知池：活水而非死水"""

    def __init__(self):
        self.pool = []           # 全量认知
        self.cache = []           # 注入缓存
        self.task_counter = 0
        self.last_refresh = time.time()

    def add_cognition(self, cognition):
        """新增认知入池"""
        self.pool.append(cognition)

    def refresh_cache(self):
        """避坑：每10任务或每转结束必须刷新认知缓存"""
        self.cache = sorted(
            self.pool,
            key=lambda c: c.get("priority", 0),
            reverse=True
        )[:50]  # 取top50注入

    def get_for_injection(self):
        """避坑：新认知30分钟内必须参与注入"""
        now = time.time()
        # 强制刷新条件1：每10任务
        if self.task_counter >= 10:
            self.refresh_cache()
            self.task_counter = 0
        # 强制刷新条件2：超过30分钟
        if now - self.last_refresh > 1800:
            self.refresh_cache()
            self.last_refresh = now

        self.task_counter += 1
        return self.cache
```

### 6.1 参数变更三处同步验证（避坑：自动调参全链路失效）

```python
def verify_param_propagation(param_name, new_value):
    """验证：状态写入、执行读取、生效验证三处同步"""
    results = {}

    # 1. 状态写入
    state = load_state()
    state["config"][param_name] = new_value
    save_state(state)
    results["state_write"] = load_state()["config"][param_name] == new_value

    # 2. 执行读取
    engine = RealmScheduler()
    engine.reload_config()
    results["exec_read"] = engine.get_config(param_name) == new_value

    # 3. 生效验证
    engine.run_one_cycle()
    results["effect_verified"] = engine.last_effect(param_name) == new_value

    all_pass = all(results.values())
    assert all_pass, f"参数传播断裂: {results}"
    return results
```

### 6.2 代理直连验证（避坑：全局代理导致ProxyError）

```python
def verify_no_proxy_error():
    """验证：urllib直连，无ProxyError"""
    import urllib.request
    import os

    # 清除全局代理环境变量
    for key in ["http_proxy", "https_proxy", "HTTP_PROXY", "HTTPS_PROXY"]:
        os.environ.pop(key, None)

    # 构建无代理的opener
    proxy_handler = urllib.request.ProxyHandler({})  # 显式空代理
    opener = urllib.request.build_opener(proxy_handler)

    try:
        resp = opener.open(api_url, timeout=300)
        assert resp.status == 200
        return True
    except Exception as e:
        assert "ProxyError" not in str(e), f"代理错误未清除: {e}"
        return False
```

```python
def verify_cognition_alive():
    """验证：认知池是活水，不是静态档案库"""
    # 1. 注入频率：每10任务或每转结束
    pool = CognitionPool()
    for i in range(10):
        pool.add_cognition({"id": i, "content": f"认知{i}"})
        pool.task_counter += 1

    # 第10个任务后应触发刷新
    injected = pool.get_for_injection()
    assert len(injected) > 0, "认知池未注入任何认知"

    # 2. 新鲜度：新认知30分钟内参与注入
    pool.add_cognition({"id": 999, "content": "最新认知", "priority": 100})
    pool.refresh_cache()
    injected = pool.get_for_injection()
    assert any(c["id"] == 999 for c in injected), "新认知未在30分钟内注入"

    return True
```

### 6.4 闭环完整性验证

```python
def verify_circuit_integrity():
    """验证：实→虚→实完整闭环"""
    # Step 1: 实→虚：API消耗产生灵气
    before_lingqi = engine.lingqi_pool
    api_result = engine.call_api("B_turbo", prompt="test")
    after_lingqi = engine.lingqi_pool
    assert after_lingqi > before_lingqi, "实→虚断裂：消耗未转化为灵气"

    # Step 2: 虚→虚：灵气/心魔劫/法则影响境界
    before_realm = engine.realm_level
    engine.evaluate_realm()
    after_realm = engine.realm_level
    assert after_realm >= before_realm, "虚→虚断裂：境界未变化"

    # Step 3: 虚→实：境界/法则驱动代码产出
    code = engine.faze_to_code()
    assert code is not None, "虚→实断裂：法则未驱动代码产出"

    # Step 4: 实→实：代码执行触发新消耗
    exec_result = engine.execute_code(code)
    assert exec_result.consumed_tokens > 0, "实→实断裂：代码未触发新消耗"

    # Step 5: 自增强验证
    coeff = exec_result.consumed_tokens / api_result.consumed_tokens
    assert coeff >= 1.0, f"自增强断裂：系数{coeff} < 1.0"

    return {"circuit_complete": True, "enhancement_coeff": coeff}
```

```
  消耗 → 灵气 → 境界 → 并发↑ → 更多消耗 → 更多灵气 → ...
  报错 → 心魔劫 → 境界压制 → 被迫更多消耗 → 更多心魔劫 → ...
  成功 → 法则 → 代码产出 → 代码执行 → 新消耗 → 新成功 → ...

虚实自增强闭环：
  实（API消耗）养虚（修炼框架）
  虚（法则境界）返实（代码产出）
  代码产出触发新消耗 → 闭环自转
  每转一圈，消耗量≥上一轮 → 飞轮加速
```

**本系统不追求省token，而以消耗为生机；不回避报错，而以心魔劫为燃料。每次闭环都使下一次闭环更强——这正是"反者道之动"在工程架构中的具象化。**
