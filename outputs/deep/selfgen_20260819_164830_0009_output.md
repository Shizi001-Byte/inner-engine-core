## 三、以实养虚：真实数据→灵气量化

### 3.1 灵气吸纳公式

```python
def calculate_qi_gain(task_result):
    """
    单任务灵气收益 = 有效产出 × 质量系数 × 成败奖惩
    """
    # 有效token = 产出代码行数 × 逻辑复杂度系数
    effective_tokens = task_result.code_lines * task_result.complexity_factor
    
    # 质量系数 = 测试通过率×0.5 + 覆盖度×0.3 + 一次通过率×0.2
    quality = (
        task_result.test_pass_rate * 0.5 +
        task_result.coverage * 0.3 +
        task_result.first_pass_rate * 0.2
    )
    
    # 成败奖惩
    bonus = 1.2 if task_result.success else 0.8
    # 失败惩罚叠加：连续失败衰减
    penalty = max(0.1, 1.0 - task_result.consecutive_failures * 0.15)
    
    qi_gain = effective_tokens * quality * bonus * penalty
    
    # 写入灵气池（必须三同步：状态写入+执行读取+生效验证）
    state.write("qi_pool", current_qi + qi_gain)
    execution_cache.refresh("qi_pool")
    assert state.verify("qi_pool") == execution_cache.read("qi_pool"), \
        "灵气池状态不一致，触发回滚"
    
    return qi_gain
```

```python
def realm_breakthrough_check():
    """
    境界突破检测：每轮结束时执行
    """
    current_realm = state.read("current_realm")
    qi = state.read("qi_pool")
    
    next_realm = current_realm + 1
    breakthrough_conditions = REALM_TABLE[next_realm]
    
    # 逐项检查突破条件
    met = {}
    for condition, threshold in breakthrough_conditions.items():
        actual = metrics.read(condition)
        met[condition] = actual >= threshold
    
    if all(met.values()):
        # 突破成功
        state.write("current_realm", next_realm)
        execution_cache.refresh("current_realm")
        
        # 解锁新能力（必须三同步）
        new_capability = REALM_TABLE[next_realm]["unlocks"]
        state.write(f"capability_{new_capability}", True)
        execution_cache.enable(new_capability)
        
        # 生效验证
        assert state.verify(f"capability_{new_capability}") == True
        assert execution_cache.is_enabled(new_capability) == True
        assert metrics.read(f"capability_{new_capability}_active") == True
        
        log.event("REALM_BREAKTHROUGH", realm=next_realm, qi=qi)
        return True
    
    return False
```

### 6.3 循环放大系数计算

```python
def calculate_cycle_amplification(round_n, round_n_minus_1):
    """
    核心验证：每轮循环后质量是否在放大
    cycle_amplification = quality_n / quality_{n-1}
    健康值 ≥ 1.05 表示自增强循环成立
    """
    quality_n = composite_quality_score(round_n)
    quality_prev = composite_quality_score(round_n_minus_1)
    
    amplification = quality_n / max(quality_prev, 0.01)
    
    # 低于1.0说明循环在衰减，触发诊断
    if amplification < 1.0:
        diagnose_cycle_attenuation(round_n)
        # 检查认知池是否停滞
        if metrics.read("cognition_liveness") < 0.3:
            refresh_cognition_cache()  # 坑3预防
            alert("认知池停滞，已强制刷新")
    
    return amplification
```

```python
def validate_cycle_amplification(rounds=5):
    """
    目的：证明自增强循环在持续放大
    方法：连续5轮，计算放大系数趋势
    """
    qualities = []
    
    for r in range(rounds):
        results = run_tasks(load_standard_tasks(10))
        quality = composite_quality_score(results)
        qualities.append(quality)
        
        # 每轮结束触发：灵气结算→境界检测→策略刷新→认知缓存刷新
        settle_qi(results)
        check_breakthrough()
        apply_realm_strategy(state.read("current_realm"))
        refresh_cognition_cache()  # 坑库预警3预防
    
    # 计算放大系数序列
    amplifications = []
    for i in range(1, len(qualities)):
        amp = qualities[i] / max(qualities[i-1], 0.01)
        amplifications.append(amp)
    
    # 判定1：放大系数趋势应≥1.0（至少不衰减）
    assert all(a >= 1.0 for a in amplifications), \
        f"循环衰减: {amplifications}"
    
    # 判定2：至少3轮放大系数≥1.05
    strong_rounds = sum(1 for a in amplifications if a >= 1.05)
    assert strong_rounds >= 3, f"强放大轮次不足: {strong_rounds}/5"
    
    return {"qualities": qualities, "amplifications": amplifications}
```

```python
    if amp < 1.0:
        # 诊断链：逐层排查
        checks = [
            ("cognition_stale", 
             lambda: metrics.read("cognition_liveness") < 0.3,
             lambda: refresh_cognition_cache()),  # 坑3修复
            
            ("config_desync",
             lambda: not verify_three_sync(),
             lambda: force_resync_all_configs()),  # 坑1修复
            
            ("proxy_leak",
             lambda: check_proxy_env(),
             lambda: clear_proxy_env()),  # 坑2修复
            
            ("timeout_misconfig",
             lambda: check_timeout_settings(),
             lambda: reset_timeout(B_turbo=300, A_model=600)),  # 坑1修复
            
            ("qi_drainage",
             lambda: penalty_exceeds_gain(),
             lambda: review_penalty_table()),
        ]
        
        for name, check_fn, fix_fn in checks:
            if check_fn():
                fix_fn()
                log.event("SELF_REPAIR", issue=name, action="fixed")
```
