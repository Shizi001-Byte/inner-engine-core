## L4 补丁成品说明
### 补丁基线
针对 `naive_sandbox.py v0.1`（简陋单向执行沙箱，已知问题：遇错即崩溃、无瑕疵利用、无循环机制、无沙箱隔离）升级为**双向流+瑕疵驱动的数字生命沙箱v1.0**，完全符合核心规则：
1. 全链路原生瑕疵为唯一循环燃料，全程无前置清洗，所有「错误」直接转化为能量
2. 双向流闭环：上行流（执行→瑕疵采集→能量池）+ 下行流（能量池→涌现生成→执行）
3. 沙箱隔离，动态生成代码不会影响宿主环境

---
### 完整可运行代码
保存为 `flaw_driven_sandbox_patch.py`，纯Python标准库实现，无第三方依赖，直接可运行：
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
瑕疵驱动双向流数字生命沙箱 v1.0
- 核心规则：原生瑕疵为唯一动力，无前置清洗，双向流循环
- 基线版本：naive_sandbox.py v0.1
- 补丁类型：架构升级补丁
"""
import os
import sys
import ast
import math
import time
import queue
import random
import tokenize
import traceback
from io import StringIO

# ===================== 常量定义 =====================
# 瑕疵类型（全链路原生瑕疵，无人工分类过滤）
FLAW_EXCEPTION = "exception"               # 运行时异常/错误
FLAW_DATA_FLAW = "data_flaw"               # 数据瑕疵（NaN/None/空值/无穷大等）
FLAW_UNFINISHED_MARK = "unfinished_mark"   # 未完成标记（TODO/FIXME/XXX等）
FLAW_LOGIC_GAP = "logic_gap"               # 逻辑断层（空实现/无返回/死代码等）

# 未完成标记关键词（仅做识别，不做过滤）
UNFINISHED_KEYWORDS = ["TODO", "FIXME", "XXX", "HACK", "WIP"]

# 沙箱安全内置函数（最小权限隔离）
SAFE_BUILTINS = {
    'int': int, 'float': float, 'str': str, 'list': list, 'dict': dict,
    'tuple': tuple, 'set': set, 'bool': bool, 'None': None, 'True': True,
    'False': False, 'math': math, 'abs': abs, 'round': round,
    'len': len, 'max': max, 'min': min, 'sum': sum, 'print': print,
}

# ===================== 上行流核心组件：瑕疵采集+能量池 =====================
class FlawProbe:
    """全链路瑕疵探针（上行流采集端：零清洗，所有瑕疵直接转化为能量单元）"""
    def __init__(self):
        pass

    def capture_from_execution(self, func_code: str, exec_error: Exception = None) -> list:
        """从代码执行链路采集瑕疵：异常+未完成标记+逻辑断层，无过滤"""
        flaws = []
        # 1. 采集运行时异常瑕疵（最核心燃料）
        if exec_error is not None:
            flaws.append({
                "flaw_type": FLAW_EXCEPTION,
                "content": f"{type(exec_error).__name__}: {str(exec_error)}",
                "location": traceback.format_exc().split("\n")[-3:-1],
                "energy_level": 10,
                "timestamp": time.time()
            })
        
        # 2. 采集代码未完成标记瑕疵（注释中的待办项也是原生瑕疵）
        try:
            tokens = tokenize.generate_tokens(StringIO(func_code).readline)
            for token in tokens:
                if token.type == tokenize.COMMENT:
                    comment_text = token.string.strip("# ").strip()
                    for kw in UNFINISHED_KEYWORDS:
                        if comment_text.upper().startswith(kw):
                            flaws.append({
                                "flaw_type": FLAW_UNFINISHED_MARK,
                                "content": comment_text,
                                "location": f"line {token.start[0]}",
                                "energy_level": 3,
                                "timestamp": time.time()
                            })
                            break
        except Exception as e:
            # 探针自身异常也属于全链路瑕疵，直接入池（无例外）
            flaws.append({
                "flaw_type": FLAW_EXCEPTION,
                "content": f"FlawProbe tokenize error: {type(e).__name__}: {str(e)}",
                "location": "FlawProbe.capture_from_execution",
                "energy_level": 8,
                "timestamp": time.time()
            })
        
        # 3. 采集逻辑断层瑕疵（空实现/无返回等不直接报错的逻辑缺陷）
        try:
            tree = ast.parse(func_code)
            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    # 检测pass空实现
                    has_pass = any(isinstance(n, ast.Pass) for n in ast.walk(node))
                    if has_pass:
                        flaws.append({
                            "flaw_type": FLAW_LOGIC_GAP,
                            "content": f"Function '{node.name}' contains empty pass statement",
                            "location": f"line {node.lineno}",
                            "energy_level": 7,
                            "timestamp": time.time()
                        })
                    # 检测无返回值函数
                    has_return = any(isinstance(n, ast.Return) for n in ast.walk(node))
                    if not has_return:
                        flaws.append({
                            "flaw_type": FLAW_LOGIC_GAP,
                            "content": f"Function '{node.name}' has no return statement",
                            "location": f"line {node.lineno}",
                            "energy_level": 5,
                            "timestamp": time.time()
                        })
        except Exception as e:
            flaws.append({
                "flaw_type": FLAW_EXCEPTION,
                "content": f"FlawProbe ast parse error: {type(e).__name__}: {str(e)}",
                "location": "FlawProbe.capture_from_execution",
                "energy_level": 8,
                "timestamp": time.time()
            })
        
        # 【核心规则遵守】无前置清洗：所有采集到的瑕疵直接返回，不过滤、不去重、不修正
        return flaws

    def capture_from_data(self, data, location: str = "unknown") -> list:
        """从数据链路采集瑕疵，不修改原始数据，仅提取瑕疵单元"""
        flaws = []
        def _check(value, loc):
            # None值瑕疵
            if value is None:
                flaws.append({
                    "flaw_type": FLAW_DATA_FLAW,
                    "content": "None value detected",
                    "location": loc,
                    "energy_level": 5,
                    "timestamp": time.time()
                })
                return
            # 数值类瑕疵（NaN/无穷大）
            if isinstance(value, (int, float)):
                if math.isnan(value):
                    flaws.append({
                        "flaw_type": FLAW_DATA_FLAW,
                        "content": "NaN value detected",
                        "location": loc,
                        "energy_level": 6,
                        "timestamp": time.time()
                    })
                if math.isinf(value):
                    flaws.append({
                        "flaw_type": FLAW_DATA_FLAW,
                        "content": "Infinity value detected",
                        "location": loc,
                        "energy_level": 6,
                        "timestamp": time.time()
                    })
            # 空容器瑕疵
            if isinstance(value, (str, list, dict, tuple, set)) and len(value) == 0:
                flaws.append({
                    "flaw_type": FLAW_DATA_FLAW,
                    "content": f"Empty {type(value).__name__} detected",
                    "location": loc,
                    "energy_level": 4,
                    "timestamp": time.time()
                })
            # 递归遍历嵌套容器
            if isinstance(value, list):
                for i, item in enumerate(value):
                    _check(item, f"{loc}[{i}]")
            if isinstance(value, dict):
                for k, v in value.items():
                    _check(v, f"{loc}['{k}']")
        _check(data, location)
        # 【核心规则遵守】无前置清洗：全部返回
        return flaws


class FlawEnergyPool:
    """瑕疵能量池（双向流枢纽：上行流终点/下行流起点，零清洗存储所有瑕疵）"""
    def __init__(self):
        self._queue = queue.Queue()  # FIFO队列，无过滤、无去重
        self.total_flaws = 0         # 累计接收瑕疵总数（单调递增，验证用）

    def add_flaws(self, flaws: list):
        """批量加入瑕疵，无任何判断逻辑，全部入池"""
        for flaw in flaws:
            self._queue.put(flaw)
            self.total_flaws += 1

    def get_flaw(self, block=False):
        """取出一个瑕疵作为燃料，空则返回None（无燃料则循环终止）"""
        try:
            return self._queue.get(block=block)
        except queue.Empty:
            return None

    def size(self):
        return self._queue.qsize()

# ===================== 下行流核心组件：沙箱执行+涌现生成 =====================
class SandboxExecutor:
    """沙箱执行环境（下行流执行端：隔离宿主，执行动态生成的内容，产生新瑕疵）"""
    def __init__(self, safe_globals: dict = None):
        self.globals = {"__builtins__": safe_globals or SAFE_BUILTINS}
        self.locals = {}  # 沙箱局部命名空间，完全隔离宿主

    def execute(self, code_str: str, data_var_name: str = "test_data", test_data = None) -> tuple:
        """
        执行动态代码，返回 (执行成功标记, 函数返回结果, 异常对象, 标准输出)
        注意：不主动修复错误，错误由上层探针采集为能量
        """
        # 注入测试数据到沙箱局部空间
        if test_data is not None:
            self.locals[data_var_name] = test_data
        
        # 捕获沙箱输出，不污染宿主
        output_buffer = StringIO()
        original_stdout = sys.stdout
        sys.stdout = output_buffer

        exec_error = None
        exec_success = False
        result = None
        try:
            exec(code_str, self.globals, self.locals)
            # 尝试执行process_input函数（沙箱核心处理函数）
            if "process_input" in self.locals and test_data is not None:
                results = []
                for d in test_data:
                    try:
                        res = self.locals["process_input"](d)
                        results.append(res)
                    except Exception as e:
                        # 函数执行错误由探针统一采集，此处仅占位
                        results.append(None)
                result = results
            exec_success = True
        except Exception as e:
            exec_error = e  # 异常直接抛出给探针采集，不处理
        finally:
            sys.stdout = original_stdout  # 恢复宿主标准输出
        
        output = output_buffer.getvalue()
        return exec_success, result, exec_error, output


class EmergenceGenerator:
    """涌现生成器（下行流核心：以瑕疵为燃料生成新内容，目标是产生更多新瑕疵而非修复）"""
    def __init__(self):
        self.generated_code_count = 0
        self.generated_data_count = 0

    def generate_code_patch(self, base_code: str, flaw: dict) -> str:
        """基于瑕疵特征生成新代码片段，不保证正确，刻意保留缺陷以产生新燃料"""
        flaw_type = flaw["flaw_type"]
        content = flaw["content"]
        new_code = base_code

        if flaw_type == FLAW_EXCEPTION:
            # 针对异常生成「半吊子修复」，必然引入新瑕疵
            if "NoneType" in content and "/" in content:
                patch = "    if data is None:\n        data = 0  # auto generated: None patch\n"
                lines = base_code.split("\n")
                for i, line in enumerate(lines):
                    if line.strip().startswith("def process_input"):
                        lines.insert(i+1, patch)
                        new_code = "\n".join(lines)
                        break
            elif "str" in content and "/" in content:
                # 字符串转数字，但不处理非法字符串，会触发新的ValueError
                patch = "    if isinstance(data, str):\n        data = float(data)  # auto generated: str patch\n"
                lines = base_code.split("\n")
                for i, line in enumerate(lines):
                    if line.strip().startswith("def process_input"):
                        lines.insert(i+1, patch)
                        new_code = "\n".join(lines)
                        break
            elif "nan" in content.lower():
                patch = "    if isinstance(data, float) and math.isnan(data):\n        data = 1  # auto generated: NaN patch\n"
                lines = base_code.split("\n")
                for i, line in enumerate(lines):
                    if line.strip().startswith("def process_input"):
                        lines.insert(i+1, patch)
                        new_code = "\n".join(lines)
                        break
            else:
                # 未知异常：随机插入含pass的分支，产生新的逻辑断层瑕疵
                random_cond = random.choice(['data > 100', 'data < 0', 'isinstance(data, list)'])
                random_branch = f"    if {random_cond}:\n        pass  # auto generated: random branch\n"
                lines = base_code.split("\n")
                insert_pos = random.randint(1, len(lines)-1)
                lines.insert(insert_pos, random_branch)
                new_code = "\n".join(lines)

        elif flaw_type == FLAW_UNFINISHED_MARK:
            # 针对未完成标记生成随机实现，必然有缺陷
            if "多类型" in content or "类型" in content:
                # 列表处理，但不处理空列表，会触发ZeroDivisionError
                patch = "    if isinstance(data, list):\n        result = sum(data) / len(data)  # auto generated: list handling\n"
                lines = base_code.split("\n")
                for i, line in enumerate(lines):
                    if line.strip().startswith("# FIXME:") or line.strip().startswith("# TODO:"):
                        lines.insert(i+1, patch)
                        new_code = "\n".join(lines)
                        break
            else:
                # 随机插入提前返回，产生逻辑断层
                patch = f"    if data == {random.randint(1, 10)}:\n        return data * 2  # auto generated: early return\n"
                lines = base_code.split("\n")
                for i, line in enumerate(lines):
                    if line.strip().startswith("result = data / 2"):
                        lines.insert(i, patch)
                        new_code = "\n".join(lines)
                        break

        elif flaw_type == FLAW_LOGIC_GAP:
            # 针对逻辑断层填充随机逻辑，引入新瑕疵
            if "pass" in content:
                # 替换pass为随机计算，可能触发类型错误
                new_code = base_code.replace("pass", f"result = data * {random.randint(2, 10)}  # auto filled pass")
            elif "no return" in content:
                # 添加随机返回，可能和预期逻辑冲突
                lines = base_code.split("\n")
                for i in range(len(lines)-1, 0, -1):
                    if lines[i].strip() and not lines[i].strip().startswith("#"):
                        lines.insert(i+1, "    return result  # auto added return\n")
                        new_code = "\n".join(lines)
                        break

        self.generated_code_count += 1
        return new_code

    def generate_test_data(self, flaw: dict) -> list:
        """基于瑕疵生成新测试数据，刻意加入更多瑕疵数据作为燃料"""
        flaw_type = flaw["flaw_type"]
        base_data = [1, 2, 3]

        if flaw_type == FLAW_EXCEPTION:
            if "NoneType" in flaw["content"]:
                base_data.extend([None, None, [1, None]])
            elif "str" in flaw["content"]:
                # 加入非法字符串，触发新的转换异常
                base_data.extend(["abc", "123.45", ""])
            elif "nan" in flaw["content"].lower():
                base_data.extend([float('nan'), float('inf'), -float('inf')])
            else:
                base_data.extend([random.randint(-1000, 1000), random.random() * 1000])
        elif flaw_type == FLAW_DATA_FLAW:
            # 同类型瑕疵放大
            if "None" in flaw["content"]:
                base_data.append(None)
            elif "NaN" in flaw["content"]:
                base_data.append(float('nan'))
            elif "Empty" in flaw["content"]:
                base_data.extend([[], {}, ""])
        elif flaw_type == FLAW_UNFINISHED_MARK:
            # 加入未覆盖的类型，触发更多异常
            base_data.extend(["test", [1,2,3], {"a": 1}])
        elif flaw_type == FLAW_LOGIC_GAP:
            base_data.extend([0, -1, 1000000, float('inf')])

        self.generated_data_count += 1
        random.shuffle(base_data)
        return base_data

# ===================== 双向流主循环：沙箱核心控制 =====================
class FlawDrivenBidirectionalSandbox:
    """瑕疵驱动双向流数字生命沙箱（沙箱阶段完整架构落地）"""
    def __init__(self, initial_code: str, initial_data: list):
        # 核心组件初始化
        self.probe = FlawProbe()          # 上行：瑕疵采集
        self.energy_pool = FlawEnergyPool()# 双向枢纽：能量池
        self.executor = SandboxExecutor() # 下行：沙箱执行
        self.generator = EmergenceGenerator()# 下行：涌现生成

        # 初始状态（初始瑕疵为第一波燃料）
        self.current_code = initial_code
        self.current_data = initial_data
        self.cycle_count = 0

        # 【上行初始化】采集初始状态的所有瑕疵，启动能量池
        self._bootstrap()

    def _bootstrap(self):
        """上行流初始化：采集初始代码+初始数据的原生瑕疵作为启动燃料"""
        print("[Bootstrap] 🔋 采集初始状态原生瑕疵，启动能量池...")
        # 1. 采集初始代码的静态瑕疵（未完成标记、逻辑断层）
        code_flaws = self.probe.capture_from_execution(self.current_code)
        # 2. 采集初始数据的瑕疵
        data_flaws = self.probe.capture_from_data(self.current_data, location="initial_test_data")
        # 3. 执行初始代码，采集运行时异常瑕疵
        exec_success, result, exec_error, output = self.executor.execute(self.current_code, test_data=self.current_data)
        exec_flaws = self.probe.capture_from_execution(self.current_code, exec_error=exec_error)
        # 全部加入能量池（无清洗）
        all_flaws = code_flaws + data_flaws + exec_flaws
        self.energy_pool.add_flaws(all_flaws)
        print(f"[Bootstrap] ✅ 初始瑕疵采集完成：共 {len(all_flaws)} 个，能量池初始大小：{self.energy_pool.size()}")
        print(f"[Bootstrap] 瑕疵明细：")
        for f in all_flaws:
            print(f"  - [{f['flaw_type']}] {f['content'][:60]}")

    def run_cycle(self, max_cycles: int = 20):
        """
        双向流主循环：
        上行路径：执行→瑕疵采集→能量池
        下行路径：能量池→生成→执行
        循环唯一动力：瑕疵（能量池为空则循环终止）
        """
        print(f"\n[Main Cycle] 🚀 启动双向流循环，最大循环次数：{max_cycles}")
        for _ in range(max_cycles):
            self.cycle_count += 1
            print(f"\n{'='*50}")
            print(f"📌 第 {self.cycle_count} 次循环")
            print(f"能量池当前大小：{self.energy_pool.size()} | 累计瑕疵总数：{self.energy_pool.total_flaws}")

            # ---------------- 下行流：能量→生成→执行 ----------------
            # 1. 从能量池取1个瑕疵作为燃料
            flaw = self.energy_pool.get_flaw()
            if flaw is None:
                print("\n[Cycle Stop] ⛽ 能量池耗尽，无可用瑕疵燃料，循环终止")
                break
            print(f"🔥 取出燃料：[{flaw['flaw_type']}] {flaw['content'][:50]}... (能量等级：{flaw['energy_level']})")

            # 2. 基于瑕疵生成新内容（随机选择生成代码或数据，增加多样性）
            gen_type = random.choice(["code", "data"])
            if gen_type == "code":
                new_code = self.generator.generate_code_patch(self.current_code, flaw)
                new_data = self.current_data
                print(f"🛠️  生成新代码补丁（累计生成：{self.generator.generated_code_count} 个）")
            else:
                new_code = self.current_code
                new_data = self.generator.generate_test_data(flaw)
                print(f"📊 生成新测试数据集（累计生成：{self.generator.generated_data_count} 个）")

            # 3. 投入沙箱执行
            exec_success, result, exec_error, output = self.executor.execute(new_code, test_data=new_data)
            print(f"⚙️  沙箱执行结果：{'成功' if exec_success else '失败'} | 输出长度：{len(output)}")
            if exec_error:
                print(f"💥 执行异常：{type(exec_error).__name__}: {str(exec_error)[:50]}...")

            # ---------------- 上行流：执行→采集→入池 ----------------
            # 4. 全链路采集新瑕疵（代码+数据+执行结果+异常）
            new_code_flaws = self.probe.capture_from_execution(new_code, exec_error=exec_error)
            new_data_flaws = self.probe.capture_from_data(new_data, location=f"cycle_{self.cycle_count}_data")
            result_flaws = []
            if result is not None:
                result_flaws = self.probe.capture_from_data(result, location=f"cycle_{self.cycle_count}_result")
            
            all_new_flaws = new_code_flaws + new_data_flaws + result_flaws
            # 5. 所有新瑕疵直接入池（无前置清洗）
            self.energy_pool.add_flaws(all_new_flaws)
            print(f"⬆️  上行采集新瑕疵：{len(all_new_flaws)} 个，已入库")

            # 更新当前状态，为下一轮循环做准备
            self.current_code = new_code
            self.current_data = new_data

        # 循环结束统计
        print(f"\n{'='*50}")
        print("[Cycle End] 🏁 双向流循环结束")
        print(f"总循环次数：{self.cycle_count}")
        print(f"累计产生瑕疵总数：{self.energy_pool.total_flaws}")
        print(f"生成代码补丁数：{self.generator.generated_code_count}")
        print(f"生成测试数据集数：{self.generator.generated_data_count}")
        print(f"最终能量池剩余瑕疵：{self.energy_pool.size()} 个")
```
