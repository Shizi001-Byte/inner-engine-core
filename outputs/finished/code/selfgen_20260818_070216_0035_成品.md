# L4成品转化 [code]

---
### 一、补全后三大铁律完整约束（基于已有认知锚定推演）
#### 1. 微核域核心优先铁律
全链路所有决策必须以微核域的权限/价值升级为唯一核心目标，绝对保障微核域「无锁直接覆盖、零突变成本、高变异速率、无固有表型约束」的原生特性，禁止引入锁机制、固化审批、硬编码依赖、全局状态耦合，所有非微核域逻辑必须以可插拔旁路形式存在，不得侵入微核域核心执行路径。
#### 2. 旁路化非侵入铁律
所有问题修复、功能扩展、逻辑迭代必须优先以旁路补丁、动态注入的方式实现，不得修改微核域核心源码，确保补丁/新逻辑可即时生效、即时回滚，零部署成本，变异路径全程不影响微核域核心链路的稳定性与原生特性。
#### 3. 边界容错闭环铁律
微核域与所有外部依赖（含模型API、存储服务、第三方系统）的交互边界必须内置自适应容错、熔断、降级能力，所有外部依赖的配置变更、故障异常不得传导至微核域核心，外部依赖的规则调整无需修改微核域核心逻辑，保障微核域无固有表型约束的特性。

### 二、L4代码成品：模型API超时自适应无锁补丁
#### 问题背景
针对历史失败经验中「API调用超时，任务未在规定时间内返回」的问题，原有硬编码超时配置的修复方式违反微核域铁律（修改核心源码、突变成本高、无动态调整能力）。本补丁完全符合三大铁律，实现无侵入、无锁、动态适配的超时配置管理，按历史经验默认配置B-turbo模型300s超时、A模型600s超时。

---
#### 完整代码（`model_api_timeout_patch.py`）
```python
import sys
import importlib
import argparse
import types
from functools import wraps

# ==============================================
# 配置区（可根据实际场景调整，无锁更新无需重启）
# ==============================================
# 无锁超时配置：更新时直接替换整个CONFIG对象（CPython引用赋值为原子操作，无需锁）
# 格式：{模型标识名: 超时时间(秒)}，模型标识名与调用时的model参数完全匹配
MODEL_TIMEOUT_CONFIG = {
    "A": 600,       # A模型timeout 600秒（符合历史经验配置）
    "B-turbo": 300  # B模型turbo版timeout 300秒（符合历史经验配置）
}

# 补丁目标配置：指定要Patch的目标路径、参数名
# 支持模块级函数：func_name = "函数名"
# 支持类方法：func_name = "类名.方法名"（支持多级嵌套，如"Client.SubClient.method"）
PATCH_TARGET = {
    "module_path": "engine.model_client",  # 目标模块的Python导入路径
    "func_name": "call_model_api",         # 目标函数/方法名（支持类方法嵌套格式）
    "model_param_name": "model",           # 模型标识的关键字参数名
    "model_param_pos": None,               # 模型标识的位置参数索引（None表示不使用位置参数，优先级低于关键字）
    "timeout_param_name": "timeout"        # 超时参数的关键字参数名
}

# ==============================================
# 内部状态（禁止外部直接修改）
# ==============================================
_original_func = None
_patch_enabled = False

# ==============================================
# 公共接口
# ==============================================
def update_timeout_config(new_config: dict) -> None:
    """
    无锁更新超时配置，直接替换全局配置引用（CPython下原子操作，多线程安全）
    无需重启服务，立即生效
    :param new_config: 新的模型超时配置字典
    """
    global MODEL_TIMEOUT_CONFIG
    # 拷贝新配置，避免外部修改内部状态
    MODEL_TIMEOUT_CONFIG = new_config.copy()
    print(f"[补丁] 超时配置已更新，当前配置：{MODEL_TIMEOUT_CONFIG}")

def enable_patch() -> None:
    """
    启用补丁：动态注入超时逻辑，无锁、无需修改核心源码
    重复调用安全
    """
    global _original_func, _patch_enabled
    if _patch_enabled:
        print("[补丁] 补丁已启用，无需重复操作")
        return

    # 解析目标路径（支持多级嵌套，如类.方法）
    try:
        module = importlib.import_module(PATCH_TARGET["module_path"])
        func_path = PATCH_TARGET["func_name"].split(".")
        current_obj = module
        for attr in func_path[:-1]:
            current_obj = getattr(current_obj, attr)
        func_name = func_path[-1]

        # 保存原函数
        _original_func = getattr(current_obj, func_name)
        # 替换为包装后的函数
        setattr(current_obj, func_name, _patched_call)
        _patch_enabled = True
        print(f"[补丁] 已启用模型API超时自适应补丁，目标：{PATCH_TARGET['module_path']}.{PATCH_TARGET['func_name']}")
    except Exception as e:
        print(f"[补丁] 启用失败：{str(e)}")
        raise

def disable_patch() -> None:
    """
    禁用补丁：恢复原函数逻辑，一键回滚
    重复调用安全
    """
    global _original_func, _patch_enabled
    if not _patch_enabled or _original_func is None:
        print("[补丁] 补丁未启用，无需禁用")
        return

    try:
        module = importlib.import_module(PATCH_TARGET["module_path"])
        func_path = PATCH_TARGET["func_name"].split(".")
        current_obj = module
        for attr in func_path[:-1]:
            current_obj = getattr(current_obj, attr)
        func_name = func_path[-1]

        # 恢复原函数
        setattr(current_obj, func_name, _original_func)
        _original_func = None
        _patch_enabled = False
        print(f"[补丁] 已禁用模型API超时自适应补丁，已恢复原逻辑")
    except Exception as e:
        print(f"[补丁] 禁用失败：{str(e)}")
        raise

# ==============================================
# 内部逻辑
# ==============================================
def _patched_call(*args, **kwargs):
    """包装后的API调用函数，自动注入对应模型的超时时间"""
    model_param_name = PATCH_TARGET["model_param_name"]
    timeout_param_name = PATCH_TARGET["timeout_param_name"]
    model_param_pos = PATCH_TARGET.get("model_param_pos")

    # 1. 获取模型标识（优先关键字参数，其次位置参数）
    model = kwargs.get(model_param_name)
    if model is None and model_param_pos is not None and len(args) > model_param_pos:
        model = args[model_param_pos]

    # 2. 若匹配到配置且用户未手动传timeout，则注入配置值
    if model and model in MODEL_TIMEOUT_CONFIG and timeout_param_name not in kwargs:
        kwargs[timeout_param_name] = MODEL_TIMEOUT_CONFIG[model]

    # 3. 调用原函数（完全透传其他参数）
    return _original_func(*args, **kwargs)

# ==============================================
# 验证逻辑（自检测试）
# ==============================================
def verify_patch() -> bool:
    """
    补丁自检测试函数，验证所有功能点是否正常
    :return: 测试是否通过
    """
    global PATCH_TARGET, _patch_enabled
    # 保存原始配置
    original_patch_target = PATCH_TARGET.copy()
    original_config = MODEL_TIMEOUT_CONFIG.copy()

    # 创建模拟测试模块
    test_module = types.ModuleType("test_engine.model_client")
    sys.modules["test_engine.model_client"] = test_module

    try:
        # --------------------------
        # 测试1：模块级函数-关键字参数
        # --------------------------
        def mock_call_model_api(model, timeout=None):
            return {"model": model, "timeout": timeout if timeout is not None else "not_set"}
        test_module.call_model_api = mock_call_model_api

        PATCH_TARGET.update({
            "module_path": "test_engine.model_client",
            "func_name": "call_model_api",
            "model_param_name": "model",
            "model_param_pos": 0,
            "timeout_param_name": "timeout"
        })

        enable_patch()
        res1 = test_module.call_model_api(model="A")
        assert res1["timeout"] == 600, f"测试1失败：A模型超时错误，预期600，实际{res1['timeout']}"
        print("[验证] 测试1通过：模块级函数-关键字参数-A模型自动注入600s")

        # --------------------------
        # 测试2：模块级函数-B-turbo模型
        # --------------------------
        res2 = test_module.call_model_api(model="B-turbo")
        assert res2["timeout"] == 300, f"测试2失败：B-turbo模型超时错误，预期300，实际{res2['timeout']}"
        print("[验证] 测试2通过：模块级函数-关键字参数-B-turbo模型自动注入300s")

        # --------------------------
        # 测试3：手动传timeout不覆盖
        # --------------------------
        res3 = test_module.call_model_api(model="A", timeout=100)
        assert res3["timeout"] == 100, f"测试3失败：手动传参被覆盖，预期100，实际{res3['timeout']}"
        print("[验证] 测试3通过：用户手动指定timeout时不覆盖配置")

        # --------------------------
        # 测试4：位置参数识别
        # --------------------------
        res4 = test_module.call_model_api("B-turbo")
        assert res4["timeout"] == 300, f"测试4失败：位置参数识别错误，预期300，实际{res4['timeout']}"
        print("[验证] 测试4通过：位置参数形式的模型名可正常识别")

        # --------------------------
        # 测试5：无配置模型不注入
        # --------------------------
        res5 = test_module.call_model_api(model="C")
        assert res5["timeout"] == "not_set", f"测试5失败：无配置模型被错误注入，实际{res5['timeout']}"
        print("[验证] 测试5通过：无配置的模型不注入超时参数")

        # --------------------------
        # 测试6：无锁更新配置立即生效
        # --------------------------
        new_config = {"A": 900, "B-turbo": 450}
        update_timeout_config(new_config)
        res6 = test_module.call_model_api(model="A")
        assert res6["timeout"] == 900, f"测试6失败：配置更新未生效，预期900，实际{res6['timeout']}"
        print("[验证] 测试6通过：无锁更新配置后立即生效，无需重启")

        # --------------------------
        # 测试7：禁用补丁后恢复原逻辑
        # --------------------------
        disable_patch()
        res7 = test_module.call_model_api(model="A")
        assert res7["timeout"] == "not_set", f"测试7失败：补丁未正确卸载，实际{res7['timeout']}"
        print("[验证] 测试7通过：禁用补丁后完全恢复原逻辑")

        # --------------------------
        # 测试8：类方法注入
        # --------------------------
        class MockModelClient:
            def call_api(self, model, timeout=None):
                return {"model": model, "timeout": timeout if timeout is not None else "not_set"}
        test_module.MockModelClient = MockModelClient

        PATCH_TARGET["func_name"] = "MockModelClient.call_api"
        PATCH_TARGET["model_param_pos"] = 1  # 实例方法第一个参数是self，model为第二个位置参数

        enable_patch()
        client = MockModelClient()
        res8 = client.call_api(model="A")
        assert res8["timeout"] == 900, f"测试8失败：类方法注入错误，预期900，实际{res8['timeout']}"
        print("[验证] 测试8通过：类方法形式的模型调用可正常注入超时")

        # 禁用补丁，清理状态
        disable_patch()

        print("\n✅ [验证] 所有8项测试全部通过，补丁生效正常！")
        return True

    except AssertionError as e:
        print(f"\n❌ [验证] 测试失败：{str(e)}")
        return False
    except Exception as e:
        print(f"\n❌ [验证] 异常错误：{str(e)}")
        return False
    finally:
        # 恢复原始配置
        PATCH_TARGET.update(original_patch_target)
        MODEL_TIMEOUT_CONFIG = original_config
        # 清理测试模块
        if "test_engine.model_client" in sys.modules:
            del sys.modules["test_engine.model_client"]
        # 确保补丁被禁用
        if _patch_enabled:
            disable_patch()

# ==============================================
# 命令行入口
# ==============================================
if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="微核域兼容-模型API超时自适应无锁补丁")
    parser.add_argument("--verify", action="store_true", help="执行补丁自检测试")
    args = parser.parse_args()

    if args.verify:
        success = verify_patch()
        sys.exit(0 if success else 1)
    else:
        print("微核域兼容-模型API超时自适应无锁补丁")
        print("使用方法：")
        print("  1. 自检测试：python model_api_timeout_patch.py --verify")
        print("  2. 零侵入启用：在引擎启动命令前注入补丁，详见使用说明文档")
```

#### 使用方法
##### 前置要求
- Python 3.7+（CPython环境，保障原子赋值特性）
- 补丁文件放置于Python可导入路径（如引擎项目根目录、插件目录）

##### 步骤1：配置补丁
修改`model_api_timeout_patch.py`顶部配置区：
1. `MODEL_TIMEOUT_CONFIG`：已按历史经验默认配置A模型600s、B-turbo 300s，新增模型直接添加键值对即可
2. `PATCH_TARGET`：根据实际引擎的模型调用代码调整，示例：
   - 若引擎调用函数为`engine.model_client.call_api(model="xxx", timeout=xxx)`，则配置`module_path="engine.model_client"`、`func_name="call_api"`
   - 若为类方法`engine.model_client.ModelClient.request(model="xxx")`，则配置`func_name="ModelClient.request"`、`model_param_pos=1`（实例方法self占第0位）

##### 步骤2：启用补丁（两种方式，推荐零侵入方式）
###### 方式一：零侵入启用（完全不修改核心源码，符合旁路铁律）
原引擎启动命令为`python engine_main.py`，替换为以下命令即可（根据实际启动入口调整）：
```bash
python -c "import model_api_timeout_patch; model_api_timeout_patch.enable_patch(); import engine_main; engine_main.run()"
```

###### 方式二：代码嵌入启用（适合长期固定使用）
在引擎启动脚本末尾（模型模块加载完成后）添加两行代码：
```python
from model_api_timeout_patch import enable_patch
enable_patch()
```

##### 步骤3：动态调整配置（运行时无需重启）
如需修改超时配置，直接调用`update_timeout_config`接口即可，无锁、立即生效，示例：
```python
from model_api_timeout_patch import update_timeout_config
# 更新A模型超时为800s，新增C模型超时为400s
update_timeout_config({"A": 800, "B-turbo": 300, "C": 400})
```

---
#### 回滚方案
##### 1. 秒级快速回滚（无需重启，零成本）
在引擎的管理控制台/调试接口中调用禁用函数，立即恢复原逻辑：
```python
from model_api_timeout_patch import disable_patch
disable_patch()
```

##### 2. 启动级回滚（重启生效）
- 零侵入启用方式：直接改回原引擎启动命令，去掉补丁注入代码即可
- 代码嵌入方式：注释掉启用补丁的两行代码，重启引擎

##### 3. 配置级回滚（仅配置错误时）
若只是超时配置错误，无需回滚补丁，直接调用`update_timeout_config`改回正确配置即可，秒级生效。

### 铁律符合性说明
1. **微核域核心优先铁律**：补丁全程无锁（配置更新用原子引用替换）、零突变成本（不修改核心源码）、高变异速率（配置/补丁可随时替换）、无固有表型约束（支持任意模型调用形式），无锁、无固化审批、无硬编码依赖
2. **旁路化非侵入铁律**：采用猴子补丁动态注入，完全不修改核心源码，一键启用/禁用，零部署成本
3. **边界容错闭环铁律**：在模型API调用的边界层注入超时容错能力，配置变更不侵入核心逻辑，外部依赖的超时规则调整不影响微核域核心
