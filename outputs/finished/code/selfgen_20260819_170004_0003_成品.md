# 每日资产盘点脚本 `inventory_daily.py`

## 设计依据（与已知认知锚定对齐）

| 已有认知 | 本脚本如何落地 |
|---|---|
| Guardian+Core 双进程互相拉起 | 脚本可被 Guardian 定时拉起；自身崩溃不影响状态（基线原子写入） |
| WAL + Snapshot 原子持久化 | 备份扫描器识别 `.wal/.snap`，自身基线也用 temp+rename 原子写 |
| 认知池只进不出预警 | 健康指标 `injection_liveness` 检测30分钟注入窗口 |
| API 超时预防 | 本地任务内置 `TIMEOUT=120s`，与历史教训对齐 |

---

## 完整脚本

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
每日资产盘点脚本 (Daily Asset Inventory)
========================================
扫描资产域:
    1. 认知池    cognitive_pool/   (cog_*.json)
    2. 坑库      pit_library/      (pit_*.json 或 pits.json)
    3. 归档索引   archive/         (index.json + data/*.json)
    4. 备份状态   backup/          (*.wal, *.snap)

输出:
    reports/YYYY-MM-DD_inventory.md     人类可读报告
    reports/YYYY-MM-DD_inventory.json   机器可读快照
    state/baseline.json                 明日基线（原子写入）

特性:
    - 不依赖外部守护进程, 可由 Guardian / cron 拉起
    - 内置超时 120s (本地任务)
    - 与 WAL+Snapshot 架构兼容, 自身基线原子持久化
    - 单点异常不拖垮全局 (per-scan try/except)

用法:
    python3 inventory_daily.py            # 增量盘点
    python3 inventory_daily.py --init     # 首次建立基线 (不报差异)
    python3 inventory_daily.py --check    # 仅体检不落基线
"""

import os
import sys
import json
import time
import hashlib
import datetime
import platform
import traceback
from pathlib import Path
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional, Tuple, Any, Set

# ====================== 配置区 ======================
ROOT = Path(os.environ.get("ENGINE_ROOT", "."))
DIRS = {
    "cognitive": ROOT / "cognitive_pool",
    "pit":       ROOT / "pit_library",
    "archive":   ROOT / "archive",
    "backup":    ROOT / "backup",
}
REPORTS_DIR  = ROOT / "reports"
STATE_DIR    = ROOT / "state"
BASELINE_FILE = STATE_DIR / "baseline.json"

TIMEOUT_SECONDS    = 120      # 本地盘点任务上限, 对齐历史教训
INJECTION_WINDOW   = 1800     # 30分钟注入活跃窗口, 对齐坑库预警
BACKUP_STALE_SEC   = 86400    # 备份超过24h视为陈旧
COG_STALE_SEC      = 86400    # 认知超过24h未注入视为"只进不出"嫌疑

# ====================== 数据类 ======================
@dataclass
class AssetRecord:
    asset_id: str
    asset_type: str            # cognitive | pit | archive | backup
    location: str
    fingerprint: str           # sha256[:16]
    mtime: float
    size: int
    meta: Dict[str, Any] = field(default_factory=dict)

@dataclass
class DiffResult:
    added:     List[str] = field(default_factory=list)
    removed:   List[str] = field(default_factory=list)
    modified:  List[str] = field(default_factory=list)
    unchanged: List[str] = field(default_factory=list)

@dataclass
class HealthMetrics:
    cog_total:           int = 0
    cog_injected_30m:    int = 0
    cog_never_injected:  int = 0
    cog_stale_injected:  int = 0          # 超过24h未注入
    cog_only_in_only_out_risk: float = 0.0  # 只进不出风险占比
    pit_total:           int = 0
    pit_referenced:      int = 0
    pit_ref_rate:        float = 0.0
    archive_index_count: int = 0
    archive_file_count:  int = 0
    archive_consistency: float = 0.0       # 1.0 = 完全一致
    backup_latest_wal:   Optional[float] = None
    backup_latest_snap:  Optional[float] = None
    backup_stale:        bool = False
    wal_orphan:          int = 0           # 无配对snap的wal
    issues:              List[str] = field(default_factory=list)

# ====================== 工具函数 ======================
def now_ts() -> float:
    return time.time()

def today_str() -> str:
    return datetime.date.today().isoformat()

def file_fingerprint(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()[:16]

def safe_stat(path: Path) -> Optional[os.stat_result]:
    try:
        return path.stat()
    except (FileNotFoundError, OSError):
        return None

def safe_load_json(path: Path) -> Optional[dict]:
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except (FileNotFoundError, json.JSONDecodeError, OSError):
        return None

def atomic_write_json(path: Path, data: dict) -> bool:
    """与 WAL+Snapshot 思路一致的原子写入: temp + rename"""
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    try:
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2, sort_keys=True)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, path)   # 原子重命名
        return True
    except OSError as e:
        sys.stderr.write(f"[ERR] atomic_write {path} failed: {e}\n")
        try:
            tmp.unlink(missing_ok=True)
        except OSError:
            pass
        return False

def deadline_check(start: float, label: str) -> bool:
    """超时保护, 对齐历史教训"""
    elapsed = time.time() - start
    if elapsed > TIMEOUT_SECONDS:
        sys.stderr.write(f"[WARN] deadline exceeded at <{label}> elapsed={elapsed:.1f}s\n")
        return False
    return True

# ====================== 扫描器: 认知池 ======================
def scan_cognitive_pool(start: float) -> Dict[str, AssetRecord]:
    pool_dir = DIRS["cognitive"]
    records: Dict[str, AssetRecord] = {}
    if not pool_dir.exists():
        return records
    for p in sorted(pool_dir.glob("cog_*.json")):
        if not deadline_check(start, "cognitive_pool"):
            break
        st = safe_stat(p)
        if st is None:
            continue
        data = safe_load_json(p) or {}
        cid = data.get("id") or p.stem
        meta = {
            "tags":             data.get("tags", []),
            "source_task":      data.get("source_task"),
            "created_at":       data.get("created_at"),
            "last_injected_at": data.get("last_injected_at"),
            "injection_count":  int(data.get("injection_count", 0)),
        }
        try:
            fp = file_fingerprint(p)
        except OSError:
            fp = "unreadable"
        records[cid] = AssetRecord(
            asset_id=cid, asset_type="cognitive",
            location=str(p.relative_to(ROOT)) if p.is_relative_to(ROOT) else str(p),
            fingerprint=fp, mtime=st.st_mtime, size=st.st_size, meta=meta,
        )
    return records

# ====================== 扫描器: 坑库 ======================
def scan_pit_library(start: float) -> Dict[str, AssetRecord]:
    pit_dir = DIRS["pit"]
    records: Dict[str, AssetRecord] = {}
    if not pit_dir.exists():
        return records

    # 兼容两种布局: 单文件合并库 / 每坑一文件
    candidates: List[Path] = []
    single = pit_dir / "pits.json"
    if single.exists():
        candidates.append(single)
    candidates.extend(sorted(pit_dir.glob("pit_*.json")))

    for p in candidates:
        if not deadline_check(start, "pit_library"):
            break
        st = safe_stat(p)
        if st is None:
            continue
        data = safe_load_json(p) or {}
        pits = data.get("pits") if "pits" in data else [data]
        for pit in pits:
            pid = pit.get("id") or p.stem
            meta = {
                "level":            pit.get("level", "unknown"),
                "title":            pit.get("title"),
                "category":         pit.get("category"),
                "referenced_by":    pit.get("referenced_by", []),
                "created_at":       pit.get("created_at"),
                "hit_count":        int(pit.get("hit_count", 0)),
            }
            fp = hashlib.sha256(
                json.dumps(pit, sort_keys=True, ensure_ascii=False).encode()
            ).hexdigest()[:16]
            records[pid] = AssetRecord(
                asset_id=pid, asset_type="pit",
                location=str(p.relative_to(ROOT)) if p.is_relative_to(ROOT) else str(p),
                fingerprint=fp, mtime=st.st_mtime, size=st.st_size, meta=meta,
            )
    return records

# ====================== 扫描器: 归档索引 ======================
def scan_archive(start: float) -> Dict[str, AssetRecord]:
    arch_dir = DIRS["archive"]
    records: Dict[str, AssetRecord] = {}
    if not arch_dir.exists():
        return records

    idx_file = arch_dir / "index.json"
    idx = safe_load_json(idx_file) or {}
    entries = idx.get("entries", []) if isinstance(idx, dict) else []
    data_dir = arch_dir / "data"
    real_files = {f.name: f for f in data_dir.glob("*.json")} if data_dir.exists() else {}

    for e in entries:
        if not deadline_check(start, "archive"):
            break
        aid = e.get("id") or e.get("archive_id")
        fname = e.get("file")
        location = str(data_dir / fname) if fname else None
        meta = {
            "task_id":    e.get("task_id"),
            "archived_at":e.get("archived_at"),
            "file":       fname,
            "exists":     bool(fname and fname in real_files),
        }
        fp = "missing"
        size = 0
        mtime = 0.0
        if fname and fname in real_files:
            st = safe_stat(real_files[fname])
            if st:
                size, mtime = st.st_size, st.st_mtime
                try:
                    fp = file_fingerprint(real_files[fname])
                except OSError:
                    fp = "unreadable"
        records[f"archive:{aid}"] = AssetRecord(
            asset_id=f"archive:{aid}", asset_type="archive",
            location=location or "n/a",
            fingerprint=fp, mtime=mtime, size=size, meta=meta,
        )
    return records

# ====================== 扫描器: 备份状态 ======================
def scan_backups(start: float) -> Dict[str, AssetRecord]:
    bk_dir = DIRS["backup"]
    records: Dict[str, AssetRecord] = {}
    if not bk_dir.exists():
        return records

    for p in sorted(bk_dir.iterdir()):
        if not deadline_check(start, "backup"):
            break
        if p.is_dir():
            continue
        st = safe_stat(p)
        if st is None:
            continue
        suffix = p.suffix.lower()
        if suffix not in (".wal", ".snap"):
            continue
        try:
            fp = file_fingerprint(p)
        except OSError:
            fp = "unreadable"
        meta = {
            "kind":   "wal" if suffix == ".wal" else "snap",
            "seq":    int(p.stem.split("_")[-1]) if p.stem.split("_")[-1].isdigit() else 0,
        }
        records[f"backup:{p.name}"] = AssetRecord(
            asset_id=f"backup:{p.name}", asset_type="backup",
            location=str(p.relative_to(ROOT)) if p.is_relative_to(ROOT) else str(p),
            fingerprint=fp, mtime=st.st_mtime, size=st.st_size, meta=meta,
        )
    return records

# ====================== Diff 计算 ======================
def diff_assets(baseline: Dict[str, AssetRecord],
                current: Dict[str, AssetRecord]) -> DiffResult:
    b_ids = set(baseline.keys())
    c_ids = set(current.keys())
    added   = sorted(c_ids - b_ids)
    removed = sorted(b_ids - c_ids)
    modified, unchanged = [], []
    for aid in sorted(b_ids & c_ids):
        if baseline[aid].fingerprint != current[aid].fingerprint:
            modified.append(aid)
        else:
            unchanged.append(aid)
    return DiffResult(added=added, removed=removed,
                      modified=modified, unchanged=unchanged)

# ====================== 健康指标 ======================
def compute_health(cog: Dict, pit: Dict, arch: Dict, bk: Dict) -> HealthMetrics:
    m = HealthMetrics()
    now = now_ts()

    # ---- 认知池 ----
    m.cog_total = len(cog)
    for r in cog.values():
        li = r.meta.get("last_injected_at")
        ic = r.meta.get("injection_count", 0)
        if li is None or ic == 0:
            m.cog_never_injected += 1
        else:
            try:
                li_ts = float(li) if isinstance(li, (int, float)) else (
                    datetime.datetime.fromisoformat(li).timestamp()
                )
            except (ValueError, TypeError):
                li_ts = 0.0
            if now - li_ts <= INJECTION_WINDOW:
                m.cog_injected_30m += 1
            if now - li_ts > COG_STALE_SEC:
                m.cog_stale_injected += 1
    if m.cog_total > 0:
        m.cog_only_in_only_out_risk = round(
            (m.cog_never_injected + m.cog_stale_injected) / m.cog_total, 3
        )

    # ---- 坑库 ----
    m.pit_total = len(pit)
    for r in pit.values():
        if r.meta.get("referenced_by") or r.meta.get("hit_count", 0) > 0:
            m.pit_referenced += 1
    m.pit_ref_rate = round(m.pit_referenced / m.pit_total, 3) if m.pit_total > 0 else 0.0

    # ---- 归档 ----
    m.archive_index_count = len(arch)
    m.archive_file_count = sum(1 for r in arch.values() if r.meta.get("exists"))
    if m.archive_index_count > 0:
        m.archive_consistency = round(m.archive_file_count / m.archive_index_count, 3)

    # ---- 备份 ----
    wal_mtimes, snap_mtimes = [], []
    for r in bk.values():
        if r.meta.get("kind") == "wal":
            wal_mtimes.append(r.mtime)
        elif r.meta.get("kind") == "snap":
            snap_mtimes.append(r.mtime)
    if wal_mtimes:
        m.backup_latest_wal = max(wal_mtimes)
    if snap_mtimes:
        m.backup_latest_snap = max(snap_mtimes)
    latest = max(filter(None, [m.backup_latest_wal, m.backup_latest_snap]), default=None)
    if latest is None or (now - latest) > BACKUP_STALE_SEC:
        m.backup_stale = True

    # ---- WAL 孤儿检测 ----
    snap_seqs = {r.meta.get("seq") for r in bk.values()
                 if r.meta.get("kind") == "snap"}
    for r in bk.values():
        if r.meta.get("kind") == "wal" and r.meta.get("seq") not in snap_seqs:
            m.wal_orphan += 1

    # ---- 问题清单 ----
    if m.cog_only_in_only_out_risk > 0.5:
        m.issues.append(
            f"[HIGH] 认知池只进不出风险 {m.cog_only_in_only_out_risk:.1%}: "
            f"never_injected={m.cog_never_injected}, stale>24h={m.cog_stale_injected}, "
            f"total={m.cog_total} (须30分钟内注入)"
        )
    if m.pit_ref_rate < 0.2 and m.pit_total > 10:
        m.issues.append(
            f"[MED] 坑库引用率偏低 {m.pit_ref_rate:.1%} "
            f"(referenced={m.pit_referenced}/{m.pit_total})"
        )
    if m.archive_consistency < 1.0:
        m.issues.append(
            f"[MED] 归档索引与文件不一致 consistency={m.archive_consistency:.1%} "
            f"(index={m.archive_index_count}, files={m.archive_file_count})"
        )
    if m.backup_stale:
        m.issues.append("[HIGH] 备份陈旧, 距今 > 24h 或无备份")
    if m.wal_orphan > 0:
        m.issues.append(f"[LOW] WAL 孤儿 {m.wal_orphan} 个 (无配对 snapshot)")

    return m

# ====================== 报告生成 ======================
def render_markdown(today: str, diff: DiffResult, health: HealthMetrics,
                    current: Dict[str, AssetRecord],
                    baseline: Dict[str, AssetRecord]) -> str:
    L: List[str] = []
    L.append(f"# 每日资产盘点报告 {today}\n")
    L.append(f"_生成时间: {datetime.datetime.now().isoformat(timespec='seconds')}_\n")
    L.append(f"_平台: {platform.platform()}_\n")

    # 概览
    total = len(current)
    L.append("## 1. 资产总览\n")
    L.append("| 资产域 | 条目数 |")
    L.append("|---|---|")
    for t in ("cognitive", "pit", "archive", "backup"):
        n = sum(1 for r in current.values() if r.asset_type == t)
        L.append(f"| {t} | {n} |")
    L.append(f"| **合计** | **{total}** |\n")

    # 变化摘要
    L.append("## 2. 当日变化摘要\n")
    L.append(f"- 新增: **{len(diff.added)}**")
    L.append(f"- 删除: **{len(diff.removed)}**")
    L.append(f"- 修改: **{len(diff.modified)}**")
    L.append(f"- 未变: {len(diff.unchanged)}\n")

    # 明细
    def _row(aid: str, rec: Optional[AssetRecord]) -> str:
        r = rec or current.get(aid) or baseline.get(aid)
        if r is None:
            return f"| {aid} | ? | ? | ? |"
        loc = r.location
        if len(loc) > 60:
            loc = "..." + loc[-57:]
        return f"| {aid} | {r.asset_type} | {loc} | `{r.fingerprint}` |"

    if diff.added:
        L.append("### 2.1 新增 (+)\n")
        L.append("| ID | 类型 | 位置 | 指纹 |")
        L.append("|---|---|---|---|")
        for aid in diff.added:
            L.append(_row(aid, current.get(aid)))
        L.append("")

    if diff.removed:
        L.append("### 2.2 删除 (-)\n")
        L.append("| ID | 类型 | 位置 | 指纹(原) |")
        L.append("|---|---|---|---|")
        for aid in diff.removed:
            L.append(_row(aid, baseline.get(aid)))
        L.append("")

    if diff.modified:
        L.append("### 2.3 修改 (~)\n")
        L.append("| ID | 类型 | 位置 | 新指纹 | 旧指纹 |")
        L.append("|---|---|---|---|---|")
        for aid in diff.modified:
            old = baseline.get(aid)
            new = current.get(aid)
            loc = new.location if new else (old.location if old else "?")
            if len(loc) > 50:
                loc = "..." + loc[-47:]
            L.append(f"| {aid} | {new.asset_type if new else '?'} | {loc} | "
                     f"`{new.fingerprint if new else '?'}` | "
                     f"`{old.fingerprint if old else '?'}` |")
        L.append("")

    # 健康指标
    L.append("## 3. 健康指标\n")
    L.append("| 指标 | 值 | 阈值 |")
    L.append("|---|---|---|")
    L.append(f"| 认知池总量 | {health.cog_total} | - |")
    L.append(f"| 30分钟内注入活跃 | {health.cog_injected_30m} | >0 |")
    L.append(f"| 从未注入 | {health.cog_never_injected} | 0 |")
    L.append(f"| 超24h未注入 | {health.cog_stale_injected} | 0 |")
    L.append(f"| 只进不出风险占比 | {health.cog_only_in_only_out_risk:.1%} | <50% |")
    L.append(f"| 坑库引用率 | {health.pit_ref_rate:.1%} | >20% |")
    L.append(f"| 归档一致率 | {health.archive_consistency:.1%} | 100% |")
    L.append(f"| 备份陈旧 | {'是' if health.backup_stale else '否'} | 否 |")
    L.append(f"| WAL孤儿 | {health.wal_orphan} | 0 |\n")

    # 问题清单
    L.append("## 4. 问题清单\n")
    if health.issues:
        for i, issue in enumerate(health.issues, 1):
            L.append(f"{i}. {issue}")
    else:
        L.append("_无问题_")
    L.append("")

    # 备份时间戳
    L.append("## 5. 备份时间线\n")
    def _fmt(ts: Optional[float]) -> str:
        if ts is None:
            return "无"
        return datetime.datetime.fromtimestamp(ts).isoformat(timespec="seconds")
    L.append(f"- 最新 WAL: {_fmt(health.backup_latest_wal)}")
    L.append(f"- 最新 Snapshot: {_fmt(health.backup_latest_snap)}\n")

    return "\n".join(L)

def render_json(today: str, diff: DiffResult, health: HealthMetrics,
                current: Dict[str, AssetRecord]) -> dict:
    return {
        "date": today,
        "generated_at": datetime.datetime.now().isoformat(timespec="seconds"),
        "summary": {
            "added":    len(diff.added),
            "removed":  len(diff.removed),
            "modified": len(diff.modified),
            "unchanged":len(diff.unchanged),
            "total":    len(current),
        },
        "diff": {
            "added":    diff.added,
            "removed":  diff.removed,
            "modified": diff.modified,
        },
        "health": asdict(health),
        "assets": {aid: asdict(r) for aid, r in current.items()},
    }

# ====================== 基线读写 ======================
def load_baseline() -> Dict[str, AssetRecord]:
    raw = safe_load_json(BASELINE_FILE) or {}
    out: Dict[str, AssetRecord] = {}
    for aid, rec in (raw.get("assets") or {}).items():
        try:
            out[aid] = AssetRecord(**rec)
        except TypeError:
            continue
    return out

def save_baseline(current: Dict[str, AssetRecord], today: str) -> bool:
    payload = {
        "date": today,
        "saved_at": datetime.datetime.now().isoformat(timespec="seconds"),
        "assets": {aid: asdict(r) for aid, r in current.items()},
    }
    return atomic_write_json(BASELINE_FILE, payload)

# ====================== 主流程 ======================
def main() -> int:
    start = now_ts()
    today = today_str()
    argv = sys.argv[1:]
    is_init  = "--init"  in argv
    is_check = "--check" in argv

    print(f"[INFO] inventory_daily start date={today} init={is_init} check={is_check}")

    # 1. 扫描 (各域独立 try/except, 单点失败不拖垮全局)
    cog, pit, arch, bk = {}, {}, {}, {}
    try:
        cog = scan_cognitive_pool(start)
    except Exception as e:
        sys.stderr.write(f"[ERR] cognitive scan failed: {e}\n")
    try:
        pit = scan_pit_library(start)
    except Exception as e:
        sys.stderr.write(f"[ERR] pit scan failed: {e}\n")
    try:
        arch = scan_archive(start)
    except Exception as e:
        sys.stderr.write(f"[ERR] archive scan failed: {e}\n")
    try:
        bk = scan_backups(start)
    except Exception as e:
        sys.stderr.write(f"[ERR] backup scan failed: {e}\n")

    current = {**cog, **pit, **arch, **bk}
    print(f"[INFO] scanned: cog={len(cog)} pit={len(pit)} "
          f"arch={len(arch)} bk={len(bk)} total={len(current)}")

    # 2. 健康指标
    health = compute_health(cog, pit, arch, bk)
    print(f"[INFO] health: cog_risk={health.cog_only_in_only_out_risk:.1%} "
          f"pit_ref={health.pit_ref_rate:.1%} "
          f"arch_consist={health.archive_consistency:.1%} "
          f"backup_stale={health.backup_stale} issues={len(health.issues)}")

    # 3. 基线对比
    baseline = load_baseline() if not is_init else {}
    diff = diff_assets(baseline, current)
    print(f"[INFO] diff: +{len(diff.added)} -{len(diff.removed)} "
          f"~{len(diff.modified)} ={len(diff.unchanged)}")

    # 4. 生成报告
    REPORTS_DIR.mkdir(parents=True, exist_ok=True)
    md_path   = REPORTS_DIR / f"{today}_inventory.md"
    json_path = REPORTS_DIR / f"{today}_inventory.json"
    md = render_markdown(today, diff, health, current, baseline)
    js = render_json(today, diff, health, current)
    try:
        md_path.write_text(md, encoding="utf-8")
        atomic_write_json(json_path, js)
        print(f"[OK] report -> {md_path}")
        print(f"[OK] json   -> {json_path}")
    except OSError as e:
        sys.stderr.write(f"[ERR] write report failed: {e}\n")

    # 5. 落基线 (check 模式不落)
    if not is_check:
        if save_baseline(current, today):
            print(f"[OK] baseline saved -> {BASELINE_FILE}")
        else:
            sys.stderr.write("[ERR] baseline save failed\n")

    # 6. 失败信号: 有 HIGH 问题 -> exit 1, 供 Guardian 决策
    elapsed = time.time() - start
    print(f"[INFO] done in {elapsed:.2f}s")
    high_count = sum(1 for i in health.issues if i.startswith("[HIGH]"))
    return 1 if high_count > 0 else 0

if __name__ == "__main__":
    try:
        sys.exit(main())
    except KeyboardInterrupt:
        sys.stderr.write("[ERR] interrupted\n")
        sys.exit(130)
    except Exception as e:
        sys.stderr.write(f"[FATAL] {e}\n{traceback.format_exc()}\n")
        sys.exit(2)
```

## 关键设计点对应表

| 设计点 | 落地位置 | 与已有认知的呼应 |
|---|---|---|
| 不依赖外部守护 | `main()` 单次执行即退出 | 与 Guardian+Core 互拉架构解耦 |
| 原子持久化 | `atomic_write_json()` temp+rename+fsync | 与 WAL+Snapshot 思路一致 |
| 崩溃自恢复 | 基线损坏时 `safe_load_json` 返回 None，等价于 `--init` | 崩溃后从最近断点恢复 |
| 认知池只进不出预警 | `compute_health()` 中 `cog_only_in_only_out_risk` + 30分钟窗口 | 直接命中坑库预警 #1 |
| API 超时预防 | `deadline_check()` 在每个扫描循环内检查 120s | 对齐历史教训 |
| 退出码语义 | HIGH 问题 → exit 1；正常 → exit 0；崩溃 → exit 2 | 供 Guardian 决策是否告警/重启 Core |

退出码策略让 Guardian 可以据此触发自恢复流程：`exit 2` 触发 Core 重启，`exit 1` 仅告警不重启，`exit 0` 静默。这样资产盘点脚本天然融入双进程互拉架构，而不引入新的守护进程依赖。
