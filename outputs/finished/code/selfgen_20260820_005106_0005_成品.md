# L4成品：任务池重复率检测与扩充建议脚本

## 设计思路（基于认知锚链推演）

将【链2 内容寻址去重】+【链4 冲突检测】思想迁移到任务池层面：
- **内容寻址** → 任务prompt的语义指纹（SimHash / TF-IDF向量）
- **块级增量** → n-gram shingle 分块比对，识别"局部雷同"
- **冲突检测** → 相似度边界态（0.6~0.8）单独标记，供人工裁决
- **异构自适应** → 区分A/B模型任务特征，B模型跳过reasoning占比误判

## 脚本主体 `task_pool_dedup.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
任务池重复率检测与扩充建议
L4成品 · 基于内容寻址去重 + 冲突检测的认知链迁移

依赖:
    pip install scikit-learn jieba numpy
"""

import json
import hashlib
import time
import re
from pathlib import Path
from dataclasses import dataclass, field
from typing import List, Dict, Tuple, Optional
from collections import defaultdict, Counter

import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import jieba

# ============================================================
# 配置层：模型差异化策略（避坑：B模型reasoning占比70-82%为正常特性）
# ============================================================
MODEL_CONFIG = {
    "A": {
        "timeout_sec": 600,
        "skip_reasoning_ratio_check": False,
        "reasoning_abnormal_threshold": 0.95,   # A模型reasoning>95%才算异常
    },
    "B_turbo": {
        "timeout_sec": 300,
        "skip_reasoning_ratio_check": True,       # 避坑：跳过B模型reasoning占比检测
        "reasoning_abnormal_threshold": None,
    },
}

# 任务维度关键词图谱（用于覆盖率分析 & 扩充建议）
TASK_DIMENSIONS = {
    "代码生成":   ["实现", "编写", "生成代码", "函数", "类", "模块"],
    "代码审查":   ["review", "审查", "检查", "优化", "重构", "bug"],
    "文档撰写":   ["文档", "说明", "README", "注释", "用户手册"],
    "测试用例":   ["测试", "test", "用例", "覆盖率", "mock", "断言"],
    "数据分析":   ["分析", "统计", "报表", "可视化", "趋势", "指标"],
    "架构设计":   ["架构", "设计", "方案", "选型", "分层", "接口"],
    "调试排障":   ["调试", "debug", "排错", "堆栈", "日志", "定位"],
    "部署运维":   ["部署", "CI/CD", "docker", "k8s", "运维", "上线"],
    "安全审计":   ["安全", "漏洞", "审计", "注入", "XSS", "权限"],
    "性能优化":   ["性能", "优化", "延迟", "吞吐", "瓶颈", "缓存"],
}

# 相似度阈值（基于冲突检测链：边界态需单独标记）
SIM_THRESHOLD_EXACT   = 0.98   # 内容寻址级精确重复
SIM_THRESHOLD_NEAR    = 0.85   # 近似重复
SIM_THRESHOLD_BORDER  = 0.60   # 冲突边界态（需人工裁决）
SIM_THRESHOLD_RELATED = 0.35   # 弱相关（扩充参考）


# ============================================================
# 数据结构
# ============================================================
@dataclass
class TaskRecord:
    task_id: str
    prompt: str
    model: str = "A"
    timestamp: float = 0.0
    reasoning_ratio: float = 0.0   # B模型可能携带，跳过检测
    tags: List[str] = field(default_factory=list)


@dataclass
class DupCluster:
    cluster_id: int
    severity: str                  # exact / near / border
    members: List[str]             # task_ids
    representative: str            # 代表性 task_id
    avg_similarity: float
    sample_prompts: Dict[str, str] = field(default_factory=dict)


@dataclass
class DedupReport:
    total_tasks: int
    exact_dups: int
    near_dups: int
    border_conflicts: int
    unique_tasks: int
    duplicate_rate: float
    clusters: List[DupCluster]
    dimension_coverage: Dict[str, float]
    expansion_suggestions: List[str]
    model_specific_notes: List[str]


# ============================================================
# 核心分析器
# ============================================================
class TaskPoolAnalyzer:
    """任务池重复率检测器"""

    def __init__(self, tasks: List[TaskRecord]):
        self.tasks = tasks
        self._tfidf_matrix = None
        self._sim_matrix = None
        self._content_hashes = {}

    # ---------- [链2] 内容寻址：精确哈希 ----------
    def _content_hash(self, prompt: str) -> str:
        """归一化后的内容寻址哈希（去空白/大小写/标点）"""
        normalized = re.sub(r"[\s\u3000，。、；：！？""''（）()\[\]{}]+", "", prompt.lower())
        return hashlib.sha256(normalized.encode("utf-8")).hexdigest()[:16]

    # ---------- 中文分词预处理 ----------
    @staticmethod
    def _tokenize(text: str) -> str:
        # 过滤单字噪声，保留有意义的词
        tokens = [w for w in jieba.cut(text) if len(w.strip()) >= 2 or w.isalnum()]
        return " ".join(tokens)

    # ---------- [链3] 块级增量：n-gram shingle特征 ----------
    def _build_tfidf(self):
        corpus = [self._tokenize(t.prompt) for t in self.tasks]
        self._tfidf_matrix = TfidfVectorizer(
            ngram_range=(1, 2),     # 块级：bigram模拟shingle
            sublinear_tf=True,
            max_features=5000,
        ).fit_transform(corpus)
        self._sim_matrix = cosine_similarity(self._tfidf_matrix)

    # ---------- [链4] 冲突检测：聚类重复簇 ----------
    def _detect_clusters(self) -> List[DupCluster]:
        clusters = []
        visited = set()
        cluster_id = 0

        # 先做内容寻址级精确重复
        hash_groups = defaultdict(list)
        for i, t in enumerate(self.tasks):
            h = self._content_hash(t.prompt)
            self._content_hashes[i] = h
            hash_groups[h].append(i)

        # 精确重复簇
        for h, idx_list in hash_groups.items():
            if len(idx_list) > 1:
                cluster_id += 1
                clusters.append(DupCluster(
                    cluster_id=cluster_id,
                    severity="exact",
                    members=[self.tasks[i].task_id for i in idx_list],
                    representative=self.tasks[idx_list[0]].task_id,
                    avg_similarity=1.0,
                    sample_prompts={self.tasks[i].task_id: self.tasks[i].prompt[:120] for i in idx_list[:3]},
                ))
                visited.update(idx_list)

        # 近似/边界态聚类（在非精确重复集合中）
        n = len(self.tasks)
        for i in range(n):
            if i in visited:
                continue
            cluster_members = [i]
            for j in range(i + 1, n):
                if j in visited:
                    continue
                sim = float(self._sim_matrix[i, j])
                if sim >= SIM_THRESHOLD_BORDER:
                    cluster_members.append(j)
            if len(cluster_members) > 1:
                # 判定严重等级
                max_sim = max(
                    float(self._sim_matrix[cluster_members[0], j])
                    for j in cluster_members[1:]
                )
                if max_sim >= SIM_THRESHOLD_NEAR:
                    severity = "near"
                else:
                    severity = "border"
                cluster_id += 1
                clusters.append(DupCluster(
                    cluster_id=cluster_id,
                    severity=severity,
                    members=[self.tasks[k].task_id for k in cluster_members],
                    representative=self.tasks[cluster_members[0]].task_id,
                    avg_similarity=max_sim,
                    sample_prompts={self.tasks[k].task_id: self.tasks[k].prompt[:120] for k in cluster_members[:3]},
                ))
                visited.update(cluster_members)
        return clusters

    # ---------- 维度覆盖率分析 ----------
    def _analyze_coverage(self) -> Dict[str, float]:
        coverage = {}
        for dim, keywords in TASK_DIMENSIONS.items():
            hit_count = 0
            for t in self.tasks:
                if any(kw.lower() in t.prompt.lower() for kw in keywords):
                    hit_count += 1
            coverage[dim] = round(hit_count / max(len(self.tasks), 1), 4)
        return coverage

    # ---------- 扩充建议生成 ----------
    def _generate_suggestions(self, coverage: Dict[str, float]) -> List[str]:
        suggestions = []
        sorted_dims = sorted(coverage.items(), key=lambda x: x[1])

        # 覆盖率<10%的维度作为优先扩充方向
        for dim, ratio in sorted_dims:
            if ratio < 0.10:
                suggestions.append(
                    f"🆕 [高优扩充] 维度「{dim}」覆盖率仅 {ratio:.1%}，"
                    f"建议新增相关任务，关键词参考：{TASK_DIMENSIONS[dim][:3]}"
                )
            elif ratio < 0.25:
                suggestions.append(
                    f"⚠️ [补充加强] 维度「{dim}」覆盖率 {ratio:.1%}，"
                    f"任务量偏少，建议适度扩充"
                )

        # 基于重复簇的语义收敛提示
        exact_clusters = [c for c in self._clusters if c.severity == "exact"]
        if exact_clusters:
            suggestions.append(
                f"♻️ [去重瘦身] 检测到 {len(exact_clusters)} 组精确重复簇，"
                f"建议合并为模板任务，通过参数化差异替代硬编码复制"
            )

        # 基于边界态冲突的语义边界探测
        border_clusters = [c for c in self._clusters if c.severity == "border"]
        if border_clusters:
            suggestions.append(
                f"🔬 [边界探索] {len(border_clusters)} 组任务处于相似度边界态(0.6-0.85)，"
                f"提示存在未充分挖掘的中间地带任务，可细化为差异化变体"
            )

        # 维度组合缺口分析
        dim_pairs = [
            ("代码生成", "测试用例"),
            ("架构设计", "性能优化"),
            ("安全审计", "部署运维"),
        ]
        for d1, d2 in dim_pairs:
            cross_count = sum(
                1 for t in self.tasks
                if any(kw.lower() in t.prompt.lower() for kw in TASK_DIMENSIONS[d1])
                and any(kw.lower() in t.prompt.lower() for kw in TASK_DIMENSIONS[d2])
            )
            if cross_count == 0:
                suggestions.append(
                    f"🔗 [组合缺口] 维度「{d1} × {d2}」无交叉任务，"
                    f"建议新增复合型任务（如：生成代码后自动产出测试用例）"
                )
        return suggestions

    # ---------- 模型差异化备注（避坑） ----------
    def _model_notes(self) -> List[str]:
        notes = []
        model_counts = Counter(t.model for t in self.tasks)
        for model, count in model_counts.items():
            cfg = MODEL_CONFIG.get(model, {})
            timeout = cfg.get("timeout_sec", 600)
            if cfg.get("skip_reasoning_ratio_check"):
                notes.append(
                    f"✅ 模型 {model} 任务 {count} 条：已跳过reasoning占比检测"
                    f"（B模型turbo特性），timeout预设 {timeout}s"
                )
            else:
                notes.append(
                    f"ℹ️ 模型 {model} 任务 {count} 条：reasoning占比检测阈值 "
                    f"{cfg.get('reasoning_abnormal_threshold')}，timeout预设 {timeout}s"
                )
        return notes

    # ---------- 主分析入口 ----------
    def analyze(self) -> DedupReport:
        self._build_tfidf()
        self._clusters = self._detect_clusters()

        exact_count = sum(len(c.members) - 1 for c in self._clusters if c.severity == "exact")
        near_count  = sum(len(c.members) - 1 for c in self._clusters if c.severity == "near")
        border_count = len([c for c in self._clusters if c.severity == "border"])

        total = len(self.tasks)
        dup_count = exact_count + near_count
        coverage = self._analyze_coverage()
        suggestions = self._generate_suggestions(coverage)
        notes = self._model_notes()

        return DedupReport(
            total_tasks=total,
            exact_dups=exact_count,
            near_dups=near_count,
            border_conflicts=border_count,
            unique_tasks=total - dup_count,
            duplicate_rate=round(dup_count / max(total, 1), 4),
            clusters=self._clusters,
            dimension_coverage=coverage,
            expansion_suggestions=suggestions,
            model_specific_notes=notes,
        )


# ============================================================
# 报告渲染器
# ============================================================
def render_report(report: DedupReport) -> str:
    lines = []
    lines.append("=" * 72)
    lines.append("📊 任务池重复率检测与扩充建议报告")
    lines.append("=" * 72)
    lines.append(f"任务总数:        {report.total_tasks}")
    lines.append(f"精确重复:        {report.exact_dups}")
    lines.append(f"近似重复:        {report.near_dups}")
    lines.append(f"边界态冲突:      {report.border_conflicts}（人工裁决区）")
    lines.append(f"有效唯一任务:    {report.unique_tasks}")
    lines.append(f"重复率:          {report.duplicate_rate:.2%}")
    lines.append("")

    # ---- 重复簇详情（前10） ----
    lines.append("─" * 72)
    lines.append("🔍 重复簇详情（Top 10）")
    lines.append("─" * 72)
    for c in report.clusters[:10]:
        lines.append(f"  [#{c.cluster_id}] {c.severity.upper():6s} | "
                     f"avg_sim={c.avg_similarity:.3f} | "
                     f"size={len(c.members)} | rep={c.representative}")
        for tid, p in c.sample_prompts.items():
            lines.append(f"        └─ {tid}: {p}...")
    lines.append("")

    # ---- 维度覆盖 ----
    lines.append("─" * 72)
    lines.append("📈 任务维度覆盖率")
    lines.append("─" * 72)
    for dim, ratio in sorted(report.dimension_coverage.items(),
                             key=lambda x: -x[1]):
        bar = "█" * int(ratio * 30) + "░" * (30 - int(ratio * 30))
        lines.append(f"  {dim:8s} {bar} {ratio:.1%}")
    lines.append("")

    # ---- 扩充建议 ----
    lines.append("─" * 72)
    lines.append("💡 扩充建议")
    lines.append("─" * 72)
    if not report.expansion_suggestions:
        lines.append("  （任务池维度覆盖均衡，暂无高优扩充项）")
    for i, s in enumerate(report.expansion_suggestions, 1):
        lines.append(f"  {i}. {s}")
    lines.append("")

    # ---- 模型差异化备注 ----
    lines.append("─" * 72)
    lines.append("⚙️ 模型差异化策略备注")
    lines.append("─" * 72)
    for n in report.model_specific_notes:
        lines.append(f"  {n}")
    lines.append("=" * 72)
    return "\n".join(lines)


# ============================================================
# CLI入口
# ============================================================
def load_tasks_from_jsonl(path: str) -> List[TaskRecord]:
    tasks = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            obj = json.loads(line)
            tasks.append(TaskRecord(
                task_id=obj["task_id"],
                prompt=obj["prompt"],
                model=obj.get("model", "A"),
                timestamp=obj.get("timestamp", 0.0),
                reasoning_ratio=obj.get("reasoning_ratio", 0.0),
                tags=obj.get("tags", []),
            ))
    return tasks


def main():
    import sys
    if len(sys.argv) < 2:
        print("用法: python task_pool_dedup.py <tasks.jsonl> [output.json]")
        sys.exit(1)

    tasks = load_tasks_from_jsonl(sys.argv[1])
    print(f"[INFO] 加载任务 {len(tasks)} 条，开始分析...")

    t0 = time.time()
    analyzer = TaskPoolAnalyzer(tasks)
    report = analyzer.analyze()
    elapsed = time.time() - t0

    # 控制台输出
    print(render_report(report))
    print(f"\n⏱ 分析耗时: {elapsed:.2f}s")

    # 可选：JSON落盘
    if len(sys.argv) >= 3:
        out = {
            "total_tasks": report.total_tasks,
            "duplicate_rate": report.duplicate_rate,
            "exact_dups": report.exact_dups,
            "near_dups": report.near_dups,
            "border_conflicts": report.border_conflicts,
            "dimension_coverage": report.dimension_coverage,
            "expansion_suggestions": report.expansion_suggestions,
            "model_notes": report.model_specific_notes,
            "clusters": [
                {
                    "cluster_id": c.cluster_id,
                    "severity": c.severity,
                    "members": c.members,
                    "avg_similarity": c.avg_similarity,
                } for c in report.clusters
            ],
        }
        Path(sys.argv[2]).write_text(
            json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8"
        )
        print(f"[INFO] 报告已保存至 {sys.argv[2]}")


if __name__ == "__main__":
    main()
```

## 关键设计要点（对应避坑与认知链）

| 设计点 | 对应认知链 / 避坑项 |
|---|---|
| SHA-256内容寻址哈希 + 归一化 | 【链2】内容寻址去重 |
| TF-IDF bigram（模拟shingle） | 【链3】块级增量同步 |
| 三级阈值（exact/near/border） | 【链4】冲突检测与异构自适应 |
