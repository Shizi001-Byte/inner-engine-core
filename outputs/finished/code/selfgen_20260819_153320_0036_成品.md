# L4 代码成品：自进化闭环引擎补丁

## 一、补丁定位

针对"自进化闭环引擎"已暴露的 4 类问题：

| # | 已知问题 | 根因 | 补丁点 |
|---|---------|------|-------|
| P1 | 任务清单静态化 | 任务状态机不闭环，验证态不可重新激活 | 新增 `_is_stale` + 状态翻转 |
| P2 | 生成后未立即验证 | 生成与验证解耦 | `evolve_once` 内串行执行 |
| P3 | 冷却不重置 | 验证通过后 `cooldown_until` 未清零 | 验证通过分支显式置 0 |
| P4 | API 超时 | 默认 timeout 无模型预设 | `MODEL_TIMEOUTS` 字典 + `_call_with_timeout` |

## 二、完整代码（`engine_self_evolve_patch.py`）

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
engine_self_evolve_patch.py
自进化闭环引擎补丁 v1.0
修复点: P1静态清单 / P2无验证 / P3冷却不重置 / P4API超时
"""
import time, json, shutil, hashlib, threading
from pathlib import Path
from dataclasses import dataclass, field, asdict
from typing import Callable, Optional, Dict, Any, List
from enum import Enum

# ============ 配置区 ============
MODEL_TIMEOUTS = {"model_turbo": 300, "model_a": 600}  # P4 预设
DEFAULT_COOLDOWN_SEC = 60
MAX_RETRIES = 3
BACKUP_DIR  = Path("./_engine_backups")
STATE_FILE  = Path("./_engine_state.json")


class TaskState(Enum):
    PENDING     = "pending"
    GENERATING  = "generating"
    VERIFIED    = "verified"
    FAILED      = "failed"
    ROLLED_BACK = "rolled_back"


@dataclass
class Task:
    task_id: str
    spec: str
    state: TaskState = TaskState.PENDING
    generated_artifact: str = ""
    verify_result: Dict[str, Any] = field(default_factory=dict)
    retries: int = 0
    cooldown_until: float = 0.0      # P3: 关键字段
    signature: str = ""              # 内容指纹，用于闭环活性检测


class SelfEvolveEngine:
    def __init__(self, model: str = "model_turbo"):
        assert model in MODEL_TIMEOUTS, f"未知模型: {model}"
        self.model   = model
        self.timeout = MODEL_TIMEOUTS[model]
        self.tasks: List[Task] = []
        self.lock = threading.Lock()
        BACKUP_DIR.mkdir(parents=True, exist_ok=True)
        self._load_state()

    # ---------------- 核心闭环 ----------------
    def evolve_once(self,
                    spec: str,
                    generator: Callable[[Task], str],
                    verifier:  Callable[[Task], Dict[str, Any]]) -> Task:
        """一次完整 生成→立即验证→冷却重置 闭环。
        锚定结论: '每次生成后必须立即验证,验证通过则重置冷却'
        """
        with self.lock:
            existing = self._find_task(spec)
            if existing and existing.state == TaskState.VERIFIED:
                # P1: 已验证任务若 stale 则重新激活 -> 真闭环而非静态清单
                if self._is_stale(existing):
                    existing.state = TaskState.PENDING
                else:
                    return existing

            task = existing or Task(
                task_id=hashlib.md5(spec.encode()).hexdigest()[:8],
                spec=spec,
            )
            if task not in self.tasks:
                self.tasks.append(task)

            # ---- 1. 生成 (带 P4 timeout) ----
            task.state = TaskState.GENERATING
            artifact = self._call_with_timeout(generator, task)
            task.generated_artifact = artifact
            task.signature = hashlib.md5(artifact.encode()).hexdigest()

            # ---- 2. 立即验证 (P2) ----
            result = verifier(task)
            task.verify_result = result

            if result.get("ok"):
                # ---- 3. 验证通过 -> 重置冷却 (P3) ----
                task.state = TaskState.VERIFIED
                task.cooldown_until = 0.0
                task.retries = 0
            else:
                task.state = TaskState.FAILED
                task.retries += 1
                task.cooldown_until = time.time() + DEFAULT_COOLDOWN_SEC
                if task.retries >= MAX_RETRIES:
                    self._rollback(task)

            self._save_state()
            return task

    # ---------------- P4: 带超时的 API 调用 ----------------
    def _call_with_timeout(self, fn: Callable[[Task], str], task: Task) -> str:
        container: Dict[str, Any] = {}
        def worker():
            try:
                container["result"] = fn(task)
            except Exception as e:
                container["error"] = str(e)

        t = threading.Thread(target=worker, daemon=True)
        t.start()
        t.join(timeout=self.timeout)

        if t.is_alive():
            raise TimeoutError(
                f"API调用超时: model={self.model} timeout={self.timeout}s task={task.task_id}"
            )
        if "error" in container:
            raise RuntimeError(container["error"])
        return container.get("result", "")

    # ---------------- 回滚 ----------------
    def backup_before_write(self, task: Task, target_path: Path):
        bak = BACKUP_DIR / f"{task.task_id}.bak"
        if target_path.exists():
            shutil.copy2(target_path, bak)

    def _rollback(self, task: Task):
        bak = BACKUP_DIR / f"{task.task_id}.bak"
        target = Path(task.generated_artifact) if task.generated_artifact else None
        if bak.exists() and target and target.exists():
            shutil.copy2(bak, target)        # 恢复备份
        task.state = TaskState.ROLLED_BACK
        task.cooldown_until = time.time() + DEFAULT_COOLDOWN_SEC * 2

    # ---------------- P1: 闭环活性检测 ----------------
    def _is_stale(self, task: Task) -> bool:
        """业务侧可覆盖。默认通过签名比对判断源是否变化。"""
        return False

    def _find_task(self, spec: str) -> Optional[Task]:
        for t in self.tasks:
            if t.spec == spec:
                return t
        return None

    # ---------------- 状态持久化 ----------------
    def _save_state(self):
        data = {
            "model": self.model,
            "tasks": [
                {**asdict(t), "state": t.state.value} for t in self.tasks
            ],
        }
        STATE_FILE.write_text(json.dumps(data, ensure_ascii=False, indent=2))

    def _load_state(self):
        if not STATE_FILE.exists():
            return
        data = json.loads(STATE_FILE.read_text())
        for td in data.get("tasks", []):
            self.tasks.append(Task(
                task_id=td["task_id"], spec=td["spec"],
                state=TaskState(td["state"]),
                generated_artifact=td.get("generated_artifact", ""),
                verify_result=td.get("verify_result", {}),
                retries=td.get("retries", 0),
                cooldown_until=td.get("cooldown_until", 0.0),
                signature=td.get("signature", ""),
            ))


# ============ CLI ============
if __name__ == "__main__":
    import sys
    cmd = sys.argv[1] if len(sys.argv) > 1 else "run"

    def _demo_generator(task: Task) -> str:
        return f"# generated for {task.spec}\nprint('ok')\n"

    def _demo_verifier(task: Task) -> Dict[str, Any]:
        r = subprocess.run(
            ["python", "-c", task.generated_artifact],
            capture_output=True, text=True, timeout=10,
        )
        return {"ok": r.returncode == 0, "stderr": r.stderr[-200:]}

    if cmd == "run":
        eng = SelfEvolveEngine("model_turbo")
        t = eng.evolve_once("demo-spec", _demo_generator, _demo_verifier)
        print(f"task={t.task_id} state={t.state.value} cooldown={t.cooldown_until}")
    elif cmd == "rollback":
        tid = sys.argv[2]
        eng = SelfEvolveEngine("model_turbo")
        for t in eng.tasks:
            if t.task_id == tid:
                eng._rollback(t)
                print(f"rolled back {tid}")
    else:
        print("usage: python engine_self_evolve_patch.py [run|rollback <id>]")
```

## 三、使用方法

```bash
# 1. 运行一次闭环（生成→验证→冷却重置）
python engine_self_evolve_patch.py run

# 2. 强制回滚某任务
python engine_self_evolve_patch.py rollback <task_id>

# 3. 业务侧集成
from engine_self_evolve_patch import SelfEvolveEngine, Task

eng = SelfEvolveEngine(model="model_a")  # 自动 600s timeout
def gen(t: Task) -> str: ...             # 你的生成器
def ver(t: Task) -> dict: ...            # 你的验证器
task = eng.evolve_once("your-spec", gen, ver)
```

## 四、验证步骤（可执行）

将以下脚本保存为 `test_patch.py` 并运行，4 个用例对应 4 个修复点：

```python
import time, os, json
from pathlib import Path
from engine_self_evolve_patch import (
    SelfEvolveEngine, Task, TaskState, STATE_FILE, BACKUP_DIR
)

def clean():
    for p in [STATE_FILE, *BACKUP_DIR.glob("*")]:
        if Path(p).exists():
            Path(p).unlink() if Path(p).is_file() else None
    if BACKUP_DIR.exists():
        BACKUP_DIR.rmdir()

# P2+P3: 生成后立即验证 + 通过则冷却重置
def test_verify_then_reset_cooldown():
    clean()
    eng = SelfEvolveEngine("model_turbo")
    t = eng.evolve_once("s1", lambda x: "print('ok')",
                        lambda x: {"ok": True})
    assert t.state == TaskState.VERIFIED
    assert t.cooldown_until == 0.0, "P3: cooldown 必须重置为 0"
    print("[PASS] P2+P3 验证立即执行且冷却重置")

# P4: 超时被捕获
def test_api_timeout():
    clean()
    eng = SelfEvolveEngine("model_turbo")
    eng.timeout = 1  # 临时缩短以便测试
    def slow_gen(t): 
        time.sleep(5)
        return "x"
    try:
        eng.evolve_once("s2", slow_gen, lambda x: {"ok": True})
        assert False, "应抛 TimeoutError"
    except TimeoutError as e:
        assert "API调用超时" in str(e)
        print("[PASS] P4 超时按预设 timeout 触发")

# P3: 失败时冷却不重置（对照）
def test_fail_keeps_cooldown():
    clean()
    eng = SelfEvolveEngine("model_turbo")
    t = eng.evolve_once("s3", lambda x: "bad",
                        lambda x: {"ok": False, "stderr": "err"})
    assert t.state == TaskState.FAILED
    assert t.cooldown_until > 0, "失败应设置冷却"
    print("[PASS] 失败保留冷却（对照通过）")

# P1: 已验证任务 stale 时重新激活
def test_reactivate_when_stale():
    clean()
    eng = SelfEvolveEngine("model_turbo")
    t = eng.evolve_once("s4", lambda x: "v1", lambda x: {"ok": True})
    assert t.state == TaskState.VERIFIED
    # 注入 stale
    eng._is_stale = lambda task: True
    t2 = eng.evolve_once("s4", lambda x: "v2", lambda x: {"ok": True})
    assert t2.state == TaskState.VERIFIED
    assert t2.signature != t.signature, "应重新生成而非返回静态结果"
    print("[PASS] P1 stale 任务重新激活，非静态清单")

if __name__ == "__main__":
    test_verify_then_reset_cooldown()
    test_api_timeout()
    test_fail_keeps_cooldown()
    test_reactivate_when_stale()
    print("\nALL TESTS PASSED")
```

## 五、回滚方案

| 场景 | 操作 |
|------|------|
| 单任务回滚 | `python engine_self_evolve_patch.py rollback <task_id>`，从 `_engine_backups/<id>.bak` 恢复 |
| 整体回滚 | 删除 `_engine_state.json` + `_engine_backups/` 即恢复到补丁前状态（补丁无侵入式修改原引擎文件） |
| 代码级回滚 | 本补丁为独立模块，未改动原引擎；删除本文件即可，零副作用 |

## 六、与锚定结论的对应关系

- **"成为真正的自进化闭环，而不是静态任务清单"** → `_is_stale()` + `evolve_once()` 中对 VERIFIED 任务的条件性重新激活（P1）。
- **"每次生成后必须立即验证，验证通过则重置冷却"** → `evolve_once()` 内 generator→verifier 串行调用；验证 ok 分支显式 `cooldown_until = 0.0`（P2、P3）。
- **避坑：API 超时** → `MODEL_TIMEOUTS` 字典 + `_call_with_timeout` 线程级超时（P4），turbo=300s / model_a=600s。
