## 完整脚本

```python
#!/usr/bin/env python3
"""
每日资产盘点脚本
功能：扫描认知池、坑库、归档索引、备份状态，输出当日变化报告
触发：每日定时 / 手动执行
输出：reports/asset_audit_YYYYMMDD.md
"""

import json
import hashlib
import os
import sys
from datetime import datetime, timedelta
from pathlib import Path
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Tuple
import yaml

# ============================================================
# 配置区
# ============================================================

BASE_DIR = Path(os.environ.get("ENGINE_BASE_DIR", "./engine_data"))

ASSET_PATHS = {
    "cognitive_pool": BASE_DIR / "cognitive_pool" / "pool.jsonl",
    "pit_library":    BASE_DIR / "pit_library" / "pits.yaml",
    "archive_index": BASE_DIR / "archive" / "index.json",
    "backup_dir":     BASE_DIR / "backups",
    "snapshot_dir":   BASE_DIR / "snapshots",
    "report_dir":     BASE_DIR / "reports",
}

# 健康检测阈值
THRESHOLDS = {
    "injection_stale_minutes": 30,     # 认知超过30分钟未注入 → 预警
    "injection_stale_max_count": 5,   # 允许的未注入认知上限
    "pit_unresolved_max_age_hours": 48, # 坑超过48小时未闭环 → 预警
    "backup_max_age_hours": 24,         # 备份超过24小时 → 预警
    "archive_orphan_tolerance": 0,     # 归档孤儿文件容忍度
    "hash_mismatch_tolerance": 0,      # 哈希不匹配容忍度
}

# API超时预设（来自历史踩坑经验）
API_TIMEOUTS = {
    "B_turbo": 300,
    "A_model": 600,
}

# ============================================================
# 数据结构
# ============================================================

@dataclass
class AssetRecord:
    """单条资产记录"""
    asset_id: str
    asset_type: str          # cognitive / pit / archive / backup
    content_hash: str       # 内容哈希（用于检测修改）
    metadata: dict          # 附加元数据
    last_injected_at: Optional[str] = None  # 最后注入时间（仅认知池）
    last_updated: str = ""


@dataclass
class DiffResult:
    """差异对比结果"""
    added: List[AssetRecord] = field(default_factory=list)
    modified: List[Tuple[AssetRecord, AssetRecord]] = field(default_factory=list)  # (old, new)
    deleted: List[AssetRecord] = field(default_factory=list)


@dataclass
class HealthAlert:
    """健康预警"""
    severity: str            # CRITICAL / WARNING / INFO
    category: str            # injection / pit / archive / backup
    message: str
    affected_ids: List[str] = field(default_factory=list)


@dataclass
class AuditReport:
    """完整盘点报告"""
    audit_date: str
    scan_started_at: str
    scan_completed_at: str
    asset_summary: Dict[str, int]
    diff: Dict[str, DiffResult]
    alerts: List[HealthAlert]
    health_score: float
    raw_records: Dict[str, Dict[str, AssetRecord]]  # 按资产类型分组的当前快照


# ============================================================
# 第一层：资产扫描
# ============================================================

class AssetScanner:
    """扫描所有资产源，构建当前快照"""

    def scan_all(self) -> Dict[str, Dict[str, AssetRecord]]:
        """执行全量扫描"""
        return {
            "cognitive_pool": self._scan_cognitive_pool(),
            "pit_library":    self._scan_pit_library(),
            "archive_index":  self._scan_archive_index(),
            "backup_status":  self._scan_backup_status(),
        }

    def _scan_cognitive_pool(self) -> Dict[str, AssetRecord]:
        """扫描认知池 JSONL"""
        records = {}
        path = ASSET_PATHS["cognitive_pool"]
        if not path.exists():
            return records

        with open(path, "r", encoding="utf-8") as f:
            for line_num, line in enumerate(f, 1):
                line = line.strip()
                if not line:
                    continue
                try:
                    entry = json.loads(line)
                    cid = entry.get("id", f"cg_line_{line_num}")
                    content = json.dumps(entry.get("content", ""), sort_keys=True, ensure_ascii=False)
                    content_hash = hashlib.sha256(content.encode("utf-8")).hexdigest()[:16]
                    records[cid] = AssetRecord(
                        asset_id=cid,
                        asset_type="cognitive",
                        content_hash=content_hash,
                        metadata={
                            "category": entry.get("category", "unknown"),
                            "source_task": entry.get("source_task", ""),
                            "confidence": entry.get("confidence", 0.0),
                        },
                        last_injected_at=entry.get("last_injected_at"),
                        last_updated=entry.get("created_at", ""),
                    )
                except json.JSONDecodeError:
                    continue
        return records

    def _scan_pit_library(self) -> Dict[str, AssetRecord]:
        """扫描坑库 YAML"""
        records = {}
        path = ASSET_PATHS["pit_library"]
        if not path.exists():
            return records

        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f) or {}

        for pit in data.get("pits", []):
            pid = pit.get("id", "")
            if not pid:
                continue
            content = json.dumps(pit, sort_keys=True, ensure_ascii=False)
            content_hash = hashlib.sha256(content.encode("utf-8")).hexdigest()[:16]
            records[pid] = AssetRecord(
                asset_id=pid,
                asset_type="pit",
                content_hash=content_hash,
                metadata={
                    "level": pit.get("level", "unknown"),       # 工程级/任务级
                    "status": pit.get("status", "open"),         # open / resolved
                    "phenomenon": pit.get("phenomenon", ""),
                    "prevention": pit.get("prevention", ""),
                    "created_at": pit.get("created_at", ""),
                    "resolved_at": pit.get("resolved_at", ""),
                },
                last_updated=pit.get("updated_at", pit.get("created_at", "")),
            )
        return records

    def _scan_archive_index(self) -> Dict[str, AssetRecord]:
        """扫描归档索引"""
        records = {}
        path = ASSET_PATHS["archive_index"]
        if not path.exists():
            return records

        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)

        for entry in data.get("archives", []):
            aid = entry.get("id", "")
            if not aid:
                continue
            content = json.dumps(entry, sort_keys=True, ensure_ascii=False)
            content_hash = hashlib.sha256(content.encode("utf-8")).hexdigest()[:16]
            records[aid] = AssetRecord(
                asset_id=aid,
                asset_type="archive",
                content_hash=content_hash,
                metadata={
                    "file_path": entry.get("file_path", ""),
                    "file_hash": entry.get("file_hash", ""),
                    "archived_at": entry.get("archived_at", ""),
                    "task_id": entry.get("task_id", ""),
                    "size_bytes": entry.get("size_bytes", 0),
                },
                last_updated=entry.get("archived_at", ""),
            )
        return records

    def _scan_backup_status(self) -> Dict[str, AssetRecord]:
        """扫描备份目录"""
        records = {}
        backup_dir = ASSET_PATHS["backup_dir"]
        if not backup_dir.exists():
            return records

        for backup_file in backup_dir.glob("*.tar.gz"):
            stat = backup_file.stat()
            file_hash = self._file_hash(backup_file)
            bid = backup_file.stem  # filename without .tar.gz
            records[bid] = AssetRecord(
                asset_id=bid,
                asset_type="backup",
                content_hash=file_hash[:16],
                metadata={
                    "file_name": backup_file.name,
                    "file_path": str(backup_file),
                    "size_bytes": stat.st_size,
                    "created_at": datetime.fromtimestamp(stat.st_mtime).isoformat(),
                },
                last_updated=datetime.fromtimestamp(stat.st_mtime).isoformat(),
            )
        return records

    @staticmethod
    def _file_hash(filepath: Path, chunk_size: int = 8192) -> str:
        """计算文件SHA256"""
        h = hashlib.sha256()
        with open(filepath, "rb") as f:
            while chunk := f.read(chunk_size):
                h.update(chunk)
        return h.hexdigest()

class DiffEngine:
    """与昨日快照对比，计算 added/modified/deleted"""

    def __init__(self, snapshot_dir: Path):
        self.snapshot_dir = snapshot_dir
        self.snapshot_dir.mkdir(parents=True, exist_ok=True)

    def get_yesterday_snapshot(self) -> Dict[str, Dict[str, AssetRecord]]:
        """加载昨日快照"""
        yesterday = (datetime.now() - timedelta(days=1)).strftime("%Y%m%d")
        snapshot_file = self.snapshot_dir / f"snapshot_{yesterday}.json"
        if not snapshot_file.exists():
            return {}
        with open(snapshot_file, "r", encoding="utf-8") as f:
            data = json.load(f)
        # 重建 AssetRecord
        result = {}
        for asset_type, records in data.items():
            result[asset_type] = {}
            for rid, rdata in records.items():
                result[asset_type][rid] = AssetRecord(**rdata)
        return result

    def save_today_snapshot(self, current: Dict[str, Dict[str, AssetRecord]]):
        """保存今日快照"""
        today = datetime.now().strftime("%Y%m%d")
        snapshot_file = self.snapshot_dir / f"snapshot_{today}.json"
        # 序列化
        serializable = {}
        for asset_type, records in current.items():
            serializable[asset_type] = {}
            for rid, record in records.items():
                serializable[asset_type][rid] = {
                    "asset_id": record.asset_id,
                    "asset_type": record.asset_type,
                    "content_hash": record.content_hash,
                    "metadata": record.metadata,
                    "last_injected_at": record.last_injected_at,
                    "last_updated": record.last_updated,
                }
        with open(snapshot_file, "w", encoding="utf-8") as f:
            json.dump(serializable, f, ensure_ascii=False, indent=2)

    def compute_diff(
        self,
        current: Dict[str, Dict[str, AssetRecord]],
        yesterday: Dict[str, Dict[str, AssetRecord]]
    ) -> Dict[str, DiffResult]:
        """计算各资产类型的差异"""
        diffs = {}
        all_types = set(list(current.keys()) + list(yesterday.keys()))

        for asset_type in all_types:
            curr = current.get(asset_type, {})
            yest = yesterday.get(asset_type, {})
            result = DiffResult()

            curr_ids = set(curr.keys())
            yest_ids = set(yest.keys())

            # 新增
            for aid in curr_ids - yest_ids:
                result.added.append(curr[aid])

            # 删除
            for aid in yest_ids - curr_ids:
                result.deleted.append(yest[aid])

            # 修改（哈希不同）
            for aid in curr_ids & yest_ids:
                if curr[aid].content_hash != yest[aid].content_hash:
                    result.modified.append((yest[aid], curr[aid]))

            diffs[asset_type] = result
        return diffs

class HealthChecker:
    """资产健康度检测，生成预警"""

    def check_all(
        self,
        current: Dict[str, Dict[str, AssetRecord]],
        diff: Dict[str, DiffResult]
    ) -> List[HealthAlert]:
        alerts = []

        # 1. 认知池"只进不出"检测
        alerts.extend(self._check_injection_staleness(current.get("cognitive_pool", {})))

        # 2. 坑库未闭环检测
        alerts.extend(self._check_pit_unresolved(current.get("pit_library", {})))

        # 3. 归档孤儿文件检测
        alerts.extend(self._check_archive_orphan(current.get("archive_index", {})))

        # 4. 备份时效性检测
        alerts.extend(self._check_backup_freshness(current.get("backup_status", {})))

        # 5. 新增认知是否在30分钟内参与注入
        alerts.extend(self._check_new_cognition_injection(diff.get("cognitive_pool")))

        return alerts

    def _check_injection_staleness(self, records: Dict[str, AssetRecord]) -> List[HealthAlert]:
        """检测认知池中长时间未注入的认知"""
        alerts = []
        now = datetime.now()
        stale_ids = []
        never_injected = []

        for rid, record in records.items():
            if record.last_injected_at is None:
                never_injected.append(rid)
                continue
            try:
                injected_time = datetime.fromisoformat(record.last_injected_at)
                stale_minutes = (now - injected_time).total_seconds() / 60
                if stale_minutes > THRESHOLDS["injection_stale_minutes"]:
                    stale_ids.append(rid)
            except (ValueError, TypeError):
                continue

        if never_injected:
            alerts.append(HealthAlert(
                severity="CRITICAL",
                category="injection",
                message=f"认知池中有 {len(never_injected)} 条认知从未被注入任何任务（只进不出）",
                affected_ids=never_injected[:20],
            ))

        if len(stale_ids) > THRESHOLDS["injection_stale_max_count"]:
            alerts.append(HealthAlert(
                severity="WARNING",
                category="injection",
                message=f"认知池中有 {len(stale_ids)} 条认知超过 {THRESHOLDS['injection_stale_minutes']} 分钟未注入",
                affected_ids=stale_ids[:20],
            ))

        return alerts

    def _check_new_cognition_injection(self, cog_diff: Optional[DiffResult]) -> List[HealthAlert]:
        """检测新增认知是否在30分钟内参与注入"""
        alerts = []
        if not cog_diff or not cog_diff.added:
            return alerts

        now = datetime.now()
        new_not_injected = []
        for record in cog_diff.added:
            if record.last_injected_at is None:
                # 检查创建时间
                try:
                    created = datetime.fromisoformat(record.last_updated) if record.last_updated else None
                    if created and (now - created).total_seconds() / 60 > THRESHOLDS["injection_stale_minutes"]:
                        new_not_injected.append(record.asset_id)
                except (ValueError, TypeError):
                    continue

        if new_not_injected:
            alerts.append(HealthAlert(
                severity="WARNING",
                category="injection",
                message=f"今日新增 {len(cog_diff.added)} 条认知，其中 {len(new_not_injected)} 条超过30分钟未注入",
                affected_ids=new_not_injected[:10],
            ))

        return alerts

    def _check_pit_unresolved(self, records: Dict[str, AssetRecord]) -> List[HealthAlert]:
        """检测坑库中长时间未闭环的坑"""
        alerts = []
        now = datetime.now()
        stale_pits = []

        for pid, record in records.items():
            if record.metadata.get("status") != "open":
                continue
            created_str = record.metadata.get("created_at", "")
            try:
                created = datetime.fromisoformat(created_str)
                age_hours = (now - created).total_seconds() / 3600
                if age_hours > THRESHOLDS["pit_unresolved_max_age_hours"]:
                    stale_pits.append(pid)
            except (ValueError, TypeError):
                continue

        if stale_pits:
            alerts.append(HealthAlert(
                severity="WARNING",
                category="pit",
                message=f"坑库中有 {len(stale_pits)} 个坑超过 {THRESHOLDS['pit_unresolved_max_age_hours']} 小时未闭环",
                affected_ids=stale_pits[:10],
            ))

        return alerts

    def _check_archive_orphan(self, records: Dict[str, AssetRecord]) -> List[HealthAlert]:
        """检测归档索引指向的文件是否存在"""
        alerts = []
        orphan_ids = []
        hash_mismatches = []

        for aid, record in records.items():
            file_path = record.metadata.get("file_path", "")
            if file_path and not Path(file_path).exists():
                orphan_ids.append(aid)
                continue
            # 哈希校验
            expected_hash = record.metadata.get("file_hash", "")
            if file_path and expected_hash and Path(file_path).exists():
                actual = AssetScanner._file_hash(Path(file_path))
                if actual != expected_hash:
                    hash_mismatches.append(aid)

        if orphan_ids:
            alerts.append(HealthAlert(
                severity="CRITICAL",
                category="archive",
                message=f"归档索引中有 {len(orphan_ids)} 条指向不存在的文件（孤儿索引）",
                affected_ids=orphan_ids[:10],
            ))

        if hash_mismatches:
            alerts.append(HealthAlert(
                severity="CRITICAL",
                category="archive",
                message=f"归档文件哈希校验失败 {len(hash_mismatches)} 个（文件可能被篡改）",
                affected_ids=hash_mismatches[:10],
            ))

        return alerts

    def _check_backup_freshness(self, records: Dict[str, AssetRecord]) -> List[HealthAlert]:
        """检测备份时效性"""
        alerts = []
        now = datetime.now()

        if not records:
            alerts.append(HealthAlert(
                severity="CRITICAL",
                category="backup",
                message="备份目录为空，无任何备份文件",
            ))
            return alerts

        latest_backup = max(records.values(), key=lambda r: r.last_updated or "")
        try:
            latest_time = datetime.fromisoformat(latest_backup.last_updated)
            age_hours = (now - latest_time).total_seconds() / 3600
            if age_hours > THRESHOLDS["backup_max_age_hours"]:
                alerts.append(HealthAlert(
                    severity="WARNING",
                    category="backup",
                    message=f"最新备份已过期 {age_hours:.1f} 小时（阈值 {THRESHOLDS['backup_max_age_hours']}h）",
                    affected_ids=[latest_backup.asset_id],
                ))
        except (ValueError, TypeError):
            alerts.append(HealthAlert(
                severity="WARNING",
                category="backup",
                message="备份文件时间戳解析失败",
            ))

        return alerts

class ReportGenerator:
    """生成 Markdown 格式的盘点报告"""

    def generate(self, report: AuditReport) -> str:
        lines = []
        lines.append(f"# 每日资产盘点报告 - {report.audit_date}")
        lines.append("")
        lines.append(f"- 扫描开始: `{report.scan_started_at}`")
        lines.append(f"- 扫描完成: `{report.scan_completed_at}`")
        lines.append(f"- 健康度评分: **{report.health_score:.1f} / 100**")
        lines.append("")

        # 资产总览
        lines.append("## 一、资产总览")
        lines.append("")
        lines.append("| 资产类型 | 当前数量 | 昨日数量 | 新增 | 修改 | 删除 |")
        lines.append("|----------|---------|---------|------|------|------|")
        for asset_type, count in report.asset_summary.items():
            d = report.diff.get(asset_type, DiffResult())
            # 从diff推断昨日数量
            yesterday_count = count - len(d.added) + len(d.deleted)
            lines.append(
                f"| {asset_type} | {count} | {yesterday_count} | "
                f"+{len(d.added)} | ~{len(d.modified)} | -{len(d.deleted)} |"
            )
        lines.append("")

        # 详细变化
        lines.append("## 二、当日变化明细")
        lines.append("")
        for asset_type, d in report.diff.items():
            if not d.added and not d.modified and not d.deleted:
                continue
            lines.append(f"### {asset_type}")
            lines.append("")

            if d.added:
                lines.append(f"#### ➕ 新增 ({len(d.added)} 条)")
                lines.append("")
                for r in d.added:
                    lines.append(f"- `{r.asset_id}` | hash={r.content_hash} | {self._fmt_meta(r.metadata)}")
                lines.append("")

            if d.modified:
                lines.append(f"#### ✏️ 修改 ({len(d.modified)} 条)")
                lines.append("")
                for old, new in d.modified:
                    lines.append(f"- `{new.asset_id}`")
                    lines.append(f"  - 旧hash: `{old.content_hash}`")
                    lines.append(f"  - 新hash: `{new.content_hash}`")
                    lines.append(f"  - 元数据: {self._fmt_meta(new.metadata)}")
                lines.append("")

            if d.deleted:
                lines.append(f"#### ❌ 删除 ({len(d.deleted)} 条)")
                lines.append("")
                for r in d.deleted:
                    lines.append(f"- `{r.asset_id}` | hash={r.content_hash} | {self._fmt_meta(r.metadata)}")
                lines.append("")

        # 健康预警
        lines.append("## 三、健康预警")
        lines.append("")
        if not report.alerts:
            lines.append("✅ 所有健康检测项通过，无预警。")
        else:
            critical = [a for a in report.alerts if a.severity == "CRITICAL"]
            warning = [a for a in report.alerts if a.severity == "WARNING"]
            info = [a for a in report.alerts if a.severity == "INFO"]

            if critical:
                lines.append(f"### 🔴 CRITICAL ({len(critical)} 项)")
                lines.append("")
                for a in critical:
                    lines.append(f"- **[{a.category}]** {a.message}")
                    if a.affected_ids:
                        lines.append(f"  - 影响ID: {', '.join(a.affected_ids)}")
                lines.append("")

            if warning:
                lines.append(f"### 🟡 WARNING ({len(warning)} 项)")
                lines.append("")
                for a in warning:
                    lines.append(f"- **[{a.category}]** {a.message}")
                    if a.affected_ids:
                        lines.append(f"  - 影响ID: {', '.join(a.affected_ids)}")
                lines.append("")

            if info:
                lines.append(f"### 🔵 INFO ({len(info)} 项)")
                lines.append("")
                for a in info:
                    lines.append(f"- **[{a.category}]** {a.message}")
                lines.append("")

        # 建议
        lines.append("## 四、处置建议")
        lines.append("")
        suggestions = self._generate_suggestions(report)
        for i, s in enumerate(suggestions, 1):
            lines.append(f"{i}. {s}")
        lines.append("")

        lines.append("---")
        lines.append(f"*本报告由 R2-维度拆解模块 / 每日资产盘点脚本 自动生成*")

        return "\n".join(lines)

    @staticmethod
    def _fmt_meta(metadata: dict) -> str:
        """格式化元数据为简短字符串"""
        items = []
        for k, v in metadata.items():
            if isinstance(v, str) and len(v) > 40:
                v = v[:37] + "..."
            items.append(f"{k}={v}")
        return " | ".join(items[:4])  # 最多展示4个字段

    @staticmethod
    def _generate_suggestions(report: AuditReport) -> List[str]:
        """根据预警生成处置建议"""
        suggestions = []
        has_injection_issue = any(a.category == "injection" for a in report.alerts)
        has_pit_issue = any(a.category == "pit" for a in report.alerts)
        has_archive_issue = any(a.category == "archive" for a in report.alerts)
        has_backup_issue = any(a.category == "backup" for a in report.alerts)

        if has_injection_issue:
            suggestions.append(
                "认知池存在'只进不出'风险：执行认知缓存刷新，将未注入认知在下一轮任务调度中强制注入。"
                "参考坑库经验：每10任务或每转结束必须刷新认知缓存。"
            )
        if has_pit_issue:
            suggestions.append(
                "坑库存在未闭环项：检查超时坑是否需要升级处理或标记为'已接受风险'。"
            )
        if has_archive_issue:
            suggestions.append(
                "归档索引与文件不一致：对孤儿索引执行清理或重建索引，对哈希不匹配文件执行恢复。"
            )
        if has_backup_issue:
            suggestions.append(
                "备份时效性不达标：立即触发一次全量备份，并检查备份调度任务是否正常运行。"
            )
        if not report.alerts:
            suggestions.append("所有资产健康度良好，维持当前调度策略。")
        return suggestions

def calculate_health_score(alerts: List[HealthAlert], total_assets: int) -> float:
    """计算健康度评分"""
    if total_assets == 0:
        return 0.0
    score = 100.0
    for alert in alerts:
        if alert.severity == "CRITICAL":
            score -= min(15, 5 + len(alert.affected_ids) * 0.5)
        elif alert.severity == "WARNING":
            score -= min(8, 2 + len(alert.affected_ids) * 0.2)
        elif alert.severity == "INFO":
            score -= 0.5
    return max(0.0, min(100.0, score))


def run_daily_audit():
    """主入口：执行每日资产盘点"""
    scan_started = datetime.now()
    print(f"[{scan_started.isoformat()}] 开始每日资产盘点...")

    # Step 1: 扫描所有资产
    scanner = AssetScanner()
    current_snapshot = scanner.scan_all()

    # Step 2: 加载昨日快照并计算差异
    diff_engine = DiffEngine(ASSET_PATHS["snapshot_dir"])
    yesterday_snapshot = diff_engine.get_yesterday_snapshot()
    diffs = diff_engine.compute_diff(current_snapshot, yesterday_snapshot)

    # Step 3: 健康检测
    health_checker = HealthChecker()
    alerts = health_checker.check_all(current_snapshot, diffs)

    # Step 4: 计算健康度
    total_assets = sum(len(records) for records in current_snapshot.values())
    health_score = calculate_health_score(alerts, total_assets)

    # Step 5: 保存今日快照
    diff_engine.save_today_snapshot(current_snapshot)

    # Step 6: 生成报告
    scan_completed = datetime.now()
    report = AuditReport(
        audit_date=scan_started.strftime("%Y-%m-%d"),
        scan_started_at=scan_started.isoformat(),
        scan_completed_at=scan_completed.isoformat(),
        asset_summary={k: len(v) for k, v in current_snapshot.items()},
        diff=diffs,
        alerts=alerts,
        health_score=health_score,
        raw_records=current_snapshot,
    )

    generator = ReportGenerator()
    markdown_report = generator.generate(report)

    # 输出报告
    ASSET_PATHS["report_dir"].mkdir(parents=True, exist_ok=True)
    report_file = ASSET_PATHS["report_dir"] / f"asset_audit_{scan_started.strftime('%Y%m%d')}.md"
    with open(report_file, "w", encoding="utf-8") as f:
        f.write(markdown_report)

    # 控制台摘要
    print(f"\n{'='*60}")
    print(f"盘点完成 | 耗时: {(scan_completed - scan_started).total_seconds():.1f}s")
    print(f"健康度: {health_score:.1f}/100")
    print(f"资产总览:")
    for asset_type, count in report.asset_summary.items():
        d = diffs.get(asset_type, DiffResult())
        print(f"  {asset_type:20s}: {count:4d} (+{len(d.added)} ~{len(d.modified)} -{len(d.deleted)})")
    print(f"预警: {len(alerts)} 条 "
          f"(CRITICAL={sum(1 for a in alerts if a.severity=='CRITICAL')}, "
          f"WARNING={sum(1 for a in alerts if a.severity=='WARNING')})")
    print(f"报告已保存: {report_file}")
    print(f"{'='*60}\n")

    # 如有CRITICAL预警，返回非零退出码
    if any(a.severity == "CRITICAL" for a in alerts):
        print("⚠️ 存在 CRITICAL 级别预警，请立即处理！")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(run_daily_audit())
```

## 关键设计说明

| 设计点 | 解决的问题 | 对应历史踩坑 |
|--------|-----------|-------------|
| **注入时效检测** (`_check_injection_staleness`) | 检测认知池中超过30分钟未注入的认知 | 坑库[工程级]认知池只进不出 |
| **新增认知注入检查** (`_check_new_cognition_injection`) | 今日新增认知是否30分钟内参与注入 | 坑库：新认知30分钟内必须参与注入 |
| **快照Diff机制** (`DiffEngine`) | 通过昨日快照对比，新增/修改/删除一目了然 | 报告需要变化可见 |
| **哈希校验** (`content_hash` / `file_hash`) | 检测内容篡改与归档文件一致性 | 备份完整性 |
| **坑未闭环检测** (`_check_pit_unresolved`) | 超过48小时未解决的坑需升级 | 坑库管理闭环 |
| **健康度评分** | 量化资产健康状态，便于趋势追踪 | 需要可度量的资产状态 |
| **CRITICAL退出码=1** | 脚本可被调度系统捕获，触发告警 | 工程级运维自动化 |
